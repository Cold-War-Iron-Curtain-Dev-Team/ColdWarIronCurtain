# ===== cwic_country_add_profit_<corp>
#@each sector=hydrocarbons
cwic_country_add_profit_<corp> = {
	# Adds this country's operating profit contribution for <corp> into the global monthly pool.
	# Requires: global.CWIC_CORP_BASE_PROFIT_UNIT, CWIC_oil_output_index, and local presence var corp_<corp>_presence_hydrocarbons (0..5)
	if = {
		limit = { has_variable = corp_<corp>_presence_hydrocarbons }

		# presence_factor = presence / 5 (v1 linear)
		set_temp_variable = { p = corp_<corp>_presence_hydrocarbons }
		divide_temp_variable = { p = 5 }
		clamp_temp_variable = { var = p min = 0 max = 1 }

		# profit_add = BASE_UNIT * sector_index * presence_factor
		set_temp_variable = { profit_add = global.CWIC_CORP_BASE_PROFIT_UNIT }
		multiply_temp_variable = { profit_add = CWIC_oil_output_index }
		multiply_temp_variable = { profit_add = p }

		add_to_variable = { global.CWIC_CORP_<corp>_profit_month = profit_add }
	}
}
#@end

# ===== cwic_finalize_corp_<corp>
#@each sector=hydrocarbons
cwic_finalize_corp_<corp> = {
	# Finalizes prior-month operating profit into cash/cap and sets this month's spend budget.

	# gross = last month's accumulated operating profit
	set_temp_variable = { gross = global.CWIC_CORP_<corp>_profit_month }

	# repatriated = gross * sector repatriation rate
	set_temp_variable = { rep = gross }
	multiply_temp_variable = { rep = global.CWIC_sector_repatriation_hydrocarbons }

	# optional macro haircut from overall corporate risk (keeps cash from exploding in high-risk worlds)
	# hair = clamp(0.50..1.00, 1 - (CWIC_corporate_risk_index/200))
	set_temp_variable = { rf = CWIC_corporate_risk_index }
	divide_temp_variable = { rf = 200 }
	set_temp_variable = { hair = 1 }
	subtract_from_temp_variable = { hair = rf }
	clamp_temp_variable = { var = hair min = 0.5 max = 1.0 }
	multiply_temp_variable = { rep = hair }

	# cash += rep (cash persists)
	add_to_variable = { global.CWIC_CORP_<corp>_cash = rep }
	clamp_variable = { var = global.CWIC_CORP_<corp>_cash min = 0 max = 1000000 }

	# cap += rep * 0.25 (retained earnings expand long-run capacity; tune later)
	set_temp_variable = { cap_add = rep }
	multiply_temp_variable = { cap_add = 0.25 }
	add_to_variable = { global.CWIC_CORP_<corp>_cap = cap_add }

	# shock factor (per corp)
	set_temp_variable = { shock_factor = 1.0 }
	if = {
		limit = { check_variable = { global.CWIC_CORP_<corp>_shock_timer > 0 } }
		set_temp_variable = { shock_factor = 0.5 }
		# tick shock timer down (1 month)
		subtract_from_variable = { global.CWIC_CORP_<corp>_shock_timer = 1 }
		clamp_variable = { var = global.CWIC_CORP_<corp>_shock_timer min = 0 max = 120 }
	}

	# desired_budget = cash * invest_rate * macro_risk_factor * shock_factor
	set_temp_variable = { desired = global.CWIC_CORP_<corp>_cash }
	multiply_temp_variable = { desired = global.CWIC_CORP_INVEST_RATE }
	multiply_temp_variable = { desired = CWIC_global_budget_risk_factor }
	multiply_temp_variable = { desired = shock_factor }

	# clamp desired budget
	clamp_temp_variable = { var = desired min = global.CWIC_CORP_BUDGET_MIN max = global.CWIC_CORP_BUDGET_MAX }

	# budget cannot exceed available cash
	set_temp_variable = { budget = desired }
	if = {
		limit = { check_variable = { budget > global.CWIC_CORP_<corp>_cash } }
		set_temp_variable = { budget = global.CWIC_CORP_<corp>_cash }
	}

	set_variable = { global.CWIC_CORP_<corp>_budget = budget }

	# book-keeping
	set_variable = { global.CWIC_CORP_<corp>_profit_last = gross }
}
#@end

# ===== cwic_corp_<corp>_invest_presence_hydrocarbons
#@each sector=hydrocarbons
cwic_corp_<corp>_invest_presence_hydrocarbons = {

	# Do nothing if no cash
	if = {
		limit = { NOT = { check_variable = { global.CWIC_CORP_<corp>_cash > 0 } } }
	}
	else = {

		# Determine current tier
		set_temp_variable = { cur = 0 }
		if = {
			limit = { has_variable = corp_<corp>_presence_hydrocarbons }
			set_temp_variable = { cur = corp_<corp>_presence_hydrocarbons }
		}

		# If already maxed (tier 5), stop
		if = {
			limit = { check_variable = { cur > 4 } }
		}
		else = {

			# target_tier = cur + 1 (or 1 if new)
			set_temp_variable = { target_tier = cur }
			add_to_temp_variable = { target_tier = 1 }
			clamp_temp_variable = { var = target_tier min = 1 max = 5 }

			# compute invest_cost for this sector in this country
			cwic_calc_presence_upgrade_cost_hydrocarbons = yes

			# Gate by cash and by budget remaining
			if = {
				limit = {
					OR = {
						check_variable = { global.CWIC_CORP_<corp>_cash > invest_cost }
						check_variable = { global.CWIC_CORP_<corp>_cash = invest_cost }
					}
					OR = {
						check_variable = { global.CWIC_CORP_<corp>_budget_remaining > invest_cost }
						check_variable = { global.CWIC_CORP_<corp>_budget_remaining = invest_cost }
					}
				}

				# Spend
				subtract_from_variable = { global.CWIC_CORP_<corp>_cash = invest_cost }
				subtract_from_variable = { global.CWIC_CORP_<corp>_budget_remaining = invest_cost }
				clamp_variable = { var = global.CWIC_CORP_<corp>_cash min = 0 max = 1000000 }
				clamp_variable = { var = global.CWIC_CORP_<corp>_budget_remaining min = 0 max = 1000000 }

				# Apply: entry or upgrade
				if = {
					limit = { NOT = { has_variable = corp_<corp>_presence_hydrocarbons } }
					set_variable = { corp_<corp>_presence_hydrocarbons = 1 }
				}
				else = {
					add_to_variable = { corp_<corp>_presence_hydrocarbons = 1 }
					clamp_variable = { var = corp_<corp>_presence_hydrocarbons min = 0 max = 5 }
				}

				# Optional: host-country effects as presence rises (stub)
				# cwic_apply_corp_presence_effects_<corp> = yes
			}
		}
	}
}
#@end

	# Targeted nationalization: <Name> (hydrocarbons)
	if = { limit = { NOT = { has_variable = corp_<corp>_presence_hydrocarbons } } }
	else_if = { limit = { has_country_flag = cwic_mnc_frozen_<corp>_hydrocarbons } }
	else = {
		set_temp_variable = { cur = corp_<corp>_presence_hydrocarbons }
		set_temp_variable = { seize_amount = cur }
		multiply_temp_variable = { seize_amount = global.CWIC_MNC_NAT_SEIZE_UNIT_HYDRO }
		set_temp_variable = { tmp = CWIC_oil_output_index }
		divide_temp_variable = { tmp = 4 }
		clamp_temp_variable = { var = tmp min = 0.5 max = 2.0 }
		multiply_temp_variable = { seize_amount = tmp }
		set_temp_variable = { corp_cash = global.CWIC_CORP_<corp>_cash }
		if = { limit = { check_variable = { seize_amount > corp_cash } } set_temp_variable = { seize_amount = corp_cash } }
		add_to_variable = { currentMoney = seize_amount }
		subtract_from_variable = { global.CWIC_CORP_<corp>_cash = seize_amount }
		clamp_variable = { var = global.CWIC_CORP_<corp>_cash min = 0 max = 100000000 }
		if = { limit = { check_variable = { cur = 1 } } clear_variable = corp_<corp>_presence_hydrocarbons }
		else = { subtract_from_variable = { corp_<corp>_presence_hydrocarbons = 1 } clamp_variable = { var = corp_<corp>_presence_hydrocarbons min = 0 max = 5 } }
		set_country_flag = cwic_mnc_frozen_<corp>_hydrocarbons
		set_country_flag = { flag = cwic_mnc_frozen_<corp>_hydrocarbons days = 180 }
		add_to_variable = { CWIC_nationalization_pressure_index = 0.15 }
		clamp_variable = { var = CWIC_nationalization_pressure_index min = 0 max = 5 }
		add_to_variable = { CWIC_instability_index_corp_factor = 0.05 }

# ===== cwic_mnc_process_invites_global
cwic_mnc_process_invites_global = {
	# Scans ONLY countries flagged as having pending MNC invites.
	# Fires at most one invite event per eligible country per month (per-country cooldown).
	# Call this from a GLOBAL monthly pulse (after budgets are set).

	every_country = {
		limit = {
			has_country_flag = cwic_has_pending_mnc_invite
			NOT = { has_country_flag = cwic_mnc_invite_evt_global_cd }
		}
		set_country_flag = { flag = cwic_mnc_invite_evt_global_cd days = 30 }

		if = { limit = { always = no } }  # chain start, keeps every company an else_if
#@each all
		else_if = {
			limit = {
				has_variable = cwic_invite_<corp>_<sector>
				NOT = { has_country_flag = cwic_mnc_invite_evt_cd_<corp>_<sector> }
			}
			country_event = { id = <invite_event> }
			set_country_flag = { flag = cwic_mnc_invite_evt_cd_<corp>_<sector> days = 30 }
		}
#@end

		else = { }

		# Cleanup: clear pending flag each run; it will be re-added when a new invite/concession is created.
		clr_country_flag = cwic_has_pending_mnc_invite
	}
}
