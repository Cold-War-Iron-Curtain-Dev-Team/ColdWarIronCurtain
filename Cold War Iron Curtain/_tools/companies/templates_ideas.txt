# Foreign investment idea slots (politics screen), one slot per sector.
# Slotting a company = the old 'invite' decision (market economies) or 'concession' decision (others, with
# cwic_allows_foreign_concessions). The AI still uses those decisions (now AI-only), so these ideas have ai_will_do 0.
ideas = {
	cwic_fi_hydrocarbons = {
#@each sector=hydrocarbons
		cwic_fi_<corp> = {
			allowed = { always = yes }
			visible = {
				OR = {
					has_free_market_bonus_max = yes
					has_moderate_free_market_bonus = yes
					has_socialist_free_market_bonus = yes
					check_variable = { economic_x_axis > 10 }
					check_variable = { economic_x_axis = 10 }
				}
			}
			available = {
				NOT = { check_variable = { CWIC_nationalization_pressure_index > 4.5 } }
				OR = {
					NOT = { has_variable = corp_<corp>_presence_<sector> }
					check_variable = { corp_<corp>_presence_<sector> < 5 }
				}
				NOT = { has_country_flag = cwic_invite_cd_<corp>_<sector> }
			}
			cost = 25
			removal_cost = 0
			picture = generic_refining_concern_1
			on_add = {
				set_variable = { cwic_invite_<corp>_<sector> = 60 } # months
				set_country_flag = cwic_has_pending_mnc_invite
				set_country_flag = { flag = cwic_invite_cd_<corp>_<sector> days = 90 value = 1 }
			}
			ai_will_do = { factor = 0 }
		}
