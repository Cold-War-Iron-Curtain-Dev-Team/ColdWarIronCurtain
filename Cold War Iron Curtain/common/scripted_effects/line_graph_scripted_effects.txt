SAMPLE_add_random_point_to_line = {
	# Adds a random point to the line graph

	# This count is displayed as an x-axis label, increment it
	add_to_variable = { SAMPLE_count = 1 }

	# We keep track of the last y position so we can calculate the slope
	set_variable = { SAMPLE_last_y_position = SAMPLE_y_position }
	
	# Random new y position, 10-90
	set_variable = { SAMPLE_y_position = random }
	multiply_variable = { SAMPLE_y_position = 80 }
	add_to_variable = { SAMPLE_y_position = 10 }
	round_variable = SAMPLE_y_position

	# Prune arrays so they hold at most 20 elements
	if = {
		limit = {
			check_variable = { SAMPLE_count > 20 }
		}
		remove_from_array = { array = SAMPLE_graph_offsets index = 0 }
		remove_from_array = { array = SAMPLE_graph_slopes index = 0 }
		remove_from_array = { array = SAMPLE_numbers index = 0 }
	}

	# The y position offset is the inverse of the y position
	set_temp_variable = { offset = SAMPLE_last_y_position }
	multiply_temp_variable = { offset = -1 }
	add_to_array = { SAMPLE_graph_offsets = offset }

	# The 'slope' here is simply the difference between the current y value and the last one, divided by 2 plus 50
	# These last two changes are needed for the shader
	set_temp_variable = { diff = SAMPLE_y_position }
	subtract_from_temp_variable = { diff = SAMPLE_last_y_position }
	divide_temp_variable = { diff = 2 }
	add_to_temp_variable = { diff = 50 }
	add_to_array = { SAMPLE_graph_slopes = diff }
	
	add_to_array = { SAMPLE_numbers = SAMPLE_count }
}


# ---------------------------------------------------------------------------
# Finance tab line graphs
#
# Each graph keeps up to 20 monthly samples in <Name>_history and is drawn from
# <Name>_graph_offsets / <Name>_graph_slopes (see the line_graph_* scripted GUIs).
# Scripted effects cannot take array names as parameters, so every graph has a small
# wrapper that copies its history into finance_graph_work, calls
# finance_graph_build_segments, and copies the result back.
#
# Month_Label (00_on_actions.txt) holds one month per sample and is seeded to the same
# length, so the x-axis labels line up with the points.
# ---------------------------------------------------------------------------

# Pads a new history to 12 copies of the first sample (flat line, not an empty chart)
# and keeps at most 20. Input: finance_graph_work. Output: finance_graph_work_offsets,
# finance_graph_work_slopes and the temp variables graph_axis_bottom / graph_axis_top.
finance_graph_build_segments = {
	clear_array = finance_graph_work_offsets
	clear_array = finance_graph_work_slopes

	find_highest_in_array = { array = finance_graph_work value = graph_max index = graph_max_index }
	find_lowest_in_array = { array = finance_graph_work value = graph_min index = graph_min_index }
	set_temp_variable = { graph_range = graph_max }
	subtract_from_temp_variable = { graph_range = graph_min }

	# Small moves must look small: the chart spans at least 10% of the largest magnitude,
	# otherwise a 0.1% change would be stretched from the bottom to the top of the chart
	set_temp_variable = { graph_floor = graph_max }
	if = {
		limit = { check_variable = { graph_floor < 0 } }
		multiply_temp_variable = { graph_floor = -1 }
	}
	set_temp_variable = { graph_abs_min = graph_min }
	if = {
		limit = { check_variable = { graph_abs_min < 0 } }
		multiply_temp_variable = { graph_abs_min = -1 }
	}
	if = {
		limit = { check_variable = { graph_abs_min > graph_floor } }
		set_temp_variable = { graph_floor = graph_abs_min }
	}
	multiply_temp_variable = { graph_floor = 0.1 }
	if = {
		limit = { check_variable = { graph_floor = 0 } }
		set_temp_variable = { graph_floor = 1 }
	}

	# Bottom of the chart: centre the data when the range is padded up to the floor
	set_temp_variable = { graph_base = graph_min }
	if = {
		limit = { check_variable = { graph_range < graph_floor } }
		set_temp_variable = { graph_pad = graph_floor }
		subtract_from_temp_variable = { graph_pad = graph_range }
		divide_temp_variable = { graph_pad = 2 }
		subtract_from_temp_variable = { graph_base = graph_pad }
		set_temp_variable = { graph_range = graph_floor }
	}
	# Values at the bottom (y 10) and top (y 90) of the chart, shown as the axis labels
	set_temp_variable = { graph_axis_bottom = graph_base }
	set_temp_variable = { graph_axis_top = graph_base }
	add_to_temp_variable = { graph_axis_top = graph_range }

	set_temp_variable = { graph_last_y = -1 }
	for_each_loop = {
		array = finance_graph_work
		value = graph_v
		# Scale into the chart's 10-90 band
		set_temp_variable = { graph_y = graph_v }
		subtract_from_temp_variable = { graph_y = graph_base }
		divide_temp_variable = { graph_y = graph_range }
		multiply_temp_variable = { graph_y = 80 }
		add_to_temp_variable = { graph_y = 10 }
		round_temp_variable = graph_y
		if = {
			limit = { check_variable = { graph_last_y = -1 } }
			set_temp_variable = { graph_last_y = graph_y }
		}
		# Segment offset is -y1; the shader slope frame is (y2 - y1) / 2 + 50
		set_temp_variable = { graph_offset = graph_last_y }
		multiply_temp_variable = { graph_offset = -1 }
		add_to_array = { finance_graph_work_offsets = graph_offset }
		set_temp_variable = { graph_slope = graph_y }
		subtract_from_temp_variable = { graph_slope = graph_last_y }
		divide_temp_variable = { graph_slope = 2 }
		add_to_temp_variable = { graph_slope = 50 }
		add_to_array = { finance_graph_work_slopes = graph_slope }
		set_temp_variable = { graph_last_y = graph_y }
	}
}

# Debt-GDP ratio is derived here and also shown by GDP_Debt_Label
GDP_Debt_Graph_Financial_Tab_Update = {
	set_variable = { GDP_Debt_Ratio = currentDebt }
	if = {
		limit = { check_variable = { grossdomesticproduct > 0 } }
		divide_variable = { GDP_Debt_Ratio = grossdomesticproduct }
	}
	Debt_Graph_Financial_Tab_Update = yes
}

GDP_Graph_Financial_Tab_Update = {
	add_to_array = { GDP_history = grossdomesticproduct }
	while_loop_effect = {
		limit = { check_variable = { GDP_history^num < 12 } }
		add_to_array = { GDP_history = grossdomesticproduct }
	}
	while_loop_effect = {
		limit = { check_variable = { GDP_history^num > 20 } }
		remove_from_array = { array = GDP_history index = 0 }
	}
	clear_array = finance_graph_work
	for_each_loop = { array = GDP_history value = graph_v add_to_array = { finance_graph_work = graph_v } }
	finance_graph_build_segments = yes
	clear_array = GDP_graph_offsets
	clear_array = GDP_graph_slopes
	for_each_loop = { array = finance_graph_work_offsets value = graph_v add_to_array = { GDP_graph_offsets = graph_v } }
	for_each_loop = { array = finance_graph_work_slopes value = graph_v add_to_array = { GDP_graph_slopes = graph_v } }
	set_variable = { GDP_axis_bottom = graph_axis_bottom }
	set_variable = { GDP_axis_top = graph_axis_top }
}
