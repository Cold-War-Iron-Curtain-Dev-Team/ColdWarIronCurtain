MLA_prepare_emergency_operation = {
	add_timed_idea = {
		idea = MLA_prepared_operation
		days = 75
	}
	# Front-load a strike package so the assaulting columns actually field artillery,
	# support weapons and transport for the border war instead of raw riflemen.
	add_manpower = 750
	add_equipment_to_stockpile = { type = infantry_equipment_0 amount = 400 producer = MLA }
	add_equipment_to_stockpile = { type = artillery_equipment_1 amount = 24 producer = CUM }
	add_equipment_to_stockpile = { type = support_equipment_1 amount = 40 producer = MLA }
	add_equipment_to_stockpile = { type = motorized_equipment_1 amount = 30 producer = ENG }
	set_country_flag = {
		flag = MLA_operation_cooldown
		days = 45
	}
}

		}

		army_experience = 3

		MLA = {
			add_to_variable = { emergency_momentum = -8 }

				days = 60
			}
		}
	}
	MLA = { MLA_clamp_emergency_momentum = yes }
}

		}

			annex_country = {
					target = MAL
				}

				every_state = {

					}
					set_party_name = { 
						ideology = neutrality

						long_name = "Eastern Command of the British Army"
					}
				}
			}
			MAL = {
				MAL_Gerald_Templer = {
					set_nationality = SGP
					add_country_leader_role = {
						promote_leader = yes
						country_leader = {
							ideology = despotism
						}

				}

				}
			}
			every_country = {
				limit = {
