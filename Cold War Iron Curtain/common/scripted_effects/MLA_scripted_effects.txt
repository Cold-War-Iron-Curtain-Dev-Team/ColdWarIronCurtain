MAL_emergency_shift_control = {
	if = {
		limit = {
			MAL_emergency_is_contest_state = yes
			MAL_emergency_contest_active = yes
		}
		custom_effect_tooltip = MAL_EMERGENCY_CONTROL_SHIFT_TT
		if = {
			limit = {
				check_variable = { emergency_control > 59 }
				check_variable = { emergency_control_shift > 0 }
			}
			multiply_temp_variable = { emergency_control_shift = 0.5 }
		}
		add_to_variable = { emergency_control = emergency_control_shift }
		clamp_variable = { var = emergency_control min = -100 max = 100 }
		MAL_emergency_set_control_frame = yes
	}
}

MAL_emergency_set_control_frame = {
	if = {
		limit = { check_variable = { emergency_control > 59 } }

	}
	else_if = {
		limit = { check_variable = { emergency_control > 19 } }
		set_variable = { Malaysia_Control = 13 }
	}
	else_if = {
		limit = { check_variable = { emergency_control > -20 } }
		set_variable = { Malaysia_Control = 12 }
	}
	else_if = {
		limit = { check_variable = { emergency_control > -60 } }
		set_variable = { Malaysia_Control = 11 }
	}
	else = {
		set_variable = { Malaysia_Control = 10 }
	}
}

MAL_emergency_recount = {
	set_variable = { emergency_liberated_states = 0 }
	set_variable = { emergency_secured_states = 0 }
	set_temp_variable = { emergency_control_sum = 0 }
	every_state = {
		limit = { MAL_emergency_is_contest_state = yes }
		add_to_temp_variable = { emergency_control_sum = emergency_control }
		if = {
			limit = { check_variable = { emergency_control > 59 } }
			PREV = { add_to_variable = { emergency_liberated_states = 1 } }
		}
		if = {
			limit = { check_variable = { emergency_control < -59 } }
			PREV = { add_to_variable = { emergency_secured_states = 1 } }
		}
	}
	set_variable = { emergency_momentum = emergency_control_sum }
	divide_variable = { emergency_momentum = 3.6 }
	MLA_clamp_emergency_momentum = yes
}

MAL_emergency_ai_turn = {
	if = {
		limit = {
			is_ai = yes
			MAL_emergency_contest_active = yes
			has_political_power > 25
			NOT = { has_country_flag = Recent_MLA_State_Decision }
		}
		set_temp_variable = { MAL_emergency_ai_skip = 0 }
		if = {
			limit = { tag = MLA is_historical_focus_on = yes }
			random = {
				chance = 70
				set_temp_variable = { MAL_emergency_ai_skip = 1 }
			}
		}
		if = {
			limit = { tag = MLA }
			random_state = {
				limit = {
					MAL_emergency_is_contest_state = yes
					check_variable = { emergency_control > -20 }
					check_variable = { emergency_control < 100 }
				}
				save_event_target_as = MAL_emergency_ai_target
			}
			if = {
				limit = { NOT = { has_event_target = MAL_emergency_ai_target } }
				random_state = {
					limit = {
						MAL_emergency_is_contest_state = yes
						check_variable = { emergency_control < 100 }
					}
					save_event_target_as = MAL_emergency_ai_target
				}
			}
		}
		else = {
			random_state = {
				limit = {
					MAL_emergency_is_contest_state = yes
					check_variable = { emergency_control > 19 }
				}
				save_event_target_as = MAL_emergency_ai_target
			}
			if = {
				limit = { NOT = { has_event_target = MAL_emergency_ai_target } }
				random_state = {
					limit = {
						MAL_emergency_is_contest_state = yes
						check_variable = { emergency_control > -100 }
					}
					save_event_target_as = MAL_emergency_ai_target
				}
			}
		}
		if = {
			limit = {
				has_event_target = MAL_emergency_ai_target
				check_variable = { MAL_emergency_ai_skip = 0 }
			}
			set_temp_variable = { MAL_emergency_ai_done = 0 }
			for_each_loop = {
				array = malaya_gui_decisions_list
				index = i
				break = MAL_emergency_ai_done
				if = {
					limit = { event_target:MAL_emergency_ai_target = { malaya_state_decision_enabled = yes } }
					log = "IC_MALAYA|AI_ACTION|date=[GetDateText]|tag=[Root.GetTag]|state=[event_target:MAL_emergency_ai_target.GetName]|action=[?malaya_gui_decisions_list^i]"
					event_target:MAL_emergency_ai_target = { mla_malaya_state_pseudo_decision_effect = yes }
					log = "IC_MALAYA|AI_ACTION_DONE|tag=[Root.GetTag]"
					add_political_power = -25
					set_country_flag = { flag = Recent_MLA_State_Decision days = 5 value = 1 }
					set_temp_variable = { MAL_emergency_ai_done = 1 }
				}
			}
		}
	}
}
