# ============================================================================
# 1953 Laos Raid mechanic
# ----------------------------------------------------------------------------
# The Phase3 block of ic_pulse (IC_scripted_effects.txt) releases the Pathet Lao
# (LAO) as a VIN satellite and throws them at the Kingdom of Laos (LOS). Left
# alone, VIN's far larger army simply overruns LOS. This file turns that scripted
# defeat into the historical over-extended raid: LOS gets a scaled CEFEO relief
# column, VIN fights under an over-extension penalty, and a timer resolves the
# raid if Hanoi cannot take a Laotian capital in time.
#
# Outcomes (decided at Laos_Raid_Timer expiry, see ic_laos_raid_resolve):
#   * attacker holds Vientiane/Luang Prabang -> conquest stands (ic_laos_raid_attacker_victory)
#   * human attacker, no capital -> one last-chance "press vs. call off" prompt (laos_raid.1)
#   * AI (or lapsed last chance), LAO still holds its zones -> stalemate, rump satellite
#   * AI (or lapsed last chance), LAO pushed out -> total failure, LOS accepts or goes total war
# ============================================================================

# --- Release the Pathet Lao ------------------------------------------------

ic_laos_raid_release_pathet_lao = {
   	set_global_flag = Laos_Pathet_Released
	every_state = {
		limit = {
			OR = {
				is_owned_by = LOS
				is_core_of = LOS
			}
		}
		add_core_of = LAO
	}
    1295 = {
        transfer_state_to = LAO
    }
   	LAO = {
  		set_capital = { state = 1295 }
  		hidden_effect = { keep_current_politics = yes }
   	}
	lao_deploy_field_force = yes	# the Pathet Lao spawned with nothing at all and fought the whole raid on VIN's divisions
	VIN = {
	    country_event = VINLAO.1
		set_autonomy = {
			target = LAO
			autonomy_state = autonomy_easternbloc
		}
	}
}

# --- The Pathet Lao's own divisions -----------------------------------------

lao_deploy_field_force = {
	if = {
		limit = { country_exists = LAO }
		set_temp_variable = { lao_force_tier = 1 }
		if = {
			limit = {
				OR = {
					NOT = { country_exists = FRE }
					FRE = { check_variable = { FRE_Metropole_Patience < 33 } }
				}
			}
			set_temp_variable = { lao_force_tier = 2 }
		}
		else_if = {
			limit = {
				country_exists = FRE
				FRE = { check_variable = { FRE_Metropole_Patience > 65 } }
			}
			set_temp_variable = { lao_force_tier = 0 }
		}

		1295 = {	# Sam Neua, the base area and the new capital
			create_unit = {
				division = "name = \"Kong Phan 1\" division_template = \"Kong Phan\" start_experience_factor = 0.3 start_equipment_factor = 0.8"
				owner = LAO
				allow_spawning_on_enemy_provs = yes
			}
			create_unit = {
				division = "name = \"Kong Phan 2\" division_template = \"Kong Phan\" start_experience_factor = 0.3 start_equipment_factor = 0.8"
				owner = LAO
				allow_spawning_on_enemy_provs = yes
			}
			create_unit = {
				division = "name = \"Kong Pathom 1\" division_template = \"Kong Pathom\" start_experience_factor = 0.35 start_equipment_factor = 0.9"
				owner = LAO
				allow_spawning_on_enemy_provs = yes
			}
		}
		if = {
			limit = { check_variable = { lao_force_tier > 0 } }
			1295 = {
				create_unit = {
					division = "name = \"Kong Phan 3\" division_template = \"Kong Phan\" start_experience_factor = 0.3 start_equipment_factor = 0.8"
					owner = LAO
					allow_spawning_on_enemy_provs = yes
				}
				create_unit = {
					division = "name = \"Kong Pathom 2\" division_template = \"Kong Pathom\" start_experience_factor = 0.35 start_equipment_factor = 0.9"
					owner = LAO
					allow_spawning_on_enemy_provs = yes
				}
			}
		}
		if = {
			limit = { check_variable = { lao_force_tier > 1 } }
			1295 = {
				create_unit = {
					division = "name = \"Kong Phan 4\" division_template = \"Kong Phan\" start_experience_factor = 0.3 start_equipment_factor = 0.8"
					owner = LAO
					allow_spawning_on_enemy_provs = yes
				}
			}
		}
		LAO = {
			add_equipment_to_stockpile = { type = infantry_equipment_1 amount = 2500 producer = VIN }
			add_equipment_to_stockpile = { type = artillery_equipment_1 amount = 200 producer = VIN }
		}
	}
}

# --- Frontier garrison for a Pathet Lao Kingdom -----------------------------

lao_deploy_frontier_garrison = {

		limit = { country_exists = LAO }

			limit = { LAO = { controls_state = 841 } }

				create_unit = {
					division = "name = \"Kong Phan Phongsaly\" division_template = \"Kong Phan\" start_experience_factor = 0.35 start_equipment_factor = 0.8"
					owner = LAO
				}
				create_unit = {
					division = "name = \"Kong Pathom Phongsaly\" division_template = \"Kong Pathom\" start_experience_factor = 0.4 start_equipment_factor = 0.9"
					owner = LAO
				}
			}
		}
		if = {
			limit = { LAO = { controls_state = 1295 } }
			1295 = {	# Sam Neua, the Annamite frontier
				create_unit = {
					division = "name = \"Kong Phan Sam Neua\" division_template = \"Kong Phan\" start_experience_factor = 0.35 start_equipment_factor = 0.8"
					owner = LAO
				}
				create_unit = {
					division = "name = \"Kong Pathom Sam Neua\" division_template = \"Kong Pathom\" start_experience_factor = 0.4 start_equipment_factor = 0.9"

				}
			}
		}
		if = {
			limit = { LAO = { controls_state = 670 } }

				create_unit = {

}

# --- Launch (called from Phase3, ROOT = VIN; all tags named explicitly) ------

ic_laos_raid_launch = {
	if = {
		limit = {
			OR = {
				has_global_flag = Laos_Raid_Launched
				has_global_flag = Laos_Raid_Active
			}
		}
		log = "[GetDateText]: ic_laos_raid_launch re-entered and refused - the relief column already exists"
	}
	else = {
		set_global_flag = Laos_Raid_Launched
		set_global_flag = Laos_Raid_Active
		clr_global_flag = Laos_Raid_Resolved
		clr_global_flag = Laos_Raid_Last_Chance
		clr_global_flag = Laos_Raid_Timer_Expired
		clr_global_flag = Laos_Raid_Stalemate
		clr_global_flag = Laos_Raid_Total_Failure

		LOS = {
			add_ideas = LOS_French_Airlift
			# The Kingdom mobilises for the emergency.
			add_ideas = extensive_conscription
			# Player-facing "hold the capitals" objective + situation briefing.
			activate_mission = Laos_Raid_Defence
			country_event = laos_raid.7
		}
		VIN = {
			add_dynamic_modifier = { modifier = laos_raid_overextension }
			activate_mission = Laos_Raid_Offensive	# the attacker had no sight of the clock at all
			# INDOCHINESE_WAR_LAOS and _LAOS_VINH (map/adjacency_rules.txt) close every
			# crossing into the Kingdom to a VIN at war with the CEFEO, which is the
			# whole war. Nothing set this flag, so Hanoi's own divisions could not follow
			# the Pathet Lao across the border it had just declared war to cross.
			set_country_flag = LAOS_BORDER_UNLOCKED
		}
		if = { limit = { country_exists = LAO } LAO = { activate_mission = Laos_Raid_Offensive } }

		# Relief column sized by how much fight Paris still has in it.
		# tier 0 = token garrison, 1 = reduced, 2 = full airlift.
		set_temp_variable = { laos_relief_tier = 1 }
		if = {
			limit = { country_exists = FRE FRE = { check_variable = { FRE_Metropole_Patience > 65 } } }
			set_temp_variable = { laos_relief_tier = 2 }
		}
		else_if = {
			limit = {
				OR = {
					NOT = { country_exists = FRE }
					FRE = { check_variable = { FRE_Metropole_Patience < 33 } }
				}
			}
			set_temp_variable = { laos_relief_tier = 0 }
		}

		670 = {
			create_unit = {
				division = "name = \"Groupement de Marche du Laos\" division_template = \"Corps Expeditionnaire Francais\" start_experience_factor = 0.4 start_equipment_factor = 1.0"
				owner = LOS
				allow_spawning_on_enemy_provs = yes
				prioritize_location = 1464
			}
		}
		1198 = {
			create_unit = {
				division = "name = \"Défense de Luang Prabang\" division_template = \"Corps Expeditionnaire Francais\" start_experience_factor = 0.4 start_equipment_factor = 1.0"
				owner = LOS
				allow_spawning_on_enemy_provs = yes
				prioritize_location = 4613
			}
		}
		if = {
			limit = { check_variable = { laos_relief_tier > 0 } }
			670 = {
				create_unit = {
					division = "name = \"1er Régiment de Chasseurs Laotiens\" division_template = \"Corps Expeditionnaire Francais\" start_experience_factor = 0.35 start_equipment_factor = 1.0"
					owner = LOS
					allow_spawning_on_enemy_provs = yes
					prioritize_location = 1464
				}
			}
		}
		if = {
			limit = { check_variable = { laos_relief_tier > 1 } }
			1198 = {
				create_unit = {
					division = "name = \"2e Groupement Aéroporté\" division_template = \"Corps Expeditionnaire Francais\" start_experience_factor = 0.45 start_equipment_factor = 1.0"
					owner = LOS
					allow_spawning_on_enemy_provs = yes
					prioritize_location = 4613
				}
			}
			670 = {
				create_unit = {
					division = "name = \"3e Groupement de Marche\" division_template = \"Corps Expeditionnaire Francais\" start_experience_factor = 0.4 start_equipment_factor = 1.0"
					owner = LOS
					allow_spawning_on_enemy_provs = yes
					prioritize_location = 1464
				}
			}
			LOS = {
				add_equipment_to_stockpile = { type = infantry_equipment_1 amount = 2000 producer = FRA }
				add_equipment_to_stockpile = { type = artillery_equipment_1 amount = 300 producer = FRA }
			}
		}

		# Arm the 90-day resolution timer on FRA (always exists, never vanishes mid-raid).
		FRA = { activate_mission = Laos_Raid_Timer }
	}
}

# --- Timer expiry dispatch (called from the mission, FRA scope) --------------

ic_laos_raid_resolve = {
	if = {
		limit = { has_global_flag = Laos_Raid_Active }
		if = {
			limit = { laos_raid_attacker_holds_capital = yes }
			ic_laos_raid_attacker_victory = yes
		}
		else_if = {
			limit = {
				laos_raid_attacker_is_human = yes
				NOT = { has_global_flag = Laos_Raid_Last_Chance }
			}
			set_global_flag = Laos_Raid_Timer_Expired
			if = { limit = { country_exists = VIN VIN = { is_ai = no } } VIN = { country_event = laos_raid.1 } }
			if = { limit = { country_exists = LAO LAO = { is_ai = no } } LAO = { country_event = laos_raid.1 } }
		}
		else = {
			ic_laos_raid_end_raid = yes
		}
	}
}

ic_laos_raid_daily_check = {
	if = {
		limit = {
			has_global_flag = Laos_Raid_Active
			laos_raid_attacker_victory_now_trigger = yes
		}
		ic_laos_raid_attacker_victory = yes
	}
}

# --- Human attacker doubles down: deepen the penalty, one short final window --

ic_laos_raid_press = {
	if = {
		limit = { has_global_flag = Laos_Raid_Active NOT = { has_global_flag = Laos_Raid_Last_Chance } }
		set_global_flag = Laos_Raid_Last_Chance
		clr_global_flag = Laos_Raid_Timer_Expired
		VIN = {
			remove_dynamic_modifier = { modifier = laos_raid_overextension }
			add_dynamic_modifier = { modifier = laos_raid_overextension_severe }
		}
		FRA = { activate_mission = Laos_Raid_LastChance_Timer }
		if = { limit = { country_exists = LOS } LOS = { activate_mission = Laos_Raid_Defence_LastChance } }
		if = { limit = { country_exists = VIN } VIN = { activate_mission = Laos_Raid_Offensive_LastChance } }
		if = { limit = { country_exists = LAO } LAO = { activate_mission = Laos_Raid_Offensive_LastChance } }
	}
}

# --- The raid culminates: stalemate or total failure -------------------------

ic_laos_raid_end_raid = {
	if = {
		limit = { has_global_flag = Laos_Raid_Active }
		if = {
			limit = { laos_raid_complete_failure = yes }
			ic_laos_raid_total_failure = yes
		}
		else = {
			ic_laos_raid_stalemate = yes
		}
	}
}

ic_laos_raid_stalemate = {
	set_global_flag = Laos_Raid_Stalemate	# distinguishing outcome flag, read by the Geneva war-outcomes tracker
	if = {
		limit = { country_exists = LAO country_exists = VIN }
		LAO = {
			transfer_state = 841
			transfer_state = 1295
			keep_current_politics = yes
		}
		VIN = {
			set_autonomy = {
				target = LAO
				autonomy_state = autonomy_easternbloc
				end_wars = no
				end_civil_wars = no
			}
		}
	}
	ic_laos_raid_ceasefire = yes
	news_event = laos_raid.2
	ic_laos_raid_cleanup = yes
}

ic_laos_raid_total_failure = {
	set_global_flag = Laos_Raid_Total_Failure	# distinguishing outcome flag; Laos_Total_War may be set on top if Vientiane escalates
	if = {
		limit = { country_exists = LOS }
		LOS = { country_event = laos_raid.3 }
	}
	else = {
		ic_laos_raid_dissolve_lao = yes
		ic_laos_raid_ceasefire = yes
		ic_laos_raid_cleanup = yes
	}
}

ic_laos_raid_attacker_victory = {
	set_global_flag = Laos_Raid_Attacker_Won
	news_event = laos_raid.6
	ic_laos_raid_cleanup = yes
}

ic_laos_raid_total_war = {
	set_global_flag = Laos_Total_War
	ic_laos_raid_dissolve_lao = yes
	news_event = laos_raid.5
	ic_laos_raid_cleanup = yes
}

# --- Shared helpers ----------------------------------------------------------

ic_laos_raid_ceasefire = {
	if = {
		limit = { country_exists = LOS }
		LOS = {
			if = { limit = { country_exists = VIN has_war_with = VIN } white_peace = VIN }
			if = { limit = { country_exists = LAO has_war_with = LAO } white_peace = LAO }
			if = { limit = { country_exists = NLF has_war_with = NLF } white_peace = NLF }
		}
	}
	ic_laos_raid_remove_relief = yes
}

ic_laos_raid_remove_relief = {
	if = {
		limit = { country_exists = LOS }
		LOS = {
			delete_unit_template_and_units = { division_template = "Corps Expeditionnaire Francais" }
		}
	}
}

ic_laos_raid_dissolve_lao = {
	if = {
		limit = { country_exists = LAO country_exists = LOS }
		LOS = { annex_country = { target = LAO } }
	}
}


# Mirrors MLA_transition_to_postwar_government.
lao_convert_to_normal_economy = {
	if = {
		limit = { has_country_flag = active_insurgency }
		clr_country_flag = active_insurgency
		drop_cosmetic_tag = yes	# no longer the Pathet Lao insurgency - the LPDR is proclaimed
		remove_ideas = closed_economy
		set_variable = { show_finance = 0 }
		set_variable = { currentMoney = 5 }
		set_variable = { currentDebt = 0 }
		set_variable = { currentDebtPayments = 0 }
		clear_current_economic_modifiers = yes
		set_variable = { economic_x_axis = 22 }
		startup_economics = yes
		lao_deploy_frontier_garrison = yes	# a government has borders to hold, and the Kingdom's army died winning them
	}
}

ic_laos_raid_cleanup = {
	clr_global_flag = Laos_Raid_Active
	clr_global_flag = Laos_Raid_Timer_Expired
	set_global_flag = Laos_Raid_Resolved
	if = {
		limit = { country_exists = LOS LOS = { has_idea = LOS_French_Airlift } }
		LOS = { remove_ideas = LOS_French_Airlift }
	}
	if = {
		limit = { country_exists = VIN VIN = { has_dynamic_modifier = { modifier = laos_raid_overextension } } }
		VIN = { remove_dynamic_modifier = { modifier = laos_raid_overextension } }
	}
	if = {
		limit = { country_exists = VIN VIN = { has_dynamic_modifier = { modifier = laos_raid_overextension_severe } } }
		VIN = { remove_dynamic_modifier = { modifier = laos_raid_overextension_severe } }
	}
}
