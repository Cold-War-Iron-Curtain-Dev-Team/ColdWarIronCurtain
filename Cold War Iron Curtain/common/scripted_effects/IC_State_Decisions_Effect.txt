set_usa_sen_map_target_jewish_percent = {
	calculate_usa_sen_state_jewish_percent = yes
	set_variable = { USA.USA_SEN_MAP_TARGET_JewishPercent = usa_sen_jewish_percent }
}

set_usa_sen_state_jewish_population_level = {
	calculate_usa_sen_state_jewish_percent = yes
	if = {
		limit = {
			check_variable = {
				usa_sen_jewish_percent > 5
			}
		}
		set_variable = {
			Jewish_Population_Level = 5
		}
	}
	if = {
		limit = {
			check_variable = {
				usa_sen_jewish_percent = 5
			}
		}
		set_variable = {
			Jewish_Population_Level = 4
		}
	}
	if = {
		limit = {
			OR = {
				check_variable = {
					usa_sen_jewish_percent = 3
				}
				check_variable = {
					usa_sen_jewish_percent = 4
				}
			}
		}
		set_variable = {
			Jewish_Population_Level = 3
		}
	}
	if = {
		limit = {
			check_variable = {
				usa_sen_jewish_percent = 2
			}
		}
		set_variable = {
			Jewish_Population_Level = 2
		}
	}
	if = {
		limit = {
			OR = {
				check_variable = {
					usa_sen_jewish_percent = 1
				}
				check_variable = {
					usa_sen_jewish_percent = 0
				}
			}
		}
		set_variable = {
			Jewish_Population_Level = 1
		}
	}
}
