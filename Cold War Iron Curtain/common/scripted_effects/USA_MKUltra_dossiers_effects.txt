# Additional CIA-charter dossiers (design 7): Hypnosis & Conditioning Claims and
# Behavioral Assessment. Both are original fictional/composite contractor cases.
#
# Hypnosis stages: 0 locked, 1 available, 2 initial research, 3 initial report,
# 4 renewal research, 5 renewal report, 6 replication research, 7 final
# appraisal, 8 archived, 9 cancelled/foreclosed. Slot in USA_mkultra_hypnosis_slot.
# Authored profile: a persuasive demonstration that does not survive independent
# replication. Its finding is the value of rejecting it (design 9, "Reliable
# internal appraisal").
#
# Behavioral stages: 0 locked, 1 available, 2 initial research, 3 initial report,
# 4 replication research, 5 final appraisal, 6 archived, 7 cancelled,
# 8 foreclosed. Slot in USA_mkultra_behavioral_slot. Profile drawn once at
# commission (design 8.4): 1 narrow useful finding, 2 ordinary explanation,
# 3 inconclusive, 4 administratively attractive but unsupported.

# Shared exposure charge for a front-grant study stage (design 8.1-8.3): +2, the
# expansion surcharge (+1 exposure, 5 PP) applied first, then compartmentalization
# (-2, minimum 1).
USA_mkultra_charge_front_grant_stage = {
	set_temp_variable = { USA_mkultra_stage_exposure = 2 }
	if = {
		limit = { check_variable = { USA_mkultra_capacity = 2 } }
		add_political_power = -5
		add_to_temp_variable = { USA_mkultra_stage_exposure = 1 }
	}
	if = {
		limit = { check_variable = { USA_mkultra_information_policy = 2 } }
		subtract_from_temp_variable = { USA_mkultra_stage_exposure = 2 }
		if = {
			limit = { check_variable = { USA_mkultra_stage_exposure < 1 } }
			set_temp_variable = { USA_mkultra_stage_exposure = 1 }
		}
	}
	add_to_variable = { USA_mkultra_exposure = USA_mkultra_stage_exposure }
}

# Shared charge for an independent replication stage: +1 exposure, +10 PP under
# compartmentalization, expansion surcharge. Replication gets no secrecy discount.
USA_mkultra_charge_replication_stage = {
	add_to_variable = { USA_mkultra_exposure = 1 }
	if = {
		limit = { check_variable = { USA_mkultra_information_policy = 2 } }
		add_political_power = -10
	}
	if = {
		limit = { check_variable = { USA_mkultra_capacity = 2 } }
		add_political_power = -5
		add_to_variable = { USA_mkultra_exposure = 1 }
	}
}

# Design 9: the Reliable internal appraisal finding makes independent review of
# two later contractor dossiers (Behavioral Assessment, MKSEARCH) 5 PP cheaper.
USA_mkultra_apply_reliable_appraisal_discount = {
	if = {
		limit = { has_country_flag = USA_mkultra_reliable_appraisal_earned }
		add_political_power = 5
	}
}

		limit = {

# Drawn once and stored; a later renewal, reload or successor never rerolls it.
# Equal weights are a provisional playtest assumption (design 8.4).
USA_mkultra_assign_behavioral_profile = {
	if = {
		limit = {
			has_country_flag = USA_mkultra_behavioral_paid
			NOT = { has_country_flag = USA_mkultra_behavioral_profile_assigned }
		}
		random_list = {
			25 = { set_variable = { USA_mkultra_behavioral_profile = 1 } }
			25 = { set_variable = { USA_mkultra_behavioral_profile = 2 } }
			25 = { set_variable = { USA_mkultra_behavioral_profile = 3 } }
			25 = { set_variable = { USA_mkultra_behavioral_profile = 4 } }
		}
		set_country_flag = USA_mkultra_behavioral_profile_assigned
	}
}
