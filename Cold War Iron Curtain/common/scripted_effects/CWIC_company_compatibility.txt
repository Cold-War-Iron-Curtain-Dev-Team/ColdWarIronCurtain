# One-time migration of the historical accented company identifiers.
# Preserve existing presence, invitations and corporate balances when loading old saves.
cwic_migrate_legacy_company_identifiers = {
    if = {
        limit = { has_variable = corp_gécamines_presence_mining }
        set_variable = { corp_gecamines_presence_mining = corp_gécamines_presence_mining }
        clear_variable = corp_gécamines_presence_mining
    }
    if = {
        limit = { has_variable = cwic_invite_gécamines_mining }
        set_variable = { cwic_invite_gecamines_mining = cwic_invite_gécamines_mining }
        clear_variable = cwic_invite_gécamines_mining
    }
    if = {
        limit = { has_variable = global.CWIC_CORP_gécamines_budget }
        set_variable = { global.CWIC_CORP_gecamines_budget = global.CWIC_CORP_gécamines_budget }
        clear_variable = global.CWIC_CORP_gécamines_budget
    }
    if = {
        limit = { has_variable = global.CWIC_CORP_gécamines_budget_remaining }
        set_variable = { global.CWIC_CORP_gecamines_budget_remaining = global.CWIC_CORP_gécamines_budget_remaining }
        clear_variable = global.CWIC_CORP_gécamines_budget_remaining
    }
    if = {
        limit = { has_variable = global.CWIC_CORP_gécamines_cap }
        set_variable = { global.CWIC_CORP_gecamines_cap = global.CWIC_CORP_gécamines_cap }
        clear_variable = global.CWIC_CORP_gécamines_cap
    }
    if = {
        limit = { has_variable = global.CWIC_CORP_gécamines_cash }
        set_variable = { global.CWIC_CORP_gecamines_cash = global.CWIC_CORP_gécamines_cash }
        clear_variable = global.CWIC_CORP_gécamines_cash
    }
    if = {
        limit = { has_variable = global.CWIC_CORP_gécamines_profit_last }
        set_variable = { global.CWIC_CORP_gecamines_profit_last = global.CWIC_CORP_gécamines_profit_last }
        clear_variable = global.CWIC_CORP_gécamines_profit_last
    }
    if = {
        limit = { has_variable = global.CWIC_CORP_gécamines_profit_month }
        set_variable = { global.CWIC_CORP_gecamines_profit_month = global.CWIC_CORP_gécamines_profit_month }
        clear_variable = global.CWIC_CORP_gécamines_profit_month
    }
    if = {
        limit = { has_variable = global.CWIC_CORP_gécamines_shock_timer }
        set_variable = { global.CWIC_CORP_gecamines_shock_timer = global.CWIC_CORP_gécamines_shock_timer }
        clear_variable = global.CWIC_CORP_gécamines_shock_timer
    }
    if = {
        limit = { has_country_flag = cwic_invite_cd_gécamines_mining }
        # Preserve a bounded cooldown; do not create a permanent blocking flag.
        set_country_flag = { flag = cwic_invite_cd_gecamines_mining days = 180 }
        clr_country_flag = cwic_invite_cd_gécamines_mining
    }
}
