fre_preparation_apply_idea = {
	fre_preparation_response_cleanup = yes
	if = {
		limit = { check_variable = { fre_preparation_tier = 2 } }
		add_ideas = FRE_Preparation_Maximum
		if = { limit = { country_exists = TAI } TAI = { add_ideas = FRE_Preparation_Maximum } }
	}
	else_if = {
		limit = { check_variable = { fre_preparation_tier = 1 } }
		add_ideas = FRE_Preparation_Supported
		if = { limit = { country_exists = TAI } TAI = { add_ideas = FRE_Preparation_Supported } }
	}
	else = {
		add_ideas = FRE_Preparation_Ready
		if = { limit = { country_exists = TAI } TAI = { add_ideas = FRE_Preparation_Ready } }
	}
}

fre_preparation_response_begin = {
	set_variable = { fre_preparation_tier = 0 }
	if = {
		limit = { VIN = { check_variable = { VIN_camp_current = 5 } } }
		if = { limit = { has_country_flag = FRE_Vinh_Yen_Prepared } set_variable = { fre_preparation_tier = 1 } }
		country_event = { id = FRE_Preparation.1 hours = 8 }
	}
	else_if = {
		limit = { VIN = { check_variable = { VIN_camp_current = 6 } } }
		if = { limit = { has_country_flag = FRE_Mao_Khe_Prepared } set_variable = { fre_preparation_tier = 1 } }
		if = { limit = { has_country_flag = FRE_Vinh_Yen_Defensive_Success } add_to_variable = { fre_preparation_tier = 1 } }
		clamp_variable = { var = fre_preparation_tier min = 0 max = 2 }
		country_event = { id = FRE_Preparation.2 hours = 8 }
	}
	else_if = {
		limit = { VIN = { check_variable = { VIN_camp_current = 7 } } }
		if = { limit = { has_country_flag = FRE_Mao_Khe_Defensive_Success } set_variable = { fre_preparation_tier = 1 } }
		country_event = { id = FRE_Preparation.3 hours = 8 }
	}
	else_if = {
		limit = { VIN = { check_variable = { VIN_camp_current = 8 } } }
		if = { limit = { has_country_flag = FRE_Na_San_Prepared } set_variable = { fre_preparation_tier = 1 } }
		if = { limit = { has_country_flag = FRE_Hoa_Binh_Mobile_Groups_Preserved } add_to_variable = { fre_preparation_tier = 1 } clr_country_flag = FRE_Hoa_Binh_Mobile_Groups_Preserved }
		if = { limit = { has_country_flag = FRE_Lorraine_Result_Withheld } add_to_variable = { fre_preparation_tier = 1 } }
		if = { limit = { has_country_flag = FRE_Nghia_Lo_Defensive_Success } add_to_variable = { fre_preparation_tier = 1 } }
		clamp_variable = { var = fre_preparation_tier min = 0 max = 2 }
		country_event = { id = FRE_Preparation.4 hours = 8 }
	}
	else_if = {
		limit = { VIN = { check_variable = { VIN_camp_current = 10 } } }
		if = { limit = { has_country_flag = FRE_De_Lattre_In_Command } set_variable = { fre_preparation_tier = 1 } }
		country_event = { id = FRE_Preparation.5 hours = 8 }
	}
	fre_preparation_apply_idea = yes
	fre_preparation_deploy_garrison = yes
}

fre_preparation_deploy_garrison = {
	fre_preparation_disband_garrison = yes
	set_temp_variable = { prep_garrison_size = 2 }
	add_to_temp_variable = { prep_garrison_size = fre_preparation_tier }
	if = {
		limit = { VIN = { check_variable = { VIN_camp_current = 5 } } 1761 = { indochina_french_held_state_trigger = yes } }
		1761 = {
			if = { limit = { is_controlled_by = FRE } set_temp_variable = { prep_garrison_owner = 1 } }
			else_if = { limit = { is_controlled_by = TAI } set_temp_variable = { prep_garrison_owner = 2 } }
			else_if = { limit = { is_controlled_by = VIE } set_temp_variable = { prep_garrison_owner = 3 } }
		}
		fre_preparation_garrison_prepare_owner = yes
		set_temp_variable = { prep_garrison_i = 0 }
		while_loop_effect = {
			limit = { check_variable = { prep_garrison_i < prep_garrison_size } check_variable = { prep_garrison_owner > 0 } }
			1761 = {
				if = { limit = { check_variable = { prep_garrison_owner = 1 } } create_unit = { division = "name = \"Groupe de Defense de Vinh Yen\" division_template = \"Groupe de Defense Locale\" start_experience_factor = 0.4 start_equipment_factor = 1.0" owner = FRE allow_spawning_on_enemy_provs = yes prioritize_location = 12075 } }
				else_if = { limit = { check_variable = { prep_garrison_owner = 2 } } create_unit = { division = "name = \"Groupe de Defense de Vinh Yen\" division_template = \"Groupe de Defense Locale\" start_experience_factor = 0.35 start_equipment_factor = 1.0" owner = TAI allow_spawning_on_enemy_provs = yes prioritize_location = 12075 } }
				else = { create_unit = { division = "name = \"Groupe de Defense de Vinh Yen\" division_template = \"Groupe de Defense Locale\" start_experience_factor = 0.3 start_equipment_factor = 1.0" owner = VIE allow_spawning_on_enemy_provs = yes prioritize_location = 12075 } }
			}
			add_to_temp_variable = { prep_garrison_i = 1 }

	}
	else_if = {
		limit = { VIN = { check_variable = { VIN_camp_current = 8 } } 671 = { indochina_french_held_state_trigger = yes } }
		671 = {
			if = { limit = { is_controlled_by = FRE } set_temp_variable = { prep_garrison_owner = 1 } }
			else_if = { limit = { is_controlled_by = TAI } set_temp_variable = { prep_garrison_owner = 2 } }
			else_if = { limit = { is_controlled_by = VIE } set_temp_variable = { prep_garrison_owner = 3 } }
		}
		fre_preparation_garrison_prepare_owner = yes
		set_temp_variable = { prep_garrison_i = 0 }
		while_loop_effect = {
			limit = { check_variable = { prep_garrison_i < prep_garrison_size } check_variable = { prep_garrison_owner > 0 } }
			671 = {
				if = { limit = { check_variable = { prep_garrison_owner = 1 } } create_unit = { division = "name = \"Groupe de Defense de Na San\" division_template = \"Groupe de Defense Locale\" start_experience_factor = 0.4 start_equipment_factor = 1.0" owner = FRE allow_spawning_on_enemy_provs = yes prioritize_location = 13757 } }
				else_if = { limit = { check_variable = { prep_garrison_owner = 2 } } create_unit = { division = "name = \"Groupe de Defense de Na San\" division_template = \"Groupe de Defense Locale\" start_experience_factor = 0.35 start_equipment_factor = 1.0" owner = TAI allow_spawning_on_enemy_provs = yes prioritize_location = 13757 } }
				else = { create_unit = { division = "name = \"Groupe de Defense de Na San\" division_template = \"Groupe de Defense Locale\" start_experience_factor = 0.3 start_equipment_factor = 1.0" owner = VIE allow_spawning_on_enemy_provs = yes prioritize_location = 13757 } }
			}
			add_to_temp_variable = { prep_garrison_i = 1 }
		}
		log = "IC_AFK|LAUNCH_GARRISON|date=[GetDateText]|campaign=8|owner_code=[?prep_garrison_owner|0]|divisions=[?prep_garrison_size|0]|tier=[?fre_preparation_tier|0]"
	}
	else_if = {
		limit = { VIN = { check_variable = { VIN_camp_current = 10 } } 1761 = { indochina_french_held_state_trigger = yes } }
		subtract_from_temp_variable = { prep_garrison_size = 1 }
		1761 = {
			if = { limit = { is_controlled_by = FRE } set_temp_variable = { prep_garrison_owner = 1 } }
			else_if = { limit = { is_controlled_by = TAI } set_temp_variable = { prep_garrison_owner = 2 } }
			else_if = { limit = { is_controlled_by = VIE } set_temp_variable = { prep_garrison_owner = 3 } }
		}
		fre_preparation_garrison_prepare_owner = yes
		set_temp_variable = { prep_garrison_i = 0 }
		while_loop_effect = {
			limit = { check_variable = { prep_garrison_i < prep_garrison_size } check_variable = { prep_garrison_owner > 0 } }
			1761 = { fre_preparation_spawn_nghia_lo_unit = yes }
			add_to_temp_variable = { prep_garrison_i = 1 }
		}
		log = "IC_AFK|LAUNCH_GARRISON|date=[GetDateText]|campaign=10|owner_code=[?prep_garrison_owner|0]|divisions=[?prep_garrison_size|0]|tier=[?fre_preparation_tier|0]"

	}
}

fre_preparation_response_cleanup = {
	if = { limit = { has_idea = FRE_Preparation_Ready } remove_ideas = FRE_Preparation_Ready }
	if = { limit = { has_idea = FRE_Preparation_Supported } remove_ideas = FRE_Preparation_Supported }
	if = { limit = { has_idea = FRE_Preparation_Maximum } remove_ideas = FRE_Preparation_Maximum }
	if = { limit = { country_exists = TAI } TAI = {
		if = { limit = { has_idea = FRE_Preparation_Ready } remove_ideas = FRE_Preparation_Ready }
		if = { limit = { has_idea = FRE_Preparation_Supported } remove_ideas = FRE_Preparation_Supported }
		if = { limit = { has_idea = FRE_Preparation_Maximum } remove_ideas = FRE_Preparation_Maximum }
	} }
}
