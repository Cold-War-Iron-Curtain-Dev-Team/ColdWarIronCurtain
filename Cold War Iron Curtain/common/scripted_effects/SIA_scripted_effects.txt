	if = {
		limit = {
			has_idea = SIA_economic_modernization
		}
		swap_ideas = {
			remove_idea = SIA_economic_modernization
			add_idea = SIA_economic_modernization_1
		}
	}
	else_if = {
		limit = {
			has_idea = SIA_economic_modernization_1
		}
		swap_ideas = {
			remove_idea = SIA_economic_modernization_1
			add_idea = SIA_economic_modernization_2
		}
	}
	else_if = {
		limit = {
			has_idea = SIA_economic_modernization_2
		}
		swap_ideas = {
			remove_idea = SIA_economic_modernization_2
			add_idea = SIA_economic_modernization_3
		}
	}
	else_if = {
		limit = {
			has_idea = SIA_economic_modernization_3
		}
		swap_ideas = {
			remove_idea = SIA_economic_modernization_3
			add_idea = SIA_economic_modernization_4
		}
	}
	else_if = {
		limit = {
