SIA_Triangle_Phibun_Action_effect = {
	if = {
		limit = {
			check_variable = {
				Phibun_Action_Count = 10
			}
		}
		set_variable = { Phibun_Action_Count = 0 }
	}
	else = {
		set_temp_variable = { Phibun_Action_Strength = 1 }
		set_temp_variable_to_random = { 
			var = Phibun_Action_Strength_Randomizer
			max = 15
			integer = yes
		}
		divide_temp_variable = {
			Phibun_Action_Strength_Randomizer = 10
		}
		add_to_temp_variable = { Phibun_Action_Strength = Phibun_Action_Strength_Randomizer }
		if = {
			limit = {
				check_variable = {
					Phibun_Action_Randomizer = 0
				}
			}
			set_temp_variable = { SIA_Phibun_Support_temp = Phibun_Action_Strength }
			SIA_Phibunsongkram_Support_effect = yes
		}
		else_if = {
			limit = {
				check_variable = {
					Phibun_Action_Randomizer = 1
				}
			}
			set_temp_variable = { SIA_Sakdina_Support_temp = Phibun_Action_Strength }
			multiply_temp_variable = { SIA_Sakdina_Support_temp = -1 }
			SIA_Sakdina_Support_effect = yes
		}
		else_if = {
			limit = {
				check_variable = {
					Phibun_Action_Randomizer = 2
				}
			}
			set_temp_variable = { SIA_Diamond_Ring_Support_temp = Phibun_Action_Strength }
			multiply_temp_variable = { SIA_Diamond_Ring_Support_temp = -1 }
			SIA_Diamond_Ring_Support_effect = yes
		}
		hidden_effect = {
			random_list = {
				30 = {
					set_variable = { Phibun_Action_Randomizer = 0 }
				}
				15 = {
					set_variable = { Phibun_Action_Randomizer = 1 }
				}
				15 = {
					set_variable = { Phibun_Action_Randomizer = 2 }
				}
			}
		}
	}
}
