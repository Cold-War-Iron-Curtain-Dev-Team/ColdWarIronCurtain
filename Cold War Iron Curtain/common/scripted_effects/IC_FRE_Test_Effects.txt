# ============================================================================
# CEFEO (FRE) QA / testing effects
# ----------------------------------------------------------------------------
# Run from the console with:   e <effect_name>
# Most tests assume you are playing FRE (`tag FRE` first); the war-score tests
# work from any tag because the struggle scores live on FRA.
#
# Companion files: IC_Struggle_Test_Effects.txt (struggle core) and
# IC_Geneva_Test_Effects.txt (Geneva conference). This file covers the CEFEO
# rework: Metropole Patience, the FRA lifeline, GCMA/VNA bonuses, the
# operation gates, Operation Castor, the Dien Bien Phu arc, command rotation
# and dissolution.
#
# Quick reference:
#   -- status / patience --
#   e test_fre_status            log patience, tier, commander bonus, GCMA, VNA
#   e test_indochina_command_status  verify the CEFEO faction, both commands, crown-domain custody, and separate wars
#   e test_indochina_command_repair  rebuild the early command model after a test/save migration
#   e test_fre_patience_set_0    patience to 0 (fires the milestone chain)
#   e test_fre_patience_set_50   patience to 50
#   e test_fre_patience_set_100  patience to 100
#   -- lifeline / allies --
#   e test_fre_request_all       fire all five FRA request events at FRA
#   e test_fre_gcma_max          GCMA network to 4 (fires Trinquier dilemma)
#   e test_fre_roll_battle_prep  log a battle-prep success roll with bonuses
#   -- operations / battles --
#   e test_fre_ops_unlock        waive date gates, clear cooldowns + activations
#   e test_fre_credits_grant     +500 War Credits so every tier is affordable
#   e test_fre_operation_launch  open the generic commitment prompt (Brochet by default)
#   e test_fre_operation_resolve fire the watchdog as if the 90-day timer expired
#   e test_fre_operation_threshold  force the deterministic outcome
#   e test_fre_operation_finish  clear all operation state, fire no outcome
#   e test_fre_operation_status  log current op, tier, states and credits
#   e test_fre_castor_start_full start the exact-province full Castor deployment
#   e test_fre_castor_advance    move an active Castor to its next daily check
#   e test_fre_castor_day_14     exercise the exact objective-completion edge
#   e test_fre_castor_fail       force an active Castor failure
#   e test_fre_castor_supersede  force an active Castor supersession
#   e test_fre_castor_status     log Castor lifecycle and hold counters
#   e test_fre_castor_reset      clear Castor test state and its camp formation
#   e test_fre_pollux_start_historical  start the historical split evacuation
#   e test_fre_pollux_start_escort      start the protected Route Pavie withdrawal
#   e test_fre_pollux_day_7      exercise the escorted/airlift completion edge
#   e test_fre_pollux_fail       force the active evacuation's column-loss result
#   e test_fre_pollux_supersede  force the theatre-superseded result
#   e test_fre_pollux_status     log posture, route, counters, and exclusive result
#   e test_fre_pollux_reset      clear Pollux test state and rebuild no formation
#   e test_fre_atlante_start_full       start the full central-Vietnam commitment
#   e test_fre_atlante_start_vietnamese start the Vietnamese-led coastal plan
#   e test_fre_atlante_choose_north     cancel Atlante and bank the northern reserve
#   e test_fre_atlante_day_14           exercise the selected objective's completion edge
#   e test_fre_atlante_fail             force the active offensive's stalled result
#   e test_fre_atlante_supersede        force the theatre-superseded result
#   e test_fre_atlante_status           log posture, objectives, counters, and result exclusivity
#   e test_fre_atlante_reset            clear Atlante test state
#   e test_fre_lorraine_status   log Lorraine posture and exclusive result
#   e test_fre_hoa_binh_status   log phase-two posture and exclusive result
#   e test_fre_hoa_binh_day_105  force a planned Amarante request at day 105
#   e test_fre_dbp_start         jump straight into the Dien Bien Phu arc
#   e test_fre_command_next      advance the command rotation one step
#   -- Tai highlands / Dien Bien Phu ownership --
#   e test_tai_status            log who owns 1761 and 671 and whether TAI is alive
#   e test_vin_tay_bac_win       fire the Northwest win at VIN (rump TAI must survive at 671)
#   e test_dbp_tai_collapse      run the camp-falls collapse (TAI annexed by VIN)
#   e test_vin_dbp_fallback      fire the no-Castor dispatcher at VIN
#   -- endings / dissolution --
#   e test_fre_win_war           push FRA scores past the victory trigger
#   e test_fre_lose_war          push FRA scores past the defeat trigger
#   e test_fre_winddown_now      start the wind-down chain immediately
#   e test_fre_dissolve_vie      dissolve the corps into VIE now
#   e test_fre_dissolve_vin      dissolve the corps into VIN now
# ============================================================================

# Tai highlands / Dien Bien Phu ownership
# ----------------------------------------------------------------------------
# The Northwest campaign now takes only Hoang Lien Son (1761). The Dien Bien Phu
# valley (671, province 4529) changes hands only through the VIN map campaign;
# its objective decides whether the rump federation survives. SWF_Indochina_War.14
# remains the fallback for a game where the CEFEO never fortified the valley.

test_tai_status = {
	if = { limit = { country_exists = TAI } log = "TAI STATUS: federation still exists" }
	else = { log = "TAI STATUS: federation is gone" }
	if = { limit = { 1761 = { is_owned_by = VIN } } log = "  1761 Hoang Lien Son: VIN" }
	else_if = { limit = { 1761 = { is_owned_by = TAI } } log = "  1761 Hoang Lien Son: TAI" }
	else = { log = "  1761 Hoang Lien Son: other" }
	if = { limit = { 671 = { is_owned_by = VIN } } log = "  671 Tay Bac Bo (Dien Bien Phu): VIN" }
	else_if = { limit = { 671 = { is_owned_by = TAI } } log = "  671 Tay Bac Bo (Dien Bien Phu): TAI" }
	else = { log = "  671 Tay Bac Bo (Dien Bien Phu): other" }
	if = { limit = { country_exists = FRE FRE = { has_country_flag = FRE_Dien_Bien_Phu_Established } } log = "  camp established: yes" }
	else = { log = "  camp established: no" }
	if = { limit = { has_global_flag = VIN_DBP_Garrison_Deployed } log = "  GONO garrison: deployed" }
	else = { log = "  GONO garrison: none" }
	if = { limit = { has_global_flag = VIN_DBP_Garrison_Is_FRE } log = "  GONO owner: FRE" }
	else_if = { limit = { has_global_flag = VIN_DBP_Garrison_Deployed } log = "  GONO owner: TAI" }
	log = "  siege days [?global.VIN_dbp_siege_days|0], works level [?global.VIN_dbp_fort_level|0]"
}

test_vin_status = {
	if = {
		limit = { NOT = { country_exists = VIN } }
		log = "VIN STATUS: no Viet Minh"
	}
	else = {
		VIN = {
			log = "VIN STATUS: political power [?political_power|0], campaign supply [?VIN_Campaign_Supply|0]"
			if = { limit = { has_idea = extensive_conscription } log = "  conscription: extensive" }
			else_if = { limit = { has_idea = limited_conscription } log = "  conscription: limited (rebalance did not apply)" }
			else = { log = "  conscription: other" }
			if = { limit = { has_idea = VIN_Resistance_War_Economy } log = "  war economy spirit: on" }
			else = { log = "  war economy spirit: OFF" }
			if = { limit = { has_idea = VIN_Indochina_AI_Support } log = "  standing AI layer: on" }
			else = { log = "  standing AI layer: off" }
			if = { limit = { has_idea = VIN_Campaign_AI_Support } log = "  campaign AI layer: on" }
			else = { log = "  campaign AI layer: off" }
			if = { limit = { has_country_flag = VIN_CMAG_Arrived } log = "  CMAG: arrived" }
			else = { log = "  CMAG: not yet" }
			log = "  divisions [?num_divisions|0]"
		}
	}
}

test_vin_fence = {
	if = {
		limit = { NOT = { country_exists = VIN } }
		log = "VIN FENCE: no Viet Minh"
	}
	else = {
		VIN = {
			log = "VIN ROUTES: northern campaign connections are unrestricted in map/adjacencies.csv"
			if = { limit = { has_country_flag = INDOCHINESE_WAR_DIEN_BIEN_UNLOCKED } log = "  Dien Bien campaign permission for the retained Laos-Vinh crossings: on" }
			else = { log = "  Dien Bien campaign permission for the retained Laos-Vinh crossings: off" }
			if = { limit = { has_country_flag = LAOS_BORDER_UNLOCKED } log = "  Laotian frontier OPEN" }
			else = { log = "  Laotian frontier shut" }
			if = { limit = { has_country_flag = VIN_Campaign_Dien_Bien_Active } log = "  valley campaign: in the field" }
			else_if = { limit = { has_country_flag = VIN_Campaign_Cooldown_Dien_Bien } log = "  valley campaign: fought and resolved" }
			else = { log = "  valley campaign: not yet run" }
			if = { limit = { controls_province = 4529 } log = "  4529 Dien Bien Phu: VIN holds it" }
			else = { log = "  4529 Dien Bien Phu: not ours" }
		}
	}
	if = { limit = { 671 = { is_owned_by = VIN } } log = "  671 Tay Bac Bo owner: VIN" }
	else_if = { limit = { 671 = { is_owned_by = TAI } } log = "  671 Tay Bac Bo owner: TAI" }
	else = { log = "  671 Tay Bac Bo owner: other" }
	if = { limit = { 1761 = { is_owned_by = VIN } } log = "  1761 Hoang Lien Son owner: VIN" }
	else_if = { limit = { 1761 = { is_owned_by = TAI } } log = "  1761 Hoang Lien Son owner: TAI" }
	else = { log = "  1761 Hoang Lien Son owner: other" }
}

# The no-Castor fallback. Clears the camp flag first so the dispatcher does not
# immediately stand down, and hands VIN the highlands so it can reach the offer.
test_vin_dbp_fallback = {
	if = { limit = { country_exists = FRE } FRE = { clr_country_flag = FRE_Dien_Bien_Phu_Established } }
	fre_dbp_disband_gono = yes	# the one-shot garrison guard is global; a re-run would otherwise spawn nothing
	set_variable = { global.VIN_dbp_siege_days = 0 }
	set_variable = { global.VIN_dbp_fort_level = 0 }
	if = {
		limit = { country_exists = VIN }
		VIN = {
			clr_country_flag = VIN_Campaign_Cao_Bac_Active
			clr_country_flag = VIN_Campaign_Tay_Bac_Active
			clr_country_flag = VIN_Campaign_Hoa_Binh_Active
			clr_country_flag = VIN_Campaign_Dien_Bien_Active
			clr_country_flag = VIN_Campaign_Cooldown_Tay_Bac
			clr_country_flag = VIN_Campaign_Cooldown_Hoa_Binh
			clr_country_flag = VIN_Campaign_Cooldown_Dien_Bien
			clr_country_flag = VIN_Campaign_Cooldown_Thap_Van_Dai
			clr_country_flag = VIN_Campaign_Cooldown_Cao_Bac
			if = { limit = { NOT = { owns_state = 1761 } } transfer_state = 1761 }
			country_event = { id = SWF_Indochina_War.14 days = 1 }
		}
	}
}

test_fre_castor_status = {
	if = {
		limit = { country_exists = FRE }
		FRE = {
			log = "[CASTOR] hold=[?FRE_Castor_Hold_Days|0]/[?FRE_Castor_Hold_Required|0] elapsed=[?FRE_Castor_Days_Elapsed|0] credits=[?FRE_War_Credits|0]"
			if = { limit = { has_country_flag = FRE_Castor_Airhead_Active } log = "[CASTOR] Airhead establishment active" }
			if = { limit = { has_country_flag = FRE_Operation_Castor_Result_Success } log = "[CASTOR] Result: airhead established" }
			if = { limit = { has_country_flag = FRE_Operation_Castor_Result_Failure } log = "[CASTOR] Result: establishment failed" }
			if = { limit = { has_country_flag = FRE_Operation_Castor_Result_Superseded } log = "[CASTOR] Result: theatre-superseded" }
			set_temp_variable = { fre_castor_test_results = 0 }
			if = { limit = { has_country_flag = FRE_Operation_Castor_Result_Success } add_to_temp_variable = { fre_castor_test_results = 1 } }
			if = { limit = { has_country_flag = FRE_Operation_Castor_Result_Failure } add_to_temp_variable = { fre_castor_test_results = 1 } }
			if = { limit = { has_country_flag = FRE_Operation_Castor_Result_Superseded } add_to_temp_variable = { fre_castor_test_results = 1 } }
			if = { limit = { check_variable = { fre_castor_test_results > 1 } } log = "[CASTOR] FAIL - more than one permanent result" }
			if = {
				limit = {
					has_country_flag = FRE_Castor_Airhead_Active
					NOT = { has_country_flag = FRE_Mission_Operation_Castor_Activated }
				}
				log = "[CASTOR] FAIL - active airhead has no active mission flag"
			}
		}
	}
}
