# ============================================================
# THE SECURITY COUNCIL, ELECTED AS IT REALLY WAS
#
# Generated from the UN's own record of elected members, 1949-2008. Each
# class serves two years, so a given year seats this year's class and last
# year's - six elected seats before 1966, ten after.
#
# WHEN THE MOD HAS NO SUCH COUNTRY the seat is not lost and is not filled at
# random: cwic_un_seat_substitute puts a UN member from the SAME CONTINENT in
# it. Ghana's seat in 1962 goes to another African member, not to Uruguay.
#
# TWO THINGS WORTH KNOWING IF YOU EDIT THIS:
#   1. cwic_seat_tag is a NORMAL variable on UNS, not a temp. var: scope on a
#      temp variable is not something the mod does anywhere else, and this
#      file would be a poor place to find out it does not work.
#   2. A country the mod lacks never goes through cwic_un_seat at all - the
#      generator calls the substitute directly, so var: is never handed a 0.
# ============================================================

# UNS.cwic_seat_tag = the country id, UNS.cwic_seat_cont = its continent:
#   1 africa  2 asia  3 europe  4 south_america  5 north_america  6 middle_east  7 australia
cwic_un_seat = {
	if = {
		limit = {
			var:UNS.cwic_seat_tag = {
				exists = yes
				has_idea = un_member
				NOT = { has_idea = UN_Security_Council_Member_Perm }
				is_subject = no
				NOT = { has_country_flag = un_sc_elected }
			}
		}
		add_to_array = { UNS.un_temp_seats = UNS.cwic_seat_tag }
		var:UNS.cwic_seat_tag = { set_country_flag = un_sc_elected }
	}
	else = { cwic_un_seat_substitute = yes }
}
