casualty_trait_impact_adjustment = {    #KIA and Captured Reduction
    #Captured Section
    meta_effect = { 
        text = {
            set_variable = { cap_reduction_rate = squad_combined_escape_chance^[CURRENT_PAGE] }
        }
        CURRENT_PAGE = "[?current_page]"
    }    
    set_variable = { cap_reduction = 0 }
    for_loop_effect = {
        end = cap_total
        value = v
        set_variable_to_random = CAP_Recovery_Chance
        if = {
            limit = {
                OR = {
                    check_variable = { CAP_Recovery_Chance < cap_reduction_rate }
                    check_variable = { CAP_Recovery_Chance = cap_reduction_rate }
                }
            }
            add_to_variable = { cap_reduction = 1 }
        }
    }
    subtract_from_variable = { cap_total = cap_reduction }
    
    #KIA Section 
    meta_effect = { 
        text = {
            set_variable = { kia_reduction_rate = squad_combined_group_kia_reduction_rate^[CURRENT_PAGE] }
        }
        CURRENT_PAGE = "[?current_page]"
    }    
    set_variable = { kia_reduction = 0 }
    for_loop_effect = {
        end = kia_total
        value = v
        set_variable_to_random = KIA_Recovery_Chance
        if = {
            limit = {
                OR = {
                    check_variable = { KIA_Recovery_Chance < kia_reduction_rate }
                    check_variable = { KIA_Recovery_Chance = kia_reduction_rate }
                }
            }
            add_to_variable = { kia_reduction = 1 }
        }
    }
    subtract_from_variable = { kia_total = kia_reduction }
    add_to_variable = { wia_total = kia_reduction }
}
casualty_trait_downgrade_impact_adjustment = {  #WIA Adjustment
    #squad_combined_recovery_duration_rate
    set_variable = { WIA_Level_1 = 0 }  #Lowest
    set_variable = { WIA_Level_2 = 0 }  #Middle
    set_variable = { WIA_Level_3 = 0 }  #Severe
    set_variable = { wia_reduction_rate = 0 }
    meta_effect = { 
        text = {
            set_variable = { wia_reduction_rate = squad_combined_recovery_duration_rate^[CURRENT_PAGE] }
        }
        CURRENT_PAGE = "[?current_page]"
    }    
    for_loop_effect = {
        end = wia_total
        value = v
        set_variable_to_random = {
            var = WIA_Level
            min = 1
            max = 3
            integer = yes
        }
        if = {
            limit = {
                check_variable = { WIA_Level = 1 }
            }
            add_to_variable = { WIA_Level_1 = 1 }
            set_variable_to_random = WIA_Recovery_Chance
            if = {
                limit = {
                    OR = {
                        check_variable = { WIA_Recovery_Chance < wia_reduction_rate }
                        check_variable = { WIA_Recovery_Chance = wia_reduction_rate }
                    }
                }
                subtract_from_variable = { WIA_Level_1 = 1 }
            }
        }
        else_if = {
            limit = {
                check_variable = { WIA_Level = 2 }
            }
            add_to_variable = { WIA_Level_1 = 2 }
            set_variable_to_random = WIA_Recovery_Chance
            if = {
                limit = {
                    OR = {
                        check_variable = { WIA_Recovery_Chance < wia_reduction_rate }
                        check_variable = { WIA_Recovery_Chance = wia_reduction_rate }
                    }
                }
                subtract_from_variable = { WIA_Level_2 = 1 }
                add_to_variable = { WIA_Level_1 = 1 }
            }
        }
        else_if = {
            limit = {
                check_variable = { WIA_Level = 3 }
            }
            add_to_variable = { WIA_Level_1 = 3 }
            set_variable_to_random = WIA_Recovery_Chance
            if = {
                limit = {
                    OR = {
                        check_variable = { WIA_Recovery_Chance < wia_reduction_rate }
                        check_variable = { WIA_Recovery_Chance = wia_reduction_rate }
                    }
                }
                subtract_from_variable = { WIA_Level_3 = 1 }
                add_to_variable = { WIA_Level_2 = 1 }
            }
        }
        else = { }
    }

}
