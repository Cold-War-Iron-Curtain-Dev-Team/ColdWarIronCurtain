# The daily owner for Hanoi's post-Dien Bien Phu strategic fork.
vin_post_dbp_southern_war_check = {
	vin_post_dbp_legacy_compatibility = yes
	if = {
		limit = {
			has_completed_focus = VIN_Prepare_Dien
			has_global_flag = geneva_dbp_outcome_applied
			check_variable = { global.geneva_outcome_dbp = 2 }
			NOT = { has_country_flag = VIN_Post_DBP_Strategy_Briefed }
		}
		set_country_flag = VIN_Post_DBP_Strategy_Briefed
		set_country_flag = VIN_Post_DBP_Southern_War_Recognized
		country_event = { id = VIN_South.1 hours = 6 }
	}
	if = {
		limit = {
			has_country_flag = VIN_Post_DBP_Escalation_Chosen
			NOT = { has_country_flag = VIN_Post_DBP_Southern_War_Ended }
			NOT = { vin_post_dbp_southern_war_active_trigger = yes }
		}
		set_country_flag = VIN_Post_DBP_Southern_War_Ended
		if = {
			limit = { vin_post_dbp_southern_resistance_defeated_trigger = yes }
			set_country_flag = VIN_Post_DBP_Funded_NLF_Defeated
		}
		else = { set_country_flag = VIN_Post_DBP_Funded_NLF_Ceasefire }
		if = {
			limit = { country_exists = NLF }
			NLF = { remove_ideas = NLF_Northern_Supply_Line }
		}
		response_package_validation_vin_south_log = yes
		country_event = { id = VIN_South.2 hours = 6 }
	}
	vin_post_dbp_escalation_cleanup_check = yes
}

# Old saves may contain a decision posture or an autonomous queue. Only an
# explicit old talks choice is allowed to inherit the focus-owned queue.
vin_post_dbp_legacy_compatibility = {
	if = {
		limit = { NOT = { has_country_flag = VIN_Post_DBP_Strategy_Compatibility_Checked } }
		set_country_flag = VIN_Post_DBP_Strategy_Compatibility_Checked
		if = {
			limit = { has_country_flag = VIN_Post_DBP_Funded_NLF }
			set_country_flag = VIN_Post_DBP_Southern_Response_Chosen
			set_country_flag = VIN_Post_DBP_Escalation_Chosen
			set_country_flag = VIN_Post_DBP_Panic_Geneva_Suppressed
			clr_country_flag = VIN_Post_DBP_Focus_Geneva_Queued
			if = {
				limit = {
					OR = {
						has_country_flag = VIN_Post_DBP_Auto_Geneva_Queued
						has_country_flag = VIN_Post_DBP_Pressed_For_Geneva
						has_country_flag = VIN_Post_DBP_Settlement_After_South
					}
				}
				clr_global_flag = Indochina_Geneva_Pursuit
			}
		}
		else_if = {
			limit = { has_country_flag = VIN_Post_DBP_Pressed_For_Geneva }
			set_country_flag = VIN_Post_DBP_Southern_Response_Chosen
			set_country_flag = VIN_Post_DBP_Negotiations_Chosen
			set_country_flag = VIN_Post_DBP_Focus_Geneva_Queued
			set_global_flag = Indochina_Geneva_Pursuit
		}
		else = {
			clr_country_flag = VIN_Post_DBP_Southern_Response_Chosen
			if = {
				limit = { has_country_flag = VIN_Post_DBP_Auto_Geneva_Queued }
				clr_global_flag = Indochina_Geneva_Pursuit
			}
		}
		clr_country_flag = VIN_Post_DBP_Auto_Geneva_Queued
	}
}

# Focus-owned queue processor. It can repair only the divisionless NLF paper
# war, and only after Hanoi itself is at peace.
vin_post_dbp_focus_geneva_queue_tick = {
	if = {
		limit = {
			has_country_flag = VIN_Post_DBP_Focus_Geneva_Queued
			has_war = no
			country_exists = VIE
			country_exists = NLF
			NLF = { num_divisions < 1 }
			VIE = { has_war_with = NLF }
		}
		set_country_flag = VIN_Post_DBP_Disarmed_NLF_Armistice
		VIE = { white_peace = NLF }
	}
	if = {
		limit = {
			has_country_flag = VIN_Post_DBP_Focus_Geneva_Queued
			geneva_conference_vietnam_at_peace_trigger = yes
			NOT = { has_global_flag = Indochina_War_Over }
			NOT = { has_global_flag = geneva_conference_announced }
			NOT = { has_global_flag = geneva_conference_in_session }
			NOT = { has_global_flag = geneva_conference_concluded }
		}
		clr_country_flag = VIN_Post_DBP_Focus_Geneva_Queued
		set_country_flag = VIN_Post_DBP_Focus_Geneva_Launched
		clr_global_flag = Indochina_Geneva_Pursuit
		indochina_struggle_invite_countries_to_geneva = yes
	}
	if = {
		limit = {
			has_country_flag = VIN_Post_DBP_Focus_Geneva_Queued
			OR = {
				has_global_flag = Indochina_War_Over
				has_global_flag = geneva_conference_announced
				has_global_flag = geneva_conference_in_session
				has_global_flag = geneva_conference_concluded
			}
		}
		clr_country_flag = VIN_Post_DBP_Focus_Geneva_Queued
	}
}

vin_post_dbp_fund_southern_war = {
	if = {
		limit = {
			OR = {
				has_country_flag = VIN_Post_DBP_Focus_Geneva_Queued
				has_country_flag = VIN_Post_DBP_Auto_Geneva_Queued
				has_country_flag = VIN_Post_DBP_Pressed_For_Geneva
				has_country_flag = VIN_Post_DBP_Settlement_After_South
			}
		}
		clr_global_flag = Indochina_Geneva_Pursuit
	}
	# A generic FRE/wrap-up pursuit can already be live without any legacy VIN
	# queue flag. The southern strategy always vetoes that pending conference.
	clr_global_flag = Indochina_Geneva_Pursuit
	set_country_flag = VIN_Post_DBP_Southern_Response_Chosen
	set_country_flag = VIN_Post_DBP_Escalation_Chosen
	set_country_flag = VIN_Post_DBP_Funded_NLF
	set_country_flag = VIN_Post_DBP_Panic_Geneva_Suppressed
	clr_country_flag = VIN_Post_DBP_Focus_Geneva_Queued
	clr_country_flag = VIN_Post_DBP_Auto_Geneva_Queued
	set_variable = { global.Indochina_Geneva_Pursuit_Source = 0 }
	if = {
		limit = { country_exists = NLF }
		NLF = {
			add_ideas = NLF_Northern_Supply_Line
			add_manpower = 15000
			add_command_power = 25
			add_equipment_to_stockpile = { type = infantry_equipment_1 amount = 4000 producer = VIN }
			add_equipment_to_stockpile = { type = support_equipment amount = 500 producer = VIN }
			add_equipment_to_stockpile = { type = artillery_equipment_1 amount = 200 producer = VIN }
		}
	}
	if = {
		limit = { NOT = { has_global_flag = Indochina_War_Over } }
		FRA = {
			add_to_variable = {
				StruggleInvolvedNationsIndochinaCommunistScore = 25
				tooltip = Add_to_Communist_Victory_Points_TT
			}
			add_to_variable = {
				global.Indochina_War_Next_Phase_A_Points = 50
				tooltip = Add_to_Escalation_Phase_TT
			}
		}
	}
	response_package_validation_vin_south_log = yes
}

vin_post_dbp_press_for_geneva = {
	set_country_flag = VIN_Post_DBP_Southern_Response_Chosen
	set_country_flag = VIN_Post_DBP_Negotiations_Chosen
	set_country_flag = VIN_Post_DBP_Pressed_For_Geneva
	set_country_flag = VIN_Post_DBP_Focus_Geneva_Queued
	set_variable = { global.Indochina_Geneva_Pursuit_Source = 3 }
	set_global_flag = Indochina_Geneva_Pursuit
	set_temp_variable = { geneva_lev_amount = 25 }
	geneva_add_communist_leverage = yes
	if = {
		limit = { NOT = { has_global_flag = Indochina_War_Over } }
		FRA = {
			add_to_variable = {
				global.Indochina_War_Next_Phase_B_Points = 150
				tooltip = Add_to_De_Escalation_Phase_TT
			}
		}
	}
	response_package_validation_vin_south_log = yes
	vin_post_dbp_focus_geneva_queue_tick = yes
}

vin_post_dbp_escalation_cleanup_check = {
	if = {
		limit = {
			has_country_flag = VIN_Post_DBP_Escalation_Chosen
			OR = {
				NOT = { country_exists = VIE }
				has_global_flag = Indochina_War_Over
				has_global_flag = geneva_conference_announced
				has_global_flag = geneva_conference_in_session
				has_global_flag = geneva_conference_concluded
			}
		}
		clr_country_flag = VIN_Post_DBP_Reunification_AI_Active
		if = { limit = { has_idea = VIN_National_Reunification_Preparation } remove_ideas = VIN_National_Reunification_Preparation }
		if = { limit = { country_exists = NLF } NLF = { remove_ideas = NLF_Northern_Supply_Line } }
		set_country_flag = VIN_Post_DBP_Preparation_Cleaned
	}
}
