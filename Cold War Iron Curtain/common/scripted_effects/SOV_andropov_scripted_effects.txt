
SOV_LaborDiscipline_calculation = {

	set_temp_variable = { LaborDiscipline_calc = LaborDiscipline }
	subtract_from_temp_variable = { LaborDiscipline_calc = 50 }
	divide_temp_variable = { LaborDiscipline_calc = 50 }

	set_temp_variable = { LaborDiscipline_calc_construction = LaborDiscipline_calc }
	set_temp_variable = { LaborDiscipline_calc_factory = LaborDiscipline_calc }
	set_temp_variable = { LaborDiscipline_calc_dockyard = LaborDiscipline_calc }
	set_temp_variable = { LaborDiscipline_calc_cap = LaborDiscipline_calc }
	set_temp_variable = { LaborDiscipline_calc_retention = LaborDiscipline_calc }
	set_temp_variable = { LaborDiscipline_calc_growth = LaborDiscipline_calc }
	set_temp_variable = { LaborDiscipline_calc_stab = LaborDiscipline_calc }
	set_temp_variable = { LaborDiscipline_calc_pp = LaborDiscipline_calc }

	divide_temp_variable = { LaborDiscipline_calc_construction = 20 }
	divide_temp_variable = { LaborDiscipline_calc_factory = 10 }
	divide_temp_variable = { LaborDiscipline_calc_dockyard = 20 }
	divide_temp_variable = { LaborDiscipline_calc_cap = 20 }
	divide_temp_variable = { LaborDiscipline_calc_retention = 10 }
	divide_temp_variable = { LaborDiscipline_calc_growth = 10 }
	divide_temp_variable = { LaborDiscipline_calc_stab = 10 }
	divide_temp_variable = { LaborDiscipline_calc_pp = 10 }

	if = {
		limit = { check_variable = { LaborDiscipline < 50 } }
		multiply_temp_variable = { LaborDiscipline_calc_construction = 3 }
		multiply_temp_variable = { LaborDiscipline_calc_cap = 2 }
		multiply_temp_variable = { LaborDiscipline_calc_retention = 2 }
		multiply_temp_variable = { LaborDiscipline_calc_growth = 2 }
	}

	multiply_temp_variable = { LaborDiscipline_calc_stab = -1 }
	multiply_temp_variable = { LaborDiscipline_calc_pp = -1 }

	set_variable = { LaborDiscipline_construction = LaborDiscipline_calc_construction }
	set_variable = { LaborDiscipline_factory = LaborDiscipline_calc_factory }
	set_variable = { LaborDiscipline_dockyard = LaborDiscipline_calc_dockyard }
	set_variable = { LaborDiscipline_cap = LaborDiscipline_calc_cap }
	set_variable = { LaborDiscipline_retention = LaborDiscipline_calc_retention }
	set_variable = { LaborDiscipline_growth = LaborDiscipline_calc_growth }
	set_variable = { LaborDiscipline_stab = LaborDiscipline_calc_stab }
	set_variable = { LaborDiscipline_pp = LaborDiscipline_calc_pp }
}

SOV_CorruptionLevel_calculation = {

	set_variable = { CorruptionLevel_100 = CorruptionLevel } #used for progressbar
	multiply_variable = { CorruptionLevel_100 = 100 }

	set_temp_variable = { CorruptionLevel_calc = CorruptionLevel }
	multiply_temp_variable = { CorruptionLevel_calc = -1 }
	set_temp_variable = { CorruptionLevel_calc_output2 = resource@rubber }
	
	set_temp_variable = { CorruptionLevel_calc_construction = CorruptionLevel_calc }
	set_temp_variable = { CorruptionLevel_calc_output = CorruptionLevel_100 }
	set_temp_variable = { CorruptionLevel_calc_labor = CorruptionLevel_calc }
	
	divide_temp_variable = { CorruptionLevel_calc_construction = 20 }
	divide_temp_variable = { CorruptionLevel_calc_output = 10 }
	divide_temp_variable = { CorruptionLevel_calc_labor = 50 }
	
	divide_temp_variable = { CorruptionLevel_calc_output2 = CorruptionLevel_calc_output }
	
	set_variable = { CorruptionLevel_construction = CorruptionLevel_calc_construction }
	set_variable = { CorruptionLevel_output = CorruptionLevel_calc_output2 }
	set_variable = { CorruptionLevel_labor = CorruptionLevel_calc_labor }
}
