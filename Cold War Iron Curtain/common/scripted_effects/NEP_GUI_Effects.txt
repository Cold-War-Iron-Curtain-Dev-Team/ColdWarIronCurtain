NEP_Monthly_Factional_Display = {
    if = {
        limit = {
            check_variable = {
                Sphere_Leader = RAJ.id
            }
        }
        set_variable = { NEP_Foriegn_Ring = 7 }
    }
    else_if = {
        limit = {
            check_variable = {
                Sphere_Leader = PRC.id
            }
        }
        set_variable = { NEP_Foriegn_Ring = 8 }
    }
    else = {
        set_variable = { NEP_Foriegn_Ring = 10 }
    }
    if = {
        limit = {
            has_stability > 0.74
        }
        set_variable = { NEP_Stability_Ring = 5 }
    }
    else_if = {
        limit = {
            has_stability > 0.39
        }
        set_variable = { NEP_Stability_Ring = 4 }
    }
    else = {
        set_variable = { NEP_Stability_Ring = 6 }
    }
}
