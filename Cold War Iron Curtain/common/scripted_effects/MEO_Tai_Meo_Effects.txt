# =============================================================================
# Tai-Meo Autonomous Region (MEO)
# =============================================================================
# These effects run in MEO scope after Hanoi approves the autonomous zone.
# Keeping the political setup separate from the focus prevents subject/autonomy
# changes from leaving MEO with the generic communist government.
# =============================================================================

meo_copy_vin_politics_for_autonomous_zone = {
	if = {
		limit = { VIN = { has_government = trotskyism } }
		meo_tai_meo_politics_trotskyism = yes
	}
	else_if = {
		limit = { VIN = { has_government = Social_Democratic } }
		meo_tai_meo_politics_socdem = yes
	}
	else = {
		meo_tai_meo_politics_communism = yes
	}
}

meo_promote_lo_van_hac_for_autonomous_zone = {
	if = {
		limit = { has_government = trotskyism }
		add_country_leader_role = {
            character = MEO_lo_van_hac
            promote_leader = yes
            country_leader = {
                ideology = ho_chi_minh_thought_revsoc
                expire = "1965.1.1.1"
                traits = { war_industrialist }
            }
        }
	}
	else_if = {
		limit = { has_government = Social_Democratic }
		add_country_leader_role = {
            character = MEO_lo_van_hac
            promote_leader = yes
            country_leader = {
                ideology = ho_chi_minh_thought_socdem
                expire = "1965.1.1.1"
                traits = { war_industrialist }
            }
        }
	}
	else = {
	    add_country_leader_role = {
            character = MEO_lo_van_hac
            promote_leader = yes
            country_leader = {
                ideology = ho_chi_minh_thought
                expire = "1965.1.1.1"
                traits = { war_industrialist }
            }
        }
	}
}
