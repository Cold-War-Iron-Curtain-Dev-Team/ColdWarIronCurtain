# ============================================================================
# Geneva Conference System
# ----------------------------------------------------------------------------
# Replaces the old Geneva_Conference_Invitation/Negotiations binary event
# chain with a round-based, multi-delegation negotiation that resolves the
# Indochina Struggle's diplomatic endings. The old outcome vocabulary
# (X_Accepted_Ceasefire / X_Accepted_Reunification / X_Rejected_* /
# USA_Tampered_With_VIE country flags, Geneva_Conference_Negotiations_Complete
# global flag) is still produced by the settlement compiler, so every
# downstream trigger and focus keeps working.
#
# Clause indices (global.geneva_clause_current^N holds the tabled option):
#   0 Ceasefire            0 = none          1 = phased          2 = immediate
#   1 Partition line       0 = unified       1 = 13th parallel   2 = 17th parallel  3 = 20th parallel
#   2 Reunification vote   0 = none          1 = postponed       2 = within two years  3 = immediate
#   3 French withdrawal    0 = retained      1 = timetabled      2 = immediate
#   4 Alignment            0 = free blocs    1 = enforced neutrality
#   5 Laos & Cambodia      0 = French Union  1 = independent     2 = communist regroupment zones
#   6 Minority peoples     0 = no provision  1 = autonomy guarantees
#
# Per-delegation country variables/arrays:
#   geneva_weight    - vote strength, seeded from struggle scores + leverage
#   geneva_patience  - 0..100, walks out at 0
#   geneva_pref^N    - preferred option per clause
#   geneva_prio^N    - how much the delegation cares about clause N (0-10)
# Flags: geneva_walked_out, geneva_core_delegation (FRA/VIE/VIN)
#
# Leverage pools (globals, accumulate before the conference convenes and can
# be fed from anywhere via the API below):
#   global.geneva_lev_communist / _profrance / _proindependence / _usa
#   global.geneva_lev_federal / _ethnic
#   global.geneva_rapprochement_vie_vin
# ============================================================================

# ----------------------------------------------------------------------------
# Modular leverage API. Callers may set temp geneva_lev_amount first;
# otherwise 50 is used. Example:
#   set_temp_variable = { geneva_lev_amount = 100 }
#   geneva_add_communist_leverage = yes
# ----------------------------------------------------------------------------
geneva_lev_default_amount = {
	if = {
		limit = { NOT = { check_variable = { geneva_lev_amount > 0 } } }
		set_temp_variable = { geneva_lev_amount = 50 }
	}
}
geneva_add_communist_leverage = {
	geneva_lev_default_amount = yes
	add_to_variable = { global.geneva_lev_communist = geneva_lev_amount }
	custom_effect_tooltip = geneva_add_communist_leverage_TT
}

# ----------------------------------------------------------------------------
# War-outcome award routers. Before the conference the points bank into the
# leverage pools; once the session is live they land on the seated delegation's
# vote weight instead. Callers set temp geneva_lev_amount first.
# ----------------------------------------------------------------------------
geneva_outcome_award_proindependence = {
	geneva_lev_default_amount = yes
	if = {
		limit = {
			has_global_flag = geneva_conference_in_session
			country_exists = VIE
			VIE = {
				is_in_array = { global.geneva_delegations = THIS.id }
				NOT = { has_country_flag = geneva_walked_out }
			}
		}
		VIE = { add_to_variable = { geneva_weight = geneva_lev_amount } }
		custom_effect_tooltip = geneva_outcome_award_vie_weight_TT
	}
	else = {
		geneva_add_proindependence_leverage = yes
	}
}

# ----------------------------------------------------------------------------
# World-state detectors. Runs daily from on_daily_FRA; every branch is
# one-shot flag guarded so the cost after firing is a handful of flag checks.
# ----------------------------------------------------------------------------
geneva_conf_world_leverage_pulse = {
	if = {
		limit = { NOT = { has_global_flag = Indochina_War_Over } }
		# --- War-balance cache for trigger-context readers (focus ai_will_do,
		# --- geneva_vie_side_*_trigger): triggers cannot do arithmetic, so the
		# --- pulse precomputes the totals daily. The settlement compiler
		# --- re-derives them from the final-score snapshot at conclusion.
		if = {
			limit = { country_exists = FRA }
			set_variable = { global.geneva_cache_comm_score = FRA.StruggleInvolvedNationsIndochinaCommunistScore }
			set_variable = { global.geneva_cache_anticomm_score = FRA.StruggleInvolvedNationsIndochinaProFranceScore }
			add_to_variable = { global.geneva_cache_anticomm_score = FRA.StruggleInvolvedNationsIndochinaProIndependenceScore }
			set_variable = { global.geneva_cache_comm_x2 = global.geneva_cache_comm_score }
			multiply_variable = { global.geneva_cache_comm_x2 = 2 }
		}
		# --- Laotian Civil War tracking (LAO = Pathet Lao, LOS = Kingdom) ---
		if = {
			limit = {
				NOT = { has_global_flag = geneva_pathet_lao_war_seen }
				country_exists = LAO
				country_exists = LOS
				LOS = { has_war_with = LAO }
			}
			set_global_flag = geneva_pathet_lao_war_seen
		}
		if = {
			limit = {
				has_global_flag = geneva_pathet_lao_war_seen
				NOT = { has_global_flag = geneva_pathet_lao_outcome_applied }
				country_exists = LAO
				OR = {
					NOT = { country_exists = LOS }
					LOS = { has_capitulated = yes }
				}
			}
			# Pathet Lao overran the Kingdom of Laos: huge communist boost.
			set_global_flag = geneva_pathet_lao_outcome_applied
			set_variable = { global.geneva_outcome_pathet_lao = 1 }
			set_temp_variable = { geneva_lev_amount = 125 }
			geneva_add_communist_leverage = yes
		}
		else_if = {
			limit = {
				has_global_flag = geneva_pathet_lao_war_seen
				NOT = { has_global_flag = geneva_pathet_lao_outcome_applied }
				country_exists = LOS
				LOS = { has_capitulated = no }
				OR = {
					NOT = { country_exists = LAO }
					LAO = { has_capitulated = yes }
				}
			}
			# The Kingdom beat back the Pathet Lao: huge anti-communist boost.
			set_global_flag = geneva_pathet_lao_outcome_applied
			set_variable = { global.geneva_outcome_pathet_lao = 2 }
			set_temp_variable = { geneva_lev_amount = 125 }
			geneva_add_profrance_leverage = yes
		}

		# --- USA commits to VIE early (Diplo tree early-intervention path) ---
		if = {
			limit = {
				NOT = { has_global_flag = geneva_usa_backs_vie_applied }
				country_exists = USA
				USA = { has_country_flag = USA_VIE_Direct_Intervention_Locked }
			}
			set_global_flag = geneva_usa_backs_vie_applied
			set_temp_variable = { geneva_lev_amount = 75 }
			geneva_add_proindependence_leverage = yes
			set_temp_variable = { geneva_lev_amount = 75 }
			geneva_add_usa_leverage = yes
		}

		# --- France hands the war to the State of Vietnam (Elysee framework) ---
		if = {
			limit = {
				NOT = { has_global_flag = geneva_handoff_applied }
				country_exists = VIE
				VIE = { has_completed_focus = VIE_Second_Elysee_Accords }
			}
			set_global_flag = geneva_handoff_applied
			set_temp_variable = { geneva_lev_amount = 75 }
			geneva_add_federal_leverage = yes
		}

		# --- USA doubles down on the French war effort ---
		if = {
			limit = {
				NOT = { has_global_flag = geneva_usa_backs_france_applied }
				country_exists = USA
				USA = {
					OR = {
						has_completed_focus = USA_50s_Operation_Vulture
						has_completed_focus = USA_50s_Military_Aid_to_France
					}
				}
			}
			set_global_flag = geneva_usa_backs_france_applied
			set_temp_variable = { geneva_lev_amount = 50 }
			geneva_add_profrance_leverage = yes
		}

		# --- USA props up the Dan Quoc government directly ---
		if = {
			limit = {
				NOT = { has_global_flag = geneva_usa_danquoc_applied }
				country_exists = USA
				USA = { has_completed_focus = USA_50s_Stabilize_DanQuoc_Regime }
			}
			set_global_flag = geneva_usa_danquoc_applied
			set_temp_variable = { geneva_lev_amount = 50 }
			geneva_add_proindependence_leverage = yes
		}

		# --- Cuong De sends Diem to talk to Ho (VIE_NH_Using_Diem_Ho_Connections):
		#     the Vietnamese closing ranks strengthens the independence camp ---
		if = {
			limit = {
				NOT = { has_global_flag = geneva_cuongde_talks_applied }
				has_global_flag = CuongDe_DiemHo_Talks
			}
			set_global_flag = geneva_cuongde_talks_applied
			set_temp_variable = { geneva_lev_amount = 50 }
			geneva_add_proindependence_leverage = yes
		}

		# --- Saigon forms the anticommunist front: reconciliation curdles.
		#     Written directly because the leverage API clamps negatives. ---
		if = {
			limit = {
				NOT = { has_global_flag = geneva_cuongde_front_applied }
				country_exists = VIE
				VIE = { has_country_flag = VIE_Hung_Formed_Anticommunist_Front }
			}
			set_global_flag = geneva_cuongde_front_applied
			add_to_variable = { global.geneva_rapprochement_vie_vin = -50 }
		}

		# --- War-outcome scenario detectors: one-shot guarded, recorders drive the
		# --- "War Outcomes" panel loc, awards route through geneva_outcome_award_*.

		# --- Fate of the Southern Viet Minh ---
		if = {
			limit = {
				NOT = { has_global_flag = geneva_nlf_outcome_applied }
				OR = {
					NOT = { country_exists = NLF }
					NLF = { has_capitulated = yes }
					NLF = { num_of_controlled_states < 1 }
	}
			}
			set_global_flag = geneva_nlf_outcome_applied
			set_global_flag = Southern_VietMinh_Defeated
			if = {
				limit = { date < 1952.1.1 }
				set_variable = { global.geneva_outcome_nlf = 1 }
				set_temp_variable = { geneva_lev_amount = 100 }
			}
			else_if = {
				limit = { date < 1953.1.1 }
				set_variable = { global.geneva_outcome_nlf = 2 }
				set_temp_variable = { geneva_lev_amount = 75 }
			}
			else = {
				set_variable = { global.geneva_outcome_nlf = 3 }
				set_temp_variable = { geneva_lev_amount = 50 }
			}
			geneva_outcome_award_proindependence = yes
			if = {
				limit = {
					has_global_flag = geneva_conference_in_session
					country_exists = VIE
					VIE = { is_in_array = { global.geneva_delegations = THIS.id } }
				}
				VIE = { geneva_conf_refresh_prefs = yes }	# a mid-conference victory restores Saigon's ambition on the partition line
			}
		}

		# The 1953 raid froze into the historical two-zone garrison.
		if = {
			limit = {
				has_global_flag = Laos_Raid_Stalemate
				check_variable = { global.geneva_outcome_pathet_lao = 0 }
			}
			set_variable = { global.geneva_outcome_pathet_lao = 3 }	# display only: points paid by the Laos Raid scenario
		}

		# --- Battle of Dien Bien Phu (result flags set by the VIN map resolver) ---
		if = {
			limit = {
				NOT = { has_global_flag = geneva_dbp_outcome_applied }
				country_exists = FRE
				FRE = {
					OR = {
						has_country_flag = FRE_DBP_Result_Victory
						has_country_flag = FRE_DBP_Result_Fall
						has_country_flag = FRE_DBP_Result_Breakout
					}
				}
			}
			set_global_flag = geneva_dbp_outcome_applied
			if = {
				limit = { FRE = { has_country_flag = FRE_DBP_Result_Victory } }
				set_variable = { global.geneva_outcome_dbp = 1 }
				set_temp_variable = { geneva_lev_amount = 75 }
				geneva_outcome_award_profrance = yes
			}
			else_if = {
				limit = { FRE = { has_country_flag = FRE_DBP_Result_Fall } }
				set_variable = { global.geneva_outcome_dbp = 2 }
				set_temp_variable = { geneva_lev_amount = 100 }
				geneva_outcome_award_communist = yes
			}
			else = {
				set_variable = { global.geneva_outcome_dbp = 3 }
				set_temp_variable = { geneva_lev_amount = 25 }
				geneva_outcome_award_profrance = yes
			}
		}

		# --- Fate of the Religious Sects ---
		if = {
			limit = {
				NOT = { has_global_flag = geneva_sects_outcome_applied }
				country_exists = VIE
				VIE = {
					OR = { has_country_flag = crushed_binh_xuyen has_country_flag = integrated_binh_xuyen }
					OR = { has_country_flag = crushed_cao_dai has_country_flag = integrated_cao_dai }
					OR = { has_country_flag = crushed_harmonism has_country_flag = integrated_harmonism }
					OR = { has_country_flag = crushed_hoa_hao has_country_flag = integrated_hoa_hao }
					OR = { has_country_flag = crushed_umdc has_country_flag = integrated_umdc }
				}
			}
			set_global_flag = geneva_sects_outcome_applied
			set_variable = { global.geneva_outcome_sects = 1 }
			set_temp_variable = { geneva_lev_amount = 75 }
			geneva_outcome_award_proindependence = yes
		}

		# --- Fate of the Highland States ---
		if = {
			limit = {
				NOT = { has_global_flag = geneva_highland_tai_applied }
				NOT = { check_variable = { global.geneva_outcome_highland_done > 1 } }
				OR = {
					TAI = { is_subject_of = VIN }
					AND = {
						NOT = { country_exists = TAI }
						OR = {
							671 = { is_owned_by = VIN }
							AND = {
								671 = { is_owned_by = MEO }
								MEO = { is_subject_of = VIN }
							}
						}
					}
				}
			}
			set_global_flag = geneva_highland_tai_applied
			add_to_variable = { global.geneva_outcome_highland = 1 }
			set_temp_variable = { geneva_lev_amount = 25 }
			geneva_outcome_award_communist = yes
		}
		if = {
			limit = {
				NOT = { has_global_flag = geneva_highland_meo_applied }
				NOT = { check_variable = { global.geneva_outcome_highland_done > 1 } }
				OR = {
					MEO = { is_subject_of = VIN }
					AND = {
						NOT = { country_exists = MEO }
						1767 = { is_owned_by = VIN }
					}
				}
			}
			set_global_flag = geneva_highland_meo_applied
			add_to_variable = { global.geneva_outcome_highland = 1 }
			set_temp_variable = { geneva_lev_amount = 25 }
			geneva_outcome_award_communist = yes
		}
		if = {
			limit = {
				NOT = { has_global_flag = geneva_highland_nun_applied }
				NOT = { check_variable = { global.geneva_outcome_highland_done > 1 } }
				OR = {
					NUN = { is_subject_of = VIN }
					AND = {
						NOT = { country_exists = NUN }
						1281 = { is_owned_by = VIN }
					}
				}
			}
			set_global_flag = geneva_highland_nun_applied
			add_to_variable = { global.geneva_outcome_highland = 1 }
			set_temp_variable = { geneva_lev_amount = 25 }
			geneva_outcome_award_communist = yes
		}
		if = {
			limit = {
				check_variable = { global.geneva_outcome_highland > 2 }
				check_variable = { global.geneva_outcome_highland_done = 0 }
			}
			set_variable = { global.geneva_outcome_highland_done = 1 }
		}

		# --- The 1953 Laos Raid (flags set by IC_Laos_Raid_Effects.txt) ---
		if = {
			limit = {
				NOT = { has_global_flag = geneva_laos_raid_outcome_applied }
				has_global_flag = Laos_Raid_Resolved
			}
			set_global_flag = geneva_laos_raid_outcome_applied
			if = {
				limit = { has_global_flag = Laos_Total_War }
				set_variable = { global.geneva_outcome_laos_raid = 4 }	# escalation: the wider war pays out via the Pathet Lao scenario
			}
			else_if = {
				limit = { has_global_flag = Laos_Raid_Attacker_Won }
				set_variable = { global.geneva_outcome_laos_raid = 1 }
				set_temp_variable = { geneva_lev_amount = 100 }
				geneva_outcome_award_communist = yes
			}
			else_if = {
				limit = { has_global_flag = Laos_Raid_Total_Failure }
				set_variable = { global.geneva_outcome_laos_raid = 3 }
				set_temp_variable = { geneva_lev_amount = 75 }
				geneva_outcome_award_profrance = yes
			}
			else_if = {
				limit = { has_global_flag = Laos_Raid_Stalemate }
				set_variable = { global.geneva_outcome_laos_raid = 2 }
				set_temp_variable = { geneva_lev_amount = 50 }
				geneva_outcome_award_communist = yes
			}
		}
	}
}

# ----------------------------------------------------------------------------
# Seals every still-pending war-outcome scenario at conference opening, before
# any delegation is seated, so the awards land in the leverage pools that
# geneva_conf_seed_weight is about to read.
# ----------------------------------------------------------------------------
geneva_conf_resolve_pending_outcomes = {
	if = {
		limit = { NOT = { has_global_flag = geneva_nlf_outcome_applied } }
		set_global_flag = geneva_nlf_outcome_applied
		set_variable = { global.geneva_outcome_nlf = 4 }	# the southern insurgency outlasted the war
		set_temp_variable = { geneva_lev_amount = 75 }
		geneva_add_communist_leverage = yes
	}
	if = {
		limit = { NOT = { has_global_flag = geneva_sects_outcome_applied } }
		set_global_flag = geneva_sects_outcome_applied
		set_variable = { global.geneva_outcome_sects = 2 }	# a fractured South weakens the anti-communist chair
		set_temp_variable = { geneva_lev_amount = 50 }
		geneva_add_communist_leverage = yes
	}
	if = {
		limit = {
			NOT = { has_global_flag = geneva_pathet_lao_outcome_applied }
			check_variable = { global.geneva_outcome_pathet_lao = 0 }
		}
		set_variable = { global.geneva_outcome_pathet_lao = 4 }	# no settlement in Laos before the conference
	}
	if = {
		limit = { check_variable = { global.geneva_outcome_highland_done = 0 } }
		set_variable = { global.geneva_outcome_highland_done = 2 }
		set_temp_variable = { g_hl_surv = 3 }
		subtract_from_temp_variable = { g_hl_surv = global.geneva_outcome_highland }
		if = {
			limit = { check_variable = { g_hl_surv > 0 } }
			set_temp_variable = { geneva_lev_amount = 25 }
			multiply_temp_variable = { geneva_lev_amount = g_hl_surv }
			geneva_add_ethnic_leverage = yes	# surviving highland polities speak through the small delegations
		}
	}
}

# ============================================================================
# Negotiation pipeline. Champion/Threaten open a visible three-day negotiation
# tracked in five parallel global arrays; the daily tick banks matured stakes
# into global.geneva_push, which the weekly vote consumes. Per-country mirrors
# exist because triggers and scripted loc cannot iterate arrays:
#   geneva_neg_mine^clause                  1 while this country negotiates it
#   geneva_neg_display_clause               clause+1 of the first active entry
#   geneva_neg_display_days                 days left on that entry
#   global.geneva_neg_clause_count^clause   entries in flight per clause
# ============================================================================

# Open a negotiation. Country scope; caller sets temps g_neg_clause,
# g_neg_option, g_neg_amount. With temp g_geneva_headless = 1 (auto-resolve)
# the stake banks instantly instead.
geneva_neg_start = {
	if = {
		limit = { check_variable = { g_geneva_headless = 1 } }
		set_temp_variable = { g_npi = g_neg_clause }
		multiply_temp_variable = { g_npi = 4 }
		add_to_temp_variable = { g_npi = g_neg_option }
		add_to_variable = { global.geneva_push^g_npi = g_neg_amount }
	}
	else_if = {
		limit = { check_variable = { geneva_neg_mine^g_neg_clause = 0 } }
		add_to_array = { global.geneva_neg_country = THIS.id }
		add_to_array = { global.geneva_neg_clause = g_neg_clause }
		add_to_array = { global.geneva_neg_option = g_neg_option }
		add_to_array = { global.geneva_neg_amount = g_neg_amount }
		add_to_array = { global.geneva_neg_days = 3 }
		geneva_neg_rebuild_mirrors = yes
		custom_effect_tooltip = geneva_neg_start_TT
	}
}

# Recompute every mirror from the arrays. Cheap: runs only while in session.
geneva_neg_rebuild_mirrors = {
	every_country = {
		limit = { is_in_array = { global.geneva_delegations = THIS.id } }
		set_variable = { geneva_neg_display_clause = 0 }

# Daily maturation sweep. Downward so removal by index never shifts an
# unvisited entry. Walked-out owners forfeit their in-flight stakes.
geneva_neg_daily_tick = {
	set_temp_variable = { g_ni = global.geneva_neg_country^num }
	subtract_from_temp_variable = { g_ni = 1 }
	while_loop_effect = {
		limit = { check_variable = { g_ni > -1 } }
		set_temp_variable = { g_nown = global.geneva_neg_country^g_ni }
		set_temp_variable = { g_ndrop = 0 }
		every_country = {
			limit = {
				check_variable = { THIS.id = g_nown }
				has_country_flag = geneva_walked_out
			}
			set_temp_variable = { g_ndrop = 1 }
		}
		if = {
			limit = { check_variable = { g_ndrop = 0 } }
			subtract_from_variable = { global.geneva_neg_days^g_ni = 1 }
			if = {
				limit = { check_variable = { global.geneva_neg_days^g_ni < 1 } }
				set_temp_variable = { g_npi = global.geneva_neg_clause^g_ni }
				multiply_temp_variable = { g_npi = 4 }
				add_to_temp_variable = { g_npi = global.geneva_neg_option^g_ni }
				add_to_variable = { global.geneva_push^g_npi = global.geneva_neg_amount^g_ni }
				set_temp_variable = { g_ndrop = 1 }
			}
		}
		if = {
			limit = { check_variable = { g_ndrop = 1 } }
			remove_from_array = { array = global.geneva_neg_country index = g_ni }
			remove_from_array = { array = global.geneva_neg_clause index = g_ni }
			remove_from_array = { array = global.geneva_neg_option index = g_ni }
			remove_from_array = { array = global.geneva_neg_amount index = g_ni }
			remove_from_array = { array = global.geneva_neg_days index = g_ni }
		}
		subtract_from_temp_variable = { g_ni = 1 }
	}
	geneva_neg_rebuild_mirrors = yes
}

	else_if = {
		limit = { OR = { tag = SOV tag = PRC } }
		set_temp_variable = { g_w = 50 }
		set_temp_variable = { g_ws = global.geneva_lev_communist }
		divide_temp_variable = { g_ws = 2 }
		add_to_temp_variable = { g_w = g_ws }

	else_if = {

		divide_temp_variable = { g_ws = 2 }

		set_temp_variable = { g_w = 70 }

			limit = { country_exists = FRA }

geneva_conf_initialize = {
	if = {
		limit = {
			NOT = { has_global_flag = geneva_conference_in_session }
			NOT = { has_global_flag = geneva_conference_concluded }
			NOT = { has_global_flag = Indochina_War_Over }
		}
		set_global_flag = geneva_conference_in_session
		set_global_flag = Geneva_Conference_Preparations
		geneva_conf_resolve_pending_outcomes = yes	# seal open war scenarios into the pools before seating reads them
		set_variable = { global.geneva_round = 0 }
		set_variable = { global.geneva_tick_days = 0 }
		set_variable = { global.geneva_outcome = 0 }
		set_variable = { global.geneva_wrecker = 0 }
		set_variable = { global.geneva_walkout_news = 0 }
		clear_array = global.geneva_delegations

		# Historical opening agenda on the table.
		clear_array = global.geneva_clause_current
		add_to_array = { global.geneva_clause_current = 2 }	# immediate ceasefire
		add_to_array = { global.geneva_clause_current = 2 }	# 17th parallel
		add_to_array = { global.geneva_clause_current = 1 }	# elections postponed
		add_to_array = { global.geneva_clause_current = 1 }	# timetabled withdrawal
		add_to_array = { global.geneva_clause_current = 1 }	# neutrality
		add_to_array = { global.geneva_clause_current = 1 }	# independent Laos/Cambodia
		add_to_array = { global.geneva_clause_current = 0 }	# no minority provision

		# Clause severity: scales action costs, AI stakes and the equal-reaction
		# backlash after a flip.
		geneva_conf_seed_severity = yes

		# Campaign pushes: one flattened 28-slot array (clause * 4 + option),
		# fed by table actions and AI campaigning, consumed and cleared by
		# each round's vote.
		clear_array = global.geneva_push
		resize_array = { array = global.geneva_push size = 28 value = 0 }
		geneva_neg_clear = yes	# no negotiations carry over from a previous conference

		# Core belligerent delegations: their walkouts can sink the conference.
		if = {
			limit = { country_exists = FRA }
			FRA = {
				geneva_conf_seat_delegation = yes
				set_country_flag = geneva_core_delegation
			}
		}
		if = {
			limit = { country_exists = VIE }
			VIE = {
				geneva_conf_seat_delegation = yes
				set_country_flag = geneva_core_delegation
			}
		}
		if = {
			limit = { country_exists = VIN }
			VIN = {
				geneva_conf_seat_delegation = yes
				set_country_flag = geneva_core_delegation
			}
		}
		# Great-power seats.
		if = { limit = { country_exists = USA } USA = { geneva_conf_seat_delegation = yes } }
		if = { limit = { country_exists = SOV } SOV = { geneva_conf_seat_delegation = yes } }
		if = { limit = { country_exists = PRC } PRC = { geneva_conf_seat_delegation = yes } }
		if = { limit = { country_exists = ENG } ENG = { geneva_conf_seat_delegation = yes } }
		# Regional minor seats.
		if = { limit = { country_exists = LOS } LOS = { geneva_conf_seat_delegation = yes } }
		if = { limit = { country_exists = CAM } CAM = { geneva_conf_seat_delegation = yes } }
		if = { limit = { country_exists = LAO } LAO = { geneva_conf_seat_delegation = yes } }
		if = { limit = { country_exists = PQC } PQC = { geneva_conf_seat_delegation = yes } }

		# Seats are taken, so the national trees can be brought level with them.
		geneva_conf_catch_up_delegation_focuses = yes

		# Broadcast to every human player: news anchored on FRA is invisible
		# to everyone else when France is AI (playtest finding).
		every_country = {
			limit = { is_ai = no }
			news_event = Geneva_Conference_Session.1
		}
	}
}

# ----------------------------------------------------------------------------
# One negotiation round. Driven weekly from on_daily_FRA while in session;
# also driven headlessly by geneva_conf_auto_resolve.
# ----------------------------------------------------------------------------
geneva_conf_round_tick = {
	set_temp_variable = { g_tick_repaired = 0 }
	if = {
		limit = {	# in session with nobody at the table: rebuild rather than voting on an empty room
			has_global_flag = geneva_conference_in_session
			NOT = { has_global_flag = Indochina_War_Over }
			NOT = { check_variable = { global.geneva_delegations^num > 0 } }
		}
		clr_global_flag = geneva_conference_in_session	# geneva_conf_initialize refuses to run while it is set
		geneva_conf_initialize = yes
		set_temp_variable = { g_tick_repaired = 1 }
	}
	if = {
		limit = {
			has_global_flag = geneva_conference_in_session
			NOT = { has_global_flag = Indochina_War_Over }
			check_variable = { g_tick_repaired = 0 }
		}
		if = {
			limit = { check_variable = { global.geneva_round > 9 } }
			set_variable = { global.geneva_round = 0 }	# a live session never persists past round 10, so anything higher is stale save data that would compile instantly
		}
		add_to_variable = { global.geneva_round = 1 }

		# --- New round: every delegation gets its table actions back.
		every_country = {
			limit = { is_in_array = { global.geneva_delegations = THIS.id } }
			set_variable = { geneva_actions_used = 0 }
		}

		# --- AI campaigning: each AI delegation contests the clauses it is losing,
		# --- worst grievance first, with an action budget scaled to its weight.
		every_country = {
			limit = {
				is_in_array = { global.geneva_delegations = THIS.id }
				NOT = { has_country_flag = geneva_walked_out }
				is_ai = yes
			}
			set_temp_variable = { g_budget = 1 }
			if = {
				limit = { check_variable = { geneva_weight > 80 } }
				set_temp_variable = { g_budget = 2 }
			}
			if = {
				limit = { check_variable = { geneva_weight > 160 } }
				set_temp_variable = { g_budget = 3 }
			}
			set_temp_variable = { g_conceded = 0 }
			set_temp_variable = { g_a0 = 0 }
			set_temp_variable = { g_a1 = 0 }
			set_temp_variable = { g_a2 = 0 }
			set_temp_variable = { g_a3 = 0 }
			set_temp_variable = { g_a4 = 0 }
			set_temp_variable = { g_a5 = 0 }
			set_temp_variable = { g_a6 = 0 }
			while_loop_effect = {
				limit = { check_variable = { g_budget > 0 } }
			set_temp_variable = { g_bc = -1 }
				set_temp_variable = { g_bcare = 0 }
			set_temp_variable = { g_ci = 0 }
			while_loop_effect = {
				limit = { check_variable = { g_ci < 7 } }
					set_temp_variable = { g_acted = 0 }
					if = { limit = { check_variable = { g_ci = 0 } } set_temp_variable = { g_acted = g_a0 } }
					else_if = { limit = { check_variable = { g_ci = 1 } } set_temp_variable = { g_acted = g_a1 } }
					else_if = { limit = { check_variable = { g_ci = 2 } } set_temp_variable = { g_acted = g_a2 } }
					else_if = { limit = { check_variable = { g_ci = 3 } } set_temp_variable = { g_acted = g_a3 } }
					else_if = { limit = { check_variable = { g_ci = 4 } } set_temp_variable = { g_acted = g_a4 } }
					else_if = { limit = { check_variable = { g_ci = 5 } } set_temp_variable = { g_acted = g_a5 } }
					else = { set_temp_variable = { g_acted = g_a6 } }
				set_temp_variable = { g_pv = geneva_pref^g_ci }
				set_temp_variable = { g_tv = global.geneva_clause_current^g_ci }
					set_temp_variable = { g_care = geneva_prio^g_ci }
					multiply_temp_variable = { g_care = global.geneva_severity^g_ci }
				if = {
					limit = {
							check_variable = { g_acted = 0 }
						NOT = { check_variable = { g_pv = g_tv } }
							check_variable = { g_care > g_bcare }
					}
						set_temp_variable = { g_bcare = g_care }
					set_temp_variable = { g_bc = g_ci }
				}
				add_to_temp_variable = { g_ci = 1 }
			}
			if = {
					limit = { check_variable = { g_bc < 0 } }
					set_temp_variable = { g_budget = 0 }	# nothing left worth fighting for this round
				}
				else = {
					if = { limit = { check_variable = { g_bc = 0 } } set_temp_variable = { g_a0 = 1 } }
					else_if = { limit = { check_variable = { g_bc = 1 } } set_temp_variable = { g_a1 = 1 } }
					else_if = { limit = { check_variable = { g_bc = 2 } } set_temp_variable = { g_a2 = 1 } }
					else_if = { limit = { check_variable = { g_bc = 3 } } set_temp_variable = { g_a3 = 1 } }
					else_if = { limit = { check_variable = { g_bc = 4 } } set_temp_variable = { g_a4 = 1 } }
					else_if = { limit = { check_variable = { g_bc = 5 } } set_temp_variable = { g_a5 = 1 } }
					else = { set_temp_variable = { g_a6 = 1 } }
					set_temp_variable = { g_bsev = global.geneva_severity^g_bc }
					set_temp_variable = { g_bprio = geneva_prio^g_bc }
					if = {
						limit = {
							check_variable = { g_bprio < 3 }
							check_variable = { geneva_weight < 60 }
							check_variable = { g_conceded = 0 }
						}
						# Hopeless minor gripe: cut losses, bank goodwill and weight.
						set_variable = { geneva_sel_clause = g_bc }
						geneva_action_concede = yes
						set_temp_variable = { g_conceded = 1 }
					}
					else_if = {
						limit = {
							check_variable = { g_bsev > 2 }
							check_variable = { g_bprio > 8 }	# only the partition line clears this: the AI puts its chair on the line for the border and nothing else
							check_variable = { geneva_patience > 60 }	# a bluff made from comfort, never from the brink
							check_variable = { global.geneva_round > 2 }	# not before the talks have properly opened
							check_variable = { geneva_threats_used < 2 }
							NOT = { check_variable = { geneva_threat_next > global.geneva_round } }
						}
						# The great clauses are worth the chair: public walkout threat.
						set_variable = { geneva_sel_clause = g_bc }
						add_to_variable = { geneva_threats_used = 1 }
						geneva_action_threaten = yes
					}
					else = {
						# Scaled campaign stake through the visible pipeline.
						set_temp_variable = { g_amt = 25 }
						set_temp_variable = { g_amt2 = g_bprio }
						multiply_temp_variable = { g_amt2 = 5 }
						add_to_temp_variable = { g_amt = g_amt2 }	# 25%..75% of weight by priority
						multiply_temp_variable = { g_amt = geneva_weight }
						divide_temp_variable = { g_amt = 100 }
						set_temp_variable = { g_neg_clause = g_bc }
						set_temp_variable = { g_neg_option = geneva_pref^g_bc }
						set_temp_variable = { g_neg_amount = g_amt }
						geneva_neg_start = yes
						set_temp_variable = { g_wcost = g_bsev }
						multiply_temp_variable = { g_wcost = 3 }
						subtract_from_variable = { geneva_weight = g_wcost }	# 3/6/9 weight by severity
				clamp_variable = {
					var = geneva_weight
							min = 15
					max = 10000
				}
			}
					subtract_from_temp_variable = { g_budget = 1 }
		}
			}
		}

		# --- Snapshot the agenda so the equal-reaction pass below can see which
		# --- clauses this vote flipped.
		set_temp_variable = { g_prev_0 = global.geneva_clause_current^0 }
		set_temp_variable = { g_prev_1 = global.geneva_clause_current^1 }
		set_temp_variable = { g_prev_2 = global.geneva_clause_current^2 }
		set_temp_variable = { g_prev_3 = global.geneva_clause_current^3 }
		set_temp_variable = { g_prev_4 = global.geneva_clause_current^4 }
		set_temp_variable = { g_prev_5 = global.geneva_clause_current^5 }
		set_temp_variable = { g_prev_6 = global.geneva_clause_current^6 }

		# --- Weighted vote per clause: strength = weight x priority. Ties keep
		# --- the currently tabled option so the agenda doesn't oscillate.
		set_temp_variable = { g_c = 0 }
		while_loop_effect = {
			limit = { check_variable = { g_c < 7 } }
			set_temp_variable = { g_opt_0 = 0 }
			set_temp_variable = { g_opt_1 = 0 }
			set_temp_variable = { g_opt_2 = 0 }
			set_temp_variable = { g_opt_3 = 0 }
			every_country = {
				limit = {
					is_in_array = { global.geneva_delegations = THIS.id }
					NOT = { has_country_flag = geneva_walked_out }
				}
				set_temp_variable = { g_vote = geneva_weight }
				multiply_temp_variable = { g_vote = geneva_prio^g_c }
				set_temp_variable = { g_p = geneva_pref^g_c }
				if = {
					limit = { check_variable = { g_p = 0 } }
					add_to_temp_variable = { g_opt_0 = g_vote }
				}
				else_if = {
					limit = { check_variable = { g_p = 1 } }
					add_to_temp_variable = { g_opt_1 = g_vote }
				}
				else_if = {
					limit = { check_variable = { g_p = 2 } }
					add_to_temp_variable = { g_opt_2 = g_vote }
				}
				else = {
					add_to_temp_variable = { g_opt_3 = g_vote }
				}
			}
			# Campaign pushes staked on this clause since the last vote.
			set_temp_variable = { g_pi0 = g_c }
			multiply_temp_variable = { g_pi0 = 4 }
			set_temp_variable = { g_pi1 = g_pi0 }
			add_to_temp_variable = { g_pi1 = 1 }
			set_temp_variable = { g_pi2 = g_pi0 }
			add_to_temp_variable = { g_pi2 = 2 }
			set_temp_variable = { g_pi3 = g_pi0 }
			add_to_temp_variable = { g_pi3 = 3 }
			add_to_temp_variable = { g_opt_0 = global.geneva_push^g_pi0 }
			add_to_temp_variable = { g_opt_1 = global.geneva_push^g_pi1 }
			add_to_temp_variable = { g_opt_2 = global.geneva_push^g_pi2 }
			add_to_temp_variable = { g_opt_3 = global.geneva_push^g_pi3 }
			set_temp_variable = { g_best = global.geneva_clause_current^g_c }
			if = {
				limit = { check_variable = { g_best = 0 } }
				set_temp_variable = { g_best_v = g_opt_0 }
			}
			else_if = {
				limit = { check_variable = { g_best = 1 } }
				set_temp_variable = { g_best_v = g_opt_1 }
			}
			else_if = {
				limit = { check_variable = { g_best = 2 } }
				set_temp_variable = { g_best_v = g_opt_2 }
			}
			else = {
				set_temp_variable = { g_best_v = g_opt_3 }
			}
			if = {
				limit = { check_variable = { g_opt_0 > g_best_v } }
				set_temp_variable = { g_best = 0 }
				set_temp_variable = { g_best_v = g_opt_0 }
			}
			if = {
				limit = { check_variable = { g_opt_1 > g_best_v } }
				set_temp_variable = { g_best = 1 }
				set_temp_variable = { g_best_v = g_opt_1 }
			}
			if = {
				limit = { check_variable = { g_opt_2 > g_best_v } }
				set_temp_variable = { g_best = 2 }
				set_temp_variable = { g_best_v = g_opt_2 }
			}
			if = {
				limit = { check_variable = { g_opt_3 > g_best_v } }
				set_temp_variable = { g_best = 3 }
				set_temp_variable = { g_best_v = g_opt_3 }
			}
			if = {	# Clause 1 is the only ordinal one: 1/2/3 run north to south along the same axis, so a plurality is not a mandate. If the table's majority sits south of the 13th or north of the 20th, the line settles on the 17th rather than handing the extreme to whichever bloc stayed concentrated while the other split. Option 0 (a unified Vietnam) is off the axis and never merged.
				limit = { check_variable = { g_c = 1 } }
				if = {
					limit = { check_variable = { g_best = 1 } }
					set_temp_variable = { g_ro_south = g_opt_2 }
					add_to_temp_variable = { g_ro_south = g_opt_3 }
					multiply_temp_variable = { g_ro_south = 1.5 }	# a plurality is not a mandate: the 13th holds only against half again the combined weight sitting south of it
					if = {
						limit = { NOT = { check_variable = { g_opt_1 > g_ro_south } } }
						set_temp_variable = { g_best = 2 }
					}
				}
				else_if = {
					limit = { check_variable = { g_best = 3 } }
					set_temp_variable = { g_ro_north = g_opt_1 }
					add_to_temp_variable = { g_ro_north = g_opt_2 }
					multiply_temp_variable = { g_ro_north = 1.5 }
					if = {
						limit = { NOT = { check_variable = { g_opt_3 > g_ro_north } } }
						set_temp_variable = { g_best = 2 }
					}
				}
			}
			set_variable = { global.geneva_clause_current^g_c = g_best }
			add_to_temp_variable = { g_c = 1 }
		}

		# --- Pushes are consumed by the vote above; reset for the next round.
		set_temp_variable = { g_pz = 0 }
		while_loop_effect = {
			limit = { check_variable = { g_pz < 28 } }
			set_variable = { global.geneva_push^g_pz = 0 }
			add_to_temp_variable = { g_pz = 1 }
		}

		# --- Equal reaction: flipping a clause hands the delegations that lost it a
		# --- weight rebate and a grievance; losing a great clause also seeds a free
		# --- counter-push for the NEXT vote (deliberately after the zeroing above).
		set_temp_variable = { g_rc = 0 }
		while_loop_effect = {
			limit = { check_variable = { g_rc < 7 } }
			set_temp_variable = { g_rprev = -1 }
			if = { limit = { check_variable = { g_rc = 0 } } set_temp_variable = { g_rprev = g_prev_0 } }
			else_if = { limit = { check_variable = { g_rc = 1 } } set_temp_variable = { g_rprev = g_prev_1 } }
			else_if = { limit = { check_variable = { g_rc = 2 } } set_temp_variable = { g_rprev = g_prev_2 } }
			else_if = { limit = { check_variable = { g_rc = 3 } } set_temp_variable = { g_rprev = g_prev_3 } }
			else_if = { limit = { check_variable = { g_rc = 4 } } set_temp_variable = { g_rprev = g_prev_4 } }
			else_if = { limit = { check_variable = { g_rc = 5 } } set_temp_variable = { g_rprev = g_prev_5 } }
			else = { set_temp_variable = { g_rprev = g_prev_6 } }
			set_temp_variable = { g_rnew = global.geneva_clause_current^g_rc }
			set_temp_variable = { g_rsev = global.geneva_severity^g_rc }
			if = {
				limit = {
					NOT = { check_variable = { g_rprev = g_rnew } }
					check_variable = { g_rsev > 1 }
				}
				set_temp_variable = { g_rreb = g_rsev }
				multiply_temp_variable = { g_rreb = 5 }	# 10/15 weight rebate for the losers
				every_country = {
					limit = {
						is_in_array = { global.geneva_delegations = THIS.id }
						NOT = { has_country_flag = geneva_walked_out }
						NOT = { check_variable = { geneva_pref^g_rc = g_rnew } }
					}
					add_to_variable = { geneva_weight = g_rreb }
					add_to_variable = { geneva_grievance = g_rsev }
					if = {
						limit = { check_variable = { g_rsev > 2 } }
						set_temp_variable = { g_rcp = geneva_weight }
						multiply_temp_variable = { g_rcp = 0.45 }
						set_temp_variable = { g_rpi = g_rc }
						multiply_temp_variable = { g_rpi = 4 }
						add_to_temp_variable = { g_rpi = geneva_pref^g_rc }
						add_to_variable = { global.geneva_push^g_rpi = g_rcp }	# the aggrieved bloc answers a flip with money and votes, not by walking: this is what makes a flipped clause flip back
					}
				}
			}
			add_to_temp_variable = { g_rc = 1 }
		}

		# --- Patience: delegations losing on the clauses they care about walk
		# --- toward the door; satisfied delegations settle back in.
		set_variable = { global.geneva_walkout_news = 0 }	# global, not temp: every_country re-scopes and a temp written in there would not survive back out
		every_country = {
			limit = {
				is_in_array = { global.geneva_delegations = THIS.id }
				NOT = { has_country_flag = geneva_walked_out }
			}
			set_temp_variable = { g_mis = 0 }
			set_temp_variable = { g_tot = 1 }
			set_temp_variable = { g_c2 = 0 }
			while_loop_effect = {
				limit = { check_variable = { g_c2 < 7 } }
				add_to_temp_variable = { g_tot = geneva_prio^g_c2 }
				set_temp_variable = { g_p2 = geneva_pref^g_c2 }
				set_temp_variable = { g_t2 = global.geneva_clause_current^g_c2 }
				if = {
					limit = { NOT = { check_variable = { g_p2 = g_t2 } } }
					set_temp_variable = { g_dist = g_p2 }
					subtract_from_temp_variable = { g_dist = g_t2 }
					if = {
						limit = { check_variable = { g_dist < 0 } }
						multiply_temp_variable = { g_dist = -1 }
					}
					set_temp_variable = { g_add = geneva_prio^g_c2 }
					if = {	# clauses 0/1/2/3/5 run along a scale, so the next notch over is a compromise rather than a defeat: this is what makes the 17th an equilibrium instead of a coin-flip. 4 and 6 are binary, where any mismatch is total
						limit = {
							check_variable = { g_dist = 1 }
							NOT = { check_variable = { g_c2 = 4 } }
							NOT = { check_variable = { g_c2 = 6 } }
						}
						divide_temp_variable = { g_add = 2 }
					}
					add_to_temp_variable = { g_mis = g_add }
				}
				add_to_temp_variable = { g_c2 = 1 }
			}
			multiply_temp_variable = { g_mis = 100 }
			divide_temp_variable = { g_mis = g_tot }
			set_temp_variable = { g_dissat = 0 }
			if = {
				limit = { check_variable = { g_mis > 54 } }
				subtract_from_variable = { geneva_patience = 9 }
				set_temp_variable = { g_dissat = 1 }
			}
			else_if = {
				limit = { check_variable = { g_mis > 34 } }
				subtract_from_variable = { geneva_patience = 4 }
				set_temp_variable = { g_dissat = 1 }
			}
			else = {
				add_to_variable = { geneva_patience = 3 }
			}
			set_temp_variable = { g_hi_mis = 0 }
			if = {
				limit = {
					check_variable = { global.geneva_severity^1 > 2 }
					NOT = { check_variable = { geneva_pref^1 = global.geneva_clause_current^1 } }
				}
				set_temp_variable = { g_hi_mis = 1 }
			}
			if = {
				limit = {
					check_variable = { global.geneva_severity^2 > 2 }
					NOT = { check_variable = { geneva_pref^2 = global.geneva_clause_current^2 } }
				}
				set_temp_variable = { g_hi_mis = 1 }
			}
			if = {
				limit = { check_variable = { g_hi_mis = 1 } }
				subtract_from_variable = { geneva_patience = 3 }
			}
			if = {	# an extreme line rammed through against a delegation that is already losing is a provocation, not merely a defeat. Option 0 is off the axis and provokes anyone who did not want it
				limit = {
					check_variable = { g_dissat = 1 }
					NOT = { check_variable = { geneva_pref^1 = global.geneva_clause_current^1 } }
					OR = {
						check_variable = { global.geneva_clause_current^1 = 0 }
						AND = {
							check_variable = { global.geneva_clause_current^1 = 1 }
							check_variable = { geneva_pref^1 > 2 }
						}
						AND = {
							check_variable = { global.geneva_clause_current^1 = 3 }
							check_variable = { geneva_pref^1 < 2 }
						}
					}
				}
				add_to_variable = { geneva_provocation = 2 }
			}
			if = {
				limit = { check_variable = { geneva_grievance > 7 } }
				subtract_from_variable = { geneva_patience = 2 }
			}
			if = {
				limit = { check_variable = { geneva_grievance > 0 } }
				subtract_from_variable = { geneva_grievance = 1 }	# last week's insult fades: only sustained churn at the table accumulates
			}
			if = {
				limit = { check_variable = { geneva_provocation > 0 } }
				subtract_from_variable = { geneva_provocation = 1 }	# pressure has to be kept up to keep a delegation at the door
			}
			set_temp_variable = { g_floor = 0 }	# human delegations always keep the right to storm out
			if = {
				limit = {
					is_ai = yes
					NOT = { has_game_rule = { rule = geneva_ai_walkout option = GENEVA_AI_WALKOUT_HISTORICAL } }
				}
				set_temp_variable = { g_floor = 15 }	# an AI does not leave over ordinary defeats at the table, however badly the clauses go
			}
			if = {
				limit = {
					is_ai = yes
					has_game_rule = { rule = geneva_ai_walkout option = DEFAULT }
					check_variable = { geneva_provocation > 2 }
				}
				set_temp_variable = { g_floor = 0 }	# actively provoked round after round: the door is open again
			}
			if = {	# clamp_variable takes literals only in this codebase, so the floor is applied by hand first
				limit = { check_variable = { geneva_patience < g_floor } }
				set_variable = { geneva_patience = g_floor }
			}
			clamp_variable = {
				var = geneva_patience
				min = 0
				max = 100
			}
			if = {
				limit = { check_variable = { geneva_patience < 1 } }
				set_country_flag = geneva_walked_out
				save_global_event_target_as = Geneva_Wrecker
				set_variable = { global.geneva_wrecker = THIS.id }
				add_to_variable = { global.geneva_walkout_news = 1 }	# reported once below, and only if the collapse check is not about to report it with .3/.4 instead
			}
		}

		# --- End conditions: two core walkouts sink it now, otherwise the
		# --- conference winds down after ten rounds and signs what it has.
		set_temp_variable = { g_core_out = 0 }
		every_country = {
			limit = {
				has_country_flag = geneva_core_delegation
				has_country_flag = geneva_walked_out
			}
			add_to_temp_variable = { g_core_out = 1 }
		}
		if = {
			limit = {
				check_variable = { global.geneva_walkout_news > 0 }
				NOT = { check_variable = { g_core_out > 1 } }
			}
			every_country = {
				limit = { is_ai = no }
				news_event = Geneva_Conference_Session.2	# one headline per round however many chairs emptied; a same-tick collapse is reported by .3 or .4 instead
			}
		}
		set_variable = { global.geneva_walkout_news = 0 }
		if = {
			limit = { check_variable = { g_core_out > 1 } }
			geneva_conf_compile_settlement = yes
		}
		else_if = {
			limit = { check_variable = { global.geneva_round > 9 } }
			geneva_conf_compile_settlement = yes
		}
	}
}

# ----------------------------------------------------------------------------
# Settlement compiler: turns the final clause states into the legacy flag
# vocabulary and hands off to the existing ending effects.
# ----------------------------------------------------------------------------
geneva_conf_compile_settlement = {
	if = {
		limit = { has_global_flag = geneva_conference_in_session }
		clr_global_flag = geneva_conference_in_session
		set_global_flag = geneva_conference_concluded
		geneva_neg_clear = yes	# lobbying still mid-timer at the close is moot
		every_country = {
			limit = { is_in_array = { global.geneva_delegations = THIS.id } }
			set_variable = { View_Geneva_Conference = 0 }
		}
		every_country = {	# the political-tab button goes with the window, as the struggle's does
			limit = { is_in_array = { array = country_politics_list value = 29 } }
			remove_from_array = { array = country_politics_list value = 29 }
		}

		set_temp_variable = { g_core_out = 0 }
		every_country = {
			limit = {
				has_country_flag = geneva_core_delegation
				has_country_flag = geneva_walked_out
			}
			add_to_temp_variable = { g_core_out = 1 }
		}

		if = {
			limit = {
				OR = {
					check_variable = { global.geneva_clause_current^0 = 0 }
					check_variable = { g_core_out > 1 }
				}
			}
			# ------------------------- COLLAPSE -------------------------
			set_variable = { global.geneva_outcome = 2 }
			# Record the refusal in the legacy vocabulary too.
			every_country = {
				limit = {
					has_country_flag = geneva_core_delegation
					has_country_flag = geneva_walked_out
				}
				if = { limit = { tag = FRA } set_country_flag = FRA_Rejected_Ceasefire }
				if = { limit = { tag = VIE } set_country_flag = VIE_Rejected_Ceasefire }
				if = { limit = { tag = VIN } set_country_flag = VIN_Rejected_Ceasefire }
			}
			if = {
				limit = { indochina_struggle_ending_failed_state_trigger = yes }
				# Nobody left with the strength to win: the depressing ending.
				every_country = {
					limit = { is_ai = no }
					news_event = Geneva_Conference_Session.4
				}
				indochina_struggle_run_failed_state_ending_effect = yes
			}
			else = {
				# The delegations went home to keep fighting: permanent war.
				every_country = {
					limit = { is_ai = no }
					news_event = Geneva_Conference_Session.3
				}
				indochina_struggle_run_never_ending_conflict_ending_effect = yes
			}
		}
		else = {
			# ------------------------- SETTLEMENT -------------------------
			set_variable = { global.geneva_outcome = 1 }
			set_global_flag = Geneva_Conference_Negotiations_Complete
			set_variable = { global.geneva_partition_line = global.geneva_clause_current^1 }
			set_variable = { global.geneva_elections_term = global.geneva_clause_current^2 }
			set_variable = { global.geneva_withdrawal_term = global.geneva_clause_current^3 }
			set_variable = { global.geneva_alignment_term = global.geneva_clause_current^4 }
			set_variable = { global.geneva_laos_cambodia_term = global.geneva_clause_current^5 }
			set_variable = { global.geneva_minorities_term = global.geneva_clause_current^6 }
			if = {
				limit = { check_variable = { global.geneva_clause_current^1 = 0 } }
				# The accords drew no partition line at all: one Vietnam, with
				# the question of who governs it left to the interim period.
				set_global_flag = geneva_unified_settlement
			}
			# Snapshot the war's final balance: the struggle scores are torn
			# down by the ending cleanup, but the all-Vietnam election that the
			# accords may schedule still needs to know who was winning.
			if = {
				limit = { country_exists = FRA }
				set_variable = { global.geneva_final_comm_score = FRA.StruggleInvolvedNationsIndochinaCommunistScore }
				set_variable = { global.geneva_final_anticomm_score = FRA.StruggleInvolvedNationsIndochinaProFranceScore }
				add_to_variable = { global.geneva_final_anticomm_score = FRA.StruggleInvolvedNationsIndochinaProIndependenceScore }
				# Freeze the trigger-context cache on the final balance: the
				# daily pulse stops once the war winds down, but the election
				# focus AI still reads geneva_vie_side_*_trigger afterwards.
				set_variable = { global.geneva_cache_comm_score = global.geneva_final_comm_score }
				set_variable = { global.geneva_cache_anticomm_score = global.geneva_final_anticomm_score }
				set_variable = { global.geneva_cache_comm_x2 = global.geneva_cache_comm_score }
				multiply_variable = { global.geneva_cache_comm_x2 = 2 }
			}

			# The signing ceremony: lists the agreed terms clause by clause
			# (the scripted-loc readers keep working after the conference).
			every_country = {
				limit = { is_ai = no }
				news_event = Geneva_Conference_Session.5
			}

			if = { limit = { country_exists = FRA } FRA = { set_country_flag = FRA_Accepted_Ceasefire } }
			if = { limit = { country_exists = VIE } VIE = { set_country_flag = VIE_Accepted_Ceasefire } }
			if = { limit = { country_exists = VIN } VIN = { set_country_flag = VIN_Accepted_Ceasefire } }

			if = {
				limit = { check_variable = { global.geneva_clause_current^2 > 1 } }
				# Reunification elections agreed: the complete-success path.
				if = { limit = { country_exists = FRA } FRA = { set_country_flag = FRA_Accepted_Reunification } }
				if = { limit = { country_exists = VIE } VIE = { set_country_flag = VIE_Accepted_Reunification } }
				if = { limit = { country_exists = VIN } VIN = { set_country_flag = VIN_Accepted_Reunification } }
			}
			else = {
				# Elections blocked or postponed indefinitely: OTL accords.
				# Each core delegation's own preference decides its flag, so
				# the aftermath correctly remembers who buried the vote.
				if = {
					limit = { country_exists = FRA }
					FRA = {
						if = {
							limit = { check_variable = { geneva_pref^2 > 1 } }
							set_country_flag = FRA_Accepted_Reunification
						}
						else = {
							set_country_flag = FRA_Rejected_Reunification
						}
					}
				}
				if = {
					limit = { country_exists = VIN }
					VIN = {
						if = {
							limit = { check_variable = { geneva_pref^2 > 1 } }
							set_country_flag = VIN_Accepted_Reunification
						}
						else = {
							set_country_flag = VIN_Rejected_Reunification
						}
					}
				}
				if = {
					limit = { country_exists = VIE }
					VIE = {
						if = {
							limit = {
								check_variable = { geneva_pref^2 > 1 }
								country_exists = USA
								USA = { has_completed_focus = USA_50s_Withhold_American_Assent }	# the table-stance focus; the block focus now sits after the accords and cannot be read here
							}
							# Saigon wanted the vote; Washington killed it.
							set_country_flag = VIE_Rejected_Reunification_USA_Intervention
							USA = { set_country_flag = USA_Tampered_With_VIE }
						}
						else_if = {
							limit = { check_variable = { geneva_pref^2 > 1 } }
							set_country_flag = VIE_Accepted_Reunification
						}
						else = {
							set_country_flag = VIE_Rejected_Reunification
						}
					}
				}
			}

			indochina_struggle_run_geneva_ending_effect = yes

			# ---------------- The interim arrangements ----------------
			# A settlement that schedules a reunification vote (or that drew no
			# partition line at all) has not answered the Vietnamese question:
			# it has set a date for it. The hidden Session.21 arbiter resolves
			# the election when it falls due; if the VIN focus-tree election
			# crisis settles the matter first, the arbiter stands down.
			if = {
				limit = {
					has_global_flag = Geneva_Conference	# set by indochina_struggle_unlock_geneva_conference in the settlement branch above and never cleared. The phase it sets alongside is 11, but the ending cleanup inside that same call overwrites it with 100 before control returns here
					country_exists = VIE
					country_exists = VIN
					OR = {
						has_global_flag = geneva_unified_settlement
						check_variable = { global.geneva_elections_term > 1 }
					}
				}
				set_global_flag = geneva_interim_government_period
				every_country = {
					limit = { is_ai = no }
					news_event = { id = Geneva_Conference_Session.20 days = 2 }
				}
				# Session.26 is the campaign trail: both capitals read the same
				# polls (the war's final balance) before the arbiter resolves
				# the vote, so nobody is blindsided by the result.
				# Session.27 opens the hidden campaign window ahead of the campaign
				# trail. Fired at FRA, which cannot vanish.
				if = {
					limit = { check_variable = { global.geneva_elections_term = 3 } }
					VIE = { country_event = { id = Geneva_Conference_Session.26 days = 95 random_days = 15 } }
					VIN = { country_event = { id = Geneva_Conference_Session.26 days = 95 random_days = 15 } }
					VIN = { country_event = { id = Geneva_Conference_Session.21 days = 150 random_days = 30 } }
					if = {
						limit = { country_exists = FRA }
						FRA = { country_event = { id = Geneva_Conference_Session.27 days = 85 } }
					}
					else = {
						VIN = { country_event = { id = Geneva_Conference_Session.27 days = 85 } }
					}
				}
				else_if = {
					limit = { check_variable = { global.geneva_elections_term = 2 } }
					VIE = { country_event = { id = Geneva_Conference_Session.26 days = 420 random_days = 30 } }
					VIN = { country_event = { id = Geneva_Conference_Session.26 days = 420 random_days = 30 } }
					VIN = { country_event = { id = Geneva_Conference_Session.21 days = 540 random_days = 90 } }
					if = {
						limit = { country_exists = FRA }
						FRA = { country_event = { id = Geneva_Conference_Session.27 days = 400 } }
					}
					else = {
						VIN = { country_event = { id = Geneva_Conference_Session.27 days = 400 } }
					}
				}
				else = {
					# Unified settlement with the vote postponed: the interim
					# council still has to produce a government eventually.
					VIE = { country_event = { id = Geneva_Conference_Session.26 days = 220 random_days = 30 } }
					VIN = { country_event = { id = Geneva_Conference_Session.26 days = 220 random_days = 30 } }
					VIN = { country_event = { id = Geneva_Conference_Session.21 days = 300 random_days = 90 } }
					if = {
						limit = { country_exists = FRA }
						FRA = { country_event = { id = Geneva_Conference_Session.27 days = 205 } }
					}
					else = {
						VIN = { country_event = { id = Geneva_Conference_Session.27 days = 205 } }
					}
				}
			}
		}
	}
}

# ----------------------------------------------------------------------------
# Headless resolution for AI-only games, called by the Indochina wrap-up
# timer. Convenes the conference if needed and plays it out immediately.
# ----------------------------------------------------------------------------
geneva_conf_auto_resolve = {
	if = {
		limit = {
			NOT = { has_global_flag = Indochina_War_Over }
			NOT = {	# an announcement is out and its six-day opener has not landed: let the interactive pipeline run instead of resolving on top of it
				AND = {
					has_global_flag = geneva_conference_announced
					NOT = { has_global_flag = geneva_conference_in_session }
					NOT = { has_global_flag = geneva_conference_concluded }
				}
			}
		}
		set_temp_variable = { g_geneva_headless = 1 }	# AI stakes bank instantly; the 3-day pipeline never stalls the loop
		geneva_neg_flush_all = yes	# a human-session save being auto-resolved keeps its in-flight stakes
		if = {
			limit = {
				NOT = { has_global_flag = geneva_conference_in_session }
				NOT = { has_global_flag = geneva_conference_concluded }
			}
			geneva_conf_initialize = yes
		}
		set_temp_variable = { g_ar = 0 }
		while_loop_effect = {
			limit = {
				check_variable = { g_ar < 12 }
				has_global_flag = geneva_conference_in_session
			}
			geneva_conf_round_tick = yes
			add_to_temp_variable = { g_ar = 1 }
		}
		if = {
			limit = { has_global_flag = geneva_conference_in_session }
			geneva_conf_compile_settlement = yes
		}
	}
}

# ----------------------------------------------------------------------------
# Table actions (player-facing, phase 3). All run in the acting delegation's
# country scope from GUI clicks. Every effect re-checks its own guards so a
# stale click can never cheat; the GUI's click_enabled triggers mirror them.
# Selection state lives on the acting country: geneva_sel_clause,
# geneva_sel_option (for Champion), geneva_sel_target (a country id set by
# the Select buttons in the delegation list).
# ----------------------------------------------------------------------------

# The shared guard (seated, at the table, session live, actions left) lives
# in common/scripted_triggers/IC_struggle_triggers.txt as
# geneva_action_possible_trigger, shared with the GUI's click_enabled checks.

# Champion: stake weight behind your selected option on the selected clause.
# Sets your own preference and adds a large one-round push to the next vote.
geneva_action_champion = {
	set_temp_variable = { g_ch_cost = global.geneva_severity^geneva_sel_clause }
	multiply_temp_variable = { g_ch_cost = 5 }	# 5/10/15 weight by clause severity
	set_temp_variable = { g_ch_cost_x2 = g_ch_cost }
	multiply_temp_variable = { g_ch_cost_x2 = 2 }
	if = {
		limit = {
			geneva_action_possible_trigger = yes
			has_political_power > 24
			check_variable = { geneva_weight > g_ch_cost_x2 }
			check_variable = { geneva_neg_mine^geneva_sel_clause = 0 }
		}
		add_political_power = -25
		add_to_variable = { geneva_actions_used = 1 }
		subtract_from_variable = { geneva_weight = g_ch_cost }
		set_variable = { geneva_pref^geneva_sel_clause = geneva_sel_option }
		if = {
			limit = { check_variable = { geneva_prio^geneva_sel_clause < 8 } }
			set_variable = { geneva_prio^geneva_sel_clause = 8 }
		}
		set_temp_variable = { g_neg_clause = geneva_sel_clause }
		set_temp_variable = { g_neg_option = geneva_sel_option }
		set_temp_variable = { g_neg_amount = geneva_weight }
		divide_temp_variable = { g_neg_amount = 2 }
		add_to_temp_variable = { g_neg_amount = 100 }
		geneva_neg_start = yes
	}
}

# Pressure: strong-arm the selected delegation into adopting your position
# on the selected clause. Costs half their weight from your pool; they
# resent it (patience) and refuse further pressure for three rounds.
geneva_action_pressure = {
	if = {
		limit = {
			geneva_action_possible_trigger = yes
			has_political_power > 24
			check_variable = { geneva_weight > 25 }
			NOT = { check_variable = { geneva_sel_target = THIS.id } }
		}
		set_temp_variable = { g_tid = geneva_sel_target }
		set_temp_variable = { g_c = geneva_sel_clause }
		set_temp_variable = { g_myp = geneva_pref^geneva_sel_clause }
		set_temp_variable = { g_cost = -1 }
		every_country = {
			limit = {
				check_variable = { THIS.id = g_tid }
				is_in_array = { global.geneva_delegations = THIS.id }
				NOT = { has_country_flag = geneva_walked_out }
				check_variable = { geneva_pressured_until < global.geneva_round }
			}
			set_temp_variable = { g_cost = geneva_weight }
			divide_temp_variable = { g_cost = 2 }
			if = {
				limit = { check_variable = { g_cost < 10 } }
				set_temp_variable = { g_cost = 10 }
			}
			set_variable = { geneva_pref^g_c = g_myp }
			subtract_from_variable = { geneva_patience = 5 }
			add_to_variable = { geneva_provocation = 3 }	# being strong-armed is the provocation that reopens the door for an AI delegation
			set_variable = { geneva_pressured_until = global.geneva_round }
			add_to_variable = { geneva_pressured_until = 2 }
		}
		# Only pay if a valid target was actually found and pressured.
		if = {
			limit = { check_variable = { g_cost > -1 } }
			add_political_power = -25
			add_to_variable = { geneva_actions_used = 1 }
			subtract_from_variable = { geneva_weight = g_cost }
			clamp_variable = {
				var = geneva_weight
				min = 5
				max = 10000
			}
		}
	}
}

# Concede: publicly drop your opposition on the selected clause. You adopt
# whatever is currently tabled, recover weight, and every delegation that
# wanted that outcome warms to the conference.
geneva_action_concede = {
	if = {
		limit = { geneva_action_possible_trigger = yes }
		add_to_variable = { geneva_actions_used = 1 }
		set_temp_variable = { g_c = geneva_sel_clause }
		set_temp_variable = { g_t = global.geneva_clause_current^g_c }
		set_variable = { geneva_pref^g_c = g_t }
		set_variable = { geneva_prio^g_c = 1 }
		add_to_variable = { geneva_weight = 15 }
		every_country = {
			limit = {
				is_in_array = { global.geneva_delegations = THIS.id }
				NOT = { has_country_flag = geneva_walked_out }
				check_variable = { geneva_pref^g_c = g_t }
			}
			add_to_variable = { geneva_patience = 8 }
			clamp_variable = {
				var = geneva_patience
				min = 0
				max = 100
			}
		}
	}
}

# Back-channel: quiet corridor talks between Saigon and Hanoi. Builds the
# rapprochement track; at 100+ both delegations soften their formal
# positions (elections become thinkable, the 17th acceptable).
geneva_action_backchannel = {
	if = {
		limit = {
			geneva_action_possible_trigger = yes
			has_political_power > 49
			NOT = { check_variable = { geneva_backchannel_next > global.geneva_round } }
			OR = {
				AND = {
					tag = VIE
					country_exists = VIN
					check_variable = { geneva_sel_target = VIN.id }
				}
				AND = {
					tag = VIN
					country_exists = VIE
					check_variable = { geneva_sel_target = VIE.id }
				}
			}
		}
		add_political_power = -50
		add_to_variable = { geneva_actions_used = 1 }
		set_variable = { geneva_backchannel_next = global.geneva_round }
		add_to_variable = { geneva_backchannel_next = 2 }
		set_temp_variable = { geneva_lev_amount = 25 }
		geneva_add_vie_vin_rapprochement = yes
		if = {
			limit = {
				tag = VIE
				country_exists = VIN
			}
			VIN = { country_event = { id = Geneva_Conference_Session.12 days = 1 } }
		}
		else_if = {
			limit = {
				tag = VIN
				country_exists = VIE
			}
			VIE = { country_event = { id = Geneva_Conference_Session.12 days = 1 } }
		}
		if = {
			limit = { check_variable = { global.geneva_rapprochement_vie_vin > 99 } }
			if = { limit = { country_exists = VIE } VIE = { geneva_conf_refresh_prefs = yes } }
			if = { limit = { country_exists = VIN } VIN = { geneva_conf_refresh_prefs = yes } }
		}
	}
}

# Threaten walkout: put your chair on the line. A huge push behind your own
# position on the selected clause, paid for with your delegation's patience.
geneva_action_threaten = {
	if = {
		limit = {
			geneva_action_possible_trigger = yes
			check_variable = { geneva_patience > 25 }
			NOT = { check_variable = { geneva_threat_next > global.geneva_round } }
			check_variable = { geneva_neg_mine^geneva_sel_clause = 0 }
		}
		add_to_variable = { geneva_actions_used = 1 }
		set_variable = { geneva_threat_next = global.geneva_round }
		add_to_variable = { geneva_threat_next = 3 }
		set_temp_variable = { g_threat_cost = 5 }
		set_temp_variable = { g_threat_sev = global.geneva_severity^geneva_sel_clause }
		multiply_temp_variable = { g_threat_sev = 3 }
		add_to_temp_variable = { g_threat_cost = g_threat_sev }
		if = {
			limit = { check_variable = { g_threat_cost < 8 } }
			set_temp_variable = { g_threat_cost = 11 }	# severity array missing on stale saves: legacy cost
		}
		subtract_from_variable = { geneva_patience = g_threat_cost }	# 8/11/14 by clause severity: roughly two bad rounds, not a quarter of the delegation's whole stamina
		set_temp_variable = { g_neg_clause = geneva_sel_clause }
		set_temp_variable = { g_neg_option = geneva_pref^geneva_sel_clause }
		set_temp_variable = { g_neg_amount = geneva_weight }
		add_to_temp_variable = { g_neg_amount = 250 }
		geneva_neg_start = yes
	}
}

# Force the vote: a core belligerent ends the conference on the current
# terms. Available once the talks have matured (round 5+).
geneva_action_force_vote = {
	if = {
		limit = {
			has_global_flag = geneva_conference_in_session
			has_country_flag = geneva_core_delegation
			NOT = { has_country_flag = geneva_walked_out }
			has_political_power > 49
			check_variable = { global.geneva_round > 4 }
		}
		add_political_power = -50
		geneva_neg_flush_all = yes	# in-flight stakes land in the final vote instead of being voided
		set_variable = { global.geneva_round = 9 }
		geneva_conf_round_tick = yes	# one genuine final vote (tick raises the round past the cap and compiles)
	}
}

# Lean on Saigon: the covert successor to the old USA tamper event. Kills
# Saigon's appetite for elections. If the two Vietnams have a warm
# back-channel (rapprochement 50+), Hanoi hears about it and the scandal
# breaks publicly.
geneva_action_lean_on_saigon = {
	if = {
		limit = {
			geneva_action_possible_trigger = yes
			tag = USA
			NOT = { has_country_flag = USA_geneva_leaned_on_saigon }
			has_political_power > 49
			country_exists = VIE
			VIE = {
				is_in_array = { global.geneva_delegations = THIS.id }
				NOT = { has_country_flag = geneva_walked_out }
			}
		}
		add_political_power = -50
		add_to_variable = { geneva_actions_used = 1 }
		set_country_flag = USA_geneva_leaned_on_saigon
		set_country_flag = USA_Tampered_With_VIE
		VIE = {
			set_variable = { geneva_pref^2 = 0 }
			set_variable = { geneva_prio^2 = 8 }
			add_to_variable = { geneva_provocation = 4 }
		}
		if = {
			limit = { check_variable = { global.geneva_rapprochement_vie_vin > 49 } }
			# Saigon's channel to Hanoi leaks the pressure: public scandal.
			every_country = {
				limit = { is_ai = no }
				news_event = Geneva_Conference_Session.13
			}
			subtract_from_variable = { geneva_weight = 25 }
			clamp_variable = {
				var = geneva_weight
				min = 5
				max = 10000
			}
			if = {
				limit = { country_exists = VIN }
				VIN = {
					subtract_from_variable = { geneva_patience = 10 }
					add_to_variable = { geneva_provocation = 2 }
				}
			}
			if = {
				limit = { country_exists = SOV }
				SOV = {
					subtract_from_variable = { geneva_patience = 10 }
					add_to_variable = { geneva_provocation = 2 }
				}
			}
		}
	}
}

# ----------------------------------------------------------------------------
# GUI selection helpers (player scope). Option counts per clause:
# c0=3, c1=4, c2=4, c3=3, c4=2, c5=3, c6=2.
# ----------------------------------------------------------------------------
geneva_ui_cycle_clause = {
	add_to_variable = { geneva_sel_clause = 1 }
	if = {
		limit = { check_variable = { geneva_sel_clause > 6 } }
		set_variable = { geneva_sel_clause = 0 }
	}
	set_variable = { geneva_sel_option = 0 }
}
geneva_ui_cycle_option = {
	add_to_variable = { geneva_sel_option = 1 }
	if = {
		limit = {
			check_variable = { geneva_sel_clause = 1 }
		}
		if = {
			limit = { check_variable = { geneva_sel_option > 3 } }
			set_variable = { geneva_sel_option = 0 }
		}
	}
	else_if = {
		limit = {
			check_variable = { geneva_sel_clause = 2 }
		}
		if = {
			limit = { check_variable = { geneva_sel_option > 3 } }
			set_variable = { geneva_sel_option = 0 }
		}
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { geneva_sel_clause = 0 }
				check_variable = { geneva_sel_clause = 3 }
				check_variable = { geneva_sel_clause = 5 }
			}
		}
		if = {
			limit = { check_variable = { geneva_sel_option > 2 } }
			set_variable = { geneva_sel_option = 0 }
		}
	}
	else = {
		if = {
			limit = { check_variable = { geneva_sel_option > 1 } }
			set_variable = { geneva_sel_option = 0 }
		}
	}
}
