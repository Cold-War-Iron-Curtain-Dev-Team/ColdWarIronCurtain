religion_proselytization_cost = {

    # Base cost from HDI
    if = {
        limit = { check_variable = { State_HDI < 0.300 } }
        set_temp_variable = { prosyletization_cost_pp = 10 }
        set_temp_variable = { prosyletization_cost_money = 0.5 }
    }
    else_if = {
        limit = { check_variable = { State_HDI > 0.299 } check_variable = { State_HDI < 0.450 } }
        set_temp_variable = { prosyletization_cost_pp = 15 }
        set_temp_variable = { prosyletization_cost_money = 0.7 }
    }
    else_if = {
        limit = { check_variable = { State_HDI > 0.449 } check_variable = { State_HDI < 0.600 } }
        set_temp_variable = { prosyletization_cost_pp = 20 }
        set_temp_variable = { prosyletization_cost_money = 0.9 }
    }
    else_if = {
        limit = { check_variable = { State_HDI > 0.599 } check_variable = { State_HDI < 0.700 } }
        set_temp_variable = { prosyletization_cost_pp = 25 }
        set_temp_variable = { prosyletization_cost_money = 1 }
    }
    else_if = {
        limit = { check_variable = { State_HDI > 0.699 } check_variable = { State_HDI < 0.800 } }
        set_temp_variable = { prosyletization_cost_pp = 30 }
        set_temp_variable = { prosyletization_cost_money = 1.2 }
    }
    else_if = {
        limit = { check_variable = { State_HDI > 0.799 } check_variable = { State_HDI < 0.900 } }
        set_temp_variable = { prosyletization_cost_pp = 35 }
        set_temp_variable = { prosyletization_cost_money = 1.5 }
    }
    else_if = {
        limit = { check_variable = { State_HDI > 0.899 } check_variable = { State_HDI < 0.970 } }
        set_temp_variable = { prosyletization_cost_pp = 40 }
        set_temp_variable = { prosyletization_cost_money = 1.75 }
    }
    else_if = {
        limit = { check_variable = { State_HDI > 0.969 } }
        set_temp_variable = { prosyletization_cost_pp = 45 }
        set_temp_variable = { prosyletization_cost_money = 2.0 }
    }

    # Adjust for population level
    if = { limit = { pop_level_1_check = yes } multiply_temp_variable = { prosyletization_cost_pp = 1.0 } multiply_temp_variable = { prosyletization_cost_money = 1.0 } }
    else_if = { limit = { pop_level_2_check = yes } multiply_temp_variable = { prosyletization_cost_pp = 1.2 } multiply_temp_variable = { prosyletization_cost_money = 1.25 } }
    else_if = { limit = { pop_level_3_check = yes } multiply_temp_variable = { prosyletization_cost_pp = 1.4 } multiply_temp_variable = { prosyletization_cost_money = 1.5 } }
    else_if = { limit = { pop_level_4_check = yes } multiply_temp_variable = { prosyletization_cost_pp = 1.5 } multiply_temp_variable = { prosyletization_cost_money = 1.75 } }
    else_if = { limit = { pop_level_5_check = yes } multiply_temp_variable = { prosyletization_cost_pp = 1.7 } multiply_temp_variable = { prosyletization_cost_money = 2 } }
    else_if = { limit = { pop_level_7_check = yes } multiply_temp_variable = { prosyletization_cost_pp = 2 } multiply_temp_variable = { prosyletization_cost_money = 2.5 } }
    else_if = { limit = { pop_level_10_check = yes } multiply_temp_variable = { prosyletization_cost_pp = 2.25 } multiply_temp_variable = { prosyletization_cost_money = 3 } }
    else_if = { limit = { pop_level_15_check = yes } multiply_temp_variable = { prosyletization_cost_pp = 2.5 } multiply_temp_variable = { prosyletization_cost_money = 4 } }
    else_if = { limit = { pop_level_20_check = yes } multiply_temp_variable = { prosyletization_cost_pp = 5 } multiply_temp_variable = { prosyletization_cost_money = 5 } }
    else = { }
    # Final application (deduct resources)
    hidden_effect = {
        set_variable = { ROOT.pros_cost_pp = temp.prosyletization_cost_pp }
        set_variable = { ROOT.pros_cost_money = temp.prosyletization_cost_money }
        #add_political_power = prosy_cost_pp
        #subtract_from_variable = { currentMoney = prosy_cost_money }
    }
    
}
