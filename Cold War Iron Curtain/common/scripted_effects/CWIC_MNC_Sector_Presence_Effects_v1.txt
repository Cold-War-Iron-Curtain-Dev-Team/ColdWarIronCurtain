cwic_mnc_apply_elite_capture_effects_country = {
	# Call from COUNTRY monthly AFTER presence tiers + instability/corruption/stability updates.

	# 1) Roll up overall footprint
	set_variable = { CWIC_mnc_presence_tier_total = 0 }
	add_to_variable = { CWIC_mnc_presence_tier_total = CWIC_mnc_presence_tier_hydrocarbons }
	add_to_variable = { CWIC_mnc_presence_tier_total = CWIC_mnc_presence_tier_mining }
	add_to_variable = { CWIC_mnc_presence_tier_total = CWIC_mnc_presence_tier_agri }
	add_to_variable = { CWIC_mnc_presence_tier_total = CWIC_mnc_presence_tier_manufacturing }
	add_to_variable = { CWIC_mnc_presence_tier_total = CWIC_mnc_presence_tier_electronics }

	# Convert tier_total into a capture_tier (0..5)
	set_variable = { CWIC_mnc_capture_tier = 0 }
	if = { limit = { check_variable = { CWIC_mnc_presence_tier_total > 2 } } set_variable = { CWIC_mnc_capture_tier = 1 } }
	if = { limit = { check_variable = { CWIC_mnc_presence_tier_total > 5 } } set_variable = { CWIC_mnc_capture_tier = 2 } }
	if = { limit = { check_variable = { CWIC_mnc_presence_tier_total > 8 } } set_variable = { CWIC_mnc_capture_tier = 3 } }
	if = { limit = { check_variable = { CWIC_mnc_presence_tier_total > 11 } } set_variable = { CWIC_mnc_capture_tier = 4 } }
	if = { limit = { check_variable = { CWIC_mnc_presence_tier_total > 14 } } set_variable = { CWIC_mnc_capture_tier = 5 } }

	# 2) Governance fragility gate
	set_temp_variable = { fragile = 0 }
	if = { limit = { check_variable = { CWIC_instability_index > 1.5 } } set_temp_variable = { fragile = 1 } }
	if = { limit = { check_variable = { CorruptionLevel > 0.45 } } set_temp_variable = { fragile = 1 } }
	if = { limit = { check_variable = { CWIC_stability_percent < 45 } } set_temp_variable = { fragile = 1 } }

	# 3) Apply / remove modifiers
	remove_ideas = cwic_mnc_elite_capture_t1
	remove_ideas = cwic_mnc_elite_capture_t2
	remove_ideas = cwic_mnc_elite_capture_t3
	remove_ideas = cwic_mnc_elite_capture_t4
	remove_ideas = cwic_mnc_elite_capture_t5

	if = {
		limit = {
			check_variable = { CWIC_mnc_capture_tier > 0 }
			check_variable = { fragile = 1 }
		}
		if = { limit = { check_variable = { CWIC_mnc_capture_tier = 1 } } add_ideas = cwic_mnc_elite_capture_t1 }
		else_if = { limit = { check_variable = { CWIC_mnc_capture_tier = 2 } } add_ideas = cwic_mnc_elite_capture_t2 }
		else_if = { limit = { check_variable = { CWIC_mnc_capture_tier = 3 } } add_ideas = cwic_mnc_elite_capture_t3 }
		else_if = { limit = { check_variable = { CWIC_mnc_capture_tier = 4 } } add_ideas = cwic_mnc_elite_capture_t4 }
		else_if = { limit = { check_variable = { CWIC_mnc_capture_tier = 5 } } add_ideas = cwic_mnc_elite_capture_t5 }
		else = { }
	}
}
