cwic_mnc_apply_hydrocarbons_presence_idea_country = {
	if = {
		limit = {
			NOT = {
				check_variable = {
					var = CWIC_mnc_presence_tier_hydrocarbons
					value = CWIC_mnc_presence_applied_tier_hydrocarbons
					compare = equals
				}
			}
		}
		if = { limit = { check_variable = { CWIC_mnc_presence_applied_tier_hydrocarbons = 1 } } remove_ideas = cwic_mnc_hydrocarbons_presence_t1 }
		else_if = { limit = { check_variable = { CWIC_mnc_presence_applied_tier_hydrocarbons = 2 } } remove_ideas = cwic_mnc_hydrocarbons_presence_t2 }

# ============================================================
# CWIC MNC: Elite Capture / Governance-Conditional Side Effects (v1)
# Sector-based, not per-company. Intended to only bite fragile states.
# ============================================================

cwic_mnc_apply_elite_capture_effects_country = {
	# Call from COUNTRY monthly AFTER presence tiers + instability/corruption/stability updates.

	# 1) Roll up overall footprint
	set_variable = {
		CWIC_mnc_presence_tier_total = {
			value = CWIC_mnc_presence_tier_hydrocarbons
			add = CWIC_mnc_presence_tier_mining
			add = CWIC_mnc_presence_tier_agri
			add = CWIC_mnc_presence_tier_manufacturing
			add = CWIC_mnc_presence_tier_electronics
		}
	}

	# Convert tier_total into a capture_tier (0..5)
	set_variable = {
		CWIC_mnc_capture_tier = {
			value = 0
			add = { value = CWIC_mnc_presence_tier_total greater_than = 2 }
			add = { value = CWIC_mnc_presence_tier_total greater_than = 5 }
			add = { value = CWIC_mnc_presence_tier_total greater_than = 8 }
			add = { value = CWIC_mnc_presence_tier_total greater_than = 11 }
			add = { value = CWIC_mnc_presence_tier_total greater_than = 14 }
		}
	}

	# 2) Governance fragility gate
	set_temp_variable = {
		fragile = {
			value = 0
			add = { value = CWIC_instability_index greater_than = 1.5 }
			add = { value = CorruptionLevel greater_than = 0.45 }
			add = { value = CWIC_stability_percent less_than = 45 }
			clamp = { min = 0 max = 1 }
		}
	}

	# 3) Apply / remove modifiers only when the visible tier changes
	set_variable = {
		CWIC_mnc_elite_capture_target_tier = {
			value = CWIC_mnc_capture_tier
			multiply = fragile
		}
	}
	cwic_mnc_apply_elite_capture_idea_country = yes
}
