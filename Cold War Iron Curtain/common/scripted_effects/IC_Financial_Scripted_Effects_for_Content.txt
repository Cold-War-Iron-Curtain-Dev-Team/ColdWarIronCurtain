        bank_name_prompt = 1
    }
    save_global_event_target_as = this_loan_country
    clear_array = purchase_length
    add_to_array = {
        purchase_length = 5
    }
    add_to_array = {
        purchase_length = 7
    }
    add_to_array = {
        purchase_length = 10
    }
    clear_array = currentInterestRateCalc
    add_to_array = {
        currentInterestRateCalc = ReserveCurrentInterestRate
    }
    add_to_array = {
        currentInterestRateCalc = ReserveCurrentInterestRate
    }
    add_to_array = {
        currentInterestRateCalc = ReserveCurrentInterestRate
    }
    set_variable = {
        max_index = 2
    }
    set_variable = {
        current_index = 0
    }
    automatic_loan_setting = yes
}

# ---------- LITERATE ----------
nep_ed_increase_literate = {
	add_to_variable = { literate_educated = literate_change }
	clamp_variable = { var = literate_educated min = 0 max = 100 }
	custom_effect_tooltip = NEP_Ed_Literate_Inc_TT
}
nep_ed_decrease_literate = {
	subtract_from_variable = { literate_educated = literate_change }
	clamp_variable = { var = literate_educated min = 0 max = 100 }
	custom_effect_tooltip = NEP_Ed_Literate_Dec_TT
}
