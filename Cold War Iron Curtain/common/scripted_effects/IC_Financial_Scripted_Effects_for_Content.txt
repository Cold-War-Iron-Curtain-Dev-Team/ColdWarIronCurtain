cwic_change_money_gdp_share = {
	set_temp_variable = { money_change_temp = grossdomesticproduct }
	multiply_temp_variable = { money_change_temp = money_change_gdp_share }
	if = {
		limit = {
			OR = {
				check_variable = { money_change_temp > 1 }
				check_variable = { money_change_temp < -1 }
			}
		}
		multiply_temp_variable = { money_change_temp = 100 }
		round_temp_variable = money_change_temp
		divide_temp_variable = { money_change_temp = 100 }
	}
	else = {
		multiply_temp_variable = { money_change_temp = 1000 }
		round_temp_variable = money_change_temp
		divide_temp_variable = { money_change_temp = 1000 }
	}
	cwic_change_money = yes
}

cwic_take_loan = {
	if = {
		limit = { check_variable = { loan_gdp_share_temp > 0 } }
		set_temp_variable = { purchase_amount_temp = grossdomesticproduct }
		multiply_temp_variable = { purchase_amount_temp = loan_gdp_share_temp }
	}
	if = {
		limit = { check_variable = { purchase_amount_temp > 0 } }
		if = {
			limit = { check_variable = { lender_temp = 11 } }
			set_temp_variable = { cwic_loan_total = loan_interest_temp }
			divide_temp_variable = { cwic_loan_total = 100 }
			multiply_temp_variable = { cwic_loan_total = loan_years_temp }
			add_to_temp_variable = { cwic_loan_total = 1 }
			multiply_temp_variable = { cwic_loan_total = purchase_amount_temp }
			set_temp_variable = { cwic_loan_payment = loan_years_temp }
			multiply_temp_variable = { cwic_loan_payment = 12 }
			clamp_temp_variable = { var = cwic_loan_payment min = 1 }
			set_temp_variable = { cwic_loan_payment_monthly = cwic_loan_total }
			divide_temp_variable = { cwic_loan_payment_monthly = cwic_loan_payment }
			set_temp_variable = { cwic_loan_interest_frac = loan_interest_temp }
			divide_temp_variable = { cwic_loan_interest_frac = 100 }
			add_to_array = { debtTotalsArray = cwic_loan_total }
			add_to_array = { debtOwnerArray = global_event_target:cwic_loan_lender }
			add_to_array = { debtOwnerDomesticStatusArray = 11 }
			add_to_array = { debtInterestArray = cwic_loan_interest_frac }
			add_to_array = { debtPaymentArray = cwic_loan_payment_monthly }
			add_to_variable = { currentDebtPayments = cwic_loan_payment_monthly }
			add_to_variable = { currentDebt = cwic_loan_total }
			global_event_target:cwic_loan_lender = {
				if = {
					limit = { NOT = { tag = GER } }
					subtract_from_variable = { currentMoney = purchase_amount_temp }
				}
			}
		}
		else = {
			add_loans_effect = yes
		}
		if = {
			limit = { NOT = { check_variable = { loan_no_payout_temp = 1 } } }
			add_to_variable = { currentMoney = purchase_amount_temp }
		}
	}
}

cwic_forgive_debt = {
	clamp_temp_variable = { var = debt_forgive_share_temp min = 0 max = 1 }
	set_temp_variable = { cwic_forgive_cost = 0 }
	for_each_loop = {
		array = debtTotalsArray
		index = i
		if = {
			limit = {
				OR = {
					NOT = { check_variable = { debt_forgive_lender_temp > 0 } }
					check_variable = { debtOwnerArray^i = debt_forgive_lender_temp }
				}
			}
			set_temp_variable = { cwic_forgive_total = debtTotalsArray^i }
			multiply_temp_variable = { cwic_forgive_total = debt_forgive_share_temp }
			set_temp_variable = { cwic_forgive_payment = debtPaymentArray^i }
			multiply_temp_variable = { cwic_forgive_payment = debt_forgive_share_temp }
			subtract_from_variable = { debtTotalsArray^i = cwic_forgive_total }
			subtract_from_variable = { debtPaymentArray^i = cwic_forgive_payment }
			subtract_from_variable = { currentDebt = cwic_forgive_total }
			subtract_from_variable = { currentDebtPayments = cwic_forgive_payment }
			add_to_temp_variable = { cwic_forgive_cost = cwic_forgive_total }
		}
	}
	clamp_variable = { var = currentDebt min = 0 }
	clamp_variable = { var = currentDebtPayments min = 0 }
	if = {
		limit = { check_variable = { debt_forgive_payer_temp = 1 } }
		global_event_target:cwic_debt_payer = {
			subtract_from_variable = { currentMoney = cwic_forgive_cost }
		}
	}
}
