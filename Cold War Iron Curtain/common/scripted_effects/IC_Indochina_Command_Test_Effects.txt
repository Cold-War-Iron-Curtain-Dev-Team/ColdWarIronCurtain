# French Indochina command QA
# Console use from any tag: e test_indochina_command_status
# Destructive early-setup repair: e test_indochina_command_repair

test_indochina_command_status = {
	if = { limit = { NOT = { country_exists = FRE } } log = "INDOCHINA COMMAND: FAIL - CEFEO does not exist" }
	if = { limit = { country_exists = FRE FRE = { NOT = { is_faction_leader = yes } } } log = "INDOCHINA COMMAND: FAIL - CEFEO is not faction leader" }
	if = { limit = { country_exists = FRE FRE = { NOT = { has_country_flag = FRE_French_Indochina_Command_Established } } } log = "INDOCHINA COMMAND: FAIL - establishment flag missing" }
	if = { limit = { country_exists = FRE FRE = { NOT = { has_idea = French_Indochina_Faction } } } log = "INDOCHINA COMMAND: FAIL - CEFEO command identity missing" }
	if = { limit = { country_exists = VIE VIE = { NOT = { has_idea = State_of_Vietnam_Faction } } } log = "INDOCHINA COMMAND: FAIL - State of Vietnam no-call identity missing" }
	if = { limit = { country_exists = VIN VIN = { NOT = { has_idea = Viet_Minh_Faction } } } log = "INDOCHINA COMMAND: FAIL - Hanoi no-call identity missing" }
	if = { limit = { country_exists = NLF NLF = { NOT = { has_idea = Viet_Minh_Faction } } } log = "INDOCHINA COMMAND: FAIL - southern resistance no-call identity missing" }

	if = { limit = { country_exists = FRE country_exists = VIE VIE = { NOT = { is_in_faction_with = FRE } } } log = "INDOCHINA COMMAND: FAIL - State of Vietnam is outside French Indochina" }
	if = { limit = { country_exists = FRE country_exists = CCC CCC = { NOT = { is_in_faction_with = FRE } } } log = "INDOCHINA COMMAND: FAIL - Cochinchina is outside French Indochina" }
	if = { limit = { country_exists = FRE country_exists = FUL FUL = { NOT = { is_in_faction_with = FRE } } } log = "INDOCHINA COMMAND: FAIL - Montagnard crown domain is outside French Indochina" }
	if = { limit = { country_exists = FRE country_exists = TAI TAI = { NOT = { is_in_faction_with = FRE } } } log = "INDOCHINA COMMAND: FAIL - Tai Federation is outside French Indochina" }
	if = { limit = { country_exists = FRE country_exists = NUN NUN = { NOT = { is_in_faction_with = FRE } } } log = "INDOCHINA COMMAND: FAIL - Nung territory is outside French Indochina" }
	if = { limit = { country_exists = FRE country_exists = TAM TAM = { NOT = { is_in_faction_with = FRE } } } log = "INDOCHINA COMMAND: FAIL - Muong Federation is outside French Indochina" }
	if = { limit = { country_exists = FRE country_exists = THO THO = { NOT = { is_in_faction_with = FRE } } } log = "INDOCHINA COMMAND: FAIL - Tho territory is outside French Indochina" }

	if = { limit = { country_exists = TAI TAI = { NOT = { is_subject_of = FRE } } } log = "INDOCHINA COMMAND: FAIL - Tai Federation is not in CEFEO custody" }
	if = { limit = { country_exists = NUN NUN = { NOT = { is_subject_of = FRE } } } log = "INDOCHINA COMMAND: FAIL - Nung territory is not in CEFEO custody" }
	if = { limit = { country_exists = TAM TAM = { NOT = { is_subject_of = FRE } } } log = "INDOCHINA COMMAND: FAIL - Muong Federation is not in CEFEO custody" }
	if = { limit = { country_exists = THO THO = { NOT = { is_subject_of = FRE } } } log = "INDOCHINA COMMAND: FAIL - Tho territory is not in CEFEO custody" }
	if = { limit = { country_exists = CCC CCC = { NOT = { is_subject_of = VIE } } } log = "INDOCHINA COMMAND: FAIL - Cochinchina is not in Saigon custody" }
	if = { limit = { country_exists = FUL FUL = { NOT = { is_subject_of = VIE } } } log = "INDOCHINA COMMAND: FAIL - Montagnard crown domain is not in Saigon custody" }

	if = { limit = { country_exists = VIE country_exists = VIN VIE = { has_war_with = VIN } } log = "INDOCHINA COMMAND: FAIL - State of Vietnam and Hanoi are direct belligerents" }
	if = { limit = { country_exists = CCC country_exists = VIN CCC = { has_war_with = VIN } } log = "INDOCHINA COMMAND: FAIL - Cochinchina leaked into Hanoi's northern war" }
	if = { limit = { country_exists = FUL country_exists = VIN FUL = { has_war_with = VIN } } log = "INDOCHINA COMMAND: FAIL - Montagnard command leaked into Hanoi's northern war" }
	if = { limit = { country_exists = NUN country_exists = VIN NUN = { has_war_with = VIN } } log = "INDOCHINA COMMAND: FAIL - Nung territory leaked into Hanoi's campaign" }
	if = { limit = { country_exists = NLF country_exists = VIN NLF = { NOT = { is_in_faction_with = VIN } } } log = "INDOCHINA COMMAND: FAIL - southern resistance is outside Hanoi's faction" }
	if = { limit = { country_exists = VIE VIE = { is_subject_of = FRA } } log = "INDOCHINA COMMAND: FAIL - State of Vietnam is a direct French subject" }
	if = { limit = { vin_nlf_coordination_active_trigger = yes } log = "INDOCHINA COMMAND: Hanoi-southern resistance coordination channel is active" }
	if = { limit = { country_exists = VIE country_exists = NLF VIE = { has_war_with = NLF } } log = "INDOCHINA COMMAND: southern war is active and separate" }
	if = { limit = { country_exists = NLF country_exists = FRE NLF = { has_war_with = FRE } } log = "INDOCHINA COMMAND: FAIL - northern campaign leaked into the southern resistance" }
	if = { limit = { country_exists = NLF country_exists = TAI NLF = { has_war_with = TAI } } log = "INDOCHINA COMMAND: FAIL - Tai campaign leaked into the southern resistance" }
	if = { limit = { country_exists = NLF country_exists = TAM NLF = { has_war_with = TAM } } log = "INDOCHINA COMMAND: FAIL - Hoa Binh campaign leaked into the southern resistance" }
	if = { limit = { country_exists = NLF country_exists = THO NLF = { has_war_with = THO } } log = "INDOCHINA COMMAND: FAIL - Cao-Bac campaign leaked into the southern resistance" }
	if = { limit = { has_global_flag = Laos_Raid_Active country_exists = NLF country_exists = LOS NLF = { has_war_with = LOS } } log = "INDOCHINA COMMAND: FAIL - Pathet Lao raid leaked into the southern resistance" }
	log = "INDOCHINA COMMAND: status check complete"
}

test_indochina_command_repair = {
	if = {
		limit = { country_exists = FRE }
		FRE = {
			clr_country_flag = FRE_French_Indochina_Command_Established
			fre_indochina_command_initialize = yes
		}
		log = "INDOCHINA COMMAND: faction and command membership rebuilt; run the status check"
	}
	else = { log = "INDOCHINA COMMAND: repair refused - CEFEO does not exist" }
}
