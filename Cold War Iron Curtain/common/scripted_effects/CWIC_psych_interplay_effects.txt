# East–West psychotronics interplay (Phase 5).
#
# Each side collects holds (documented leverage on the other side's program)
# and spends them on a leak, which costs the other side, or on a budget
# briefing, which pays Political Power now but raises the shared mirror race.
# Nothing here moves research, Research Record, proof or dossier progress:
# claims affect urgency and pressure only. Directorate globals move only for a
# member of a running Directorate (WP_psych_extra_directorate_member_trigger).
#
# global.CWIC_interplay_west_holds   0-3  American leverage on the Soviet program
# global.CWIC_interplay_east_holds   0-3  Soviet leverage on the American program
# global.CWIC_interplay_mirror_race  0-5  mutual alarm
# global.CWIC_interplay_west_leaks   0-2  leaks used by the United States
# global.CWIC_interplay_east_leaks   0-2  leaks used by the Soviet Union

CWIC_interplay_clamp = {
	if = { limit = { has_variable = global.CWIC_interplay_west_holds } clamp_variable = { var = global.CWIC_interplay_west_holds min = 0 max = 3 } }
	if = { limit = { has_variable = global.CWIC_interplay_east_holds } clamp_variable = { var = global.CWIC_interplay_east_holds min = 0 max = 3 } }
	if = { limit = { has_variable = global.CWIC_interplay_mirror_race } clamp_variable = { var = global.CWIC_interplay_mirror_race min = 0 max = 5 } }
	if = { limit = { has_variable = global.CWIC_interplay_west_leaks } clamp_variable = { var = global.CWIC_interplay_west_leaks min = 0 max = 2 } }
	if = { limit = { has_variable = global.CWIC_interplay_east_leaks } clamp_variable = { var = global.CWIC_interplay_east_leaks min = 0 max = 2 } }
}

CWIC_interplay_shift_mirror_race = {
	if = {
		limit = { CWIC_interplay_enabled_trigger = yes }
		if = {
			limit = { check_variable = { cwic_interplay_race_delta > 0 } }
			CWIC_interplay_raise_race = yes
		}
		else_if = {
			limit = { check_variable = { cwic_interplay_race_delta < 0 } }
			CWIC_interplay_lower_race = yes
		}
	}
}

CWIC_interplay_reconcile = {
	# 1972: the DIA study reaches the program offices.
	if = {
		limit = {
			date > 1972.6.30
			NOT = { has_global_flag = CWIC_interplay_dia_report_fired }
		}
		set_global_flag = CWIC_interplay_dia_report_fired
		CWIC_interplay_open_file = yes
		country_event = { id = cwic_interplay.1 days = 1 }
	}
	# The Soviet appraisal reaches the Soviet Division.
	if = {
		limit = { has_global_flag = cwic_psychotronic_claims_circulating }
		CWIC_interplay_open_file = yes
	}
	# Urgency adjustments either program already applied; each counts once.
	if = {
		limit = {
			has_country_flag = USA_mkultra_soviet_claims_urgency_applied
			NOT = { has_global_flag = CWIC_interplay_counted_rv_urgency }
		}
		set_global_flag = CWIC_interplay_counted_rv_urgency
		CWIC_interplay_raise_race = yes
	}
	if = {
		limit = {
			has_global_flag = WP_psych_US_urgency_applied
			NOT = { has_global_flag = CWIC_interplay_counted_soviet_urgency }
		}
		set_global_flag = CWIC_interplay_counted_soviet_urgency
		CWIC_interplay_raise_race = yes
	}
	if = {
		limit = {
			has_country_flag = USA_mkultra_grill_flame_soviet_urgency
			NOT = { has_global_flag = CWIC_interplay_counted_grill_flame_urgency }
		}
		set_global_flag = CWIC_interplay_counted_grill_flame_urgency
		CWIC_interplay_raise_race = yes
	}
	if = {
		limit = {
			has_country_flag = USA_mkultra_grill_flame_gap_cited
			NOT = { has_global_flag = CWIC_interplay_counted_grill_flame_gap }
		}
		set_global_flag = CWIC_interplay_counted_grill_flame_gap
		CWIC_interplay_raise_race = yes
	}
	if = {
		limit = {
			has_country_flag = USA_mkultra_grill_flame_skeptical_estimate
			NOT = { has_global_flag = CWIC_interplay_counted_grill_flame_skeptical }
		}
		set_global_flag = CWIC_interplay_counted_grill_flame_skeptical
		CWIC_interplay_lower_race = yes
	}
	# 1977: the Moscow correspondent and the Petukhov manuscript (Toth incident).
	if = {
		limit = {
			date > 1977.6.1
			date < 1980.1.1
			NOT = { has_global_flag = CWIC_interplay_toth_incident_fired }
			CWIC_interplay_soviet_target_live_trigger = yes
		}
		set_global_flag = CWIC_interplay_toth_incident_fired
		CWIC_interplay_open_file = yes
		SOV = { country_event = { id = cwic_interplay.50 days = 1 } }
	}
	# 1988: the Esalen-Moscow citizen-diplomacy exchange delegation.
	if = {
		limit = {
			date > 1988.5.1
			date < 1991.8.1
			NOT = { has_global_flag = CWIC_interplay_1988_exchange_fired }
			CWIC_interplay_soviet_target_live_trigger = yes
		}
		set_global_flag = CWIC_interplay_1988_exchange_fired
		CWIC_interplay_open_file = yes
		country_event = { id = cwic_interplay.60 days = 1 }
	}
	# Budget pressure, once per game.
	if = {
		limit = {
			check_variable = { global.CWIC_interplay_mirror_race > 2 }
			NOT = { has_global_flag = CWIC_interplay_budget_pressure_fired }
		}
		set_global_flag = CWIC_interplay_budget_pressure_fired
		country_event = { id = cwic_interplay.20 days = 1 }
		if = {
			limit = { SOV = { SOV_psych_charter_active_trigger = yes } }
			SOV = { country_event = { id = cwic_interplay.21 days = 1 } }
		}
	}
}
