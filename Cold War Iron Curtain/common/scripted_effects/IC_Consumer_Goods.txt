calculate_civilian_factory_production_total_factories = {

    # Start with factories the player can use for construction
    set_variable = {
        Actual_Civilian_Factories = num_of_available_civilian_factories
    }

    # Compute factories locked to consumer goods
	set_temp_variable = {
		total_milcivs = {
			value = num_of_military_factories
			add = num_of_civilian_factories
			subtract = cwic_tn_offsite
			subtract = cwic_tn_export_civs
		}
	}
    set_temp_variable = { 
		cg_locked = {
			value = modifier@consumer_goods_factor 
			add = 1
			multiply = modifier@consumer_goods_expected_value
			max = 0.1
			multiply = total_milcivs
		}
	}
    round_temp_variable = cg_locked

    # Add locked consumer goods factories back in
    add_to_variable = {
        Actual_Civilian_Factories = cg_locked
    }

    # Safety clamp (rounding can occasionally overshoot by 1)
    if = {
        limit = { check_variable = { Actual_Civilian_Factories > num_of_civilian_factories } }
        set_variable = { Actual_Civilian_Factories = num_of_civilian_factories }
    }

    clear_variable = cg_locked
}


cwic_calculate_consumer_goods_monthly_demand = {

	# ============================================================
	# Inputs expected on ROOT:
	# - National_HDI (0.0..1.0)
	# - state_population_k of owned core states
	# - global.year
	# Outputs:
	# - cg_*_demand (float)
	# - cg_*_expected_tier (int 0..5; electronics 1..5)
	# ============================================================

	# ----------------------------
	# Clamp HDI defensively
	# ----------------------------
	set_temp_variable = { cg_hdi = ROOT.National_HDI }
	if = { limit = { check_variable = { cg_hdi < 0 } } set_temp_variable = { cg_hdi = 0 } }
	if = { limit = { check_variable = { cg_hdi > 1 } } set_temp_variable = { cg_hdi = 1 } }

	# Workforce (thousands)
	set_variable = { cg_pop_k_sum = 0 }
	every_owned_state = {
		limit = {
			is_core_of = PREV
			state_population > 10000
		}
		add_to_variable = { PREV.cg_pop_k_sum = state_population_k }
	}
	set_temp_variable = { cg_pop_k = cg_pop_k_sum }
	multiply_temp_variable = { cg_pop_k = 0.078 }
	if = { limit = { check_variable = { cg_pop_k < 0 } } set_temp_variable = { cg_pop_k = 0 } }

	# ============================================================
	# 1) Base decade tier from year
	# Tier mapping:
	# 0 = Prewar
	# 1 = 1950s
	# 2 = 1960s
	# 3 = 1970s
	# 4 = 1980s
	# 5 = 1990s+
	# ============================================================
	set_temp_variable = { cg_decade_tier = 1 } # default 1950s baseline

	if = {
		limit = { check_variable = { global.year < 1945 } }
		set_temp_variable = { cg_decade_tier = 0 }
	}
	else_if = {
		limit = { check_variable = { global.year < 1960 } }
		set_temp_variable = { cg_decade_tier = 1 }
	}
	else_if = {
		limit = { check_variable = { global.year < 1970 } }
		set_temp_variable = { cg_decade_tier = 2 }
	}
	else_if = {
		limit = { check_variable = { global.year < 1980 } }
		set_temp_variable = { cg_decade_tier = 3 }
	}
	else_if = {
		limit = { check_variable = { global.year < 1990 } }
		set_temp_variable = { cg_decade_tier = 4 }
	}
	else = {
		set_temp_variable = { cg_decade_tier = 5 }
	}

	# ============================================================
	# 2) HDI adjustment to expected tier (soft push)
	# Low HDI tends to accept older standards, high HDI expects newer.
	# - HDI < 0.35 : -1 tier
	# - HDI > 0.80 : +1 tier
	# ============================================================
	set_temp_variable = { cg_hdi_tier_adj = 0 }

	if = {
		limit = { check_variable = { cg_hdi < 0.35 } }
		set_temp_variable = { cg_hdi_tier_adj = -1 }
	}
	else_if = {
		limit = { check_variable = { cg_hdi > 0.80 } }
		set_temp_variable = { cg_hdi_tier_adj = 1 }
	}

	# ============================================================
	# 3) Expected tiers by good type
	# Basics/durables/vehicles can be tier 0..5.
	# Consumer electronics starts at tier 1 (no tier 0 expectations).
	# ============================================================
	set_variable = { cg_basic_expected_tier = cg_decade_tier }
	add_to_variable = { cg_basic_expected_tier = cg_hdi_tier_adj }
	if = { limit = { check_variable = { cg_basic_expected_tier < 0 } } set_variable = { cg_basic_expected_tier = 0 } }
	if = { limit = { check_variable = { cg_basic_expected_tier > 5 } } set_variable = { cg_basic_expected_tier = 5 } }

	set_variable = { cg_durable_expected_tier = cg_decade_tier }
	add_to_variable = { cg_durable_expected_tier = cg_hdi_tier_adj }
	if = { limit = { check_variable = { cg_durable_expected_tier < 0 } } set_variable = { cg_durable_expected_tier = 0 } }
	if = { limit = { check_variable = { cg_durable_expected_tier > 5 } } set_variable = { cg_durable_expected_tier = 5 } }

	set_variable = { cg_vehicles_expected_tier = cg_decade_tier }
	add_to_variable = { cg_vehicles_expected_tier = cg_hdi_tier_adj }
	if = { limit = { check_variable = { cg_vehicles_expected_tier < 0 } } set_variable = { cg_vehicles_expected_tier = 0 } }
	if = { limit = { check_variable = { cg_vehicles_expected_tier > 5 } } set_variable = { cg_vehicles_expected_tier = 5 } }

	set_variable = { cg_consumer_electronics_expected_tier = cg_decade_tier }
	add_to_variable = { cg_consumer_electronics_expected_tier = cg_hdi_tier_adj }
	if = { limit = { check_variable = { cg_consumer_electronics_expected_tier < 1 } } set_variable = { cg_consumer_electronics_expected_tier = 1 } }
	if = { limit = { check_variable = { cg_consumer_electronics_expected_tier > 5 } } set_variable = { cg_consumer_electronics_expected_tier = 5 } }


	# ============================================================
	# 4) Demand calculation framework
	# Demand = pop_k * base_rate * hdi_scale * tier_multiplier
	#
	# Notes:
	# - base_rate is "units per thousand employable people per month"
	# - hdi_scale is linear here for performance and tuning simplicity
	# - tier_multiplier increases expectations over time (and modern categories rise faster)
	# ============================================================

	# Shared HDI scaling:
	# hdi_scale = 0.60 + (0.80 * HDI)  => 0.60..1.40
	set_temp_variable = { cg_hdi_scale = cg_hdi }
	multiply_temp_variable = { cg_hdi_scale = 0.80 }
	add_to_temp_variable = { cg_hdi_scale = 0.60 }


	# ----------------------------
	# BASIC GOODS demand
	# base_rate_basic = 0.12
	# tier multipliers: 0.80, 1.00, 1.05, 1.10, 1.15, 1.20
	# ----------------------------
	set_temp_variable = { cg_mult = 1.00 }
	if = { limit = { check_variable = { cg_basic_expected_tier = 0 } } set_temp_variable = { cg_mult = 0.80 } }
	else_if = { limit = { check_variable = { cg_basic_expected_tier = 1 } } set_temp_variable = { cg_mult = 1.00 } }
	else_if = { limit = { check_variable = { cg_basic_expected_tier = 2 } } set_temp_variable = { cg_mult = 1.05 } }
	else_if = { limit = { check_variable = { cg_basic_expected_tier = 3 } } set_temp_variable = { cg_mult = 1.10 } }
	else_if = { limit = { check_variable = { cg_basic_expected_tier = 4 } } set_temp_variable = { cg_mult = 1.15 } }
	else = { set_temp_variable = { cg_mult = 1.20 } }

	set_variable = { cg_basic_demand_raw = cg_pop_k }
	multiply_variable = { cg_basic_demand_raw = 0.12 }
	multiply_variable = { cg_basic_demand_raw = cg_hdi_scale }
	multiply_variable = { cg_basic_demand_raw = cg_mult }


	# ----------------------------
	# DURABLE GOODS demand
	# base_rate_durable = 0.05
	# tier multipliers: 0.50, 0.80, 1.00, 1.15, 1.30, 1.45
	# ----------------------------
	set_temp_variable = { cg_mult = 1.00 }
	if = { limit = { check_variable = { cg_durable_expected_tier = 0 } } set_temp_variable = { cg_mult = 0.50 } }
	else_if = { limit = { check_variable = { cg_durable_expected_tier = 1 } } set_temp_variable = { cg_mult = 0.80 } }
	else_if = { limit = { check_variable = { cg_durable_expected_tier = 2 } } set_temp_variable = { cg_mult = 1.00 } }
	else_if = { limit = { check_variable = { cg_durable_expected_tier = 3 } } set_temp_variable = { cg_mult = 1.15 } }
	else_if = { limit = { check_variable = { cg_durable_expected_tier = 4 } } set_temp_variable = { cg_mult = 1.30 } }
	else = { set_temp_variable = { cg_mult = 1.45 } }

	set_variable = { cg_durable_demand_raw = cg_pop_k }
	multiply_variable = { cg_durable_demand_raw = 0.05 }
	multiply_variable = { cg_durable_demand_raw = cg_hdi_scale }
	multiply_variable = { cg_durable_demand_raw = cg_mult }


	# ----------------------------
	# VEHICLES demand
	# base_rate_vehicles = 0.02
	# tier multipliers: 0.40, 0.70, 0.90, 1.05, 1.20, 1.35
	# ----------------------------
	set_temp_variable = { cg_mult = 1.00 }
	if = { limit = { check_variable = { cg_vehicles_expected_tier = 0 } } set_temp_variable = { cg_mult = 0.40 } }
	else_if = { limit = { check_variable = { cg_vehicles_expected_tier = 1 } } set_temp_variable = { cg_mult = 0.70 } }
	else_if = { limit = { check_variable = { cg_vehicles_expected_tier = 2 } } set_temp_variable = { cg_mult = 0.90 } }
	else_if = { limit = { check_variable = { cg_vehicles_expected_tier = 3 } } set_temp_variable = { cg_mult = 1.05 } }
	else_if = { limit = { check_variable = { cg_vehicles_expected_tier = 4 } } set_temp_variable = { cg_mult = 1.20 } }
	else = { set_temp_variable = { cg_mult = 1.35 } }

	set_variable = { cg_vehicles_demand_raw = cg_pop_k }
	multiply_variable = { cg_vehicles_demand_raw = 0.02 }
	multiply_variable = { cg_vehicles_demand_raw = cg_hdi_scale }
	multiply_variable = { cg_vehicles_demand_raw = cg_mult }

	# ----------------------------
	# CONSUMER ELECTRONICS demand
	# base_rate_electronics = 0.015
	# tier multipliers: (tier1..5): 0.60, 0.90, 1.15, 1.40, 1.70
	# Electronics demand is much more HDI-sensitive, so we apply an extra HDI lift:
	# electronics_hdi_scale = 0.30 + (1.20 * HDI) => 0.30..1.50
	# ----------------------------
	set_temp_variable = { cg_mult = 1.00 }
	if = { limit = { check_variable = { cg_consumer_electronics_expected_tier = 1 } } set_temp_variable = { cg_mult = 0.60 } }
	else_if = { limit = { check_variable = { cg_consumer_electronics_expected_tier = 2 } } set_temp_variable = { cg_mult = 0.90 } }
	else_if = { limit = { check_variable = { cg_consumer_electronics_expected_tier = 3 } } set_temp_variable = { cg_mult = 1.15 } }
	else_if = { limit = { check_variable = { cg_consumer_electronics_expected_tier = 4 } } set_temp_variable = { cg_mult = 1.40 } }
	else = { set_temp_variable = { cg_mult = 1.70 } }

	set_temp_variable = { cg_elec_hdi_scale = cg_hdi }
	multiply_temp_variable = { cg_elec_hdi_scale = 1.20 }
	add_to_temp_variable = { cg_elec_hdi_scale = 0.30 }

	set_variable = { cg_consumer_electronics_demand_raw = cg_pop_k }
	multiply_variable = { cg_consumer_electronics_demand_raw = 0.015 }
	multiply_variable = { cg_consumer_electronics_demand_raw = cg_elec_hdi_scale }
	multiply_variable = { cg_consumer_electronics_demand_raw = cg_mult }


	set_variable = { cg_vehicles_readiness = cg_hdi }
	subtract_from_variable = { cg_vehicles_readiness = 0.35 }
	multiply_variable = { cg_vehicles_readiness = 4 }
	clamp_variable = { var = cg_vehicles_readiness min = 0 max = 1 }
	set_variable = { cg_consumer_electronics_readiness = cg_hdi }
	subtract_from_variable = { cg_consumer_electronics_readiness = 0.45 }
	multiply_variable = { cg_consumer_electronics_readiness = 4 }
	clamp_variable = { var = cg_consumer_electronics_readiness min = 0 max = 1 }

	if = {
		limit = { has_variable = cg_basic_demand_anchor }
		set_temp_variable = { cg_demand_mult = modifier@cg_demand_factor }
		clamp_temp_variable = { var = cg_demand_mult min = -0.5 max = 0.5 }
		add_to_temp_variable = { cg_demand_mult = 1 }
		set_temp_variable = { cg_target = cg_basic_demand_raw }
		multiply_temp_variable = { cg_target = cg_basic_demand_anchor }
		multiply_temp_variable = { cg_target = 0.05 }
		if = { limit = { has_variable = global.cg_basic_demand_scale } multiply_temp_variable = { cg_target = global.cg_basic_demand_scale } }
		multiply_temp_variable = { cg_target = cg_demand_mult }
		multiply_variable = { cg_basic_demand = 0.95 }
		add_to_variable = { cg_basic_demand = cg_target }

		set_temp_variable = { cg_target = cg_durable_demand_raw }
		multiply_temp_variable = { cg_target = cg_durable_demand_anchor }
		multiply_temp_variable = { cg_target = 0.05 }
		if = { limit = { has_variable = global.cg_durable_demand_scale } multiply_temp_variable = { cg_target = global.cg_durable_demand_scale } }
		multiply_temp_variable = { cg_target = cg_demand_mult }
		multiply_variable = { cg_durable_demand = 0.95 }
		add_to_variable = { cg_durable_demand = cg_target }

		set_temp_variable = { cg_target = cg_vehicles_demand_raw }
		multiply_temp_variable = { cg_target = cg_vehicles_demand_anchor }
		multiply_temp_variable = { cg_target = 0.05 }
		if = { limit = { has_variable = global.cg_vehicles_demand_scale } multiply_temp_variable = { cg_target = global.cg_vehicles_demand_scale } }
		multiply_temp_variable = { cg_target = cg_demand_mult }
		multiply_variable = { cg_vehicles_demand = 0.95 }
		add_to_variable = { cg_vehicles_demand = cg_target }

		set_temp_variable = { cg_target = cg_consumer_electronics_demand_raw }
		multiply_temp_variable = { cg_target = cg_consumer_electronics_demand_anchor }
		multiply_temp_variable = { cg_target = 0.05 }
		if = { limit = { has_variable = global.cg_electronics_demand_scale } multiply_temp_variable = { cg_target = global.cg_electronics_demand_scale } }
		multiply_temp_variable = { cg_target = cg_demand_mult }
		multiply_variable = { cg_consumer_electronics_demand = 0.95 }
		add_to_variable = { cg_consumer_electronics_demand = cg_target }
	}
	else = {
		set_variable = { cg_basic_demand = 0 }
		set_variable = { cg_durable_demand = 0 }
		set_variable = { cg_vehicles_demand = 0 }
		set_variable = { cg_consumer_electronics_demand = 0 }
	}

	# Cleanup temps
	clear_variable = cg_hdi
	clear_variable = cg_pop_k
	clear_variable = cg_decade_tier
	clear_variable = cg_hdi_tier_adj
	clear_variable = cg_hdi_scale
	clear_variable = cg_elec_hdi_scale
	clear_variable = cg_mult
}
cwic_cg_monthly_production_and_allotment = {

    cwic_cg_sync_resource_pools_from_vanilla = yes


	# ============================================================
	# 0) Get "Actual_Civilian_Factories" (consumer goods + available civs)
	# ============================================================
	calculate_civilian_factory_production_total_factories = yes
	if = { limit = { check_variable = { ROOT.Actual_Civilian_Factories < 0 } } set_variable = { Actual_Civilian_Factories = 0 } }

	# ============================================================
	# 1) Clamp HDI (0..1)
	# ============================================================
	set_temp_variable = { cg_hdi = ROOT.National_HDI }
	if = { limit = { check_variable = { cg_hdi < 0 } } set_temp_variable = { cg_hdi = 0 } }
	if = { limit = { check_variable = { cg_hdi > 1 } } set_temp_variable = { cg_hdi = 1 } }


	# ============================================================
	# 2) Resolve producible tier (best tier available)
	# ============================================================

	# BASIC
	set_variable = { cg_basic_producible_tier = 0 }
	if = { limit = { cwic_can_produce_cg_basic_tier_5 = yes } set_variable = { cg_basic_producible_tier = 5 } }
	else_if = { limit = { cwic_can_produce_cg_basic_tier_4 = yes } set_variable = { cg_basic_producible_tier = 4 } }
	else_if = { limit = { cwic_can_produce_cg_basic_tier_3 = yes } set_variable = { cg_basic_producible_tier = 3 } }
	else_if = { limit = { cwic_can_produce_cg_basic_tier_2 = yes } set_variable = { cg_basic_producible_tier = 2 } }
	else_if = { limit = { cwic_can_produce_cg_basic_tier_1 = yes } set_variable = { cg_basic_producible_tier = 1 } }

	# DURABLE
	set_variable = { cg_durable_producible_tier = 0 }
	if = { limit = { cwic_can_produce_cg_durable_tier_5 = yes } set_variable = { cg_durable_producible_tier = 5 } }
	else_if = { limit = { cwic_can_produce_cg_durable_tier_4 = yes } set_variable = { cg_durable_producible_tier = 4 } }
	else_if = { limit = { cwic_can_produce_cg_durable_tier_3 = yes } set_variable = { cg_durable_producible_tier = 3 } }
	else_if = { limit = { cwic_can_produce_cg_durable_tier_2 = yes } set_variable = { cg_durable_producible_tier = 2 } }
	else_if = { limit = { cwic_can_produce_cg_durable_tier_1 = yes } set_variable = { cg_durable_producible_tier = 1 } }

	# VEHICLES
	set_variable = { cg_vehicles_producible_tier = 0 }
	if = { limit = { cwic_can_produce_cg_vehicles_tier_5 = yes } set_variable = { cg_vehicles_producible_tier = 5 } }
	else_if = { limit = { cwic_can_produce_cg_vehicles_tier_4 = yes } set_variable = { cg_vehicles_producible_tier = 4 } }
	else_if = { limit = { cwic_can_produce_cg_vehicles_tier_3 = yes } set_variable = { cg_vehicles_producible_tier = 3 } }
	else_if = { limit = { cwic_can_produce_cg_vehicles_tier_2 = yes } set_variable = { cg_vehicles_producible_tier = 2 } }
	else_if = { limit = { cwic_can_produce_cg_vehicles_tier_1 = yes } set_variable = { cg_vehicles_producible_tier = 1 } }

	# CONSUMER ELECTRONICS (no tier 0)
	set_variable = { cg_consumer_electronics_producible_tier = 1 }
	if = { limit = { cwic_can_produce_cg_consumer_electronics_tier_5 = yes } set_variable = { cg_consumer_electronics_producible_tier = 5 } }
	else_if = { limit = { cwic_can_produce_cg_consumer_electronics_tier_4 = yes } set_variable = { cg_consumer_electronics_producible_tier = 4 } }
	else_if = { limit = { cwic_can_produce_cg_consumer_electronics_tier_3 = yes } set_variable = { cg_consumer_electronics_producible_tier = 3 } }
	else_if = { limit = { cwic_can_produce_cg_consumer_electronics_tier_2 = yes } set_variable = { cg_consumer_electronics_producible_tier = 2 } }


	# ============================================================
	# 2B) Hard gates (disable category until tech prereqs)
	# ============================================================
	set_temp_variable = { allow_vehicles = 0 }
	if = { limit = { cwic_can_produce_cg_vehicles_tier_0 = yes } set_temp_variable = { allow_vehicles = 1 } }

	set_temp_variable = { allow_elec = 0 }
	if = { limit = { cwic_can_produce_cg_consumer_electronics_tier_1 = yes } set_temp_variable = { allow_elec = 1 } }


	# ============================================================
	# 2C) Office Park gating (uses ROOT.office_park_total)
	# - Electronics: >= 2 parks, full at 10
	# - Vehicles:    >= 1 park, full at 6
	# ============================================================
	set_temp_variable = { office_parks = 0 }
	every_owned_state = {
		add_to_temp_variable = { office_parks = building_level@office_park }
	}
	if = { limit = { check_variable = { office_parks < 0 } } set_temp_variable = { office_parks = 0 } }

	# Electronics office gate threshold (>= 2)
	set_temp_variable = { elec_office_gate = 0 }
	if = {
		limit = {
			OR = {
				check_variable = { office_parks = 2 }
				check_variable = { office_parks > 2 }
			}
		}
		set_temp_variable = { elec_office_gate = office_parks }
		subtract_from_temp_variable = { elec_office_gate = 2 }      # 0 at 2
		divide_temp_variable = { elec_office_gate = 8 }             # 1.0 at 10
		if = { limit = { check_variable = { elec_office_gate < 0 } } set_temp_variable = { elec_office_gate = 0 } }
		if = { limit = { check_variable = { elec_office_gate > 1 } } set_temp_variable = { elec_office_gate = 1 } }
	}

	# Vehicles office gate threshold (>= 1)
	set_temp_variable = { veh_office_gate = 0 }
	if = {
		limit = {
			OR = {
				check_variable = { office_parks = 1 }
				check_variable = { office_parks > 1 }
			}
		}
		set_temp_variable = { veh_office_gate = office_parks }
		subtract_from_temp_variable = { veh_office_gate = 1 }       # 0 at 1
		divide_temp_variable = { veh_office_gate = 5 }              # 1.0 at 6
		if = { limit = { check_variable = { veh_office_gate < 0 } } set_temp_variable = { veh_office_gate = 0 } }
		if = { limit = { check_variable = { veh_office_gate > 1 } } set_temp_variable = { veh_office_gate = 1 } }
	}

	# Apply office gates to allow flags (if gate is effectively 0, block category)
	if = { limit = { check_variable = { elec_office_gate < 0.01 } } set_temp_variable = { allow_elec = 0 } }
	if = { limit = { check_variable = { veh_office_gate < 0.01 } } set_temp_variable = { allow_vehicles = 0 } }


	# ============================================================
	# 2D) Rare Earth gate for Consumer Electronics (resource@tungsten is INTEGER)
	# - hard minimum: >= 1 (implemented as = OR >)
	# - scaling: 16 reaches full gate (1.0)
	# ============================================================
	set_temp_variable = { elec_rare_gate = 0 }
	if = {
		limit = {
			OR = {
				check_variable = { cwic_res_tungsten_pool = 1 }
				check_variable = { cwic_res_tungsten_pool > 1 }
			}
		}
		set_temp_variable = { elec_rare_gate = cwic_res_tungsten_pool }
		subtract_from_temp_variable = { elec_rare_gate = 1 }        # 0 at 1
		divide_temp_variable = { elec_rare_gate = 15 }              # 1.0 at 16
		if = { limit = { check_variable = { elec_rare_gate < 0 } } set_temp_variable = { elec_rare_gate = 0 } }
		if = { limit = { check_variable = { elec_rare_gate > 1 } } set_temp_variable = { elec_rare_gate = 1 } }
	}
	add_to_temp_variable = { elec_rare_gate = modifier@cg_rare_earth_relief }
	clamp_temp_variable = { var = elec_rare_gate min = 0 max = 1 }
	if = { limit = { check_variable = { elec_rare_gate < 0.01 } } set_temp_variable = { allow_elec = 0 } }


	# ============================================================
	# 3) Shortage pressure per category (0..1) from last month's coverage
	# ============================================================
	set_temp_variable = { sh_basic = 0 }
	if = {
		limit = { check_variable = { ROOT.cg_basic_demand > 0.01 } }
		set_temp_variable = { sh_basic = 1 }
		subtract_from_temp_variable = { sh_basic = ROOT.cg_basic_coverage }
		clamp_temp_variable = { var = sh_basic min = 0 max = 1 }
	}

	set_temp_variable = { sh_durable = 0 }
	if = {
		limit = { check_variable = { ROOT.cg_durable_demand > 0.01 } }
		set_temp_variable = { sh_durable = 1 }
		subtract_from_temp_variable = { sh_durable = ROOT.cg_durable_coverage }
		clamp_temp_variable = { var = sh_durable min = 0 max = 1 }
	}

	set_temp_variable = { sh_vehicles = 0 }
	if = {

### ============================================================
### MONTHLY: Allocate production points by CG category
### Expects (temp vars): w_basic, w_durable, w_vehicles, w_elec
### Writes (country vars): cg_monthly_points_*
### ============================================================

cwic_cg_allocate_monthly_production_points = {
    set_variable = { cg_points_per_factory = 25 }
	# ------------------------------------------------------------
	# 0) Resolve tuning knob (points per factory)
	# ------------------------------------------------------------
	clear_variable = cg_ppf
	set_temp_variable = { cg_ppf = ROOT.cg_points_per_factory }

	# Default if missing or too small
	if = {
		limit = { check_variable = { cg_ppf < 0.1 } }
		set_temp_variable = { cg_ppf = 10 }
	}

	# ------------------------------------------------------------
	# 1) Total monthly points from "Actual_Civilian_Factories"
	# ------------------------------------------------------------
	clear_variable = total_points
	set_temp_variable = { total_points = ROOT.Actual_Civilian_Factories }
	if = { limit = { check_variable = { total_points < 0 } } set_temp_variable = { total_points = 0 } }

	multiply_temp_variable = { total_points = cg_ppf }

	# Keep points as whole numbers for downstream math
	round_temp_variable = total_points

	set_variable = { cg_monthly_points_total = total_points }

	# ------------------------------------------------------------
	# 2) Allocate points by normalized weights
	# ------------------------------------------------------------
	clear_variable = pts_basic
	clear_variable = pts_durable
	clear_variable = pts_vehicles
	clear_variable = pts_elec

	set_temp_variable = { pts_basic = total_points }
	multiply_temp_variable = { pts_basic = w_basic }

	set_temp_variable = { pts_durable = total_points }
	multiply_temp_variable = { pts_durable = w_durable }

	set_temp_variable = { pts_vehicles = total_points }
	multiply_temp_variable = { pts_vehicles = w_vehicles }

	set_temp_variable = { pts_elec = total_points }
	multiply_temp_variable = { pts_elec = w_elec }

	round_temp_variable = pts_basic
	round_temp_variable = pts_durable
	round_temp_variable = pts_vehicles
	round_temp_variable = pts_elec

	# ------------------------------------------------------------
	# 3) Rounding reconciliation
	# If rounding lost or gained points, shove the difference into BASIC.
	# ------------------------------------------------------------
	clear_variable = pts_sum
	clear_variable = pts_diff

	set_temp_variable = { pts_sum = pts_basic }
	add_to_temp_variable = { pts_sum = pts_durable }
	add_to_temp_variable = { pts_sum = pts_vehicles }
	add_to_temp_variable = { pts_sum = pts_elec }

	set_temp_variable = { pts_diff = total_points }
	subtract_from_temp_variable = { pts_diff = pts_sum }

	# If diff > 0, add to basic
	if = {
		limit = { check_variable = { pts_diff > 0 } }
		add_to_temp_variable = { pts_basic = pts_diff }
	}

	# If diff < 0, subtract from basic (clamped at 0)
	if = {
		limit = { check_variable = { pts_diff < 0 } }
		add_to_temp_variable = { pts_basic = pts_diff }
		if = { limit = { check_variable = { pts_basic < 0 } } set_temp_variable = { pts_basic = 0 } }
	}

	# ------------------------------------------------------------
	# 4) Write country variables
	# ------------------------------------------------------------
	set_variable = { cg_monthly_points_basic = pts_basic }
	set_variable = { cg_monthly_points_durable = pts_durable }
	set_variable = { cg_monthly_points_vehicles = pts_vehicles }
	set_variable = { cg_monthly_points_electronics = pts_elec }

	# ------------------------------------------------------------
	# 5) Cleanup temps used here (optional, matches your hygiene style)
	# ------------------------------------------------------------
	clear_variable = cg_ppf
	clear_variable = total_points
	clear_variable = pts_basic
	clear_variable = pts_durable
	clear_variable = pts_vehicles
	clear_variable = pts_elec
	clear_variable = pts_sum
	clear_variable = pts_diff
}
cwic_cg_calibrate_demand = {
	if = {
		limit = {
			check_variable = { cg_monthly_points_total > 0 }
			check_variable = { cg_basic_demand_raw > 0 }
			check_variable = { cg_basic_produced_this_month > 0 }
		}
		set_temp_variable = { cg_t = 0.92 }
		if = { limit = { check_variable = { National_HDI > 0.499 } } set_temp_variable = { cg_t = 0.8 } }
		set_temp_variable = { cg_home = 1 }
		subtract_from_temp_variable = { cg_home = modifier@consumer_goods_export_rate }
		clamp_temp_variable = { var = cg_home min = 0.1 max = 1 }
		multiply_temp_variable = { cg_t = cg_home }

		set_variable = { cg_basic_demand_anchor = cg_basic_produced_this_month }
		multiply_variable = { cg_basic_demand_anchor = cg_t }
		divide_variable = { cg_basic_demand_anchor = cg_basic_demand_raw }

		set_variable = { cg_durable_demand_anchor = cg_basic_demand_anchor }
		multiply_variable = { cg_durable_demand_anchor = 0.6 }
		if = {
			limit = {
				check_variable = { cg_durable_produced_this_month > 0 }
				check_variable = { cg_durable_demand_raw > 0 }
			}
			set_variable = { cg_durable_demand_anchor = cg_durable_produced_this_month }
			multiply_variable = { cg_durable_demand_anchor = cg_t }
			divide_variable = { cg_durable_demand_anchor = cg_durable_demand_raw }
		}

		set_variable = { cg_vehicles_demand_anchor = cg_basic_demand_anchor }
		multiply_variable = { cg_vehicles_demand_anchor = 0.35 }
		if = {
			limit = {
				check_variable = { cg_vehicles_produced_this_month > 0 }
				check_variable = { cg_vehicles_demand_raw > 0 }
			}
			set_variable = { cg_vehicles_demand_anchor = cg_vehicles_produced_this_month }
			multiply_variable = { cg_vehicles_demand_anchor = cg_t }
			divide_variable = { cg_vehicles_demand_anchor = cg_vehicles_demand_raw }
		}

		set_variable = { cg_consumer_electronics_demand_anchor = cg_basic_demand_anchor }
		multiply_variable = { cg_consumer_electronics_demand_anchor = 0.35 }
		if = {
			limit = {
				check_variable = { cg_consumer_electronics_produced_this_month > 0 }
				check_variable = { cg_consumer_electronics_demand_raw > 0 }
			}
			set_variable = { cg_consumer_electronics_demand_anchor = cg_consumer_electronics_produced_this_month }
			multiply_variable = { cg_consumer_electronics_demand_anchor = cg_t }
			divide_variable = { cg_consumer_electronics_demand_anchor = cg_consumer_electronics_demand_raw }
		}

		set_variable = { cg_basic_demand = cg_basic_demand_raw }
		multiply_variable = { cg_basic_demand = cg_basic_demand_anchor }
		set_variable = { cg_durable_demand = cg_durable_demand_raw }
		multiply_variable = { cg_durable_demand = cg_durable_demand_anchor }
		set_variable = { cg_vehicles_demand = cg_vehicles_demand_raw }
		multiply_variable = { cg_vehicles_demand = cg_vehicles_demand_anchor }
		set_variable = { cg_consumer_electronics_demand = cg_consumer_electronics_demand_raw }
		multiply_variable = { cg_consumer_electronics_demand = cg_consumer_electronics_demand_anchor }

		set_variable = { cg_basic_coverage = 1 }
		set_variable = { cg_durable_coverage = 1 }
		set_variable = { cg_vehicles_coverage = 1 }
		set_variable = { cg_consumer_electronics_coverage = 1 }
		set_variable = { cg_basic_supplied_tier = cg_basic_expected_tier }
		set_variable = { cg_durable_supplied_tier = cg_durable_expected_tier }
		set_variable = { cg_vehicles_supplied_tier = cg_vehicles_expected_tier }
		set_variable = { cg_consumer_electronics_supplied_tier = cg_consumer_electronics_expected_tier }
	}
}

# Permanent consumer-industry base for content. Set any of these temp variables
# first, then call: ci_light, ci_precision, ci_auto, ci_elec, ci_mgmt (capability
# points, techs give 1-3 each) and ci_basic_out, ci_durable_out, ci_vehicles_out,
# ci_elec_out (output factor, 0.02-0.06 per step). Values accumulate in the
# cwic_consumer_industry dynamic modifier.
cwic_add_consumer_industry = {
	if = { limit = { check_variable = { ci_light > 0 } } custom_effect_tooltip = CWIC_CI_LIGHT_TT }
	if = { limit = { check_variable = { ci_precision > 0 } } custom_effect_tooltip = CWIC_CI_PRECISION_TT }
	if = { limit = { check_variable = { ci_auto > 0 } } custom_effect_tooltip = CWIC_CI_AUTO_TT }
	if = { limit = { check_variable = { ci_elec > 0 } } custom_effect_tooltip = CWIC_CI_ELEC_TT }
	if = { limit = { check_variable = { ci_mgmt > 0 } } custom_effect_tooltip = CWIC_CI_MGMT_TT }
	if = { limit = { check_variable = { ci_basic_out > 0 } } custom_effect_tooltip = CWIC_CI_BASIC_OUT_TT }
	if = { limit = { check_variable = { ci_durable_out > 0 } } custom_effect_tooltip = CWIC_CI_DURABLE_OUT_TT }
	if = { limit = { check_variable = { ci_vehicles_out > 0 } } custom_effect_tooltip = CWIC_CI_VEHICLES_OUT_TT }
	if = { limit = { check_variable = { ci_elec_out > 0 } } custom_effect_tooltip = CWIC_CI_ELEC_OUT_TT }
	hidden_effect = {
		add_to_variable = { cwic_ci_cap_light = ci_light }
		add_to_variable = { cwic_ci_cap_precision = ci_precision }
		add_to_variable = { cwic_ci_cap_auto = ci_auto }
		add_to_variable = { cwic_ci_cap_elec = ci_elec }
		add_to_variable = { cwic_ci_cap_mgmt = ci_mgmt }
		add_to_variable = { cwic_ci_basic_out = ci_basic_out }
		add_to_variable = { cwic_ci_durable_out = ci_durable_out }
		add_to_variable = { cwic_ci_vehicles_out = ci_vehicles_out }
		add_to_variable = { cwic_ci_elec_out = ci_elec_out }
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = cwic_consumer_industry } } }
			add_dynamic_modifier = { modifier = cwic_consumer_industry }
		}
	}
}

# Opens one small native import deal for consumer vehicles (cic 0.25, so 4 units
# is one civilian factory). Set ci_import_from to the exporter's id first. Skips
# when the exporter is not offering enough or the pair is embargoed.
cwic_import_consumer_vehicles = {
	custom_effect_tooltip = CWIC_CI_VEHICLE_IMPORT_TT
	hidden_effect = {
		var:ci_import_from = {
			if = {
				limit = {
					check_variable = { cg_vehicles_offered > 3 }
					NOT = { check_variable = { ROOT.trade_embargo_check^THIS = 1 } }
					NOT = { check_variable = { trade_embargo_check^ROOT = 1 } }
				}
				save_event_target_as = ci_vehicle_exporter
			}
		}
		if = {
			limit = { has_event_target = ci_vehicle_exporter }
			meta_effect = {
				text = {
					create_import = { resource = cg_vehicles amount = 4 exporter = [TAG] }
				}
				TAG = "[?event_target:ci_vehicle_exporter.GetTag]"
			}
		}
	}
}

cwic_cg_ensure_office_parks = {
	custom_effect_tooltip = CG_ensure_office_parks_tt
	hidden_effect = {
		set_temp_variable = { cg_parks = 0 }
		every_owned_state = {
			add_to_temp_variable = { cg_parks = building_level@office_park }
		}
		for_loop_effect = {
			start = 0
			end = 3
			if = {
				limit = { check_variable = { cg_parks < 3 } }
				capital_scope = {
					add_extra_state_shared_building_slots = 1
					add_building_construction = {
						type = office_park
						level = 1
						instant_build = yes
					}
				}
				add_to_temp_variable = { cg_parks = 1 }
			}
		}
	}
}
