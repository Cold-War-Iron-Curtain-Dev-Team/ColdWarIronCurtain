cwic_apply_consumer_goods_pressure_master = {

	# ============================================================
	# Reset output variables (dynamic modifier inputs)
	# ============================================================
	set_variable = { CWIC_DM_CG_STABILITY_FACTOR = 0 }
	set_variable = { CWIC_DM_CG_POLITICAL_POWER_GAIN = 0 }

	set_variable = { CWIC_DM_CG_WORKFORCE_RATIO = 0 }
	set_variable = { CWIC_DM_CG_PRODUCTIVITY_FACTOR = 0 }

	set_variable = { CWIC_DM_CG_WELFARE_SPENDING_MOD = 0 }
	set_variable = { CWIC_DM_CG_POLICING_SPENDING_MOD = 0 }
	set_variable = { CWIC_DM_CG_PUBLIC_SERVICES_SPENDING_MOD = 0 }

	set_variable = { CWIC_DM_CG_CONSTRUCTION_SPEED_FACTOR = 0 }
	set_variable = { CWIC_DM_CG_OTHER_RESOURCES_INCOME_BONUS = 0 }

	set_variable = { CWIC_DM_CG_INDUSTRIAL_PARK_INCOME_BONUS = 0 }

	# Clear temps
	# ============================================================
	clear_variable = cg_shortage
	clear_variable = cg_reliance
	clear_variable = cg_gap
	clear_variable = cg_outdated
	clear_variable = cg_severity

	clear_variable = sev_basic
	clear_variable = sev_durable
	clear_variable = sev_vehicles
	clear_variable = sev_electronics

	clear_variable = master_scalar
	clear_variable = cg_tmp


	# ============================================================
	# BASIC GOODS severity (outdated weight 0.50)
	# ============================================================
	set_temp_variable = { cg_shortage = 1 }
	subtract_from_temp_variable = { cg_shortage = ROOT.cg_basic_coverage }
	if = { limit = { check_variable = { cg_shortage < 0 } } set_temp_variable = { cg_shortage = 0 } }
	if = { limit = { check_variable = { cg_shortage > 1 } } set_temp_variable = { cg_shortage = 1 } }

	if = { limit = { check_variable = { cg_reliance < 0 } } set_temp_variable = { cg_reliance = 0 } }
	if = { limit = { check_variable = { cg_reliance > 1 } } set_temp_variable = { cg_reliance = 1 } }

	subtract_from_temp_variable = { cg_gap = ROOT.cg_basic_supplied_tier }
	if = { limit = { check_variable = { cg_gap < 0 } } set_temp_variable = { cg_gap = 0 } }
	if = { limit = { check_variable = { cg_gap > 5 } } set_temp_variable = { cg_gap = 5 } }

	divide_temp_variable = { cg_outdated = 5 }

	set_temp_variable = { cg_severity = cg_outdated }
	multiply_temp_variable = { cg_severity = cg_reliance }
	multiply_temp_variable = { cg_severity = 0.50 }
	add_to_temp_variable = { cg_severity = cg_shortage }
	if = { limit = { check_variable = { cg_severity > 1 } } set_temp_variable = { cg_severity = 1 } }

	set_temp_variable = { sev_basic = cg_severity }


	# ============================================================
	# DURABLE GOODS severity (outdated weight 0.80)
	# ============================================================
	set_temp_variable = { cg_shortage = 1 }
	subtract_from_temp_variable = { cg_shortage = ROOT.cg_durable_coverage }
	if = { limit = { check_variable = { cg_shortage < 0 } } set_temp_variable = { cg_shortage = 0 } }
	if = { limit = { check_variable = { cg_shortage > 1 } } set_temp_variable = { cg_shortage = 1 } }

	set_temp_variable = { cg_reliance = ROOT.cg_durable_coverage }
	if = { limit = { check_variable = { cg_reliance < 0 } } set_temp_variable = { cg_reliance = 0 } }
	if = { limit = { check_variable = { cg_reliance > 1 } } set_temp_variable = { cg_reliance = 1 } }

	if = { limit = { check_variable = { cg_gap > 5 } } set_temp_variable = { cg_gap = 5 } }

	divide_temp_variable = { cg_outdated = 5 }

	set_temp_variable = { cg_severity = cg_outdated }
	multiply_temp_variable = { cg_severity = cg_reliance }
	multiply_temp_variable = { cg_severity = 0.80 }
	add_to_temp_variable = { cg_severity = cg_shortage }
	if = { limit = { check_variable = { cg_severity > 1 } } set_temp_variable = { cg_severity = 1 } }

	set_temp_variable = { sev_durable = cg_severity }


	# ============================================================
	# VEHICLES severity (outdated weight 0.70)
	# ============================================================
	set_temp_variable = { cg_shortage = 1 }
	subtract_from_temp_variable = { cg_shortage = ROOT.cg_vehicles_coverage }
	if = { limit = { check_variable = { cg_shortage < 0 } } set_temp_variable = { cg_shortage = 0 } }

	if = { limit = { check_variable = { cg_reliance > 1 } } set_temp_variable = { cg_reliance = 1 } }

	set_temp_variable = { cg_gap = ROOT.cg_vehicles_expected_tier }
	subtract_from_temp_variable = { cg_gap = ROOT.cg_vehicles_supplied_tier }
	if = { limit = { check_variable = { cg_gap < 0 } } set_temp_variable = { cg_gap = 0 } }
	if = { limit = { check_variable = { cg_gap > 5 } } set_temp_variable = { cg_gap = 5 } }

	set_temp_variable = { cg_outdated = cg_gap }
	divide_temp_variable = { cg_outdated = 5 }

	multiply_temp_variable = { cg_severity = 0.70 }

	# ============================================================
	set_temp_variable = { cg_shortage = 1 }

	if = { limit = { check_variable = { cg_shortage < 0 } } set_temp_variable = { cg_shortage = 0 } }
	if = { limit = { check_variable = { cg_shortage > 1 } } set_temp_variable = { cg_shortage = 1 } }

	if = { limit = { check_variable = { cg_reliance < 0 } } set_temp_variable = { cg_reliance = 0 } }
	if = { limit = { check_variable = { cg_reliance > 1 } } set_temp_variable = { cg_reliance = 1 } }

	if = { limit = { check_variable = { cg_gap > 5 } } set_temp_variable = { cg_gap = 5 } }

	set_temp_variable = { cg_outdated = cg_gap }
	divide_temp_variable = { cg_outdated = 5 }

	multiply_temp_variable = { cg_severity = cg_reliance }
	multiply_temp_variable = { cg_severity = 1.20 }
	add_to_temp_variable = { cg_severity = cg_shortage }
	if = { limit = { check_variable = { cg_severity > 1 } } set_temp_variable = { cg_severity = 1 } }

	set_temp_variable = { sev_electronics = cg_severity }


	# ============================================================
	# MASTER scalar (overall legitimacy pressure)
	# ============================================================
	set_temp_variable = { master_scalar = 0 }

	set_temp_variable = { cg_tmp = sev_basic }
	multiply_temp_variable = { cg_tmp = 0.35 }
	add_to_temp_variable = { master_scalar = cg_tmp }

	set_temp_variable = { cg_tmp = sev_durable }
	multiply_temp_variable = { cg_tmp = 0.20 }
	add_to_temp_variable = { master_scalar = cg_tmp }

	set_temp_variable = { cg_tmp = sev_vehicles }
	multiply_temp_variable = { cg_tmp = 0.25 }
	add_to_temp_variable = { master_scalar = cg_tmp }

	set_temp_variable = { cg_tmp = sev_electronics }
	multiply_temp_variable = { cg_tmp = 0.20 }
	add_to_temp_variable = { master_scalar = cg_tmp }

	if = { limit = { check_variable = { master_scalar > 1 } } set_temp_variable = { master_scalar = 1 } }


	# ============================================================
	# Write dynamic modifier variables
	# ============================================================

	# Politics
	set_variable = { CWIC_DM_CG_STABILITY_FACTOR = master_scalar }
	multiply_variable = { CWIC_DM_CG_STABILITY_FACTOR = -0.18 }

	# --------------------------------------------
	# Black market stability penalty
	# CWIC_bm_stability_penalty is a 0..1-ish penalty value (ex: 0.015)
	# We add it directly to the stability factor variable as a negative.
	# --------------------------------------------

	# Safe init
	if = { limit = { NOT = { has_variable = CWIC_bm_stability_penalty } } set_variable = { CWIC_bm_stability_penalty = 0 } }

	# Clamp safety (keep it sane if tuning goes wild)
	if = { limit = { check_variable = { CWIC_bm_stability_penalty < 0 } } set_variable = { CWIC_bm_stability_penalty = 0 } }
	if = { limit = { check_variable = { CWIC_bm_stability_penalty > 0.20 } } set_variable = { CWIC_bm_stability_penalty = 0.20 } }

	# Apply as a direct negative contribution to stability factor
	subtract_from_variable = { CWIC_DM_CG_STABILITY_FACTOR = CWIC_bm_stability_penalty }

	set_variable = { CWIC_DM_CG_POLITICAL_POWER_GAIN = master_scalar }
	multiply_variable = { CWIC_DM_CG_POLITICAL_POWER_GAIN = -0.20 }

	# Workforce / welfare (Basic)
	set_variable = { CWIC_DM_CG_WORKFORCE_RATIO = sev_basic }
	multiply_variable = { CWIC_DM_CG_WORKFORCE_RATIO = -0.15 }

	set_variable = { CWIC_DM_CG_WELFARE_SPENDING_MOD = sev_basic }
	multiply_variable = { CWIC_DM_CG_WELFARE_SPENDING_MOD = 0.18 }

	# Vehicles / state reach
	set_variable = { CWIC_DM_CG_CONSTRUCTION_SPEED_FACTOR = sev_vehicles }
	multiply_variable = { CWIC_DM_CG_CONSTRUCTION_SPEED_FACTOR = -0.18 }

	set_variable = { CWIC_DM_CG_POLICING_SPENDING_MOD = sev_vehicles }
	multiply_variable = { CWIC_DM_CG_POLICING_SPENDING_MOD = 0.15 }

	set_variable = { CWIC_DM_CG_PUBLIC_SERVICES_SPENDING_MOD = sev_vehicles }
	multiply_variable = { CWIC_DM_CG_PUBLIC_SERVICES_SPENDING_MOD = 0.10 }

	set_variable = { CWIC_DM_CG_OTHER_RESOURCES_INCOME_BONUS = sev_vehicles }
	multiply_variable = { CWIC_DM_CG_OTHER_RESOURCES_INCOME_BONUS = -0.08 }

	# Durables / capital base
	set_variable = { CWIC_DM_CG_INDUSTRIAL_PARK_INCOME_BONUS = sev_durable }
	multiply_variable = { CWIC_DM_CG_INDUSTRIAL_PARK_INCOME_BONUS = -0.085 }

	# Electronics / services
	set_variable = { CWIC_DM_CG_OFFICE_PARK_INCOME_BONUS = sev_electronics }
	multiply_variable = { CWIC_DM_CG_OFFICE_PARK_INCOME_BONUS = -0.15 }

	set_variable = { CWIC_DM_CG_EDUCATION_GAIN_RATE = sev_electronics }
	multiply_variable = { CWIC_DM_CG_EDUCATION_GAIN_RATE = -0.0024 }

	# Productivity (blended)
	set_variable = { CWIC_DM_CG_PRODUCTIVITY_FACTOR = 0 }

	set_temp_variable = { cg_tmp = sev_basic }
	multiply_temp_variable = { cg_tmp = 0.40 }
	add_to_variable = { CWIC_DM_CG_PRODUCTIVITY_FACTOR = cg_tmp }

	set_temp_variable = { cg_tmp = sev_durable }
	multiply_temp_variable = { cg_tmp = 0.30 }
	add_to_variable = { CWIC_DM_CG_PRODUCTIVITY_FACTOR = cg_tmp }

	set_temp_variable = { cg_tmp = sev_vehicles }
	multiply_temp_variable = { cg_tmp = 0.20 }
	add_to_variable = { CWIC_DM_CG_PRODUCTIVITY_FACTOR = cg_tmp }

	set_temp_variable = { cg_tmp = sev_electronics }
	multiply_temp_variable = { cg_tmp = 0.10 }
	add_to_variable = { CWIC_DM_CG_PRODUCTIVITY_FACTOR = cg_tmp }

	if = { limit = { check_variable = { CWIC_DM_CG_PRODUCTIVITY_FACTOR > 1 } } set_variable = { CWIC_DM_CG_PRODUCTIVITY_FACTOR = 1 } }
	multiply_variable = { CWIC_DM_CG_PRODUCTIVITY_FACTOR = -0.16 }

	#black market strains enforcement and services
	add_to_variable = { CWIC_DM_CG_POLICING_SPENDING_MOD = CWIC_bm_tax_leak_mult }
	add_to_variable = { CWIC_DM_CG_PUBLIC_SERVICES_SPENDING_MOD = CWIC_bm_tax_leak_mult }


	# ============================================================
	# Apply dynamic modifier
	# ============================================================
	remove_dynamic_modifier = cwic_dm_consumer_goods_pressure_master
	add_dynamic_modifier = {
		modifier = cwic_dm_consumer_goods_pressure_master
		value = 1
	}
}

calculate_civilian_factory_production_total_factories = {

    # Start with factories the player can use for construction
    set_variable = {
        Actual_Civilian_Factories = num_of_available_civilian_factories
    }

    # Compute factories locked to consumer goods
    set_temp_variable = { cg_locked = num_of_civilian_factories }
    multiply_temp_variable = { cg_locked = modifier@consumer_goods_factor }
    round_temp_variable = cg_locked

    # Add locked consumer goods factories back in
    add_to_variable = {
        Actual_Civilian_Factories = cg_locked
    }

    # Safety clamp (rounding can occasionally overshoot by 1)
    if = {
        limit = { check_variable = { Actual_Civilian_Factories > num_of_civilian_factories } }
        set_variable = { Actual_Civilian_Factories = num_of_civilian_factories }
    }

    clear_variable = cg_locked
}


cwic_calculate_consumer_goods_monthly_demand = {

	# ============================================================
	# Inputs expected on ROOT:
	# - National_HDI (0.0..1.0)
	# - employable_population_sum (in thousands)
	# - current_year (int, e.g. 1952)
	# Outputs:
	# - cg_*_demand (float)
	# - cg_*_expected_tier (int 0..5; electronics 1..5)
	# ============================================================

	# ----------------------------
	# Clamp HDI defensively
	# ----------------------------
	set_temp_variable = { cg_hdi = ROOT.National_HDI }
	if = { limit = { check_variable = { cg_hdi < 0 } } set_temp_variable = { cg_hdi = 0 } }
	if = { limit = { check_variable = { cg_hdi > 1 } } set_temp_variable = { cg_hdi = 1 } }

	# Workforce (thousands)
	set_temp_variable = { cg_pop_k = ROOT.employable_population_sum }
	if = { limit = { check_variable = { cg_pop_k < 0 } } set_temp_variable = { cg_pop_k = 0 } }

	# ============================================================
	# 1) Base decade tier from year
	# Tier mapping:
	# 0 = Prewar
	# 1 = 1950s
	# 2 = 1960s
	# 3 = 1970s
	# 4 = 1980s
	# 5 = 1990s+
	# ============================================================
	set_temp_variable = { cg_decade_tier = 1 } # default 1950s baseline

	if = {
		limit = { check_variable = { ROOT.current_year < 1945 } }
		set_temp_variable = { cg_decade_tier = 0 }
	}
	else_if = {
		limit = { check_variable = { ROOT.current_year < 1960 } }
		set_temp_variable = { cg_decade_tier = 1 }
	}
	else_if = {
		limit = { check_variable = { ROOT.current_year < 1970 } }
		set_temp_variable = { cg_decade_tier = 2 }
	}
	else_if = {
		limit = { check_variable = { ROOT.current_year < 1980 } }
		set_temp_variable = { cg_decade_tier = 3 }
	}
	else_if = {
		limit = { check_variable = { ROOT.current_year < 1990 } }
		set_temp_variable = { cg_decade_tier = 4 }
	}
	else = {
		set_temp_variable = { cg_decade_tier = 5 }
	}

	# ============================================================
	# 2) HDI adjustment to expected tier (soft push)
	# Low HDI tends to accept older standards, high HDI expects newer.
	# - HDI < 0.35 : -1 tier
	# - HDI > 0.80 : +1 tier
	# ============================================================
	set_temp_variable = { cg_hdi_tier_adj = 0 }

	if = {
		limit = { check_variable = { cg_hdi < 0.35 } }
		set_temp_variable = { cg_hdi_tier_adj = -1 }
	}
	else_if = {
		limit = { check_variable = { cg_hdi > 0.80 } }
		set_temp_variable = { cg_hdi_tier_adj = 1 }
	}

	# ============================================================
	# 3) Expected tiers by good type
	# Basics/durables/vehicles can be tier 0..5.
	# Consumer electronics starts at tier 1 (no tier 0 expectations).
	# ============================================================
	set_variable = { cg_basic_expected_tier = cg_decade_tier }
	add_to_variable = { cg_basic_expected_tier = cg_hdi_tier_adj }
	if = { limit = { check_variable = { cg_basic_expected_tier < 0 } } set_variable = { cg_basic_expected_tier = 0 } }
	if = { limit = { check_variable = { cg_basic_expected_tier > 5 } } set_variable = { cg_basic_expected_tier = 5 } }

	set_variable = { cg_durable_expected_tier = cg_decade_tier }
	add_to_variable = { cg_durable_expected_tier = cg_hdi_tier_adj }
	if = { limit = { check_variable = { cg_durable_expected_tier < 0 } } set_variable = { cg_durable_expected_tier = 0 } }
	if = { limit = { check_variable = { cg_durable_expected_tier > 5 } } set_variable = { cg_durable_expected_tier = 5 } }

	set_variable = { cg_vehicles_expected_tier = cg_decade_tier }
	add_to_variable = { cg_vehicles_expected_tier = cg_hdi_tier_adj }
	if = { limit = { check_variable = { cg_vehicles_expected_tier < 0 } } set_variable = { cg_vehicles_expected_tier = 0 } }
	if = { limit = { check_variable = { cg_vehicles_expected_tier > 5 } } set_variable = { cg_vehicles_expected_tier = 5 } }

	set_variable = { cg_consumer_electronics_expected_tier = cg_decade_tier }
	add_to_variable = { cg_consumer_electronics_expected_tier = cg_hdi_tier_adj }
	if = { limit = { check_variable = { cg_consumer_electronics_expected_tier < 1 } } set_variable = { cg_consumer_electronics_expected_tier = 1 } }
	if = { limit = { check_variable = { cg_consumer_electronics_expected_tier > 5 } } set_variable = { cg_consumer_electronics_expected_tier = 5 } }


	# ============================================================
	# 4) Demand calculation framework
	# Demand = pop_k * base_rate * hdi_scale * tier_multiplier
	#
	# Notes:
	# - base_rate is "units per thousand employable people per month"
	# - hdi_scale is linear here for performance and tuning simplicity
	# - tier_multiplier increases expectations over time (and modern categories rise faster)
	# ============================================================

	# Shared HDI scaling:
	# hdi_scale = 0.60 + (0.80 * HDI)  => 0.60..1.40
	set_temp_variable = { cg_hdi_scale = cg_hdi }
	multiply_temp_variable = { cg_hdi_scale = 0.80 }
	add_to_temp_variable = { cg_hdi_scale = 0.60 }


	# ----------------------------
	# BASIC GOODS demand
	# base_rate_basic = 0.12
	# tier multipliers: 0.80, 1.00, 1.05, 1.10, 1.15, 1.20
	# ----------------------------
	set_temp_variable = { cg_mult = 1.00 }
	if = { limit = { check_variable = { cg_basic_expected_tier = 0 } } set_temp_variable = { cg_mult = 0.80 } }
	else_if = { limit = { check_variable = { cg_basic_expected_tier = 1 } } set_temp_variable = { cg_mult = 1.00 } }
	else_if = { limit = { check_variable = { cg_basic_expected_tier = 2 } } set_temp_variable = { cg_mult = 1.05 } }
	else_if = { limit = { check_variable = { cg_basic_expected_tier = 3 } } set_temp_variable = { cg_mult = 1.10 } }
	else_if = { limit = { check_variable = { cg_basic_expected_tier = 4 } } set_temp_variable = { cg_mult = 1.15 } }
	else = { set_temp_variable = { cg_mult = 1.20 } }

	set_variable = { cg_basic_demand = cg_pop_k }
	multiply_variable = { cg_basic_demand = 0.12 }
	multiply_variable = { cg_basic_demand = cg_hdi_scale }
	multiply_variable = { cg_basic_demand = cg_mult }

    multiply_variable = { cg_basic_demand = 0.01 }
    if = { limit = { check_variable = { cg_basic_demand < 1 } } set_variable = { cg_basic_demand = 1 } }


	# ----------------------------
	# DURABLE GOODS demand
	# base_rate_durable = 0.05
	# tier multipliers: 0.50, 0.80, 1.00, 1.15, 1.30, 1.45
	# ----------------------------
	set_temp_variable = { cg_mult = 1.00 }
	if = { limit = { check_variable = { cg_durable_expected_tier = 0 } } set_temp_variable = { cg_mult = 0.50 } }
	else_if = { limit = { check_variable = { cg_durable_expected_tier = 1 } } set_temp_variable = { cg_mult = 0.80 } }
	else_if = { limit = { check_variable = { cg_durable_expected_tier = 2 } } set_temp_variable = { cg_mult = 1.00 } }
	else_if = { limit = { check_variable = { cg_durable_expected_tier = 3 } } set_temp_variable = { cg_mult = 1.15 } }
	else_if = { limit = { check_variable = { cg_durable_expected_tier = 4 } } set_temp_variable = { cg_mult = 1.30 } }
	else = { set_temp_variable = { cg_mult = 1.45 } }

	set_variable = { cg_durable_demand = cg_pop_k }
	multiply_variable = { cg_durable_demand = 0.05 }
	multiply_variable = { cg_durable_demand = cg_hdi_scale }
	multiply_variable = { cg_durable_demand = cg_mult }

    multiply_variable = { cg_durable_demand = 0.01 }


	# ----------------------------
	# VEHICLES demand
	# base_rate_vehicles = 0.02
	# tier multipliers: 0.40, 0.70, 0.90, 1.05, 1.20, 1.35
	# ----------------------------
	set_temp_variable = { cg_mult = 1.00 }
	if = { limit = { check_variable = { cg_vehicles_expected_tier = 0 } } set_temp_variable = { cg_mult = 0.40 } }
	else_if = { limit = { check_variable = { cg_vehicles_expected_tier = 1 } } set_temp_variable = { cg_mult = 0.70 } }
	else_if = { limit = { check_variable = { cg_vehicles_expected_tier = 2 } } set_temp_variable = { cg_mult = 0.90 } }
	else_if = { limit = { check_variable = { cg_vehicles_expected_tier = 3 } } set_temp_variable = { cg_mult = 1.05 } }
	else_if = { limit = { check_variable = { cg_vehicles_expected_tier = 4 } } set_temp_variable = { cg_mult = 1.20 } }

	else_if = { limit = { check_variable = { cg_consumer_electronics_expected_tier = 4 } } set_temp_variable = { cg_mult = 1.40 } }
	else = { set_temp_variable = { cg_mult = 1.70 } }

	set_temp_variable = { cg_elec_hdi_scale = cg_hdi }
	multiply_temp_variable = { cg_elec_hdi_scale = 1.20 }
	add_to_temp_variable = { cg_elec_hdi_scale = 0.30 }

	set_variable = { cg_consumer_electronics_demand = cg_pop_k }
	multiply_variable = { cg_consumer_electronics_demand = 0.015 }
	multiply_variable = { cg_consumer_electronics_demand = cg_elec_hdi_scale }
	multiply_variable = { cg_consumer_electronics_demand = cg_mult }

    multiply_variable = { cg_consumer_electronics_demand = 0.01 }

	# Cleanup temps
	clear_variable = cg_hdi
	clear_variable = cg_pop_k
	clear_variable = cg_decade_tier
	clear_variable = cg_hdi_tier_adj
	clear_variable = cg_hdi_scale

cwic_calculate_consumer_goods_monthly_demand = {

	# Inputs expected on ROOT:
	# - National_HDI (0.0..1.0)
	# - employable_population_sum (in thousands)
	# Outputs:
	# - cg_*_demand (float)
	# - cg_*_expected_tier (int 0..5; electronics 1..5)

	# ----------------------------
	# Clamp HDI defensively
	# ----------------------------
	set_temp_variable = { cg_hdi = ROOT.National_HDI }
	if = { limit = { check_variable = { cg_hdi < 0 } } set_temp_variable = { cg_hdi = 0 } }
	if = { limit = { check_variable = { cg_hdi > 1 } } set_temp_variable = { cg_hdi = 1 } }

	# Workforce (thousands)
	set_temp_variable = { cg_pop_k = ROOT.employable_population_sum }
	if = { limit = { check_variable = { cg_pop_k < 0 } } set_temp_variable = { cg_pop_k = 0 } }


	# ============================================================
	# 1) Base decade tier from date
	# Tier mapping:
	# 0 = Prewar (< 1950)
	# 1 = 1950s
	# 2 = 1960s
	# 3 = 1970s
	# 4 = 1980s
	# 5 = 1990s+
	# ============================================================
	set_temp_variable = { cg_decade_tier = 1 }

	if = {
		limit = { date < 1950.01.01 }
		set_temp_variable = { cg_decade_tier = 0 }
	}
	else_if = {
		limit = { date < 1960.01.01 }
		set_temp_variable = { cg_decade_tier = 1 }
	}
	else_if = {
		limit = { date < 1970.01.01 }
		set_temp_variable = { cg_decade_tier = 2 }
	}
	else_if = {
		limit = { date < 1980.01.01 }
		set_temp_variable = { cg_decade_tier = 3 }
	}
	else_if = {
		limit = { date < 1990.01.01 }
		set_temp_variable = { cg_decade_tier = 4 }
	}
	else = {
		set_temp_variable = { cg_decade_tier = 5 }
	}


	# ============================================================
	# 2) HDI adjustment to expected tier (soft push)
	# ============================================================
	set_temp_variable = { cg_hdi_tier_adj = 0 }

	if = {
		limit = { check_variable = { cg_hdi < 0.35 } }
		set_temp_variable = { cg_hdi_tier_adj = -1 }
	}
	else_if = {
		limit = { check_variable = { cg_hdi > 0.80 } }
		set_temp_variable = { cg_hdi_tier_adj = 1 }
	}


	# ============================================================
	# 3) Expected tiers by good type
	# ============================================================
	set_variable = { cg_basic_expected_tier = cg_decade_tier }
	add_to_variable = { cg_basic_expected_tier = cg_hdi_tier_adj }
	if = { limit = { check_variable = { cg_basic_expected_tier < 0 } } set_variable = { cg_basic_expected_tier = 0 } }
	if = { limit = { check_variable = { cg_basic_expected_tier > 5 } } set_variable = { cg_basic_expected_tier = 5 } }

	set_variable = { cg_durable_expected_tier = cg_decade_tier }
	add_to_variable = { cg_durable_expected_tier = cg_hdi_tier_adj }
	if = { limit = { check_variable = { cg_durable_expected_tier < 0 } } set_variable = { cg_durable_expected_tier = 0 } }
	if = { limit = { check_variable = { cg_durable_expected_tier > 5 } } set_variable = { cg_durable_expected_tier = 5 } }

	set_variable = { cg_vehicles_expected_tier = cg_decade_tier }
	add_to_variable = { cg_vehicles_expected_tier = cg_hdi_tier_adj }
	if = { limit = { check_variable = { cg_vehicles_expected_tier < 0 } } set_variable = { cg_vehicles_expected_tier = 0 } }
	if = { limit = { check_variable = { cg_vehicles_expected_tier > 5 } } set_variable = { cg_vehicles_expected_tier = 5 } }

	set_variable = { cg_consumer_electronics_expected_tier = cg_decade_tier }
	add_to_variable = { cg_consumer_electronics_expected_tier = cg_hdi_tier_adj }
	if = { limit = { check_variable = { cg_consumer_electronics_expected_tier < 1 } } set_variable = { cg_consumer_electronics_expected_tier = 1 } }
	if = { limit = { check_variable = { cg_consumer_electronics_expected_tier > 5 } } set_variable = { cg_consumer_electronics_expected_tier = 5 } }


	# ============================================================
	# 4) Demand calculation framework
	# Demand = pop_k * base_rate * hdi_scale * tier_multiplier
	# ============================================================

	# Shared HDI scaling: 0.60..1.40
	set_temp_variable = { cg_hdi_scale = cg_hdi }
	multiply_temp_variable = { cg_hdi_scale = 0.80 }
	add_to_temp_variable = { cg_hdi_scale = 0.60 }


	# BASIC
	set_temp_variable = { cg_mult = 1.00 }
	if = { limit = { check_variable = { cg_basic_expected_tier = 0 } } set_temp_variable = { cg_mult = 0.80 } }
	else_if = { limit = { check_variable = { cg_basic_expected_tier = 1 } } set_temp_variable = { cg_mult = 1.00 } }
	else_if = { limit = { check_variable = { cg_basic_expected_tier = 2 } } set_temp_variable = { cg_mult = 1.05 } }
	else_if = { limit = { check_variable = { cg_basic_expected_tier = 3 } } set_temp_variable = { cg_mult = 1.10 } }
	else_if = { limit = { check_variable = { cg_basic_expected_tier = 4 } } set_temp_variable = { cg_mult = 1.15 } }
	else = { set_temp_variable = { cg_mult = 1.20 } }

	multiply_variable = { cg_basic_demand = 0.12 }
	multiply_variable = { cg_basic_demand = cg_hdi_scale }

	else_if = { limit = { check_variable = { cg_durable_expected_tier = 4 } } set_temp_variable = { cg_mult = 1.30 } }
	else = { set_temp_variable = { cg_mult = 1.45 } }

	set_variable = { cg_durable_demand = cg_pop_k }
	multiply_variable = { cg_durable_demand = 0.05 }
	multiply_variable = { cg_durable_demand = cg_hdi_scale }
	multiply_variable = { cg_durable_demand = cg_mult }


	# VEHICLES
	set_temp_variable = { cg_mult = 1.00 }
	if = { limit = { check_variable = { cg_vehicles_expected_tier = 0 } } set_temp_variable = { cg_mult = 0.40 } }
	else_if = { limit = { check_variable = { cg_vehicles_expected_tier = 1 } } set_temp_variable = { cg_mult = 0.70 } }
	else_if = { limit = { check_variable = { cg_vehicles_expected_tier = 2 } } set_temp_variable = { cg_mult = 0.90 } }
	else_if = { limit = { check_variable = { cg_vehicles_expected_tier = 3 } } set_temp_variable = { cg_mult = 1.05 } }
	else_if = { limit = { check_variable = { cg_vehicles_expected_tier = 4 } } set_temp_variable = { cg_mult = 1.20 } }
	else = { set_temp_variable = { cg_mult = 1.35 } }

	set_variable = { cg_vehicles_demand = cg_pop_k }
	multiply_variable = { cg_vehicles_demand = 0.02 }
	multiply_variable = { cg_vehicles_demand = cg_hdi_scale }
	multiply_variable = { cg_vehicles_demand = cg_mult }

	else_if = { limit = { check_variable = { cg_consumer_electronics_expected_tier = 3 } } set_temp_variable = { cg_mult = 1.15 } }
	else_if = { limit = { check_variable = { cg_consumer_electronics_expected_tier = 4 } } set_temp_variable = { cg_mult = 1.40 } }
	else = { set_temp_variable = { cg_mult = 1.70 } }

cwic_calculate_consumer_goods_coverage_and_consume = {

	# ============================================================
	# Helper mini-block pattern (inline, repeated):
	# - cg_need: remaining demand
	# - cg_take: amount pulled from a tier stockpile
	# - cg_stock: the tier stockpile value
	# ============================================================


	# ============================================================
	# BASIC GOODS
	# ============================================================
	set_variable = { cg_basic_provided = 0 }
	set_variable = { cg_basic_supplied_tier = 0 }
	set_variable = { cg_basic_coverage = 1 }

	set_temp_variable = { cg_need = ROOT.cg_basic_demand }
	if = { limit = { check_variable = { cg_need < 0 } } set_temp_variable = { cg_need = 0 } }

	set_temp_variable = { cg_weighted_sum = 0 }

	# If no demand, keep coverage = 1 and tier = expected
	if = {
		limit = { check_variable = { cg_need < 0.01 } }
		set_variable = { cg_basic_supplied_tier = ROOT.cg_basic_expected_tier }
	}
	else = {

		# Tier 5
		set_temp_variable = { cg_stock = ROOT.cg_basic_stock_t5 }
		if = { limit = { check_variable = { cg_stock < 0 } } set_temp_variable = { cg_stock = 0 } }
		set_temp_variable = { cg_take = cg_stock }
		if = { limit = { check_variable = { cg_take > cg_need } } set_temp_variable = { cg_take = cg_need } }
		subtract_from_variable = { cg_basic_stock_t5 = cg_take }
		subtract_from_temp_variable = { cg_need = cg_take }
		add_to_variable = { cg_basic_provided = cg_take }
		set_temp_variable = { cg_tmp = cg_take }
		multiply_temp_variable = { cg_tmp = 5 }
		add_to_temp_variable = { cg_weighted_sum = cg_tmp }

		# Tier 4
		if = { limit = { check_variable = { cg_need > 0 } }
			set_temp_variable = { cg_stock = ROOT.cg_basic_stock_t4 }
			if = { limit = { check_variable = { cg_stock < 0 } } set_temp_variable = { cg_stock = 0 } }
			set_temp_variable = { cg_take = cg_stock }
			if = { limit = { check_variable = { cg_take > cg_need } } set_temp_variable = { cg_take = cg_need } }
			subtract_from_variable = { cg_basic_stock_t4 = cg_take }
			subtract_from_temp_variable = { cg_need = cg_take }
			add_to_variable = { cg_basic_provided = cg_take }
			set_temp_variable = { cg_tmp = cg_take }
			multiply_temp_variable = { cg_tmp = 4 }
			add_to_temp_variable = { cg_weighted_sum = cg_tmp }
		}

		# Tier 3
		if = { limit = { check_variable = { cg_need > 0 } }
			set_temp_variable = { cg_stock = ROOT.cg_basic_stock_t3 }
			if = { limit = { check_variable = { cg_stock < 0 } } set_temp_variable = { cg_stock = 0 } }
			set_temp_variable = { cg_take = cg_stock }
			if = { limit = { check_variable = { cg_take > cg_need } } set_temp_variable = { cg_take = cg_need } }
			subtract_from_variable = { cg_basic_stock_t3 = cg_take }
			subtract_from_temp_variable = { cg_need = cg_take }
			add_to_variable = { cg_basic_provided = cg_take }
			set_temp_variable = { cg_tmp = cg_take }
			multiply_temp_variable = { cg_tmp = 3 }
			add_to_temp_variable = { cg_weighted_sum = cg_tmp }
		}

		# Tier 2
		if = { limit = { check_variable = { cg_need > 0 } }
			set_temp_variable = { cg_stock = ROOT.cg_basic_stock_t2 }
			if = { limit = { check_variable = { cg_stock < 0 } } set_temp_variable = { cg_stock = 0 } }
			set_temp_variable = { cg_take = cg_stock }
			if = { limit = { check_variable = { cg_take > cg_need } } set_temp_variable = { cg_take = cg_need } }
			subtract_from_variable = { cg_basic_stock_t2 = cg_take }
			subtract_from_temp_variable = { cg_need = cg_take }
			add_to_variable = { cg_basic_provided = cg_take }
			set_temp_variable = { cg_tmp = cg_take }
			multiply_temp_variable = { cg_tmp = 2 }
			add_to_temp_variable = { cg_weighted_sum = cg_tmp }
		}

		# Tier 1
		if = { limit = { check_variable = { cg_need > 0 } }
			set_temp_variable = { cg_stock = ROOT.cg_basic_stock_t1 }
			if = { limit = { check_variable = { cg_stock < 0 } } set_temp_variable = { cg_stock = 0 } }
			set_temp_variable = { cg_take = cg_stock }
			if = { limit = { check_variable = { cg_take > cg_need } } set_temp_variable = { cg_take = cg_need } }
			subtract_from_variable = { cg_basic_stock_t1 = cg_take }
			subtract_from_temp_variable = { cg_need = cg_take }
			add_to_variable = { cg_basic_provided = cg_take }
			set_temp_variable = { cg_tmp = cg_take }
			multiply_temp_variable = { cg_tmp = 1 }
			add_to_temp_variable = { cg_weighted_sum = cg_tmp }
		}

		# Tier 0
		if = { limit = { check_variable = { cg_need > 0 } }
			set_temp_variable = { cg_stock = ROOT.cg_basic_stock_t0 }
			if = { limit = { check_variable = { cg_stock < 0 } } set_temp_variable = { cg_stock = 0 } }
			set_temp_variable = { cg_take = cg_stock }
			if = { limit = { check_variable = { cg_take > cg_need } } set_temp_variable = { cg_take = cg_need } }
			subtract_from_variable = { cg_basic_stock_t0 = cg_take }
			subtract_from_temp_variable = { cg_need = cg_take }
			add_to_variable = { cg_basic_provided = cg_take }
			# tier 0 adds nothing to weighted sum
		}

		# coverage = provided / demand
		set_variable = { cg_basic_coverage = cg_basic_provided }
		divide_variable = { cg_basic_coverage = ROOT.cg_basic_demand }
		if = { limit = { check_variable = { cg_basic_coverage < 0 } } set_variable = { cg_basic_coverage = 0 } }
		if = { limit = { check_variable = { cg_basic_coverage > 2 } } set_variable = { cg_basic_coverage = 2 } }

		# supplied tier = weighted_sum / provided
		if = {
			limit = { check_variable = { ROOT.cg_basic_provided > 0 } }
			set_variable = { cg_basic_supplied_tier = cg_weighted_sum }
			divide_variable = { cg_basic_supplied_tier = ROOT.cg_basic_provided }
			round_variable = cg_basic_supplied_tier
			if = { limit = { check_variable = { cg_basic_supplied_tier < 0 } } set_variable = { cg_basic_supplied_tier = 0 } }
			if = { limit = { check_variable = { cg_basic_supplied_tier > 5 } } set_variable = { cg_basic_supplied_tier = 5 } }
		}
		else = {
			set_variable = { cg_basic_supplied_tier = 0 }
			set_variable = { cg_basic_coverage = 0 }
		}
	}

	# ============================================================
	# DURABLE GOODS
	# ============================================================
	set_variable = { cg_durable_provided = 0 }
	set_variable = { cg_durable_supplied_tier = 0 }
	set_variable = { cg_durable_coverage = 1 }

	set_temp_variable = { cg_need = ROOT.cg_durable_demand }

		add_to_variable = { cg_durable_provided = cg_take }
		set_temp_variable = { cg_tmp = cg_take }
		multiply_temp_variable = { cg_tmp = 5 }
		add_to_temp_variable = { cg_weighted_sum = cg_tmp }

		# Tier 4
		if = { limit = { check_variable = { cg_need > 0 } }
			set_temp_variable = { cg_stock = ROOT.cg_durable_stock_t4 }
			if = { limit = { check_variable = { cg_stock < 0 } } set_temp_variable = { cg_stock = 0 } }
			set_temp_variable = { cg_take = cg_stock }
			if = { limit = { check_variable = { cg_take > cg_need } } set_temp_variable = { cg_take = cg_need } }
			subtract_from_variable = { cg_durable_stock_t4 = cg_take }
			subtract_from_temp_variable = { cg_need = cg_take }
			add_to_variable = { cg_durable_provided = cg_take }
			set_temp_variable = { cg_tmp = cg_take }
			multiply_temp_variable = { cg_tmp = 4 }
			add_to_temp_variable = { cg_weighted_sum = cg_tmp }
		}

		# Tier 3
		if = { limit = { check_variable = { cg_need > 0 } }
			set_temp_variable = { cg_stock = ROOT.cg_durable_stock_t3 }
			if = { limit = { check_variable = { cg_stock < 0 } } set_temp_variable = { cg_stock = 0 } }
			set_temp_variable = { cg_take = cg_stock }
			if = { limit = { check_variable = { cg_take > cg_need } } set_temp_variable = { cg_take = cg_need } }
			subtract_from_variable = { cg_durable_stock_t3 = cg_take }
			subtract_from_temp_variable = { cg_need = cg_take }
			add_to_variable = { cg_durable_provided = cg_take }
			set_temp_variable = { cg_tmp = cg_take }
			multiply_temp_variable = { cg_tmp = 3 }
			add_to_temp_variable = { cg_weighted_sum = cg_tmp }
		}

		# Tier 2
		if = { limit = { check_variable = { cg_need > 0 } }
			set_temp_variable = { cg_stock = ROOT.cg_durable_stock_t2 }
			if = { limit = { check_variable = { cg_stock < 0 } } set_temp_variable = { cg_stock = 0 } }
			set_temp_variable = { cg_take = cg_stock }
			if = { limit = { check_variable = { cg_take > cg_need } } set_temp_variable = { cg_take = cg_need } }
			subtract_from_variable = { cg_durable_stock_t2 = cg_take }
			subtract_from_temp_variable = { cg_need = cg_take }
			add_to_variable = { cg_durable_provided = cg_take }

	# - Tier 2: 1960s
	# - Tier 3: 1970s
	# - Tier 4: 1980s
	# - Tier 5: 1990+
	# Consumer Electronics has no Tier 0 (min expected tier is 1)
	# ============================================================

	# BASIC expected tier
	set_temp_variable = { exp_basic = 0 }
	if = { limit = { date < 1950.01.01 } set_temp_variable = { exp_basic = 0 } }

# - CG_M_PRODUCTIVITY                     (negative or small positive)

# - CG_M_OFFICE_PARK_INCOME_BONUS         (negative, cap -0.15)

		add_to_temp_variable = { cap_total = ROOT.cg_consumer_electronics_stock_t2 }

		add_to_temp_variable = { cap_total = ROOT.cg_consumer_electronics_stock_t4 }
		add_to_temp_variable = { cap_total = ROOT.cg_consumer_electronics_stock_t5 }

		if = {
			limit = { check_variable = { cap_total > cap_limit } }

			set_temp_variable = { cap_over = cap_total }
			subtract_from_temp_variable = { cap_over = cap_limit }

			set_temp_variable = { cap_take = ROOT.cg_consumer_electronics_stock_t1 }
			if = { limit = { check_variable = { cap_take > cap_over } } set_temp_variable = { cap_take = cap_over } }

				set_temp_variable = { cap_take = ROOT.cg_consumer_electronics_stock_t2 }
				if = { limit = { check_variable = { cap_take > cap_over } } set_temp_variable = { cap_take = cap_over } }
				subtract_from_variable = { cg_consumer_electronics_stock_t2 = cap_take }
				subtract_from_temp_variable = { cap_over = cap_take }
			}
			if = { limit = { check_variable = { cap_over > 0 } }
				set_temp_variable = { cap_take = ROOT.cg_consumer_electronics_stock_t3 }
				if = { limit = { check_variable = { cap_take > cap_over } } set_temp_variable = { cap_take = cap_over } }
				subtract_from_variable = { cg_consumer_electronics_stock_t3 = cap_take }
				subtract_from_temp_variable = { cap_over = cap_take }
			}
			if = { limit = { check_variable = { cap_over > 0 } }
				set_temp_variable = { cap_take = ROOT.cg_consumer_electronics_stock_t4 }
				if = { limit = { check_variable = { cap_take > cap_over } } set_temp_variable = { cap_take = cap_over } }
				subtract_from_variable = { cg_consumer_electronics_stock_t4 = cap_take }
				subtract_from_temp_variable = { cap_over = cap_take }
			}
			if = { limit = { check_variable = { cap_over > 0 } }
				set_temp_variable = { cap_take = ROOT.cg_consumer_electronics_stock_t5 }
				if = { limit = { check_variable = { cap_take > cap_over } } set_temp_variable = { cap_take = cap_over } }
				subtract_from_variable = { cg_consumer_electronics_stock_t5 = cap_take }
				subtract_from_temp_variable = { cap_over = cap_take }
			}
		}

		# Cleanup cap temps
		clear_variable = cap_total
		clear_variable = cap_limit
		clear_variable = cap_over
		clear_variable = cap_take
	}
}

### ============================================================
### MONTHLY: Allocate production points by CG category
### Expects (temp vars): w_basic, w_durable, w_vehicles, w_elec
### Writes (country vars): cg_monthly_points_*
### ============================================================

cwic_cg_allocate_monthly_production_points = {
    set_variable = { cg_points_per_factory = 25 }
	# ------------------------------------------------------------
	# 0) Resolve tuning knob (points per factory)
	# ------------------------------------------------------------
	clear_variable = cg_ppf
	set_temp_variable = { cg_ppf = ROOT.cg_points_per_factory }

	# Default if missing or too small
	if = {
		limit = { check_variable = { cg_ppf < 0.1 } }
		set_temp_variable = { cg_ppf = 10 }
	}

	# ------------------------------------------------------------
	# 1) Total monthly points from "Actual_Civilian_Factories"
	# ------------------------------------------------------------
	clear_variable = total_points
	set_temp_variable = { total_points = ROOT.Actual_Civilian_Factories }
	if = { limit = { check_variable = { total_points < 0 } } set_temp_variable = { total_points = 0 } }

	multiply_temp_variable = { total_points = cg_ppf }

	# Keep points as whole numbers for downstream math
	round_temp_variable = total_points

	set_variable = { cg_monthly_points_total = total_points }

	# ------------------------------------------------------------
	# 2) Allocate points by normalized weights
	# ------------------------------------------------------------
	clear_variable = pts_basic
	clear_variable = pts_durable
	clear_variable = pts_vehicles
	clear_variable = pts_elec

	set_temp_variable = { pts_basic = total_points }
	multiply_temp_variable = { pts_basic = w_basic }

	set_temp_variable = { pts_durable = total_points }
	multiply_temp_variable = { pts_durable = w_durable }

	set_temp_variable = { pts_vehicles = total_points }
	multiply_temp_variable = { pts_vehicles = w_vehicles }

	set_temp_variable = { pts_elec = total_points }
	multiply_temp_variable = { pts_elec = w_elec }

	round_temp_variable = pts_basic
	round_temp_variable = pts_durable
	round_temp_variable = pts_vehicles
	round_temp_variable = pts_elec

	# ------------------------------------------------------------
	# 3) Rounding reconciliation
	# If rounding lost or gained points, shove the difference into BASIC.
	# ------------------------------------------------------------
	clear_variable = pts_sum
	clear_variable = pts_diff

	set_temp_variable = { pts_sum = pts_basic }
	add_to_temp_variable = { pts_sum = pts_durable }
	add_to_temp_variable = { pts_sum = pts_vehicles }
	add_to_temp_variable = { pts_sum = pts_elec }

	set_temp_variable = { pts_diff = total_points }
	subtract_from_temp_variable = { pts_diff = pts_sum }

	# If diff > 0, add to basic
	if = {
		limit = { check_variable = { pts_diff > 0 } }
		add_to_temp_variable = { pts_basic = pts_diff }
	}

	# If diff < 0, subtract from basic (clamped at 0)
	if = {
		limit = { check_variable = { pts_diff < 0 } }
		add_to_temp_variable = { pts_basic = pts_diff }
		if = { limit = { check_variable = { pts_basic < 0 } } set_temp_variable = { pts_basic = 0 } }
	}

	# ------------------------------------------------------------
	# 4) Write country variables
	# ------------------------------------------------------------
	set_variable = { cg_monthly_points_basic = pts_basic }
	set_variable = { cg_monthly_points_durable = pts_durable }
	set_variable = { cg_monthly_points_vehicles = pts_vehicles }
	set_variable = { cg_monthly_points_electronics = pts_elec }

	# ------------------------------------------------------------
	# 5) Cleanup temps used here (optional, matches your hygiene style)
	# ------------------------------------------------------------
	clear_variable = cg_ppf
	clear_variable = total_points
	clear_variable = pts_basic
	clear_variable = pts_durable
	clear_variable = pts_vehicles
	clear_variable = pts_elec
	clear_variable = pts_sum
	clear_variable = pts_diff
}
