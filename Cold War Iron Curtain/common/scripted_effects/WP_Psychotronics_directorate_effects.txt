# Warsaw Pact Psychotronics Directorate
#
# Shared values describe the quality and politics of a finite archive. They do
# not measure paranormal power. Every dossier contribution is guarded by a
# stable one-time flag so migration and save/load reconciliation are idempotent.

WP_psych_clamp_global_variables = {
	if = { limit = { has_variable = global.WP_psych_directorate_state } clamp_variable = { var = global.WP_psych_directorate_state min = 0 max = 4 } }
	if = { limit = { has_variable = global.WP_psych_record } clamp_variable = { var = global.WP_psych_record min = 0 max = 100 } }
	if = { limit = { has_variable = global.WP_psych_scrutiny } clamp_variable = { var = global.WP_psych_scrutiny min = 0 max = 100 } }
	if = { limit = { has_variable = global.WP_psych_coordination } clamp_variable = { var = global.WP_psych_coordination min = 0 max = 100 } }
	if = { limit = { has_variable = global.WP_psych_review_slot } clamp_variable = { var = global.WP_psych_review_slot min = 0 max = 9 } }
	if = {
		limit = { SOV_project_x_enabled_trigger = yes has_variable = global.WP_psych_anomaly_record }
		clamp_variable = { var = global.WP_psych_anomaly_record min = 0 max = 100 }
	}
}

WP_psych_evaluate_thresholds = {
	if = {
		limit = { has_global_flag = WP_psych_global_initialized }
		WP_psych_clamp_global_variables = yes
		if = {
			limit = {
				check_variable = { global.WP_psych_record > 24 }
				NOT = { has_global_flag = WP_psych_second_dossier_wave_open }
			}
			set_global_flag = WP_psych_second_dossier_wave_open
		}
		if = {
			limit = {
				check_variable = { global.WP_psych_scrutiny > 49 }
				check_variable = { global.WP_psych_directorate_state = 2 }
			}
			set_variable = { global.WP_psych_directorate_state = 3 }
			set_global_flag = WP_psych_formal_review_active
		}
		if = {
			limit = {
				check_variable = { global.WP_psych_scrutiny > 74 }
				check_variable = { global.WP_psych_directorate_state > 0 }
				check_variable = { global.WP_psych_directorate_state < 4 }
				NOT = { has_global_flag = WP_psych_funding_crisis_fired }
			}
			set_variable = { global.WP_psych_directorate_state = 3 }
			set_global_flag = WP_psych_formal_review_active
			set_global_flag = WP_psych_funding_crisis_active
			set_global_flag = WP_psych_funding_crisis_fired
			if = {
				limit = { country_exists = SOV }
				SOV = { country_event = { id = cwic_wp_directorate.20 } }
			}
		}
	}
}

WP_psych_global_initialize = {
	if = {
		limit = { NOT = { has_global_flag = WP_psych_global_initialized } }
		set_variable = { global.WP_psych_directorate_state = 0 }

		set_variable = { global.WP_psych_scrutiny = 0 }

		}
		set_global_flag = WP_psych_global_initialized

	WP_psych_clamp_global_variables = yes
}
