# ============================================================
# CWIC Division Limit System - Scripted Effects
# Based on Kaiserreich's unit limit system (used with credit)
# ============================================================

# --------------------------------------------------
# Calculate the maximum allowed divisions for a country
# Result stored in CWIC_max_divisions
# --------------------------------------------------
CWIC_calculate_maximum_divisions = {
	# Wartime: no limit - armies expand freely
	if = {
		limit = { has_war = yes }
		set_variable = { CWIC_max_divisions = 9999 }
	}
	else = {
		# --- Tiered base from GDP ---
		# Calibrated against: USA=2589, USSR=660, FRA=444, ENG=398, CHI=204, PRC=117
		set_variable = {
			CWIC_max_divisions = {
				value = 16										# Minor
				if = { limit = { value = grossdomesticproduct greater_than_or_equals = 50 } add = 14 }		# Secondary power: 30
				if = { limit = { value = grossdomesticproduct greater_than_or_equals = 150 } add = 20 }		# Regional power: 50
				if = { limit = { value = grossdomesticproduct greater_than_or_equals = 400 } add = 240 }		# Major power: 290
				if = { limit = { value = grossdomesticproduct greater_than_or_equals = 2000 } add = 50 }		# Superpower: 340

				# Army model law bonus:
				# volunteer_professional: +0   (small quality force, no bonus)
				# national_service:       +15  (mixed system)
				# mass_conscription:      +30  (large standing army)
				add = modifier@unit_limit_bonus

				# Percentage modifier from national spirits / focuses
				multiply = {
					value = 1
					add = modifier@unit_limit_factor
				}

				# Hard floor: no peacetime country drops below 5 divisions
				max = 5
				round = yes
			}
		}
	}
}

# --------------------------------------------------
# Calculate the mobilisation penalty scaling
# Penalty = how far over limit / 4, capped at 25%
# --------------------------------------------------
CWIC_calculate_mobilisation_penalty = {
	set_variable = {
		CWIC_mobilisation_penalty_var_positive = {
			value = CWIC_num_actual_divisions
			divide = CWIC_max_divisions
			subtract = 1
			multiply = 0.25
			clamp = { min = -0.25 max = 0.25 }
		}
	}
	# Negative version for org/reinforce debuffs
	set_variable = { CWIC_mobilisation_penalty_var = { value = CWIC_mobilisation_penalty_var_positive multiply = -1 } }
}

# --------------------------------------------------
# Daily recalculation - run for every country on_daily
# Applies/removes penalty for human player
# --------------------------------------------------
CWIC_unit_limit_calculation_effect = {
	CWIC_calculate_maximum_divisions = yes

	set_variable = { CWIC_num_actual_divisions = { value = num_divisions clamp = { min = 0 max = 10000 } } }

	# Only apply the penalty modifier to human players
	if = {
		limit = { is_ai = no }
		if = {
			limit = { has_dynamic_modifier = { modifier = CWIC_mobilisation_penalty_modifier } }
			if = {
				# At war or back under limit - remove penalty
				limit = {
					OR = {
						has_war = yes
						CWIC_has_exceeded_maximum_divisions = no
					}
				}
				clear_variable = CWIC_mobilisation_penalty_var
				clear_variable = CWIC_mobilisation_penalty_var_positive
				remove_dynamic_modifier = { modifier = CWIC_mobilisation_penalty_modifier }
			}
			else = {
				# Still over limit in peacetime - update scaling
				CWIC_calculate_mobilisation_penalty = yes
			}
		}
		else_if = {
			# Not at war, over limit, no penalty yet - apply it
			limit = {
				has_war = no
				CWIC_has_exceeded_maximum_divisions = yes
			}
			CWIC_calculate_mobilisation_penalty = yes
			add_dynamic_modifier = { modifier = CWIC_mobilisation_penalty_modifier }
		}
	}

	# GUI properties variable: 1 = under/at limit  |  2 = over limit
	set_variable = { CWIC_unit_limit_status_frame = { value = CWIC_num_actual_divisions subtract = CWIC_max_divisions greater_than = 0 add = 1 } }
}

# --------------------------------------------------
# Disband helper: use the permanent UNS utility tag as a unit
# "graveyard" instead of spawning a fresh dynamic country.
#
# Why: create_dynamic_country measured ~0.9s PER CALL in the profiler
# (8.9s over a single 80-day run) and left an orphan country behind on
# every disband, which also inflated the engine's native AI/intel passes.
# UNS already exists and owns territory at game start, so there is no
# allocation and no state to transfer back and forth. We just move the
# units in, disband them, and return the freed manpower to the owner.
#
# NOTE: delete_unit (in CWIC_disband_units_transfer_manpower) disbands
# ALL units owned by the scope. This is safe only because UNS holds no
# army units of its own. If UNS is ever given combat divisions, this
# trick must change (track transferred unit ids instead).
# Based on Kaiserreich's disband_units_setup (concept used with credit)
# --------------------------------------------------
CWIC_disband_units_setup = {
	set_temp_variable = { CWIC_disband_country = THIS }
	UNS = {
		set_temp_variable = { CWIC_disband_instance = THIS }
		set_temp_variable = { CWIC_old_manpower = manpower_k }
	}
}

CWIC_disband_units_transfer_manpower = {
	delete_unit = { disband = yes }

	set_temp_variable = { CWIC_manpower_diff = { value = manpower_k subtract = CWIC_old_manpower } }

	# Avoid overflow
	while_loop_effect = {
		limit = { check_variable = { CWIC_manpower_diff > 100 } }
		subtract_from_temp_variable = { CWIC_manpower_diff = 100 }
		var:CWIC_disband_country = { add_manpower = 100000 }
		add_manpower = -100000
	}

	multiply_temp_variable = { CWIC_manpower_diff = 1000 }
	var:CWIC_disband_country = { add_manpower = CWIC_manpower_diff }

	multiply_temp_variable = { CWIC_manpower_diff = -1 }
	add_manpower = CWIC_manpower_diff
}

# NOTE: CWIC_disband_units_finalise is no longer needed - the old version
# transferred the temporary impassable state back to its owner, but the
# UNS graveyard approach never moves a state, so there is nothing to undo.

# --------------------------------------------------
# Disband ~1% of the army (roughly 1-3 divisions)
# Called by AI event loop every 7 days until under limit
# Based on Kaiserreich's disband_one_to_three_divisions (used with credit)
# --------------------------------------------------
CWIC_disband_one_to_three_divisions = {
	hidden_effect = {
		CWIC_disband_units_setup = yes
		if = {
			limit = { CWIC_indochina_demob_pilot = yes }
			transfer_units_fraction = {
				target = var:CWIC_disband_instance
				target_organization = 1
				source_organization = 1
				size = 0
				stockpile_ratio = 0
				army_ratio = 0.12
				navy_ratio = 0
				air_ratio = 0
				keep_unit_leaders_trigger = { always = yes }
			}
		}
		else = {
			transfer_units_fraction = {
				target = var:CWIC_disband_instance
				target_organization = 1
				source_organization = 1
				size = 0
				stockpile_ratio = 0
				army_ratio = 0.03
				navy_ratio = 0
				air_ratio = 0
				keep_unit_leaders_trigger = { always = yes }
			}
		}
		var:CWIC_disband_instance = {
			CWIC_disband_units_transfer_manpower = yes
			send_equipment_fraction = { target = var:CWIC_disband_country value = 1 }
		}
	}
}
