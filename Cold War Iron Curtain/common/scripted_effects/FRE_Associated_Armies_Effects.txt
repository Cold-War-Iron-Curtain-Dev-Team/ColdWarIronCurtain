fre_associated_armies_commit = {
	if = {
		limit = {
			NOT = { has_country_flag = FRE_Associated_Armies_Chosen }
			NOT = { has_completed_focus = FRE_Arm_the_National_Army }
			NOT = { has_global_flag = Indochina_War_Over }
			NOT = { has_global_flag = geneva_conference_concluded }
			check_variable = { global.Indochina_War_Active_Phase < 9 }
			NOT = { check_variable = { FRE_War_Credits < 60 } }
			country_exists = LOS
			country_exists = CAM
			LOS = { any_owned_state = { is_controlled_by = LOS } }
			CAM = { any_owned_state = { is_controlled_by = CAM } }
		}
		set_country_flag = FRE_Associated_Armies_Chosen
		set_temp_variable = { fre_credit_amount = -60 }
		fre_change_war_credits = yes
		LOS = { fre_associated_armies_raise_los = yes }
		CAM = { fre_associated_armies_raise_cam = yes }
		if = {
			limit = { country_exists = VIE }
			VIE = { set_country_flag = VIE_Associated_Armies_Cap_Reduced }
		}
		set_temp_variable = { geneva_lev_amount = 30 }
		geneva_add_ethnic_leverage = yes
		set_temp_variable = { ic_trace_writer = 31 }
		set_temp_variable = { ic_trace_amount = 10 }
		ic_afk_trace_score_pro_france = yes
		FRA = {
			add_to_variable = {
				StruggleInvolvedNationsIndochinaProFranceScore = 10
				tooltip = Add_to_French_Victory_Points_TT
			}
			indochina_struggle_recalculate_total_anti_communist_score = yes
		}
		log = "IC_AFK|ASSOCIATED_ARMIES|date=[GetDateText]|choice=associated|los_divisions=2|cam_divisions=2|credits=[?FRE_War_Credits|0]|vie_cap=25"
	}
}
