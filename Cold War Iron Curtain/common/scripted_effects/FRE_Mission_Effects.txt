# FRE Mission Success Calculation Effects

fre_mission_calculate_rc4_success = {
	# Base chance: 50% (reasonable starting point for FRE)
	set_temp_variable = { mission_success_chance = 50 }
	
	# Political power helps (up to +15%)
	if = {
		limit = { check_variable = { political_power > 50 } }
		add_to_temp_variable = { mission_success_chance = 5 }
	}
	if = {
		limit = { check_variable = { political_power > 100 } }
		add_to_temp_variable = { mission_success_chance = 5 }
	}
	if = {
		limit = { check_variable = { political_power > 150 } }
		add_to_temp_variable = { mission_success_chance = 5 }
	}
	
	# Stability helps (up to +10%) - lowered thresholds for FRE
	if = {
		limit = { has_stability > 0.25 }
		add_to_temp_variable = { mission_success_chance = 3 }
	}
	if = {
		limit = { has_stability > 0.40 }
		add_to_temp_variable = { mission_success_chance = 4 }
	}
	if = {
		limit = { has_stability > 0.55 }
		add_to_temp_variable = { mission_success_chance = 3 }
	}
	
	# War support helps (up to +8%) - lowered thresholds
	if = {
		limit = { has_war_support > 0.20 }
		add_to_temp_variable = { mission_success_chance = 3 }
	}
	if = {
		limit = { has_war_support > 0.35 }
		add_to_temp_variable = { mission_success_chance = 3 }
	}
	if = {
		limit = { has_war_support > 0.50 }
		add_to_temp_variable = { mission_success_chance = 2 }
	}
	
	# Equipment stockpile helps (up to +12%) - lowered thresholds
	if = {
		limit = { has_equipment = { infantry_equipment_1 > 5000 } }
		add_to_temp_variable = { mission_success_chance = 4 }
	}
	if = {
		limit = { has_equipment = { infantry_equipment_1 > 10000 } }
		add_to_temp_variable = { mission_success_chance = 4 }
	}
	if = {
		limit = { has_equipment = { infantry_equipment_1 > 20000 } }
		add_to_temp_variable = { mission_success_chance = 4 }
	}
	
	# Clamp to 0-100
	if = {
		limit = { check_variable = { temp.mission_success_chance > 100 } }
		set_temp_variable = { mission_success_chance = 100 }
	}
	if = {
		limit = { check_variable = { temp.mission_success_chance < 0 } }
		set_temp_variable = { mission_success_chance = 0 }
	}
}
