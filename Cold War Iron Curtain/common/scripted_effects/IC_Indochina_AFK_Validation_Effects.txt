ic_afk_validation_vin_balance_contract = {
	if = {
		limit = { OR = { has_idea = VIN_Indochina_AI_Support has_idea = VIN_Campaign_AI_Support } }
		log = "IC_AFK|FAIL|date=[GetDateText]|check=VIN_hidden_AI_buffs_removed|campaign=[?VIN_camp_current|0]"
	}
	else = { log = "IC_AFK|PASS|date=[GetDateText]|check=VIN_hidden_AI_buffs_removed|campaign=[?VIN_camp_current|0]" }
	if = {
		limit = {
			country_exists = FRE
			has_war_with = FRE
			OR = {
				AND = { check_variable = { VIN_camp_current = 1 } country_exists = TAI has_war_with = TAI }
				AND = { check_variable = { VIN_camp_current = 2 } country_exists = TAM has_war_with = TAM }
				AND = { check_variable = { VIN_camp_current = 3 } country_exists = TAI has_war_with = TAI }
				AND = { check_variable = { VIN_camp_current = 4 } country_exists = THO has_war_with = THO }
				AND = { check_variable = { VIN_camp_current = 5 } country_exists = TAI has_war_with = TAI }
				check_variable = { VIN_camp_current = 6 }
				check_variable = { VIN_camp_current = 7 }
				AND = { check_variable = { VIN_camp_current = 8 } country_exists = TAI has_war_with = TAI }
			}
			OR = { check_variable = { VIN_camp_current = 2 } NOT = { country_exists = TAM } TAM = { NOT = { has_war_with = ROOT } } }
			OR = { check_variable = { VIN_camp_current = 4 } NOT = { country_exists = THO } THO = { NOT = { has_war_with = ROOT } } }
			OR = { check_variable = { VIN_camp_current = 1 } check_variable = { VIN_camp_current = 3 } check_variable = { VIN_camp_current = 5 } check_variable = { VIN_camp_current = 8 } NOT = { country_exists = TAI } TAI = { NOT = { has_war_with = ROOT } } }
			OR = { NOT = { country_exists = NUN } NUN = { NOT = { has_war_with = ROOT } } }
		}
		log = "IC_AFK|PASS|date=[GetDateText]|check=VIN_campaign_participant_envelope|campaign=[?VIN_camp_current|0]"
	}
	else = { log = "IC_AFK|FAIL|date=[GetDateText]|check=VIN_campaign_participant_envelope|campaign=[?VIN_camp_current|0]" }
	clr_global_flag = IC_AFK_VIN_Overextension_Was_Active
}
