ger_state_level_gdp_sorting = {
	if = {
		limit = {
			check_variable = { state_gdp > ROOT.german_gdp_level_1_cutoff }
		}
		set_variable = { German_GDP = 30  }
	}
	else_if = {
		limit = {
			check_variable = { state_gdp > ROOT.german_gdp_level_2_cutoff }
		}
		set_variable = { German_GDP = 31 }
	}
	else_if = {
		limit = {
			check_variable = { state_gdp > ROOT.german_gdp_level_3_cutoff }
		}
		set_variable = { German_GDP = 32 }
	}
	else_if = {
		limit = {
			check_variable = { state_gdp > ROOT.german_gdp_level_4_cutoff }
		}
		set_variable = { German_GDP = 33 }
	}
	else = {
		set_variable = { German_GDP = 34 }
	}

	#Capita Section
	if = {
		limit = {
			check_variable = { state_gdp_capita > ROOT.german_gdp_level_1_capita_cutoff }
		}
		set_variable = { German_GDP_Capita = 30  }
	}
	else_if = {
		limit = {
			check_variable = { state_gdp_capita > ROOT.german_gdp_level_2_capita_cutoff }
		}
		set_variable = { German_GDP_Capita = 31 }
	}
	else_if = {
		limit = {
			check_variable = { state_gdp_capita > ROOT.german_gdp_level_3_capita_cutoff }
		}
		set_variable = { German_GDP_Capita = 32 }
	}
	else_if = {
		limit = {
			check_variable = { state_gdp_capita > ROOT.german_gdp_level_4_capita_cutoff }
		}
		set_variable = { German_GDP_Capita = 33 }
	}
	else = {
		set_variable = { German_GDP_Capita = 34 }
	}
}
