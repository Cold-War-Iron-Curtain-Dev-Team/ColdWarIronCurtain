indonesia_party_support_refresh = {
    # Creation of temp arrays.
    # The arrays are "copied" from INO_Party_ID_LOCAL and INO_Party_SUPPORT_LOCAL for sorting
    resize_temp_array = {
        array = INO_Party_SUPPORT_LOCAL_temp
        size = INO_Party_SUPPORT_LOCAL^num
    }
    resize_temp_array = {
        array = INO_Party_ID_LOCAL_temp
        size = INO_Party_SUPPORT_LOCAL^num
    }
    # Sort the values by largest support
    for_loop_effect = {
        end = INO_Party_SUPPORT_LOCAL^num
        value = v
        find_highest_in_array = {
            array = INO_Party_SUPPORT_LOCAL
            value = max
            index = max_index
        }
        set_temp_variable = { INO_Party_SUPPORT_LOCAL_temp^v = max } #sets biggest at the earliest available position
        set_temp_variable = { INO_Party_ID_LOCAL_temp^v = INO_Party_ID_LOCAL^max_index } #sets biggest TAG at the earliest available position
        set_variable = { INO_Party_SUPPORT_LOCAL^max_index = -1 } #removes it from selection
    }

    # Clears the Values
    clear_array = INO_Party_SUPPORT_LOCAL
    clear_array = INO_Party_ID_LOCAL

    set_temp_variable = { influence_total = 0 }
    # Restores destroyed arrays in the above process - i.e. if they were deleted due to being less then 1%
    for_each_loop = {
        array = INO_Party_SUPPORT_LOCAL_temp
        add_to_array = { INO_Party_SUPPORT_LOCAL = v }
        add_to_temp_variable = { influence_total = v }
    }
    for_each_loop = {
        array = INO_Party_ID_LOCAL_temp
        add_to_array = { INO_Party_ID_LOCAL = v }
    }
    clear_temp_array = INO_Party_SUPPORT_LOCAL_temp
    clear_temp_array = INO_Party_ID_LOCAL_temp
}
