# Consumer Goods AI / Automated Trading Effects

cwic_ai_consumer_goods_trade_master = {
	# ------------------------------------------------------------
	# Automated consumer goods trade.
	# Recommended placement:
	# 1) monthly production
	# 2) apply existing imports to domestic stock
	# 3) calculate persistent export stock
	# 4) run this AI trade master
	# 5) optionally apply new imports to domestic stock if trades should deliver immediately
	# 6) consume/allot domestic consumer goods
	# ------------------------------------------------------------
	if = {
		limit = {
            OR = {
                is_ai = yes
                check_variable = { trade_automation = 2 }
            }
			
			check_variable = { trade_infrastructure_total_available > 0.99 }
		}

		cwic_ai_import_cg_basic = yes
		cwic_ai_import_cg_durable = yes
		cwic_ai_import_cg_vehicles = yes
		cwic_ai_import_cg_electronics = yes
	}

	set_variable = { purchase_amount = 0 }
	set_variable = { purchase_amount_prior = 0 }
	if = {
		limit = { is_ai = yes }
		clear_consumer_goods_sort_cache_arrays = yes
	}
}

cwic_ai_import_cg_basic_t0 = {
	if = {
		limit = {
			check_variable = { cg_basic_ai_need > 0.99 }
			check_variable = { trade_infrastructure_total_available > 0.99 }
		}

		generate_available_cg_basic_t0_import_array = yes
		sort_price_number_cg_basic_t0 = yes

		set_temp_variable = { cheapest_brk = 0 }

		for_each_scope_loop = {
			array = ROOT.cgBasicT0Exporters
			break = cheapest_brk

			if = {
				limit = {
					check_variable = { ROOT.cg_basic_ai_need > 0.99 }
					check_variable = { ROOT.trade_infrastructure_total_available > 0.99 }
					check_variable = { currentAvailableCgBasicT0Export > 0.99 }
				}

				set_variable = { ROOT.purchase_amount = ROOT.cg_basic_ai_need }

				if = {
					limit = { check_variable = { currentAvailableCgBasicT0Export < ROOT.purchase_amount } }
					set_variable = { ROOT.purchase_amount = currentAvailableCgBasicT0Export }
				}

				if = {
					limit = { check_variable = { ROOT.trade_infrastructure_total_available < ROOT.purchase_amount } }
					set_variable = { ROOT.purchase_amount = ROOT.trade_infrastructure_total_available }
				}

				round_variable = ROOT.purchase_amount

				if = {
					limit = { check_variable = { ROOT.purchase_amount > 0.99 } }

					subtract_from_variable = { ROOT.cg_basic_ai_need = ROOT.purchase_amount }

					# Write import/export relationship arrays
					for_loop_effect = {
						end = ROOT.CgBasicT0_Import_List
						value = v
						if = {
							limit = {
								check_variable = { CgBasicT0_Import_List^v = THIS.id }
							}
							set_country_flag = found
						}
					}
					if = {
						limit = { NOT = { has_country_flag = found } }
						add_to_array = { ROOT.CgBasicT0_Import_List = THIS.id }
					}
					clr_country_flag = found

					for_loop_effect = {
						end = CgBasicT0_Export_List
						value = v
						if = {
							limit = {
								check_variable = { CgBasicT0_Export_List^v = ROOT.id }
							}
							set_country_flag = found
						}
					}
					if = {
						limit = { NOT = { has_country_flag = found } }
						add_to_array = { CgBasicT0_Export_List = ROOT.id }
					}
					clr_country_flag = found

					# Apply trade contract
					subtract_from_variable = {
						currentAvailableCgBasicT0Export = ROOT.purchase_amount
					}
					add_to_variable = {
						ROOT.trade_infrastructure_total_used = ROOT.purchase_amount
					}
					subtract_from_variable = {
						ROOT.trade_infrastructure_total_available = ROOT.purchase_amount
					}
					add_to_variable = {
						currentExportedCgBasicT0 = ROOT.purchase_amount
					}
					add_to_variable = {
						ROOT.cgBasicT0Imports = ROOT.purchase_amount
					}

					set_variable = {
						CgBasicT0ExportTarget^ROOT = ROOT.id
					}
					add_to_variable = {
						CgBasicT0ExportTargetAmount^ROOT = ROOT.purchase_amount
					}

					set_variable = {
						price_of_goods = cgBasicT0ExporterPrice^ROOT
					}
					multiply_variable = {
						price_of_goods = ROOT.purchase_amount
					}

					add_to_variable = {
						CgBasicT0ExportTargetMonthly^ROOT = price_of_goods
					}
					add_to_variable = {
						ROOT.CgBasicT0ImportTargetMonthly^THIS = price_of_goods
					}
					set_variable = {
						ROOT.CgBasicT0ImportTarget^THIS = THIS.id
					}
					add_to_variable = {
						ROOT.CgBasicT0ImportTargetAmount^THIS = ROOT.purchase_amount
					}

					set_variable = { price_of_goods = 0 }

					if = {
						limit = {
							check_variable = { currentAvailableCgBasicT0Export < 0 }
						}
						country_event = trade.40
					}

					set_variable = { ROOT.purchase_amount = 0 }
				}
			}

			if = {
				limit = {
					OR = {
						NOT = { check_variable = { ROOT.cg_basic_ai_need > 0.99 } }
						NOT = { check_variable = { ROOT.trade_infrastructure_total_available > 0.99 } }
					}
				}
				set_temp_variable = { cheapest_brk = 1 }
			}
		}
	}

	clear_variable = cheapest_brk
}

				if = {
					limit = { check_variable = { ROOT.purchase_amount > 0.99 } }

					subtract_from_variable = { ROOT.cg_electronics_ai_need = ROOT.purchase_amount }

					# Write import/export relationship arrays
					for_loop_effect = {
						end = ROOT.CgElectronicsT5_Import_List
						value = v
						if = {
							limit = {
								check_variable = { CgElectronicsT5_Import_List^v = THIS.id }
							}
							set_country_flag = found
						}
					}
					if = {
						limit = { NOT = { has_country_flag = found } }
						add_to_array = { ROOT.CgElectronicsT5_Import_List = THIS.id }
					}
					clr_country_flag = found

					for_loop_effect = {
						end = CgElectronicsT5_Export_List
						value = v
						if = {
							limit = {
								check_variable = { CgElectronicsT5_Export_List^v = ROOT.id }
							}
							set_country_flag = found
						}
					}
					if = {
						limit = { NOT = { has_country_flag = found } }
						add_to_array = { CgElectronicsT5_Export_List = ROOT.id }
					}
					clr_country_flag = found

					# Apply trade contract
					subtract_from_variable = {
						currentAvailableCgElectronicsT5Export = ROOT.purchase_amount
					}
					add_to_variable = {
						ROOT.trade_infrastructure_total_used = ROOT.purchase_amount
					}
					subtract_from_variable = {
						ROOT.trade_infrastructure_total_available = ROOT.purchase_amount
					}
					add_to_variable = {
						currentExportedCgElectronicsT5 = ROOT.purchase_amount
					}
					add_to_variable = {
						ROOT.cgElectronicsT5Imports = ROOT.purchase_amount
					}

					set_variable = {
						CgElectronicsT5ExportTarget^ROOT = ROOT.id
					}
					add_to_variable = {
						CgElectronicsT5ExportTargetAmount^ROOT = ROOT.purchase_amount
					}

					set_variable = {
						price_of_goods = cgElectronicsT5ExporterPrice^ROOT
					}
					multiply_variable = {
						price_of_goods = ROOT.purchase_amount
					}

					add_to_variable = {
						CgElectronicsT5ExportTargetMonthly^ROOT = price_of_goods
					}
					add_to_variable = {
						ROOT.CgElectronicsT5ImportTargetMonthly^THIS = price_of_goods
					}
					set_variable = {
						ROOT.CgElectronicsT5ImportTarget^THIS = THIS.id
					}
					add_to_variable = {
						ROOT.CgElectronicsT5ImportTargetAmount^THIS = ROOT.purchase_amount
					}

					set_variable = { price_of_goods = 0 }

					if = {
						limit = {
							check_variable = { currentAvailableCgElectronicsT5Export < 0 }
						}
						country_event = trade.39
					}

					set_variable = { ROOT.purchase_amount = 0 }
				}
			}

			if = {
				limit = {
					OR = {
						NOT = { check_variable = { ROOT.cg_electronics_ai_need > 0.99 } }
						NOT = { check_variable = { ROOT.trade_infrastructure_total_available > 0.99 } }
					}
				}
				set_temp_variable = { cheapest_brk = 1 }
			}
		}
	}

	clear_variable = cheapest_brk
}
