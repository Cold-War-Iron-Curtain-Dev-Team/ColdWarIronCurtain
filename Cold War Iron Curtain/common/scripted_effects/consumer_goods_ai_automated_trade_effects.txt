# Consumer Goods AI / Automated Trading Effects

cwic_ai_consumer_goods_trade_master = {
	# ------------------------------------------------------------
	# Automated consumer goods trade.
	# Recommended placement:
	# 1) monthly production
	# 2) apply existing imports to domestic stock
	# 3) calculate persistent export stock
	# 4) run this AI trade master
	# 5) optionally apply new imports to domestic stock if trades should deliver immediately
	# 6) consume/allot domestic consumer goods
	# ------------------------------------------------------------
	if = {
		limit = {
            OR = {
                is_ai = yes
                check_variable = { trade_automation = 2 }
            }
			
			check_variable = { trade_infrastructure_total_available > 0.99 }
		}

		cwic_ai_import_cg_basic = yes
		cwic_ai_import_cg_durable = yes
		cwic_ai_import_cg_vehicles = yes
		cwic_ai_import_cg_electronics = yes
	}

	set_variable = { purchase_amount = 0 }
	set_variable = { purchase_amount_prior = 0 }
	if = {
		limit = { is_ai = yes }
		clear_consumer_goods_sort_cache_arrays = yes
	}
}

cwic_ai_import_cg_basic = {
	# ------------------------------------------------------------
	# Basic Consumer Goods AI import target
	# Need = monthly demand - current domestic stock.
	# Existing imports are already included if cwic_apply_consumer_goods_imports_to_stock ran earlier this tick.
	# ------------------------------------------------------------
	set_variable = { cg_basic_ai_stock_total = 0 }
	add_to_variable = { cg_basic_ai_stock_total = cg_basic_stock_t0 }
	add_to_variable = { cg_basic_ai_stock_total = cg_basic_stock_t1 }
	add_to_variable = { cg_basic_ai_stock_total = cg_basic_stock_t2 }
	add_to_variable = { cg_basic_ai_stock_total = cg_basic_stock_t3 }
	add_to_variable = { cg_basic_ai_stock_total = cg_basic_stock_t4 }
	add_to_variable = { cg_basic_ai_stock_total = cg_basic_stock_t5 }

	set_variable = { cg_basic_ai_need = cg_basic_demand }
	subtract_from_variable = { cg_basic_ai_need = cg_basic_ai_stock_total }

	if = {
		limit = { check_variable = { cg_basic_ai_need < 1 } }
		set_variable = { cg_basic_ai_need = 0 }
	}

	if = {
		limit = {
			check_variable = { cg_basic_ai_need > 0.99 }
			check_variable = { trade_infrastructure_total_available > 0.99 }
		}
		if = {
			limit = {
				check_variable = { cg_basic_ai_need > 0.99 }
				check_variable = { trade_infrastructure_total_available > 0.99 }
				check_variable = { cg_basic_expected_tier > 4 }
			}
			cwic_ai_import_cg_basic_t5 = yes
		}
		if = {
			limit = {
				check_variable = { cg_basic_ai_need > 0.99 }
				check_variable = { trade_infrastructure_total_available > 0.99 }
				check_variable = { cg_basic_expected_tier > 3 }
			}
			cwic_ai_import_cg_basic_t4 = yes
		}
		if = {
			limit = {
				check_variable = { cg_basic_ai_need > 0.99 }
				check_variable = { trade_infrastructure_total_available > 0.99 }
				check_variable = { cg_basic_expected_tier > 2 }
			}
			cwic_ai_import_cg_basic_t3 = yes
		}
		if = {
			limit = {
				check_variable = { cg_basic_ai_need > 0.99 }
				check_variable = { trade_infrastructure_total_available > 0.99 }
				check_variable = { cg_basic_expected_tier > 1 }
			}
			cwic_ai_import_cg_basic_t2 = yes
		}
		if = {
			limit = {
				check_variable = { cg_basic_ai_need > 0.99 }
				check_variable = { trade_infrastructure_total_available > 0.99 }
				check_variable = { cg_basic_expected_tier > 0 }
			}
			cwic_ai_import_cg_basic_t1 = yes
		}
		if = {
			limit = {
				check_variable = { cg_basic_ai_need > 0.99 }
				check_variable = { trade_infrastructure_total_available > 0.99 }
			}
			cwic_ai_import_cg_basic_t0 = yes
		}
	}

	clear_variable = cg_basic_ai_stock_total
	clear_variable = cg_basic_ai_need
}
