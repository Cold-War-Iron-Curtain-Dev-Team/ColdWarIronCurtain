cwic_sync_trade_embargoes = {
	every_country = {
		clear_temp_array = cwic_emb_lift
		for_each_scope_loop = {
			array = cwic_embargo_applied
			if = {
				limit = {
					NOT = { check_variable = { PREV.trade_embargo_check^THIS = 1 } }
					PREV = { NOT = { check_variable = { PREV.trade_embargo_check^THIS = 1 } } }
				}
				add_to_temp_array = { cwic_emb_lift = THIS.id }
			}
		}
		for_each_scope_loop = {
			array = cwic_emb_lift
			remove_opinion_modifier = { target = PREV modifier = cwic_trade_embargo }
			remove_from_array = { cwic_embargo_applied = PREV.id }
			PREV = {
				remove_opinion_modifier = { target = PREV modifier = cwic_trade_embargo }
				remove_from_array = { cwic_embargo_applied = PREV.id }
			}
		}
		every_other_country = {
			limit = {
				OR = {
					check_variable = { PREV.trade_embargo_check^THIS = 1 }
					PREV = { check_variable = { PREV.trade_embargo_check^THIS = 1 } }
				}
				NOT = { is_in_array = { array = PREV.cwic_embargo_applied value = THIS.id } }
			}
			add_opinion_modifier = { target = PREV modifier = cwic_trade_embargo }
			add_to_array = { cwic_embargo_applied = PREV.id }
			PREV = {
				add_opinion_modifier = { target = PREV modifier = cwic_trade_embargo }
				add_to_array = { cwic_embargo_applied = PREV.id }
			}
		}
	}
}

cwic_update_market_refunds = {
	set_temp_variable = { cwic_mr_share = modifier@min_export }
	clamp_temp_variable = { var = cwic_mr_share min = 0 max = 0.9 }
	set_temp_variable = { cwic_mr_law = 0 }
	set_temp_variable = { cwic_mr_food_law = 0 }
	set_temp_variable = { cwic_mr_chromium_law = 0 }
	if = {
		limit = { has_idea = free_trade }
		set_temp_variable = { cwic_mr_law = 0.7 }
		set_temp_variable = { cwic_mr_food_law = 0.45 }
		set_temp_variable = { cwic_mr_chromium_law = 0.25 }
	}
	else_if = {
		limit = { has_idea = export_focus }
		set_temp_variable = { cwic_mr_law = 0.6 }
		set_temp_variable = { cwic_mr_food_law = 0.3 }
		set_temp_variable = { cwic_mr_chromium_law = 0.2 }
	}
	else_if = {
		limit = { has_idea = limited_exports }
		set_temp_variable = { cwic_mr_law = 0.4 }
		set_temp_variable = { cwic_mr_food_law = 0.15 }
		set_temp_variable = { cwic_mr_chromium_law = 0.1 }
	}
	else_if = {
		limit = { has_idea = closed_economy }
		set_temp_variable = { cwic_mr_law = 0.2 }
		set_temp_variable = { cwic_mr_food_law = 0.05 }
		set_temp_variable = { cwic_mr_chromium_law = 0 }
	}
	set_temp_variable = { cwic_mr_keep = 1 }
	subtract_from_temp_variable = { cwic_mr_keep = cwic_mr_share }

	set_temp_variable = { cwic_mr_base = resource_produced@coal }
	subtract_from_temp_variable = { cwic_mr_base = cwic_coal_market_refund }
	clamp_temp_variable = { var = cwic_mr_base min = 0 }
	set_variable = { cwic_coal_market_refund = cwic_mr_base }
	multiply_variable = { cwic_coal_market_refund = cwic_mr_share }
	divide_variable = { cwic_coal_market_refund = cwic_mr_keep }

	set_temp_variable = { cwic_mr_base = resource_produced@rubber }
	subtract_from_temp_variable = { cwic_mr_base = cwic_rubber_market_refund }
	clamp_temp_variable = { var = cwic_mr_base min = 0 }
	set_temp_variable = { cwic_mr_food_rate = cwic_mr_food_law }
	add_to_temp_variable = { cwic_mr_food_rate = cwic_mr_share }
	subtract_from_temp_variable = { cwic_mr_food_rate = cwic_mr_law }
	clamp_temp_variable = { var = cwic_mr_food_rate min = 0 max = 0.9 }
	set_temp_variable = { cwic_mr_food_pool = agri_industrial_complex_quantity }
	multiply_temp_variable = { cwic_mr_food_pool = cwic_mr_food_rate }
	set_variable = { cwic_rubber_market_refund = cwic_mr_base }
	multiply_variable = { cwic_rubber_market_refund = cwic_mr_share }
	subtract_from_variable = { cwic_rubber_market_refund = cwic_mr_food_pool }
	clamp_variable = { var = cwic_rubber_market_refund min = 0 }
	divide_variable = { cwic_rubber_market_refund = cwic_mr_keep }

	set_temp_variable = { cwic_mr_base = resource_produced@chromium }
	subtract_from_temp_variable = { cwic_mr_base = cwic_chromium_market_refund }
	clamp_temp_variable = { var = cwic_mr_base min = 0 }
	set_temp_variable = { cwic_mr_k = cwic_mr_law }
	subtract_from_temp_variable = { cwic_mr_k = cwic_mr_chromium_law }
	clamp_temp_variable = { var = cwic_mr_k min = 0 }
	set_variable = { cwic_chromium_market_refund = cwic_mr_base }
	multiply_variable = { cwic_chromium_market_refund = cwic_mr_k }
	divide_variable = { cwic_chromium_market_refund = cwic_mr_keep }

	force_update_dynamic_modifier = yes
}

cwic_trade_tier_calc = {
	set_temp_variable = { cwic_tier = 0 }
	if = {
		limit = {
			OR = {
				is_subject_of = PREV
				PREV = { is_subject_of = PREV }
			}
		}
		set_temp_variable = { cwic_tier = 3 }
	}
	else_if = {
		limit = {
			OR = {
				AND = {
					has_idea = EEC_Currency
					PREV = { has_idea = EEC_Currency }
				}
				AND = {
					has_idea = european_free_trade_association
					PREV = { has_idea = european_free_trade_association }
				}
				AND = {
					OR = {
						tag = SOV
						is_in_array = { array = SOV.COMECON_MEMBERSHIP value = THIS.id }
					}
					PREV = {
						OR = {
							tag = SOV
							is_in_array = { array = SOV.COMECON_MEMBERSHIP value = THIS.id }
						}
					}
				}
			}
		}
		set_temp_variable = { cwic_tier = 2 }
	}
	else_if = {
		limit = { is_in_faction_with = PREV }
		set_temp_variable = { cwic_tier = 1 }
	}
}

cwic_sync_trade_partner_discounts = {
	every_country = {
		every_other_country = {
			cwic_trade_tier_calc = yes
			if = {
				limit = {
					NOT = { check_variable = { var = PREV.cwic_td^THIS value = cwic_tier compare = equals } }
				}
				PREV = {
					if = {
						limit = { check_variable = { cwic_td^PREV = 1 } }
						remove_relation_modifier = { target = PREV modifier = cwic_trade_partner_faction }
					}
					else_if = {
						limit = { check_variable = { cwic_td^PREV = 2 } }
						remove_relation_modifier = { target = PREV modifier = cwic_trade_partner_bloc }
					}
					else_if = {
						limit = { check_variable = { cwic_td^PREV = 3 } }
						remove_relation_modifier = { target = PREV modifier = cwic_trade_partner_subject }
					}
					if = {
						limit = { check_variable = { cwic_tier = 1 } }
						add_relation_modifier = { target = PREV modifier = cwic_trade_partner_faction }
					}
					else_if = {
						limit = { check_variable = { cwic_tier = 2 } }
						add_relation_modifier = { target = PREV modifier = cwic_trade_partner_bloc }
					}
					else_if = {
						limit = { check_variable = { cwic_tier = 3 } }
						add_relation_modifier = { target = PREV modifier = cwic_trade_partner_subject }
					}
					set_variable = { cwic_td^PREV = cwic_tier }
				}
			}
		}
	}
}
