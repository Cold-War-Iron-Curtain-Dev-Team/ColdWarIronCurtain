cwic_sync_trade_embargoes = {
	every_country = {
		every_other_country = {
			PREV = { cwic_sync_native_embargo = yes }
			if = {
				limit = {
					OR = {
						check_variable = { PREV.trade_embargo_check^THIS = 1 }
						check_variable = { trade_embargo_check^PREV = 1 }
					}
					NOT = { is_in_array = { array = PREV.cwic_emb_block value = THIS.id } }
				}
				cwic_apply_embargo_block = yes
			}
			else_if = {
				limit = {
					NOT = { check_variable = { PREV.trade_embargo_check^THIS = 1 } }
					NOT = { check_variable = { trade_embargo_check^PREV = 1 } }
					is_in_array = { array = PREV.cwic_emb_block value = THIS.id }
				}
				cwic_lift_embargo_block = yes
			}
		}
	}
}

cwic_apply_embargo_block = {
	add_relation_modifier = { target = PREV modifier = cwic_trade_embargo_block }
	add_to_array = { cwic_emb_block = PREV.id }
	PREV = {
		add_relation_modifier = { target = PREV modifier = cwic_trade_embargo_block }
		add_to_array = { cwic_emb_block = PREV.id }
	}
	if = {
		limit = { NOT = { has_global_flag = cwic_emb_quiet } }
		if = {
			limit = { PREV = { is_ai = no } }
			PREV = {
				set_variable = { cwic_emb_partner_tmp = PREV.id }
				country_event = trade.51
			}
		}
		if = {
			limit = { is_ai = no }
			set_variable = { cwic_emb_partner_tmp = PREV.id }
			country_event = trade.51
		}
	}
}

cwic_lift_embargo_block = {
	remove_relation_modifier = { target = PREV modifier = cwic_trade_embargo_block }
	remove_from_array = { cwic_emb_block = PREV.id }
	PREV = {
		remove_relation_modifier = { target = PREV modifier = cwic_trade_embargo_block }
		remove_from_array = { cwic_emb_block = PREV.id }
	}
	if = {
		limit = { PREV = { is_ai = no } }
		PREV = {
			set_variable = { cwic_emb_partner_tmp = PREV.id }
		country_event = trade.52
		}
	}
	if = {
		limit = { is_ai = no }
		set_variable = { cwic_emb_partner_tmp = PREV.id }
		country_event = trade.52
	}
}

cwic_update_market_refunds = {
	set_temp_variable = { cwic_mr_share = modifier@min_export }
	clamp_temp_variable = { var = cwic_mr_share min = 0.01 max = 0.9 }
	set_temp_variable = { cwic_mr_law = 0 }
	set_temp_variable = { cwic_mr_food_law = 0 }
	set_temp_variable = { cwic_mr_chromium_law = 0 }
	if = {
		limit = { has_idea = free_trade }
		set_temp_variable = { cwic_mr_law = 0.7 }
		set_temp_variable = { cwic_mr_food_law = 0.45 }
		set_temp_variable = { cwic_mr_chromium_law = 0.25 }
	}
	else_if = {
		limit = { has_idea = export_focus }
		set_temp_variable = { cwic_mr_law = 0.6 }
		set_temp_variable = { cwic_mr_food_law = 0.3 }
		set_temp_variable = { cwic_mr_chromium_law = 0.2 }
	}
	else_if = {
		limit = { has_idea = limited_exports }
		set_temp_variable = { cwic_mr_law = 0.4 }
		set_temp_variable = { cwic_mr_food_law = 0.15 }
		set_temp_variable = { cwic_mr_chromium_law = 0.1 }
	}
	else_if = {
		limit = { has_idea = closed_economy }
		set_temp_variable = { cwic_mr_law = 0.2 }
		set_temp_variable = { cwic_mr_food_law = 0.05 }
		set_temp_variable = { cwic_mr_chromium_law = 0 }
	}
	set_temp_variable = { cwic_mr_keep = 1 }
	subtract_from_temp_variable = { cwic_mr_keep = cwic_mr_share }

	set_variable = { cwic_mr_gross = 0 }
	every_controlled_state = { add_to_variable = { PREV.cwic_mr_gross = resource@coal } }
	set_temp_variable = { cwic_mr_base = cwic_mr_gross }
	add_to_temp_variable = { cwic_mr_base = modifier@country_resource_coal }
	subtract_from_temp_variable = { cwic_mr_base = cwic_coal_market_refund }
	subtract_from_temp_variable = { cwic_mr_base = modifier@country_resource_cost_coal }
	clamp_temp_variable = { var = cwic_mr_base min = 0 }
	set_variable = { cwic_coal_market_refund = cwic_mr_base }
	multiply_variable = { cwic_coal_market_refund = cwic_mr_share }
	divide_variable = { cwic_coal_market_refund = cwic_mr_keep }

	set_variable = { cwic_mr_gross = 0 }
	every_controlled_state = { add_to_variable = { PREV.cwic_mr_gross = resource@rubber } }
	set_temp_variable = { cwic_mr_base = cwic_mr_gross }
	add_to_temp_variable = { cwic_mr_base = modifier@country_resource_rubber }
	subtract_from_temp_variable = { cwic_mr_base = cwic_rubber_market_refund }
	subtract_from_temp_variable = { cwic_mr_base = modifier@country_resource_cost_rubber }
	clamp_temp_variable = { var = cwic_mr_base min = 0 }
	set_temp_variable = { cwic_mr_food_rate = cwic_mr_food_law }
	add_to_temp_variable = { cwic_mr_food_rate = cwic_mr_share }
	subtract_from_temp_variable = { cwic_mr_food_rate = cwic_mr_law }
	clamp_temp_variable = { var = cwic_mr_food_rate min = 0 max = 0.9 }
	set_temp_variable = { cwic_mr_food_pool = agri_industrial_complex_quantity }
	multiply_temp_variable = { cwic_mr_food_pool = cwic_mr_food_rate }
	set_variable = { cwic_rubber_market_refund = cwic_mr_base }
	multiply_variable = { cwic_rubber_market_refund = cwic_mr_share }
	subtract_from_variable = { cwic_rubber_market_refund = cwic_mr_food_pool }
	clamp_variable = { var = cwic_rubber_market_refund min = 0 }
	divide_variable = { cwic_rubber_market_refund = cwic_mr_keep }

	set_variable = { cwic_mr_gross = 0 }
	every_controlled_state = { add_to_variable = { PREV.cwic_mr_gross = resource@chromium } }
	set_temp_variable = { cwic_mr_base = cwic_mr_gross }
	add_to_temp_variable = { cwic_mr_base = modifier@country_resource_chromium }
	subtract_from_temp_variable = { cwic_mr_base = cwic_chromium_market_refund }
	subtract_from_temp_variable = { cwic_mr_base = modifier@country_resource_cost_chromium }
	clamp_temp_variable = { var = cwic_mr_base min = 0 }
	set_temp_variable = { cwic_mr_k = cwic_mr_law }
	subtract_from_temp_variable = { cwic_mr_k = cwic_mr_chromium_law }
	clamp_temp_variable = { var = cwic_mr_k min = 0 }
	set_variable = { cwic_chromium_market_refund = cwic_mr_base }
	multiply_variable = { cwic_chromium_market_refund = cwic_mr_k }
	divide_variable = { cwic_chromium_market_refund = cwic_mr_keep }

	force_update_dynamic_modifier = yes
}

cwic_build_native_seller_list_oil = {
	clear_array = ROOT.oilExporters
	clear_array = ROOT.oilExporterInfo
	every_other_country = {
		limit = {
			NOT = { has_war_with = ROOT }
			check_variable = { ROOT.trade_embargo_check^THIS = 0 }
		}
		set_variable = { cwicSellAvail = resource_produced@oil }
		multiply_variable = { cwicSellAvail = modifier@min_export }
		set_variable = { cwicSellExported = resource_exported@oil }
		subtract_from_variable = { cwicSellAvail = resource_exported@oil }
		round_variable = cwicSellAvail
		set_variable = { cwicSellPrice = global.oilBasePrice }
		if = {
			limit = { check_variable = { cwicSellAvail > 0 } }
			add_to_array = { ROOT.oilExporters = THIS.id }
			add_to_array = { ROOT.oilExporterInfo = cwicSellAvail }
		}
	}
}

		set_variable = { global.oilMinPrice = { value = global.oilBasePrice multiply = 0.3 } }

		limit = {
			OR = {
				is_subject_of = PREV
				PREV = { is_subject_of = PREV }
			}
		}
		set_temp_variable = { cwic_tier = 3 }
	}
	else_if = {
		limit = {
			OR = {
				AND = {
					has_idea = EEC_Currency
					PREV = { has_idea = EEC_Currency }
				}
				AND = {
					has_idea = european_free_trade_association
					PREV = { has_idea = european_free_trade_association }
				}
				AND = {
					OR = {
						tag = SOV
						is_in_array = { array = SOV.COMECON_MEMBERSHIP value = THIS.id }
					}
					PREV = {
						OR = {
							tag = SOV
							is_in_array = { array = SOV.COMECON_MEMBERSHIP value = THIS.id }
						}
					}
				}
			}
		}
		set_temp_variable = { cwic_tier = 2 }
	}
	else_if = {
		limit = { is_in_faction_with = PREV }
		set_temp_variable = { cwic_tier = 1 }
	}
}

cwic_cg_auto_import = {
	if = {
		limit = {
			OR = {
				is_ai = yes
				check_variable = { trade_automation = 2 }
			}
		}
		if = {
			limit = {
				check_variable = { cg_basic_monthly_balance < -2 }
				NOT = { has_country_flag = cg_import_cd_basic }
			}
			clear_array = cg_seller_list
			clear_array = cg_seller_info
			every_other_country = {
				limit = {
					NOT = { has_war_with = ROOT }
					check_variable = { ROOT.trade_embargo_check^THIS = 0 }
					NOT = { is_in_array = { array = ROOT.cg_basic_tried_sellers value = THIS.id } }
				}
				set_temp_variable = { cg_free = cg_basic_offer_applied }
				multiply_temp_variable = { cg_free = 0.8 }
				subtract_from_temp_variable = { cg_free = cg_basic_promised }
				if = {
					limit = { check_variable = { cg_free > 4.99 } }
					add_to_array = { ROOT.cg_seller_list = THIS.id }
					add_to_array = { ROOT.cg_seller_info = cg_free }
				}
			}
			if = {
				limit = { check_variable = { cg_seller_list^num > 0 } }
				find_highest_in_array = { array = cg_seller_info value = cwic_ai_best index = cwic_ai_best_i }
				set_variable = { currentPurchaser = cg_seller_list^cwic_ai_best_i }
				add_to_array = { cg_basic_tried_sellers = currentPurchaser }
				set_variable = { purchase_amount = cg_basic_monthly_balance }
				divide_variable = { purchase_amount = -5 }
				round_variable = purchase_amount
				clamp_variable = { var = purchase_amount min = 1 }
				divide_variable = { cwic_ai_best = 5 }
				subtract_from_variable = { cwic_ai_best = 0.49 }
				round_variable = cwic_ai_best
				clamp_variable = { var = purchase_amount max = cwic_ai_best }
				multiply_variable = { purchase_amount = 5 }
				var:currentPurchaser = { add_to_variable = { cg_basic_promised = PREV.purchase_amount } }
				set_country_flag = { flag = cg_import_cd_basic days = 90 value = 1 }
				meta_effect = {
					text = {
						create_import = { resource = cg_basic amount = [AMT] exporter = [TAG] }
					}
					AMT = "[?purchase_amount|0]"
					TAG = "[?var:currentPurchaser.GetTag]"
