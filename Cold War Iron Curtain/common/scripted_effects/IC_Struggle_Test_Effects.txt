# Console command: ic_reset
# Resets all Indochina Struggle variables and flags to default state
d_ic_reset = {
    FROM = {
        set_global_flag = Indochina_Struggle_Test_Mode
        clr_global_flag = Indochina_War_Over
        clr_global_flag = Geneva_Conference
        clr_global_flag = dan_quoc_peace
        clr_global_flag = diem_ho_chi_minh_reunified
        clr_global_flag = USA_VIN_Reunification_Complete
        set_variable = { global.Indochina_War_Active_Phase = 5 }
        set_variable = { global.Indochina_War_Next_Phase_A_Points = 0 }
        set_variable = { global.Indochina_War_Next_Phase_B_Points = 0 }
        indochina_struggle_update_phase_transitions = yes
        set_variable = { FRA.StruggleInvolvedNationsIndochinaCommunistScore = 0 }
        set_variable = { FRA.StruggleInvolvedNationsIndochinaProFranceScore = 0 }
        set_variable = { FRA.StruggleInvolvedNationsIndochinaProIndependenceScore = 0 }
        set_variable = { FRA.StruggleInvolvedNationsIndochinaProEthnicScore = 0 }
        set_variable = { FRA.StruggleInvolvedNationsIndochinaKuomintangScore = 0 }
        set_variable = { FRA.StruggleInvolvedNationsIndochinaTotalAntiCommunistScore = 0 }
        set_variable = { global.indochina_phase_check = 0 }
        set_variable = { global.indochina_border_war_check = 0 }
        set_variable = { show_indochina_end_gui = 0 }
        set_variable = { show_indochina_phase_gui = 0 }
        log = "Indochina Struggle reset complete"
    }
}

test_indochina_manual_escalate = {
    if = {
        limit = {
            check_variable = { global.Indochina_War_Active_Phase < 8 }
        }
        set_variable = { global.Indochina_War_Active_Phase = global.Indochina_War_Next_Phase_A }
        set_variable = { global.Indochina_War_Next_Phase_A_Points = 0 }
        set_variable = { global.Indochina_War_Next_Phase_B_Points = 0 }
        indochina_struggle_update_phase_transitions = yes
        every_country = {
            limit = {
                OR = {
                    is_in_array = { StruggleInvolvedNationsIndochinaCommunist = THIS.id }
                    is_in_array = { StruggleInvolvedNationsIndochinaProFrance = THIS.id }
                    is_in_array = { StruggleInvolvedNationsIndochinaProIndependence = THIS.id }
                    is_in_array = { StruggleInvolvedNationsIndochinaKuomintang = THIS.id }
                    is_in_array = { StruggleInvolvedNationsIndochinaProEthnic = THIS.id }
                }
            }
            indochina_struggle_starting_idea_phase_change = yes
            set_variable = { global.indochina_phase_check = 1 }
        }
        log = "Phase escalated manually"
    }
    else = {
        log = "Cannot escalate - phase is 8 or higher (ending phases)"
    }
}

                add_opinion_modifier = {
                    target = VIN
                    modifier = medium_decrease
                }

        }

            limit = { check_variable = { args.0 = "american_north" } }
