test_indochina_manual_escalate = {
    if = {
        limit = {
            check_variable = { global.Indochina_War_Active_Phase < 8 }
        }
        set_variable = { global.Indochina_War_Active_Phase = global.Indochina_War_Next_Phase_A }
        set_variable = { global.Indochina_War_Next_Phase_A_Points = 0 }
        set_variable = { global.Indochina_War_Next_Phase_B_Points = 0 }
        indochina_struggle_update_phase_transitions = yes
        every_country = {
            limit = {
                OR = {
                    is_in_array = { StruggleInvolvedNationsIndochinaCommunist = THIS.id }
                    is_in_array = { StruggleInvolvedNationsIndochinaProFrance = THIS.id }
                    is_in_array = { StruggleInvolvedNationsIndochinaProIndependence = THIS.id }
                    is_in_array = { StruggleInvolvedNationsIndochinaKuomintang = THIS.id }
                    is_in_array = { StruggleInvolvedNationsIndochinaProEthnic = THIS.id }
                }
            }
            indochina_struggle_starting_idea_phase_change = yes
            set_variable = { global.indochina_phase_check = 1 }
        }
        log = "Phase escalated manually"
    }
    else = {
        log = "Cannot escalate - phase is 8 or higher (ending phases)"
    }
}

    set_variable = { FRA.StruggleInvolvedNationsIndochinaProFranceScore = 600 }

        limit = {
            indochina_struggle_ending_dan_quoc_peace_trigger = yes
            NOT = { has_global_flag = Indochina_War_Over }

    }
    else = {
        log = "Dan Quoc Peace conditions not met!"

            limit = { has_global_flag = Indochina_War_Over }
            log = "Reason: Indochina War already over"
        }
        if = {
            limit = { NOT = { has_global_flag = dan_quoc_peace } }
            log = "Reason: dan_quoc_peace flag not set"
        }
        if = {
            limit = { 
                VIE = {
                    NOT = {
                        has_country_leader = {
                            character = VIE_Ngo_Dinh_Diem
                            ruling_only = yes
                        }
                    }
                }
            }
            log = "Reason: Ngo Dinh Diem not in power in South Vietnam"
        }
        if = {
            limit = { 
                VIN = {
                    NOT = {
                        has_country_leader = {
                            character = VIN_Ho_Chi_Minh
                            ruling_only = yes
                        }
                    }
                }
            }
            log = "Reason: Ho Chi Minh not in power in North Vietnam"
        }
        if = {
            limit = { 
                NOT = { check_variable = { FRA.StruggleInvolvedNationsIndochinaProIndependenceScore > FRA.StruggleInvolvedNationsIndochinaProFrance } }
            }
            log = "Reason: ProIndependence score not higher than ProFrance"
        }
        if = {
            limit = { 
                check_variable = { FRA.StruggleInvolvedNationsIndochinaProIndependenceScore < 500 }
            }
            log = "Reason: ProIndependence score below 500"
        }
    }
}
