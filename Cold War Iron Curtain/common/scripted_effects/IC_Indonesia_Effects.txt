CalculateIndonesiaReligionTotals = {
	#GDP Stuff
	set_variable = { state_gdp = 0 }
	add_to_variable = { state_gdp = net_factory_earnings }
	add_to_variable = { state_gdp = pop_lowerclass_net_wage }
	add_to_variable = { state_gdp = pop_middleclass_net_wage }
	add_to_variable = { state_gdp = pop_substinents_gross_wage }
	add_to_variable = { state_gdp = pop_artisan_gross_wage }
	add_to_variable = { state_gdp = pop_upperclass_net_wage }

	set_variable = { state_gdp_capita = state_gdp }
	multiply_variable = { state_gdp_capita = 1000 }
	divide_variable = { state_gdp_capita = state_population_k }

	#Religion
    set_variable = { StateChristianTotal = StateChristianityProtestant }
    add_to_variable = { StateChristianTotal = StateChristianityOrthodox }
    add_to_variable = { StateChristianTotal = StateChristianityOriental }
    add_to_variable = { StateChristianTotal = StateZionistChristian }

    set_variable = { StateMuslimTotal = StateIslamSunni }
    add_to_variable = { StateMuslimTotal = StateIslamShia }
    add_to_variable = { StateMuslimTotal = StateIslamIbadi }

    set_variable = { StateBuddhismTotal = StateBuddhismVajrayana }
    add_to_variable = { StateBuddhismTotal = StateBuddhismMahayana }
    add_to_variable = { StateBuddhismTotal = StateBuddhismThereveda }
    add_to_variable = { StateBuddhismTotal = StateNavayana }

    set_variable = { StateHinduTotal = StateHindu }

    set_variable = { StateOtherTotal = StateJudaism }
    add_to_variable = { StateOtherTotal = StateShinto  }
    add_to_variable = { StateOtherTotal = StateTraditionalEastern }
    add_to_variable = { StateOtherTotal = StateConfucian }
    add_to_variable = { StateOtherTotal = StateSikh }
    add_to_variable = { StateOtherTotal = StateTraditionalIranian }
    add_to_variable = { StateOtherTotal = StateIrreligious }
    add_to_variable = { StateOtherTotal = StateAtheist }
    add_to_variable = { StateOtherTotal = StateStatist }
    add_to_variable = { StateOtherTotal = StateCultistSafe }
    add_to_variable = { StateOtherTotal = StateCultistFatal }
    add_to_variable = { StateOtherTotal = StateTraditionalAfrican }
    add_to_variable = { StateOtherTotal = StateTraditionalAmerican }
    add_to_variable = { StateOtherTotal = StateTraditionalOceanic }
    add_to_variable = { StateOtherTotal = StateChineseSalvationist }
    add_to_variable = { StateOtherTotal = StateMessianic }
    add_to_variable = { StateOtherTotal = StateOther }

    #Control Map Setting
	if = {	
		limit = {
			check_variable = { controller = INO } 
					
		}
		set_variable = { Indonesia_Control = 25 }
	}
	else_if = {
		limit = {
            OR = {
                check_variable = { controller = INS }
                check_variable = { controller = HOL }
                check_variable = { controller = WPA }
            }

		}
		set_variable = { Indonesia_Control = 11 }
	}
	else_if = {
		limit = {
			MLA = {
				has_completed_focus = MLA_Found_the_Peoples_Republic_of_Malaysia
			}
			OR = {
				check_variable = { owner = SWK } 
				check_variable = { owner = NBO } 
                check_variable = { owner = MAL } 
                check_variable = { owner = MLA } 
			}
		}
		set_variable = { Indonesia_Control = 26 }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { owner = POR } 
				check_variable = { owner = TIM } 
			}
		}
		set_variable = { Indonesia_Control = 23 }
	}
    else = {     
        set_variable = { Indonesia_Control = 24 }
    }

    #Religion Map Setting
    if = {  #6 - Muslim
        limit = {
            check_variable = { StateMuslimTotal > StateBuddhismTotal }
            check_variable = { StateMuslimTotal > StateChristianTotal }
            check_variable = { StateMuslimTotal > StateOtherTotal }
            check_variable = { StateMuslimTotal > StateHinduTotal }
        }
        set_variable = { Indonesia_Religion = 31 }
    }
    else_if = {  #7 - Buddhist
        limit = {
            check_variable = { StateBuddhismTotal > StateMuslimTotal }
            check_variable = { StateBuddhismTotal > StateChristianTotal }
            check_variable = { StateBuddhismTotal > StateOtherTotal }
            check_variable = { StateBuddhismTotal > StateHinduTotal }
        }
        set_variable = { Indonesia_Religion = 32 }
    }
    else_if = {  #8 - Chrsitian
        limit = {
            check_variable = { StateChristianTotal > StateBuddhismTotal }
            check_variable = { StateChristianTotal > StateMuslimTotal }
            check_variable = { StateChristianTotal > StateOtherTotal }
            check_variable = { StateChristianTotal > StateHinduTotal }
        }
        set_variable = { Indonesia_Religion = 33 }
    }
    else_if = {  #9 - Hindu
        limit = {
            check_variable = { StateHinduTotal > StateBuddhismTotal }
            check_variable = { StateHinduTotal > StateChristianTotal }
            check_variable = { StateHinduTotal > StateMuslimTotal }
            check_variable = { StateHinduTotal > StateOtherTotal }
        }
        set_variable = { Indonesia_Religion = 35 }
    }
    else_if = {  #10 - Other
        limit = {
            check_variable = { StateOtherTotal > StateBuddhismTotal }
            check_variable = { StateOtherTotal > StateChristianTotal }
            check_variable = { StateOtherTotal > StateMuslimTotal }
            check_variable = { StateOtherTotal > StateHinduTotal }
        }
        set_variable = { Indonesia_Religion = 10 }
    }
    else = { set_variable = { Indonesia_Religion = 31 } }

        

    if = { limit = { check_variable = { culture^0 = 66 } } set_variable = { Indonesia_Ethnicity = 36 } }
    else_if = { limit = { check_variable = { culture^0 = 435 } } set_variable = { Indonesia_Ethnicity = 37 } }              
    else_if = { limit = { check_variable = { culture^0 = 68 } } set_variable = { Indonesia_Ethnicity = 38 } }              
    else_if = { limit = { check_variable = { culture^0 = 67 } } set_variable = { Indonesia_Ethnicity = 39 } }              
    else_if = { limit = { check_variable = { culture^0 = 273 } } set_variable = { Indonesia_Ethnicity = 40 } }              
    else_if = { limit = { check_variable = { culture^0 = 275 } } set_variable = { Indonesia_Ethnicity = 41 } }              
    else_if = { limit = { check_variable = { culture^0 = 74 } } set_variable = { Indonesia_Ethnicity = 42 } }              
    else_if = { limit = { check_variable = { culture^0 = 279 } } set_variable = { Indonesia_Ethnicity = 43 } }              
    else_if = { limit = { check_variable = { culture^0 = 280 } } set_variable = { Indonesia_Ethnicity = 44 } }              
    else_if = { limit = { check_variable = { culture^0 = 69 } } set_variable = { Indonesia_Ethnicity = 45 } }              
    else_if = { limit = { check_variable = { culture^0 = 65 } } set_variable = { Indonesia_Ethnicity = 46 } }              
    else_if = { limit = { check_variable = { culture^0 = 121 } } set_variable = { Indonesia_Ethnicity = 47 } }              
    else_if = { limit = { check_variable = { culture^0 = 73 } } set_variable = { Indonesia_Ethnicity = 48 } }
