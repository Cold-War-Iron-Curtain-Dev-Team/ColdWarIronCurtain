nlf_dak_doa_update_hold = {
	if = {
		limit = { country_exists = FRE FRE = { has_country_flag = FRE_Dak_Doa_Posture_Evacuate } }
		add_to_variable = { NLF_Dak_Doa_Hold_Days = 1 }
	}
	else_if = {
		limit = { NLF_dak_doa_objective_held_trigger = yes }
		add_to_variable = { NLF_Dak_Doa_Hold_Days = 1 }
	}
	else = { set_variable = { NLF_Dak_Doa_Hold_Days = 0 } }
}

nlf_dak_doa_check_resolution = {
	if = {
		limit = { NLF_dak_doa_superseded_trigger = yes }
		set_temp_variable = { nlf_dak_doa_outcome = 5 }
		nlf_dak_doa_resolve = yes
	}
	else_if = {
		limit = { country_exists = FRE FRE = { has_country_flag = FRE_Dak_Doa_Posture_Evacuate } check_variable = { NLF_Dak_Doa_Hold_Days > 6 } }
		set_temp_variable = { nlf_dak_doa_outcome = 1 }
		nlf_dak_doa_resolve = yes
	}
	else_if = {
		limit = { check_variable = { NLF_Dak_Doa_Hold_Days > 6 } NOT = { check_variable = { NLF_Dak_Doa_Days_Elapsed > 45 } } }
		if = { limit = { NOT = { check_variable = { NLF_Dak_Doa_Days_Elapsed > 21 } } } set_temp_variable = { nlf_dak_doa_outcome = 1 } }
		else = { set_temp_variable = { nlf_dak_doa_outcome = 2 } }
		nlf_dak_doa_resolve = yes
	}
	else_if = {
		limit = { check_variable = { NLF_Dak_Doa_Days_Elapsed > 45 } }
		set_temp_variable = { nlf_dak_doa_outcome = 4 }
		nlf_dak_doa_resolve = yes
	}
	else_if = {
		limit = { NLF_dak_doa_war_abandoned_trigger = yes }
		set_temp_variable = { nlf_dak_doa_outcome = 3 }
		nlf_dak_doa_resolve = yes
	}
}
