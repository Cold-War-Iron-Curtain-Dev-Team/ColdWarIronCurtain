# ============================================================================
# Indochina failsafe QA harness
# ----------------------------------------------------------------------------
# Console only. Run from any country scope with "e <name>". The break_* effects
# each reproduce one of the edge cases the layer exists to catch; the router is
# then either left to the on_actions or invoked directly with
# test_ic_failsafe_route_now. Read the results out of the game log.
#
# The reported bug is test_ic_failsafe_break_vin_to_fre.
# ============================================================================

test_ic_failsafe_status = {
	log = "IC FAILSAFE: phase=[?global.Indochina_War_Active_Phase|0] ending_available=[?global.indochina_ending_available_check|0]"
	log = "IC FAILSAFE SCORES: comm=[?global.ic_score_comm|0] profr=[?global.ic_score_profr|0] proind=[?global.ic_score_proind|0] proeth=[?global.ic_score_proeth|0] kmt=[?global.ic_score_kmt|0] anticomm=[?global.ic_score_anticomm|0]"
	log = "IC FAILSAFE STALLS: carrier=[?global.ic_stall_carrier_days|0] decided=[?global.ic_stall_decided_days|0] geneva=[?global.ic_stall_geneva_days|0] latch=[?global.ic_stall_latch_days|0] unclicked=[?global.ic_stall_unclicked_days|0] nowar=[?global.ic_stall_nowar_days|0]"
	if = { limit = { has_global_flag = ic_failsafe_theatre_present } log = "IC FAILSAFE: armed" }
	else = { log = "IC FAILSAFE: NOT armed - ic_failsafe_theatre_present is unset" }
	if = { limit = { has_global_flag = Indochina_War_Over } log = "IC FAILSAFE: Indochina_War_Over is SET" }
	else = { log = "IC FAILSAFE: Indochina_War_Over is unset" }
	if = { limit = { has_global_flag = ic_failsafe_fired } log = "IC FAILSAFE: the router has run at least once" }
	if = { limit = { ic_failsafe_militarily_decided_trigger = yes } log = "IC FAILSAFE: militarily decided = TRUE" }
	if = { limit = { ic_failsafe_geneva_latched_dead_trigger = yes } log = "IC FAILSAFE: Geneva latch is DEAD" }
	if = { limit = { ic_failsafe_geneva_live_trigger = yes } log = "IC FAILSAFE: a conference is genuinely running" }
	if = { limit = { ic_failsafe_no_theatre_war_trigger = yes } log = "IC FAILSAFE: nobody in the theatre is at war" }
	if = { limit = { ic_failsafe_ending_owner_is_human_trigger = yes } log = "IC FAILSAFE: a human can take the ending (90 day grace)" }
	else = { log = "IC FAILSAFE: no human can take the ending (60 day grace)" }
}

# The reported bug. The corps takes Tonkin outright and the Viet Minh is wiped
# off the map without VIE ever having been the belligerent, which is exactly the
# case the old VIE_on_actions hook could not see.
test_ic_failsafe_break_vin_to_fre = {
	test_ic_failsafe_arm = yes
	if = {
		limit = { country_exists = FRE }
		FRE = {
			transfer_state = 1760
			transfer_state = 786
			transfer_state = 838
			transfer_state = 881
			if = {
				limit = { country_exists = VIN }
				annex_country = { target = VIN transfer_troops = no }
			}
		}
	}
	log = "IC FAILSAFE TEST: FRE has taken Tonkin and annexed VIN"
	test_ic_failsafe_status = yes
}

# France dies in Europe: on_daily_FRA stops and every FRA-scope score read goes
# with it. The carrier chain should pick the tick up on VIN.
test_ic_failsafe_kill_fra = {
	test_ic_failsafe_arm = yes
	if = {
		limit = { country_exists = FRA }
		if = {
			limit = { country_exists = SOV }
			SOV = { annex_country = { target = FRA transfer_troops = no } }
		}
		else_if = {
			limit = { country_exists = ENG }
			ENG = { annex_country = { target = FRA transfer_troops = no } }
		}
	}
	log = "IC FAILSAFE TEST: France annexed - the watchdog should route in 14 days"
	test_ic_failsafe_status = yes
}

test_ic_failsafe_kill_all_principals = {
	test_ic_failsafe_arm = yes
	if = {
		limit = { country_exists = PRC }
		PRC = {

			if = { limit = { country_exists = VIE } annex_country = { target = VIE transfer_troops = no } }
		}

	test_ic_failsafe_status = yes
}

# The half of the reported bug that left two French-aligned Vietnams.
test_ic_failsafe_puppet_vin_to_fra = {
	test_ic_failsafe_arm = yes
	if = {
		limit = {
			country_exists = FRA
			country_exists = VIN
		}
		FRA = {
			puppet = { target = VIN }
			set_autonomy = {
				target = VIN
				autonomy_state = autonomy_indon_union
				end_wars = no
				end_civil_wars = no
			}
		}
	}
	log = "IC FAILSAFE TEST: VIN is now a French puppet - break_bad_subjects should free it"
	test_ic_failsafe_status = yes
}

# The one-way latch with nothing behind it: every non-Geneva ending is now shut
# and the conference will never convene.
test_ic_failsafe_geneva_latch = {
	test_ic_failsafe_arm = yes
	set_global_flag = Geneva_Conference
	clr_global_flag = geneva_conference_announced
	clr_global_flag = geneva_conference_in_session
	clr_global_flag = geneva_conference_concluded
	clr_global_flag = Geneva_Conference_Negotiations_Complete
	set_variable = { global.ic_stall_latch_days = 0 }
	log = "IC FAILSAFE TEST: dead Geneva latch set - abort at 60 days, route at 90"
	test_ic_failsafe_status = yes
}
