# MKSEARCH continuation (design 4 / 7): FY1966-FY1972 successor remit inside the
# CIA charter, opened only when the Inspector General review chose reorganization.
# Historical anchor dossier with an authored, inconclusive evidence profile: it
# has no renewal (narrower remit) and its only finding is a records inventory.
#
# MKSEARCH stages: 0 locked, 1 available, 2 initial research, 3 initial report,
# 4 replication research, 5 final appraisal, 6 archived, 7 cancelled,
# 8 foreclosed (funding ended or charter closed before commission).
# The slot it occupies is stored in USA_mkultra_mksearch_slot.

USA_mkultra_open_mksearch_remit = {
	if = {
		limit = {
			USA_mkultra_charter_active_trigger = yes
			NOT = { has_country_flag = USA_mkultra_mksearch_remit }
		}
		set_country_flag = USA_mkultra_mksearch_remit
		# The narrower remit has no room for a safehouse program nobody opened.
		# Work already under way keeps its committed stages but cannot renew.
		if = {
			limit = { check_variable = { USA_mkultra_midnight_climax_stage = 1 } }
			set_variable = { USA_mkultra_midnight_climax_stage = 9 }
		}
		if = {
			limit = { check_variable = { USA_mkultra_mksearch_stage = 0 } }
			set_variable = { USA_mkultra_mksearch_stage = 1 }
		}
		USA_mkultra_gui_reconcile = yes
		USA_mkultra_mark_gui_dirty = yes
	}
}

USA_mkultra_start_mksearch = {
	if = {
		limit = { USA_mkultra_can_start_mksearch_trigger = yes }
		set_country_flag = USA_mkultra_mksearch_paid
		set_country_flag = USA_mkultra_mksearch_owner_cia
		set_variable = { USA_mkultra_mksearch_stage = 2 }
		if = {
			limit = { check_variable = { USA_mkultra_slot_1_state = 0 } }
			set_variable = { USA_mkultra_mksearch_slot = 1 }
		}
		else = {
			set_variable = { USA_mkultra_mksearch_slot = 2 }
		}

		if = {
			limit = { check_variable = { USA_mkultra_information_policy = 2 } }
			add_to_variable = { USA_mkultra_exposure = 6 }
		}
		else = {
			add_to_variable = { USA_mkultra_exposure = 8 }
		}
		add_to_variable = { USA_mkultra_harm = 8 }

		if = {
			limit = { check_variable = { USA_mkultra_capacity = 2 } }
			add_political_power = -5
			add_to_variable = { USA_mkultra_exposure = 1 }
		}

		USA_mkultra_clamp_variables = yes
		USA_mkultra_evaluate_basic_inquiry = yes
		activate_mission = USA_mkultra_mksearch_research
		USA_mkultra_recompute_slot_states = yes
		USA_mkultra_mark_gui_dirty = yes
	}
}

USA_mkultra_complete_mksearch = {
	if = {
		limit = {
			USA_mkultra_charter_active_trigger = yes
			check_variable = { USA_mkultra_mksearch_stage = 2 }
			has_country_flag = USA_mkultra_mksearch_paid
			has_country_flag = USA_mkultra_mksearch_owner_cia
			NOT = { has_country_flag = USA_mkultra_mksearch_record_awarded }
		}
		set_country_flag = USA_mkultra_mksearch_record_awarded
		add_to_variable = { USA_mkultra_progress = 16 }
		set_variable = { USA_mkultra_mksearch_stage = 3 }
		if = {
			limit = { has_active_mission = USA_mkultra_mksearch_research }
			remove_mission = USA_mkultra_mksearch_research
		}
		USA_mkultra_clamp_variables = yes
		activate_mission = USA_mkultra_mksearch_review
		USA_mkultra_recompute_slot_states = yes
		USA_mkultra_mark_gui_dirty = yes
	}
}

USA_mkultra_start_mksearch_replication = {
	if = {
		limit = { USA_mkultra_can_replicate_mksearch_trigger = yes }
		set_country_flag = USA_mkultra_mksearch_replication_paid
		set_variable = { USA_mkultra_mksearch_stage = 4 }
		USA_mkultra_charge_replication_stage = yes
		USA_mkultra_apply_reliable_appraisal_discount = yes
		if = {
			limit = { has_active_mission = USA_mkultra_mksearch_review }
			remove_mission = USA_mkultra_mksearch_review
		}
		USA_mkultra_clamp_variables = yes
		USA_mkultra_evaluate_basic_inquiry = yes
		activate_mission = USA_mkultra_mksearch_replication_research
		USA_mkultra_recompute_slot_states = yes
		USA_mkultra_mark_gui_dirty = yes
	}
}

USA_mkultra_complete_mksearch_replication = {
	if = {
		limit = {
			USA_mkultra_charter_active_trigger = yes
			check_variable = { USA_mkultra_mksearch_stage = 4 }
			has_country_flag = USA_mkultra_mksearch_replication_paid
			NOT = { has_country_flag = USA_mkultra_mksearch_replication_record_awarded }
		}
		set_country_flag = USA_mkultra_mksearch_replication_record_awarded
		add_to_variable = { USA_mkultra_progress = 6 }
		set_variable = { USA_mkultra_mksearch_stage = 5 }
		if = {
			limit = { has_active_mission = USA_mkultra_mksearch_replication_research }
			remove_mission = USA_mkultra_mksearch_replication_research
		}
		USA_mkultra_clamp_variables = yes
		activate_mission = USA_mkultra_mksearch_final_review
		USA_mkultra_recompute_slot_states = yes
		USA_mkultra_mark_gui_dirty = yes
	}
}

# The only finding: an independently checked inventory of which lines survived
# reorganization. It is the "preserved transferable archive" (design 9) a later
# related program can draw on, and it is lost if the working files are destroyed.
USA_mkultra_apply_mksearch_inventory = {
	if = {
		limit = { USA_mkultra_can_apply_mksearch_inventory_trigger = yes }
		set_country_flag = USA_mkultra_mksearch_inventory_earned
		USA_mkultra_archive_mksearch = yes
	}
}

USA_mkultra_archive_mksearch = {
	if = {
		limit = {
			USA_mkultra_enabled_trigger = yes
			OR = {
				check_variable = { USA_mkultra_mksearch_stage = 3 }
				check_variable = { USA_mkultra_mksearch_stage = 5 }
			}
		}
		set_country_flag = USA_mkultra_mksearch_archived
		set_variable = { USA_mkultra_mksearch_stage = 6 }
		if = {
			limit = { has_active_mission = USA_mkultra_mksearch_review }
			remove_mission = USA_mkultra_mksearch_review
		}
		if = {
			limit = { has_active_mission = USA_mkultra_mksearch_final_review }
			remove_mission = USA_mkultra_mksearch_final_review
		}
		USA_mkultra_recompute_slot_states = yes
		USA_mkultra_gui_reconcile = yes
		USA_mkultra_mark_gui_dirty = yes
	}
}

# No refund; exposure and harm committed at the start remain.
USA_mkultra_cancel_mksearch = {
	if = {
		limit = {
			OR = {
				check_variable = { USA_mkultra_mksearch_stage = 2 }
				check_variable = { USA_mkultra_mksearch_stage = 4 }
			}
		}
		set_variable = { USA_mkultra_mksearch_stage = 7 }
		if = {
			limit = { has_active_mission = USA_mkultra_mksearch_research }
			remove_mission = USA_mkultra_mksearch_research
		}
		if = {
			limit = { has_active_mission = USA_mkultra_mksearch_replication_research }
			remove_mission = USA_mkultra_mksearch_replication_research
		}
		USA_mkultra_recompute_slot_states = yes
		USA_mkultra_gui_reconcile = yes
		USA_mkultra_mark_gui_dirty = yes
	}
}

# Settles every MKSEARCH stage once: used when FY1972 funding ends and when the
# CIA charter closes. Running work is cancelled, a pending report is archived
# without its finding, and an uncommissioned dossier is foreclosed.
USA_mkultra_settle_mksearch = {
	USA_mkultra_cancel_mksearch = yes
	USA_mkultra_archive_mksearch = yes
	if = {
		limit = { check_variable = { USA_mkultra_mksearch_stage = 1 } }
		set_variable = { USA_mkultra_mksearch_stage = 8 }
		USA_mkultra_gui_reconcile = yes
		USA_mkultra_mark_gui_dirty = yes
	}
}

USA_mkultra_end_mksearch_funding = {
	if = {
		limit = {
			has_country_flag = USA_mkultra_mksearch_remit
			NOT = { has_country_flag = USA_mkultra_mksearch_funding_ended }
		}
		set_country_flag = USA_mkultra_mksearch_funding_ended
		USA_mkultra_settle_mksearch = yes
	}
}

# Called from USA_mkultra_phase_2_reconcile. Repairs only derived state from
# durable flags; never replays a payment, Research Record award, or finding.
USA_mkultra_mksearch_reconcile = {
	if = {
		limit = { has_country_flag = USA_mkultra_mksearch_archived }
		set_variable = { USA_mkultra_mksearch_stage = 6 }
		if = {
			limit = { has_active_mission = USA_mkultra_mksearch_review }
			remove_mission = USA_mkultra_mksearch_review
		}
		if = {
			limit = { has_active_mission = USA_mkultra_mksearch_final_review }
			remove_mission = USA_mkultra_mksearch_final_review
		}
	}
	if = {
		limit = {
			has_country_flag = USA_mkultra_mksearch_funding_ended
			OR = {
				check_variable = { USA_mkultra_mksearch_stage = 1 }
				check_variable = { USA_mkultra_mksearch_stage = 2 }
				check_variable = { USA_mkultra_mksearch_stage = 3 }
				check_variable = { USA_mkultra_mksearch_stage = 4 }
				check_variable = { USA_mkultra_mksearch_stage = 5 }
			}
		}
		USA_mkultra_settle_mksearch = yes
	}
	if = {
		limit = { USA_mkultra_charter_active_trigger = yes }
		if = {
			limit = {
				check_variable = { USA_mkultra_mksearch_stage = 2 }
				has_country_flag = USA_mkultra_mksearch_record_awarded
			}
			set_variable = { USA_mkultra_mksearch_stage = 3 }
		}
		if = {
			limit = {
				check_variable = { USA_mkultra_mksearch_stage = 4 }
				has_country_flag = USA_mkultra_mksearch_replication_record_awarded
			}
			set_variable = { USA_mkultra_mksearch_stage = 5 }
		}
		if = {
			limit = {
				check_variable = { USA_mkultra_mksearch_stage = 2 }
				NOT = { has_active_mission = USA_mkultra_mksearch_research }
			}
			activate_mission = USA_mkultra_mksearch_research
		}
		if = {
			limit = {
				check_variable = { USA_mkultra_mksearch_stage = 3 }
				NOT = { has_active_mission = USA_mkultra_mksearch_review }
			}
			if = {
				limit = { has_active_mission = USA_mkultra_mksearch_research }
				remove_mission = USA_mkultra_mksearch_research
			}
			activate_mission = USA_mkultra_mksearch_review
		}
		if = {
			limit = {
				check_variable = { USA_mkultra_mksearch_stage = 4 }
				NOT = { has_active_mission = USA_mkultra_mksearch_replication_research }
			}
			activate_mission = USA_mkultra_mksearch_replication_research
		}
		if = {
			limit = {
				check_variable = { USA_mkultra_mksearch_stage = 5 }
				NOT = { has_active_mission = USA_mkultra_mksearch_final_review }
			}
			if = {
				limit = { has_active_mission = USA_mkultra_mksearch_replication_research }
				remove_mission = USA_mkultra_mksearch_replication_research
			}
			activate_mission = USA_mkultra_mksearch_final_review
		}
	}
}
