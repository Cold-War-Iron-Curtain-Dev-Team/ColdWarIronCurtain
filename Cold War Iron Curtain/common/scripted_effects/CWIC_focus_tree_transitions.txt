# Repair countries caught by the former blanket January 1960 reset.
cwic_repair_1960_focus_tree = {
	if = {
		limit = {
			tag = SAF
			OR = {
				has_focus_tree = SAF_1950s
				AND = {
					has_country_flag = focus_tree_60_reset
					has_focus_tree = empty_focus
				}
			}
		}
		load_focus_tree = { tree = SAF_1960s keep_completed = yes }
		set_country_flag = focus_tree_60_reset
	}
	else_if = {
		limit = {
			tag = VIN
			has_country_flag = focus_tree_60_reset
			has_focus_tree = empty_focus
		}
		load_focus_tree = { tree = VIN_1950s keep_completed = yes }
	}
}

# Older 1980 saves can still have the automatic starter assigned. Later and
# alternate trees are handled by their campaign effects.
cwic_repair_1980_focus_tree = {
	if = {
		limit = { tag = ENG has_focus_tree = UK50 }
		load_focus_tree = { tree = ENG_60s keep_completed = yes }
	}
	else_if = {
		limit = { tag = WGR has_focus_tree = WGR_50s_Starting_Tree }
		load_focus_tree = { tree = WGR_60s keep_completed = yes }
	}
	else_if = {
		limit = { tag = RAJ has_focus_tree = INDIA_50s }
		load_focus_tree = { tree = INDIA_70s keep_completed = yes }
	}
	else_if = {
		limit = { tag = PRC has_focus_tree = PRC_National_Focus }
		load_focus_tree = { tree = PRC_70s keep_completed = yes }
	}
	else_if = {
		limit = { tag = KOR has_focus_tree = KOR_Syngman_Rhee_1948 }
		load_focus_tree = { tree = KOR_60s keep_completed = yes }
	}
	else_if = {
		limit = { tag = SAF has_focus_tree = SAF_1950s }
		load_focus_tree = { tree = SAF_1960s keep_completed = yes }
	}
	else_if = {
		limit = { tag = BUL has_focus_tree = 1950s_BUL }
		load_focus_tree = { tree = BUL_60s keep_completed = yes }
	}
}
