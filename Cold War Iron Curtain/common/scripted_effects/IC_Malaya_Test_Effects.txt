# QA harness for the Malayan Emergency. Run from the console with `e <effect_name>`.

# Liberates four contested states: one short of the General Uprising threshold.
test_mla_contest_near_win = {
    MLA = {
        set_country_flag = active_insurgency
        set_country_flag = MLA_final_perak_ready
        set_country_flag = MLA_final_kuala_lumpur_ready
        set_country_flag = MLA_final_terengganu_ready
    }
    791 = { set_variable = { emergency_control = 80 } }
    960 = { set_variable = { emergency_control = 80 } }
    1676 = { set_variable = { emergency_control = 80 } }
    1895 = { set_variable = { emergency_control = 80 } }
    every_state = {
        limit = { MAL_emergency_is_contest_state = yes }
        MAL_emergency_set_control_frame = yes
    }
    MLA = { MAL_emergency_recount = yes }
}

# Liberates five contested states: the General Uprising threshold.
test_mla_contest_win = {
    test_mla_contest_near_win = yes
    1894 = { set_variable = { emergency_control = 80 } MAL_emergency_set_control_frame = yes }
    MLA = { MAL_emergency_recount = yes }
}

# Secures seven contested states: the Final Sweep threshold.
test_mal_contest_win = {
    MLA = { set_country_flag = active_insurgency }
    every_state = {
        limit = {
            MAL_emergency_is_contest_state = yes
            NOT = { state = 791 }
            NOT = { state = 960 }
        }
        set_variable = { emergency_control = -80 }
        MAL_emergency_set_control_frame = yes
    }
    MLA = { MAL_emergency_recount = yes }
}

# Seeds cells across the peninsula so the uprising's set_state_controller sweep has targets.
test_mla_seed_cells = {
    every_state = {
        limit = {
            is_core_of = MLA
            owner = { OR = { tag = MAL tag = ENG } }
        }
        set_state_flag = MLA_cell_present
    }
}

# Full MNLA path: seed cells, bank a plausible operation record, then declare the war.
test_mla_launch_uprising = {
    test_mla_seed_cells = yes
    test_mla_contest_win = yes
    MLA_Launch_General_Uprising = yes
}
