# Stage 9a: closeout handlers and builders for the two scripted projects that
# were listed in the Black Operations window but could never be launched:
#   Cuban Regime Change (USA, tokens 100-116)
#   Pacifying Western Ukraine (SOV, tokens 500-506)
#
# CWIC_espionage_dispatch_project_outcome runs at the end of
# CWIC_espionage_dispatch_operation_outcome, in the launching country's scope,
# after operation_closeout_effect has rolled slot_one_successful and
# slot_one_detected. current_token is the temp variable set by the closeout.
# Token 100 keeps its handler in IC_Espionage_effects.txt.
#
# Every flag and variable written here is one that
# current_entry_visible_trigger (scripted_triggers/IC_Espionage.txt) or the
# project panel already reads. Nothing wrote them before this file.

CWIC_espionage_dispatch_project_outcome = {
    if = {
        limit = {
            check_variable = { current_token > 100 }
            check_variable = { current_token < 117 }
        }
        if = {
            limit = { check_variable = { current_token = 116 } }
            CWIC_espionage_bop_roll_liberation = yes
        }
        if = {
            limit = { check_variable = { slot_one_successful = 1 } }
            CWIC_espionage_bop_success = yes
            slot_1_operation_success_shared_effect = yes
        }
        else = {
            CWIC_espionage_bop_failure = yes
            slot_1_operation_failure_effect = yes
        }
        CWIC_espionage_bop_recount = yes
        CWIC_espionage_refresh_operation_status = yes
    }
    if = {
        limit = {
            check_variable = { current_token > 499 }
            check_variable = { current_token < 507 }
        }
        if = {
            limit = { check_variable = { slot_one_successful = 1 } }
            CWIC_espionage_uia_success = yes
            slot_1_operation_success_shared_effect = yes
        }
        else = {
            CWIC_espionage_uia_failure = yes
            slot_1_operation_failure_effect = yes
        }
        CWIC_espionage_uia_check_end = yes
        CWIC_espionage_refresh_operation_status = yes
    }
}

CWIC_espionage_bop_success = {
    if = {
        limit = { check_variable = { current_token = 101 } }    # Establish Brigade 2506
        set_country_flag = Brigade_2506_Token
        add_to_variable = { Bay_of_Pigs_Recruits = 200 }
    }
    else_if = {
        limit = { check_variable = { current_token = 102 } }    # Pilot Recruitment
        set_country_flag = Recruited_Cuban_Pilots_Token
        add_to_variable = { Bay_of_Pigs_Pilots = 25 }
    }
    else_if = {
        limit = { check_variable = { current_token = 103 } }    # Acquire Aircraft
        set_country_flag = Acquired_Aircraft_for_Cuban_Project_Token
        add_to_variable = { Bay_of_Pigs_B26s = 8 }
        add_to_variable = { Bay_of_Pigs_C46s = 4 }
    }
    else_if = {
        limit = { check_variable = { current_token = 104 } }    # Cuban Liberation Air Force
        set_country_flag = Cuban_Liberation_Air_Force_Token
    }
    else_if = {
        limit = { check_variable = { current_token = 105 } }    # Rayo Base
        set_country_flag = Rayo_Base_Token
        if = {
            limit = { GUA = { exists = yes } }
            GUA = { add_opinion_modifier = { target = ROOT modifier = CWIC_espionage_bop_base_hosted } }
        }
    }
    else_if = {
        limit = { check_variable = { current_token = 106 } }    # Pilot Training
        set_country_flag = Pilots_Trained_Cuba_Project_Token
        add_to_variable = { Bay_of_Pigs_Pilots = 30 }
    }
    else_if = {
        limit = { check_variable = { current_token = 107 } }    # Recruit Cuban Emigres
        add_to_variable = { Bay_of_Pigs_Recruits = 400 }
    }
    else_if = {
        limit = { check_variable = { current_token = 108 } }    # Divert Infantry Equipment
        add_equipment_to_stockpile = { type = infantry_equipment amount = -1000 }
        add_to_variable = { Bay_of_Pigs_Infantry_Equipment = 1000 }
    }
    else_if = {
        limit = { check_variable = { current_token = 109 } }    # Train Invasion Force Members
        set_variable = { CWIC_esp_bop_trainees = Bay_of_Pigs_Recruits }
        clamp_variable = { var = CWIC_esp_bop_trainees min = 0 max = 400 }
        subtract_from_variable = { Bay_of_Pigs_Recruits = CWIC_esp_bop_trainees }
        add_to_variable = { Bay_of_Pigs_Trained_Recruits = CWIC_esp_bop_trainees }
        add_to_variable = { Bay_of_Pigs_Experience = 15 }
    }
    else_if = {
        limit = { check_variable = { current_token = 110 } }    # Landing craft
        add_to_variable = { Bay_of_Pigs_Ships = 4 }
    }
    else_if = {
        limit = { check_variable = { current_token = 111 } }    # Infiltrate Cuba
        set_country_flag = Infilitrated_Cuba_Cuba_Project_Token
        set_variable = { CWIC_esp_bop_landing_site = 1 }
    }
    else_if = {
        limit = { check_variable = { current_token = 112 } }    # Contact insurgents in Cuba
        set_country_flag = Contacted_Insurgents_Cuba_Project_Token
        set_variable = { CWIC_esp_bop_landing_site = 2 }
        if = {
            limit = { CUB = { exists = yes } }
            CUB = {
                every_owned_state = {
                    limit = { has_resistance = yes }
                    add_resistance = 10
                }
            }
        }
    }
    else_if = {
        limit = { check_variable = { current_token = 113 } }    # Operation Puma
        set_country_flag = Operation_Puma_Token
        if = {
            limit = { CUB = { exists = yes } }
            CUB = { add_dynamic_modifier = { modifier = CWIC_espionage_bop_air_strikes days = 90 } }
        }
    }
    else_if = {
        limit = { check_variable = { current_token = 114 } }    # Psychological warfare over Havana
        set_country_flag = Psychological_Warfare_Havana_Token
        if = {
            limit = { CUB = { exists = yes } }
            CUB = {
                add_stability = -0.05
                add_war_support = -0.05
            }
        }
    }
    else_if = {
        limit = { check_variable = { current_token = 115 } }    # Diversionary strikes
        set_country_flag = Diversionary_Strikes_Cuba_Project_Token
    }
    else_if = {
        limit = { check_variable = { current_token = 116 } }    # The landing
        set_country_flag = Bay_of_Pigs_Start_Token
        set_variable = { CWIC_esp_bop_outcome = 2 }
        set_global_flag = bay_of_pigs_successful
        if = {
            limit = { CUB = { exists = yes } }
            CUB = {
                news_event = bay_of_pigs.4
                if = {
                    limit = { has_government = communism }
                    set_politics = { ruling_party = Social_Democratic elections_allowed = yes }
                    start_civil_war = { ideology = communism size = 0.35 }
                }
                else = {
                    add_stability = -0.15
                    add_war_support = -0.1
                }
            }
        }
    }
}

CWIC_espionage_bop_failure = {
    if = {
        limit = { check_variable = { current_token = 101 } }
        if = {
            limit = { check_variable = { slot_one_detected = 1 } }
            add_political_power = -10
        }
    }
    else_if = {
        limit = { check_variable = { current_token = 102 } }
        add_to_variable = { Bay_of_Pigs_Pilots = 5 }
    }
    else_if = {
        limit = { check_variable = { current_token = 103 } }
        add_to_variable = { Bay_of_Pigs_B26s = 4 }
    }
    else_if = {
        limit = { check_variable = { current_token = 105 } }
        if = {
            limit = { GUA = { exists = yes } }
            GUA = { add_opinion_modifier = { target = ROOT modifier = CWIC_espionage_bop_base_exposed } }
        }
    }
    else_if = {
        limit = { check_variable = { current_token = 106 } }
        subtract_from_variable = { Bay_of_Pigs_Pilots = 5 }
    }
    else_if = {
        limit = { check_variable = { current_token = 107 } }
        add_to_variable = { Bay_of_Pigs_Recruits = 100 }
    }
    else_if = {
        limit = { check_variable = { current_token = 108 } }
        add_equipment_to_stockpile = { type = infantry_equipment amount = -300 }
        add_to_variable = { Bay_of_Pigs_Infantry_Equipment = 300 }
    }
    else_if = {
        limit = { check_variable = { current_token = 109 } }
        add_to_variable = { Bay_of_Pigs_Experience = 5 }
    }
    else_if = {
        limit = { check_variable = { current_token = 110 } }
        add_to_variable = { Bay_of_Pigs_Ships = 1 }
    }
    else_if = {
        limit = { check_variable = { current_token = 111 } }
        if = {
            limit = { CUB = { exists = yes } }
            CUB = { add_opinion_modifier = { target = ROOT modifier = CWIC_espionage_bop_infiltrators_caught } }
        }
    }
    else_if = {
        limit = { check_variable = { current_token = 113 } }
        subtract_from_variable = { Bay_of_Pigs_B26s = 2 }
    }
    else_if = {
        limit = { check_variable = { current_token = 115 } }
        subtract_from_variable = { Bay_of_Pigs_B26s = 2 }
    }
    else_if = {
        limit = { check_variable = { current_token = 116 } }
        # One attempt only: the landing is spent whether or not it works.
        set_country_flag = Bay_of_Pigs_Start_Token
        set_variable = { CWIC_esp_bop_outcome = 1 }
        set_global_flag = bay_of_pigs_disaster
        add_political_power = -100
        if = {
            limit = { CUB = { exists = yes } }
            CUB = {
                news_event = bay_of_pigs.3
                add_stability = 0.1
            }
        }
    }
}

# The landing does not use the generic mission table: its odds come from how
# well the brigade was prepared. 10% base, up to +45% from training level,
# +5% each for Operation Puma and the diversionary strikes, +10% if the
# insurgents inside Cuba were contacted. Maximum 75%.
CWIC_espionage_bop_roll_liberation = {
    set_variable = { CWIC_esp_bop_chance = Bay_of_Pigs_Experience }
    multiply_variable = { CWIC_esp_bop_chance = 0.0045 }
    clamp_variable = { var = CWIC_esp_bop_chance min = 0 max = 0.45 }
    add_to_variable = { CWIC_esp_bop_chance = 0.1 }
    if = {
        limit = { has_country_flag = Operation_Puma_Token }
        add_to_variable = { CWIC_esp_bop_chance = 0.05 }
    }
    if = {
        limit = { has_country_flag = Diversionary_Strikes_Cuba_Project_Token }
        add_to_variable = { CWIC_esp_bop_chance = 0.05 }
    }
    if = {
        limit = { has_country_flag = Contacted_Insurgents_Cuba_Project_Token }
        add_to_variable = { CWIC_esp_bop_chance = 0.1 }
    }
    set_variable_to_random = CWIC_esp_bop_roll
    set_variable = { slot_one_successful = 0 }
    if = {
        limit = { check_variable = { CWIC_esp_bop_roll < CWIC_esp_bop_chance } }
        set_variable = { slot_one_successful = 1 }
    }
}

    clamp_variable = { var = Bay_of_Pigs_C54s min = 0 max = 999 }

    }
    clear_array = bay_of_pigs_operation_list_verification

    if = {
        limit = { has_country_flag = Brigade_2506_Token }

    }
    if = {
        limit = { has_country_flag = Cuban_Liberation_Air_Force_Token }

    if = {

    if = {
        limit = { check_variable = { Bay_of_Pigs_Trained_Recruits > 1499 } }

    }

# The operation list has an "in progress" icon and a "completed" tick that read
# local_possible_operations_completed_status (1 and 2). Nothing filled that
# array before. It is filled only while one of the two projects above is the
# open list; any other list keeps it empty (CWIC_espionage_reset_project_views
# clears it), so no stale marker can land on another project's rows.
CWIC_espionage_refresh_operation_status = {
    clear_array = local_possible_operations_completed_status
    if = {
        limit = {
            OR = {
                check_variable = { show_bay_of_pigs = 1 }
                check_variable = { show_uia = 1 }
            }
        }
        for_each_loop = {
            array = local_possible_operations_id_token
            index = CWIC_esp_status_index
            set_temp_variable = { CWIC_esp_status_token = local_possible_operations_id_token^CWIC_esp_status_index }
            set_temp_variable = { CWIC_esp_status = 0 }
            if = {
                limit = {
                    is_in_array = { array = current_operation_list value = CWIC_esp_status_token }
                    # Raids share token 501 across oblasts; do not mark every row.
                    NOT = { check_variable = { CWIC_esp_status_token = 501 } }
                }
                set_temp_variable = { CWIC_esp_status = 1 }
            }
            else_if = {
                limit = { CWIC_espionage_project_token_done_trigger = yes }
                set_temp_variable = { CWIC_esp_status = 2 }
            }
            add_to_array = { local_possible_operations_completed_status = CWIC_esp_status }
        }
    }
}
