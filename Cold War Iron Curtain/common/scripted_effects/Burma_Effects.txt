BUR_update_phase = {
	if = {
		limit = {
			check_variable = { global.BUR_deescalation > 499 }
			check_variable = { global.BUR_phase < 5 }
		}
		add_to_variable = { global.BUR_phase = 1 }
		set_variable = { global.BUR_deescalation = 0 }
		set_variable = { global.BUR_escalation = 0 }
		set_variable = { global.BUR_phase_change_dir = 1 }
		set_global_flag = { flag = BUR_phase_changed days = 7 }
	}
	else_if = {
		limit = {
			check_variable = { global.BUR_escalation > 499 }
			check_variable = { global.BUR_phase > 1 }
		}
		subtract_from_variable = { global.BUR_phase = 1 }
		set_variable = { global.BUR_deescalation = 0 }
		set_variable = { global.BUR_escalation = 0 }
		set_variable = { global.BUR_phase_change_dir = -1 }
		set_global_flag = { flag = BUR_phase_changed days = 7 }
	}
}

BUR_sync_dead_groups = {
	set_temp_variable = { BUR_i = 1 }
	while_loop_effect = {
		limit = { check_variable = { BUR_i < 13 } }
		set_temp_variable = { BUR_idx = BUR_i }
		if = {
			limit = {
				NOT = { check_variable = { BUR_i = 11 } }
				OR = {
					check_variable = { global.BUR_status^BUR_i = 0 }
					check_variable = { global.BUR_status^BUR_i = 4 }
				}
				NOT = { BUR_group_tag_exists = yes }
			}
			set_variable = { global.BUR_status^BUR_i = 2 }
			if = {
				limit = { BUR_logging_on = yes }
				log = "BUR_AFK|RESOLVE|day=[?global.num_days]|kind=tag_gone|group=[?BUR_i]"
			}
		}
		add_to_temp_variable = { BUR_i = 1 }
	}
	if = {
		limit = {
			has_global_flag = BUR_kmt_spawned
			check_variable = { global.BUR_status^11 = 0 }
			NOT = { country_exists = YAN }
		}
		set_variable = { global.BUR_status^11 = 2 }
	}
}

BUR_monthly_drift = {
	set_temp_variable = { BUR_hostile = 0 }
	set_temp_variable = { BUR_i = 1 }
	while_loop_effect = {
		limit = { check_variable = { BUR_i < 13 } }
		if = {
			limit = { check_variable = { global.BUR_status^BUR_i = 0 } }
			add_to_temp_variable = { BUR_hostile = global.BUR_strength^BUR_i }
		}
		add_to_temp_variable = { BUR_i = 1 }
	}
	set_temp_variable = { BUR_load = BUR_hostile }
	divide_temp_variable = { BUR_load = 100 }
	set_temp_variable = { BUR_de = 50 }
	subtract_from_temp_variable = { BUR_de = BUR_load }
	set_temp_variable = { BUR_es = BUR_load }
	subtract_from_temp_variable = { BUR_es = 20 }
	if = {
		limit = {
			OR = {
				NOT = { country_exists = BRM }
				BRM = { has_stability < 0.4 }
			}
		}
		add_to_temp_variable = { BUR_es = 10 }
	}
	clamp_temp_variable = { var = BUR_de min = 0 max = 50 }
	clamp_temp_variable = { var = BUR_es min = 0 max = 100 }
	add_to_variable = { global.BUR_deescalation = BUR_de }
	add_to_variable = { global.BUR_escalation = BUR_es }
	BUR_clamp_meters = yes
}

BUR_kmt_pulse = {
	if = {
		limit = {
			check_variable = { global.BUR_status^11 = 5 }
			country_exists = YAN
			OR = {
				YAN = { any_owned_state = { always = yes } }
				date > 1950.1.1
			}
		}
		set_variable = { global.BUR_status^11 = 0 }
		set_variable = { global.BUR_strength^11 = 150 }
		set_global_flag = BUR_kmt_spawned
		if = {
			limit = { BUR_logging_on = yes }
			log = "BUR_AFK|KMT_SPAWN|day=[?global.num_days]"
		}
	}
	else_if = {
		limit = {
			check_variable = { global.BUR_status^11 = 0 }
			check_variable = { global.BUR_strength^11 < 600 }
			date < 1952.1.1
		}
		add_to_variable = { global.BUR_strength^11 = 19 }
		BUR_clamp_meters = yes
	}
}

BUR_update_ending_available = {
	set_variable = { global.BUR_n_sub = 0 }
	set_variable = { global.BUR_n_fed = 0 }
	set_variable = { global.BUR_n_strong = 0 }
	set_temp_variable = { BUR_i = 1 }
	while_loop_effect = {
		limit = { check_variable = { BUR_i < 13 } }
		if = {
			limit = { check_variable = { global.BUR_status^BUR_i = 2 } }
			add_to_variable = { global.BUR_n_sub = 1 }
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { global.BUR_status^BUR_i = 1 }
					check_variable = { global.BUR_status^BUR_i = 3 }
				}
			}
			add_to_variable = { global.BUR_n_fed = 1 }
		}
		if = {
			limit = {
				check_variable = { BUR_i > 1 }
				NOT = { check_variable = { BUR_i = 11 } }
				check_variable = { global.BUR_status^BUR_i = 0 }
				check_variable = { global.BUR_strength^BUR_i > 399 }
			}
			add_to_variable = { global.BUR_n_strong = 1 }
		}
		add_to_temp_variable = { BUR_i = 1 }
	}
	if = {
		limit = { BUR_ending_land_to_the_tillers_trigger = yes }
		set_variable = { global.BUR_ending_available = 4 }
	}
	else_if = {
		limit = { BUR_ending_blood_of_the_tyrants_trigger = yes }
		set_variable = { global.BUR_ending_available = 2 }
	}
	else_if = {
		limit = { BUR_ending_orphaned_no_more_trigger = yes }
		set_variable = { global.BUR_ending_available = 5 }
	}
	else_if = {
		limit = { BUR_ending_bamar_supremacy_trigger = yes }
		set_variable = { global.BUR_ending_available = 1 }
	}
	else_if = {
		limit = { BUR_ending_federal_burma_trigger = yes }
		set_variable = { global.BUR_ending_available = 3 }
	}
	else = {
		set_variable = { global.BUR_ending_available = 0 }
	}
}

BUR_gui_sync_involvement = {
	if = {
		limit = { check_variable = { global.BUR_struggle_active = 1 } }
		BUR_sync_display = yes
		every_country = {
			limit = {
				OR = {
					tag = BRM
					tag = BCP
					tag = CPB
					tag = KAY
					tag = KND
					tag = KNN
					tag = MNL
					tag = SHA
					tag = ARA
					tag = ROH
					tag = KOK
					tag = KAC
					tag = CHN
					tag = YAN
					tag = PAO
					tag = PRC
					tag = USA
					tag = CHI
					tag = PAK
					tag = ENG
					tag = RAJ
				}
				NOT = { is_in_array = { array = country_politics_list value = 32 } }
			}
			add_to_array = { country_politics_list = 32 }
		}
	}
	else = {
		every_country = {
			limit = { is_in_array = { array = country_politics_list value = 32 } }
			remove_from_array = { array = country_politics_list value = 32 }
			set_variable = { View_Burma_Struggle = 0 }
		}
	}
}

	while_loop_effect = {
		limit = { check_variable = { BUR_i < 13 } }
		if = {
			limit = {
				NOT = { check_variable = { BUR_i = 11 } }
				check_variable = { global.BUR_status^BUR_i = 0 }
				check_variable = { global.BUR_friendliness^BUR_i > BUR_ai_best }
			}
			set_temp_variable = { BUR_ai_t = BUR_i }
			set_temp_variable = { BUR_ai_best = global.BUR_friendliness^BUR_i }
		}
		add_to_temp_variable = { BUR_i = 1 }
	}
}

BUR_ai_run_action = {
	set_temp_variable = { BUR_ai_prev = global.BUR_selected_group }
	set_variable = { global.BUR_selected_group = BUR_ai_t }
	if = {
		limit = { check_variable = { BUR_ai_act = 1 } }
		BUR_action_dialogue = yes
	}
	else_if = {
		limit = { check_variable = { BUR_ai_act = 2 } }
		BUR_action_amnesty = yes
	}
	else_if = {
		limit = { check_variable = { BUR_ai_act = 3 } }
		BUR_action_burmanize = yes
	}
	else_if = {
		limit = { check_variable = { BUR_ai_act = 4 } }
		BUR_action_sweep = yes
	}
	else_if = {
		limit = { check_variable = { BUR_ai_act = 5 } }
		BUR_action_frontier = yes
	}
	set_variable = { global.BUR_selected_group = BUR_ai_prev }
	BUR_clamp_meters = yes
	if = {
		limit = { BUR_logging_on = yes }
		log = "BUR_AFK|AI_ACTION|day=[?global.num_days]|action=[?BUR_ai_act]|group=[?BUR_ai_t]|str=[?global.BUR_strength^BUR_ai_t]|friend=[?global.BUR_friendliness^BUR_ai_t]|burm=[?global.BUR_burmanization^BUR_ai_t]"
	}
}

BUR_ai_pulse_brm = {
	set_temp_variable = { BUR_ai_act = 0 }
	set_temp_variable = { BUR_dbg_hist = 0 }
	if = { limit = { BUR_route_historical = yes } set_temp_variable = { BUR_dbg_hist = 1 } }
	if = { limit = { BUR_route_federal = yes } set_temp_variable = { BUR_dbg_hist = 2 } }
	if = { limit = { BUR_route_bamar = yes } set_temp_variable = { BUR_dbg_hist = 3 } }
	if = {
		limit = { BUR_logging_on = yes }
		log = "BUR_AFK|AI_PULSE|day=[?global.num_days]|pp=[?political_power]|cp=[?command_power]|mp=[?manpower_k]k|historical=[?BUR_dbg_hist]"
	}
	if = {
		limit = { BUR_route_brm_passive = yes }
		if = {
			limit = { BUR_logging_on = yes }
			log = "BUR_AFK|AI_PASSIVE|day=[?global.num_days]"
		}
	}
	else_if = {
		limit = { BUR_route_federal = yes }
		BUR_ai_pick_friendliest = yes
		if = {
			limit = { check_variable = { BUR_ai_t > 0 } }
			set_temp_variable = { BUR_idx = BUR_ai_t }
			if = {
				limit = { BUR_group_can_integrate = yes }
				BUR_resolve_integrate = yes
			}
			else_if = {
				limit = { BUR_group_can_autonomy = yes }
				BUR_resolve_autonomy = yes
			}
			else_if = {
				limit = {
					has_political_power > 49
					check_variable = { global.num_days > global.BUR_cd_dialogue^BUR_idx }
				}
				set_temp_variable = { BUR_ai_act = 1 }
			}
			else_if = {
				limit = {
					has_political_power > 99
					check_variable = { global.num_days > global.BUR_cd_frontier^BUR_idx }
				}
				set_temp_variable = { BUR_ai_act = 5 }
			}
			else_if = {
				limit = {
					has_political_power > 74
					check_variable = { global.BUR_friendliness^BUR_idx > 19 }
					check_variable = { global.num_days > global.BUR_cd_amnesty^BUR_idx }
				}
				set_temp_variable = { BUR_ai_act = 2 }
			}
		}
	}
	else = {
		BUR_ai_pick_strongest = yes
		if = {
			limit = { check_variable = { BUR_ai_t > 0 } }
			set_temp_variable = { BUR_idx = BUR_ai_t }
			if = {
				limit = {
					NOT = { BUR_route_historical = yes }
					BUR_group_can_subjugate = yes
				}
				BUR_resolve_subjugate = yes
			}
			else_if = {
				limit = {
					command_power > 9
					has_manpower > 1999
					check_variable = { global.num_days > global.BUR_cd_sweep^BUR_idx }
				}
				set_temp_variable = { BUR_ai_act = 4 }
			}
			else_if = {
				limit = {
					NOT = { BUR_route_historical = yes }
					check_variable = { BUR_ai_t < 11 }
					has_political_power > 49
					check_variable = { global.num_days > global.BUR_cd_burmanize^BUR_idx }
				}
				set_temp_variable = { BUR_ai_act = 3 }
			}
		}
	}
	if = {
		limit = { check_variable = { BUR_ai_act > 0 } }
		BUR_ai_run_action = yes
	}
}
