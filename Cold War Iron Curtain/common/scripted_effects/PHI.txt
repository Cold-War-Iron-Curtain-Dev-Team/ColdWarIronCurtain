
PHI_Recalc_Hacienda_Modifier = {
    set_variable = { 
		PHI_Hacienda_Control_Modifier = PHI_Hacienda_Control
	}
	multiply_variable = {
		PHI_Hacienda_Control_Modifier = 0.25
	}
}
PHI_Hacienda_Init = {
	PHI = {
		every_core_state = {
			limit = {
				NOT = {
					state = 327
				}
			}
			set_variable = { PHI_Hacienda_Control = 3 }
			PHI_Recalc_Hacienda_Modifier = yes
		}
		every_owned_state = {
			add_dynamic_modifier = {
				modifier = PHI_Hacienda_Control
			}
		}
	}
}
