# Phase 2 finite-state extension
# Missing Baseline stages: 0 available, 1 initial research, 2 initial report,
# 3 archived, 4 initial cancelled, 5 renewal research, 6 renewal report,
# 7 replication research, 8 final appraisal, 9 applied/closed,
# 10 renewal cancelled, 11 replication cancelled.
# Countermeasure stages: 0 locked, 1 available, 2 research, 3 report,
# 4 archived, 5 finding applied, 6 research cancelled, 7 foreclosed.

USA_mkultra_phase_2_initialize = {
	if = {
		limit = { USA_mkultra_enabled_trigger = yes }
		if = {
			limit = { NOT = { has_variable = USA_mkultra_schema_version } }
			set_variable = { USA_mkultra_schema_version = 3 }
		}
		if = {
			limit = { check_variable = { USA_mkultra_schema_version < 3 } }
			set_variable = { USA_mkultra_schema_version = 3 }
		}
		if = {
			limit = { NOT = { has_variable = USA_mkultra_countermeasure_stage } }
			set_variable = { USA_mkultra_countermeasure_stage = 0 }
		}
		if = {
			limit = { NOT = { has_variable = USA_mkultra_inquiry_state } }
			set_variable = { USA_mkultra_inquiry_state = 0 }
		}
		if = {
			limit = { NOT = { has_variable = USA_mkultra_security_scare_state } }
			set_variable = { USA_mkultra_security_scare_state = 0 }
		}
		set_country_flag = USA_mkultra_phase_2_migrated
		USA_mkultra_assign_missing_baseline_profile = yes
		USA_mkultra_gui_initialize = yes
	}
}

USA_mkultra_assign_missing_baseline_profile = {
	if = {
		limit = {
			USA_mkultra_enabled_trigger = yes
			has_country_flag = USA_mkultra_missing_baseline_paid
			NOT = { has_country_flag = USA_mkultra_missing_baseline_profile_assigned }
		}
		if = {
			limit = { has_variable = USA_mkultra_missing_baseline_profile }
			clamp_variable = {
				var = USA_mkultra_missing_baseline_profile
				min = 1
				max = 4
			}
		}
		else = {
			# Original fictional dossier. Equal 25% weights are a provisional,
			# explicit playtest assumption; the stored profile never rerolls.
			random_list = {
				25 = { set_variable = { USA_mkultra_missing_baseline_profile = 1 } }
				25 = { set_variable = { USA_mkultra_missing_baseline_profile = 2 } }
				25 = { set_variable = { USA_mkultra_missing_baseline_profile = 3 } }
				25 = { set_variable = { USA_mkultra_missing_baseline_profile = 4 } }
			}
		}
		set_country_flag = USA_mkultra_missing_baseline_profile_assigned
		USA_mkultra_mark_gui_dirty = yes
	}
}

USA_mkultra_phase_2_unlock_countermeasure = {
	if = {
		limit = {
			USA_mkultra_charter_active_trigger = yes
			has_country_flag = USA_mkultra_missing_baseline_archived
			check_variable = { USA_mkultra_countermeasure_stage = 0 }
			NOT = { has_country_flag = USA_mkultra_countermeasure_introduced }
		}
		set_country_flag = USA_mkultra_countermeasure_introduced
		set_variable = { USA_mkultra_countermeasure_stage = 1 }
		USA_mkultra_gui_reconcile = yes
		USA_mkultra_mark_gui_dirty = yes
	}
}

USA_mkultra_phase_2_archive_missing_baseline = {
	if = {
		limit = {
			USA_mkultra_enabled_trigger = yes
			check_variable = { USA_mkultra_slot_1_state = 2 }
			OR = {
				check_variable = { USA_mkultra_missing_baseline_stage = 2 }
				check_variable = { USA_mkultra_missing_baseline_stage = 6 }
				check_variable = { USA_mkultra_missing_baseline_stage = 8 }
			}
		}
		if = {
			limit = { check_variable = { USA_mkultra_missing_baseline_stage = 2 } }
			USA_mkultra_archive_missing_baseline = yes
		}
		else = {
			set_country_flag = USA_mkultra_missing_baseline_archived
			set_variable = { USA_mkultra_missing_baseline_stage = 3 }
			set_variable = { USA_mkultra_slot_1_state = 0 }
		}
		if = {
			limit = { has_active_mission = USA_mkultra_missing_baseline_review }
			remove_mission = USA_mkultra_missing_baseline_review
		}
		if = {
			limit = { has_active_mission = USA_mkultra_missing_baseline_renewal_review }
			remove_mission = USA_mkultra_missing_baseline_renewal_review
		}
		if = {
			limit = { has_active_mission = USA_mkultra_missing_baseline_final_review }
			remove_mission = USA_mkultra_missing_baseline_final_review
		}
		USA_mkultra_phase_2_unlock_countermeasure = yes
		USA_mkultra_gui_reconcile = yes
		USA_mkultra_mark_gui_dirty = yes
	}
}

USA_mkultra_schedule_security_scare = {
	if = {
		limit = {
			USA_mkultra_enabled_trigger = yes
			NOT = { has_country_flag = USA_mkultra_security_scare_scheduled }
			NOT = { has_country_flag = USA_mkultra_security_scare_resolved }
		}
		set_country_flag = USA_mkultra_security_scare_scheduled
		set_variable = { USA_mkultra_security_scare_state = 1 }
		# Original fictional pacing: a visible, save-persistent mission owns the
		# 60-day clock. Old schema-2 delayed events remain harmless because the
		# event and resolution effects are guarded by this durable state.
		if = {
			limit = { NOT = { has_active_mission = USA_mkultra_security_scare_countdown } }
			activate_mission = USA_mkultra_security_scare_countdown
		}
		USA_mkultra_mark_gui_dirty = yes
	}
}

USA_mkultra_resolve_security_scare_enhanced = {
	if = {
		limit = {
			tag = USA
			check_variable = { USA_mkultra_security_scare_state = 1 }
			has_country_flag = USA_mkultra_countermeasure_lessons_available
			NOT = { has_country_flag = USA_mkultra_countermeasure_lessons_used }
			has_political_power > 14
		}
		add_political_power = -15
		add_stability = -0.01
		clr_country_flag = USA_mkultra_countermeasure_lessons_available
		set_country_flag = USA_mkultra_countermeasure_lessons_used
		USA_mkultra_finish_security_scare = yes
	}
}

USA_mkultra_evaluate_basic_inquiry = {
	if = {
		limit = {
			tag = USA
			has_country_flag = USA_mkultra_initialized
			NOT = { has_country_flag = USA_mkultra_basic_inquiry_scheduled }
			NOT = { has_country_flag = USA_mkultra_basic_inquiry_resolved }
			OR = {
				has_country_flag = USA_mkultra_missing_baseline_paid
				has_country_flag = USA_mkultra_countermeasure_paid
			}
			OR = {
				check_variable = { USA_mkultra_exposure > 24 }
				has_country_flag = USA_mkultra_basic_inquiry_basis_security_scare
			}
		}
		set_country_flag = USA_mkultra_basic_inquiry_scheduled
		# The visible mission separates the inquiry from the scare popup and gives
		# any active research stage an explicit, persisted interruption context.
		if = {
			limit = { NOT = { has_active_mission = USA_mkultra_basic_inquiry_notice } }
			activate_mission = USA_mkultra_basic_inquiry_notice
		}
		USA_mkultra_mark_gui_dirty = yes
	}
}

USA_mkultra_phase_2_settle_charter_closure = {
	if = {
		limit = { tag = USA }
		if = {
			limit = { check_variable = { USA_mkultra_missing_baseline_stage = 5 } }
			USA_mkultra_cancel_missing_baseline_renewal = yes
		}
		if = {
			limit = { check_variable = { USA_mkultra_missing_baseline_stage = 7 } }
			USA_mkultra_cancel_missing_baseline_replication = yes
		}
		if = {
			limit = {
				OR = {
					check_variable = { USA_mkultra_missing_baseline_stage = 6 }
					check_variable = { USA_mkultra_missing_baseline_stage = 8 }
				}
			}
			USA_mkultra_phase_2_archive_missing_baseline = yes
		}
		if = {
			limit = { check_variable = { USA_mkultra_countermeasure_stage = 1 } }
			set_variable = { USA_mkultra_countermeasure_stage = 7 }
		}
		if = {
			limit = { check_variable = { USA_mkultra_countermeasure_stage = 2 } }
			USA_mkultra_cancel_countermeasure = yes
		}
		if = {
			limit = { check_variable = { USA_mkultra_countermeasure_stage = 3 } }
			USA_mkultra_archive_countermeasure = yes
		}
		set_variable = { USA_mkultra_slot_1_state = 0 }
		if = {
			limit = { has_active_mission = USA_mkultra_missing_baseline_renewal_research }
			remove_mission = USA_mkultra_missing_baseline_renewal_research
		}
		if = {
			limit = { has_active_mission = USA_mkultra_missing_baseline_renewal_review }
			remove_mission = USA_mkultra_missing_baseline_renewal_review
		}
		if = {
			limit = { has_active_mission = USA_mkultra_missing_baseline_replication_research }
			remove_mission = USA_mkultra_missing_baseline_replication_research
		}
		if = {
			limit = { has_active_mission = USA_mkultra_missing_baseline_final_review }
			remove_mission = USA_mkultra_missing_baseline_final_review
		}
		if = {
			limit = { has_active_mission = USA_mkultra_countermeasure_research }
			remove_mission = USA_mkultra_countermeasure_research
		}
		if = {
			limit = { has_active_mission = USA_mkultra_countermeasure_review }
			remove_mission = USA_mkultra_countermeasure_review
		}
		# Consequence clocks are intentionally not removed here. Once scheduled,
		# recorded scrutiny survives charter closure.
		USA_mkultra_gui_reconcile = yes
		USA_mkultra_mark_gui_dirty = yes
	}
}

USA_mkultra_phase_2_reconcile = {
	if = {
		limit = { USA_mkultra_enabled_trigger = yes }
		USA_mkultra_assign_missing_baseline_profile = yes

		if = {
			limit = { has_country_flag = USA_mkultra_charter_terminated }
			USA_mkultra_phase_2_settle_charter_closure = yes
		}

		if = {
			limit = { USA_mkultra_charter_active_trigger = yes }
			USA_mkultra_phase_2_unlock_countermeasure = yes

			if = {
				limit = {
					check_variable = { USA_mkultra_missing_baseline_stage = 5 }
					has_country_flag = USA_mkultra_missing_baseline_renewal_record_awarded
				}
				set_variable = { USA_mkultra_missing_baseline_stage = 6 }
				set_variable = { USA_mkultra_slot_1_state = 2 }
				if = {
					limit = { has_active_mission = USA_mkultra_missing_baseline_renewal_research }
					remove_mission = USA_mkultra_missing_baseline_renewal_research
				}
			}
			if = {
				limit = {
					check_variable = { USA_mkultra_missing_baseline_stage = 7 }
					has_country_flag = USA_mkultra_missing_baseline_replication_record_awarded
				}
				set_variable = { USA_mkultra_missing_baseline_stage = 8 }
				set_variable = { USA_mkultra_slot_1_state = 2 }
				if = {
					limit = { has_active_mission = USA_mkultra_missing_baseline_replication_research }
					remove_mission = USA_mkultra_missing_baseline_replication_research
				}
			}
			if = {
				limit = {
					check_variable = { USA_mkultra_countermeasure_stage = 2 }
					has_country_flag = USA_mkultra_countermeasure_record_awarded
				}
				set_variable = { USA_mkultra_countermeasure_stage = 3 }
				set_variable = { USA_mkultra_slot_1_state = 2 }
				if = {
					limit = { has_active_mission = USA_mkultra_countermeasure_research }
					remove_mission = USA_mkultra_countermeasure_research
				}
			}

			if = {
				limit = { check_variable = { USA_mkultra_missing_baseline_stage = 5 } }
				set_variable = { USA_mkultra_slot_1_state = 1 }
				if = {
					limit = { NOT = { has_active_mission = USA_mkultra_missing_baseline_renewal_research } }
					activate_mission = USA_mkultra_missing_baseline_renewal_research
				}
			}
			if = {
				limit = { check_variable = { USA_mkultra_missing_baseline_stage = 6 } }
				set_variable = { USA_mkultra_slot_1_state = 2 }
				if = {
					limit = { NOT = { has_active_mission = USA_mkultra_missing_baseline_renewal_review } }
					activate_mission = USA_mkultra_missing_baseline_renewal_review
				}
			}
			if = {
				limit = { check_variable = { USA_mkultra_missing_baseline_stage = 7 } }
				set_variable = { USA_mkultra_slot_1_state = 1 }
				if = {
					limit = { NOT = { has_active_mission = USA_mkultra_missing_baseline_replication_research } }
					activate_mission = USA_mkultra_missing_baseline_replication_research
				}
			}
			if = {
				limit = { check_variable = { USA_mkultra_missing_baseline_stage = 8 } }
				set_variable = { USA_mkultra_slot_1_state = 2 }
				if = {
					limit = { NOT = { has_active_mission = USA_mkultra_missing_baseline_final_review } }
					activate_mission = USA_mkultra_missing_baseline_final_review
				}
			}

			# Countermeasure work owns the single slot only after the front-grant
			# dossier has reached a durable closed disposition.
			if = {
				limit = {
					check_variable = { USA_mkultra_countermeasure_stage = 2 }
					OR = {
						check_variable = { USA_mkultra_missing_baseline_stage = 3 }
						check_variable = { USA_mkultra_missing_baseline_stage = 9 }
					}
				}
				set_variable = { USA_mkultra_slot_1_state = 1 }
				if = {
					limit = { NOT = { has_active_mission = USA_mkultra_countermeasure_research } }
					activate_mission = USA_mkultra_countermeasure_research
				}
			}
			if = {
				limit = {
					check_variable = { USA_mkultra_countermeasure_stage = 3 }
					OR = {
						check_variable = { USA_mkultra_missing_baseline_stage = 3 }
						check_variable = { USA_mkultra_missing_baseline_stage = 9 }
					}
				}
				set_variable = { USA_mkultra_slot_1_state = 2 }
				if = {
					limit = { NOT = { has_active_mission = USA_mkultra_countermeasure_review } }
					activate_mission = USA_mkultra_countermeasure_review
				}
			}
		}

		if = {
			limit = {
				has_country_flag = USA_mkultra_countermeasure_archived
				NOT = { has_country_flag = USA_mkultra_security_scare_scheduled }
				NOT = { has_country_flag = USA_mkultra_security_scare_resolved }
			}
			USA_mkultra_schedule_security_scare = yes
		}
		USA_mkultra_evaluate_basic_inquiry = yes

		# Schema-3 migration and safe state recovery. Recreate only a still-pending
		# visible clock; never replay an already resolved consequence.
		if = {
			limit = {
				has_country_flag = USA_mkultra_security_scare_scheduled
				NOT = { has_country_flag = USA_mkultra_security_scare_resolved }
			}
			set_variable = { USA_mkultra_security_scare_state = 1 }
			if = {
				limit = { NOT = { has_active_mission = USA_mkultra_security_scare_countdown } }
				activate_mission = USA_mkultra_security_scare_countdown
			}
		}
		else_if = {
			limit = { has_active_mission = USA_mkultra_security_scare_countdown }
			remove_mission = USA_mkultra_security_scare_countdown
		}

		if = {
			limit = {
				has_country_flag = USA_mkultra_basic_inquiry_scheduled
				NOT = { has_country_flag = USA_mkultra_basic_inquiry_resolved }
				check_variable = { USA_mkultra_inquiry_state = 0 }
			}
			if = {
				limit = { NOT = { has_active_mission = USA_mkultra_basic_inquiry_notice } }
				activate_mission = USA_mkultra_basic_inquiry_notice
			}
		}
		else_if = {
			limit = { has_active_mission = USA_mkultra_basic_inquiry_notice }
			remove_mission = USA_mkultra_basic_inquiry_notice
		}

		USA_mkultra_gui_reconcile = yes
		USA_mkultra_mark_gui_dirty = yes
	}
}
