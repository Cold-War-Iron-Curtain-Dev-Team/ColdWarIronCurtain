get_best_alliance_match_democratic_effect = { #arguably a trigger, but it does stuff so we keep it in here. If this is your first time seeing this, welcome to my personal hell.
	set_temp_variable = { best_leader = 0 }
	set_temp_variable = { best_leader_score = 0 }
	if = {
		limit = {
			all_other_country = {
				is_faction_leader = yes
				is_major = yes
				has_government = democratic
				not = { has_war_with = ROOT }
				set_temp_variable = { this.leader_score = num_of_factories }
				add_to_temp_variable = { this.leader_score = num_battalions }
				all_of_scopes = {
					array = allies
					add_to_temp_variable = { PREV.leader_score = num_of_factories }
					add_to_temp_variable = { PREV.leader_score = num_battalions }
				}
				if = {
					limit = {
						capital_scope = {
							is_on_continent = ROOT
						}
					}
					multiply_temp_variable = { leader_score = 2 }
				}
				if = {
					limit = {
						any_neighbor_country = { tag = ROOT }
					}
					multiply_temp_variable = { leader_score = 1.25 }
				}
				if = {
					limit = {
						check_variable = { ROOT.best_leader_score < leader_score }
						set_temp_variable = { ROOT.best_leader = this }
						set_temp_variable = { ROOT.best_leader_score = leader_score }
					}
				}
			}
		}
	}		
}

coalition_init = {
	set_country_flag = coalition_leader
	set_country_flag = coalition_founder
	set_variable = { coalition_objective = coalition_new_objective }
	set_variable = { coalition_target = coalition_new_target }
	if = {
		limit = { check_variable = { coalition_new_protected > 0 } }
		set_variable = { coalition_protected = coalition_new_protected }
	}
	set_variable = { coalition_founders_total = 1 }
	set_variable = { coalition_founders_present = 1 }
	if = {
		limit = { check_variable = { coalition_objective = 1 } }
		add_faction_goal = faction_goal_coalition_repel_invasion
	}
	else_if = {
		limit = { check_variable = { coalition_objective = 2 } }
		add_faction_goal = faction_goal_coalition_preserve_government
	}
	else_if = {
		limit = { check_variable = { coalition_objective = 3 } }
		add_faction_goal = faction_goal_coalition_secure_withdrawal
	}
	else_if = {
		limit = { check_variable = { coalition_objective = 4 } }
		add_faction_goal = faction_goal_coalition_defeat_target
	}
	if = {
		limit = {
			check_variable = { coalition_target > 0 }
			has_war_with = var:coalition_target
		}
		set_country_flag = coalition_war_seen
	}
}

coalition_add_member = {
	if = {
		limit = {
			has_country_flag = coalition_leader
			var:coalition_new_member = { is_in_faction = no }
		}
		var:coalition_new_member = {
			PREV = { add_to_faction = PREV }
			set_country_flag = coalition_founder
		}
		add_to_variable = { coalition_founders_total = 1 }
		add_to_variable = { coalition_founders_present = 1 }
	}
}

coalition_recount_founders = {
	set_variable = { coalition_founders_present = 0 }
	every_faction_member = {
		limit = {
			has_country_flag = coalition_founder
			has_capitulated = no
		}
		PREV = { add_to_variable = { coalition_founders_present = 1 } }
	}
	if = {
		limit = {
			NOT = { has_country_flag = coalition_war_seen }
