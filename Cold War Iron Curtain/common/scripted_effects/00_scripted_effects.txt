coalition_init = {
	set_country_flag = coalition_leader
	set_country_flag = coalition_founder
	set_variable = { coalition_objective = coalition_new_objective }
	set_variable = { coalition_target = coalition_new_target }
	if = {
		limit = { check_variable = { coalition_new_protected > 0 } }
		set_variable = { coalition_protected = coalition_new_protected }
	}
	set_variable = { coalition_founders_total = 1 }
	set_variable = { coalition_founders_present = 1 }
	if = {
		limit = { check_variable = { coalition_objective = 1 } }
		add_faction_goal = faction_goal_coalition_repel_invasion
	}
	else_if = {
		limit = { check_variable = { coalition_objective = 2 } }
		add_faction_goal = faction_goal_coalition_preserve_government
	}
	else_if = {
		limit = { check_variable = { coalition_objective = 3 } }
		add_faction_goal = faction_goal_coalition_secure_withdrawal
	}
	else_if = {
		limit = { check_variable = { coalition_objective = 4 } }
		add_faction_goal = faction_goal_coalition_defeat_target
	}
	if = {
		limit = {
			check_variable = { coalition_target > 0 }
			has_war_with = var:coalition_target
		}
		set_country_flag = coalition_war_seen
	}
}

coalition_add_member = {
	if = {
		limit = {
			has_country_flag = coalition_leader
			var:coalition_new_member = { is_in_faction = no }
		}
		var:coalition_new_member = {
			PREV = { add_to_faction = PREV }
			set_country_flag = coalition_founder
		}
		add_to_variable = { coalition_founders_total = 1 }
		add_to_variable = { coalition_founders_present = 1 }
	}
}

coalition_recount_founders = {
	set_variable = { coalition_founders_present = 0 }
	every_faction_member = {
		limit = {
			has_country_flag = coalition_founder
			has_capitulated = no
		}
		PREV = { add_to_variable = { coalition_founders_present = 1 } }
	}
	if = {
		limit = {
			NOT = { has_country_flag = coalition_war_seen }
			check_variable = { coalition_target > 0 }
			var:coalition_target = { exists = yes }
			has_war_with = var:coalition_target
		}
		set_country_flag = coalition_war_seen
	}
}

coalition_dissolve = {
	if = {
		limit = { has_country_flag = coalition_leader }
		every_faction_member = {
			clr_country_flag = coalition_founder
			clr_country_flag = coalition_liaison_funded
		}
		clr_country_flag = coalition_leader
		clr_country_flag = coalition_war_seen
		clr_country_flag = coalition_campaign_done
		clr_country_flag = coalition_settlement_enabled
		clear_variable = coalition_objective
		clear_variable = coalition_target
		clear_variable = coalition_protected
		clear_variable = coalition_founders_total
		clear_variable = coalition_founders_present
		dismantle_faction = yes
	}
}

GCC_lifecycle_monthly = {
	if = {
		limit = {
			has_global_flag = cwic_bookmark_1980_start
			NOT = { has_global_flag = GCC_1980_campaign }
		}
		set_global_flag = GCC_1980_campaign
	}
	if = {
		limit = {
			has_global_flag = GCC_1980_campaign
			NOT = { has_global_flag = GCC_founded }
			date > 1981.5.24
			country_exists = SAU
			country_exists = KUW
			country_exists = BHR
			country_exists = QAT
			country_exists = UAE
			country_exists = OMA
			SAU = {
				is_subject = no
				is_in_faction = no
			}
			KUW = {
				is_subject = no
				is_in_faction = no
			}
			BHR = {
				is_subject = no
				is_in_faction = no
			}
			QAT = {
				is_subject = no
				is_in_faction = no
			}
			UAE = {
				is_subject = no
				is_in_faction = no
			}
			OMA = {
				is_subject = no
				is_in_faction = no
			}
		}
		SAU = {
			country_event = { id = GCC_Peninsula_Shield.1 }
		}
	}
	if = {
		limit = {
			has_global_flag = GCC_founded
		}
		if = {
			limit = { country_exists = SAU }
			SAU = { GCC_recount_members = yes }
		}
		every_country = {
			limit = {
				has_country_flag = GCC_force_contributor
				NOT = { tag = SAU }
				NOT = { is_in_faction_with = SAU }
			}
			clr_country_flag = GCC_force_contributor
			add_to_variable = { global.GCC_force_contributor_count = -1 }
			set_global_flag = GCC_force_break
		}
		if = {
			limit = {
				has_global_flag = GCC_force_break
			}
			set_variable = { global.GCC_force_months = 0 }
			clr_global_flag = GCC_force_break
		}
		else_if = {
			limit = {
				has_global_flag = GCC_force_established
			}
			add_to_variable = { global.GCC_force_months = 1 }
		}
	}
}

	}
}

add_potential_special_forces_tree = {
	custom_effect_tooltip = add_potential_special_forces_tree_tt
	if = {
		limit = { NOT = { has_variable = sf_trees } }
		set_variable = { sf_trees = 2 }
	}
	else = {
		add_to_variable = { sf_trees = 1 }
	}
}

remove_potential_special_forces_tree = {
	if = {
		limit = { NOT = { has_variable = sf_trees } }
		set_variable = { sf_trees = 0 }
	}
	else = {
		subtract_from_variable = { sf_trees = 1 }
	}
}
