# ============================================================================
# Viet Minh campaign wrapper (Tay Bac, Hoa Binh, Dien Bien Phu, Cao-Bac)
# ----------------------------------------------------------------------------
# A campaign spends VIN_Campaign_Supply to set a commitment tier, and the tier
# decides how many regular formations march with it. Mirrors the CEFEO side in
# FRE_Operation_Effects.txt.
#
# Campaign flag contract (VIN_indochina_campaign_triggers.txt): the *_Active
# flag is set before the campaign war; the common resolver clears it and books
# the cooldown under a limit = { tag = VIN } guard.
#
# Campaign ids: 1 Tay Bac, 2 Hoa Binh, 3 Dien Bien Phu, 4 Cao-Bac.
# Commitment tiers: 0 Light, 1 Standard, 2 Maximum.
# ============================================================================

vin_campaign_initialize = {
	if = {
		limit = { NOT = { has_variable = VIN_Campaign_Supply } }
		set_variable = { VIN_Campaign_Supply = 100 }
		set_variable = { VIN_Supply_Accrual_Counter = 0 }
	}
}

vin_change_campaign_supply = {
	add_to_variable = { VIN_Campaign_Supply = vin_supply_amount }
	clamp_variable = {
		var = VIN_Campaign_Supply
		min = 0
		max = 1000
	}
	custom_effect_tooltip = VIN_Campaign_Supply_Change_tt
}

vin_accrue_monthly_supply = {
	set_temp_variable = { vin_supply_amount = 15 }
	if = {
		limit = { has_global_flag = PRC_Victory }
		set_temp_variable = { vin_supply_amount = 30 }	# the Chinese border is open
	}
	if = {
		limit = { has_country_flag = Enable_Larger_Raids }
		add_to_temp_variable = { vin_supply_amount = 10 }
	}
	hidden_effect = { vin_change_campaign_supply = yes }
}

# --- Commitment -------------------------------------------------------------

vin_campaign_begin = {
	vin_campaign_initialize = yes
	set_variable = { VIN_camp_current = vin_camp_id }
	set_variable = { VIN_camp_price_light = 40 }
	set_variable = { VIN_camp_price_standard = 80 }
	set_variable = { VIN_camp_price_maximum = 140 }
	if = {
		limit = { check_variable = { VIN_camp_current = 3 } }	# Dien Bien Phu is the whole army's campaign
		set_variable = { VIN_camp_price_light = 60 }
		set_variable = { VIN_camp_price_standard = 120 }
		set_variable = { VIN_camp_price_maximum = 200 }
	}
	vin_ai_prepare_campaign = yes
	country_event = { id = VIN_Camp.1 hours = 6 }
}

vin_ai_prepare_campaign = {
	if = {
		limit = {
			is_ai = yes
			check_variable = { VIN_Campaign_Supply < VIN_camp_price_maximum }
		}
		set_variable = { VIN_Campaign_Supply = VIN_camp_price_maximum }
		add_to_variable = { VIN_Campaign_Supply = 10 }	# the option needs a strict >
	}
	if = {
		limit = { is_ai = yes }
		add_equipment_to_stockpile = {
			type = infantry_equipment_1
			amount = 3000
			producer = PRC
		}
		add_equipment_to_stockpile = {
			type = artillery_equipment_1
			amount = 400
			producer = PRC
		}
		add_manpower = 25000
		add_command_power = 50
	}
}

vin_indochina_ai_initialize = {
	if = {
		limit = {
			vin_indochina_ai_support_trigger = yes
			NOT = { has_idea = VIN_Indochina_AI_Support }
			NOT = { has_global_flag = Indochina_War_Over }
		}
		add_ideas = VIN_Indochina_AI_Support
	}
	if = {
		limit = {
			NOT = { vin_indochina_ai_support_trigger = yes }
			has_idea = VIN_Indochina_AI_Support
		}
		remove_ideas = VIN_Indochina_AI_Support
	}
}

vin_campaign_commit = {
	set_variable = { VIN_camp_commitment_tier = vin_camp_tier }
	set_temp_variable = { vin_supply_amount = VIN_camp_price_light }
	if = {
		limit = { check_variable = { VIN_camp_commitment_tier = 1 } }
		set_temp_variable = { vin_supply_amount = VIN_camp_price_standard }
	}
	else_if = {
		limit = { check_variable = { VIN_camp_commitment_tier = 2 } }
		set_temp_variable = { vin_supply_amount = VIN_camp_price_maximum }
	}
	multiply_temp_variable = { vin_supply_amount = -1 }
	vin_change_campaign_supply = yes
	vin_campaign_spawn_columns = yes
}

vin_campaign_spawn_columns = {
	if = {
		limit = { check_variable = { VIN_camp_commitment_tier > 0 } }
		custom_effect_tooltip = VIN_Camp_Spawn_Regiment_tt
		random_owned_controlled_state = {
			limit = { OR = { state = 881  state = 1762  state = 1761 } }
			create_unit = {
				division = "name = \"Trung doan Chu luc\" division_template = \"Trung doan Bo binh Infantry\" start_experience_factor = 0.4 start_equipment_factor = 0.9"
				owner = VIN
				allow_spawning_on_enemy_provs = yes
			}
		}
	}
	if = {
		limit = { check_variable = { VIN_camp_commitment_tier > 1 } }
		custom_effect_tooltip = VIN_Camp_Spawn_Artillery_tt
		random_owned_controlled_state = {
			limit = { OR = { state = 881  state = 1762  state = 1761 } }
			create_unit = {
				division = "name = \"Dai doan Cong Phao\" division_template = \"Tieu doan Phao binh Artillery\" start_experience_factor = 0.5 start_equipment_factor = 1.0"
				owner = VIN
				allow_spawning_on_enemy_provs = yes
			}
		}
		add_equipment_to_stockpile = { type = infantry_equipment_1 amount = 1500 producer = PRC }
		add_equipment_to_stockpile = { type = artillery_equipment_1 amount = 200 producer = PRC }
	}
}

vin_campaign_open_corridors = {
	if = {
		# Northern campaign routes are now unrestricted. This remaining flag is
		# consumed only by the retained Laos-Vinh geopolitical boundary rule.
		limit = { check_variable = { VIN_camp_current = 3 } }
		set_country_flag = INDOCHINESE_WAR_DIEN_BIEN_UNLOCKED
	}
}

# Rebuild the initialization baseline before selecting the ground prepared for
# the current operation. Older campaign code cleared the persistent state flag
# as well as the modifier across the entire northern theatre, so every later
# campaign inherited a progressively more open front. The flag is theatre
# ownership data and must survive a temporary planned offensive.
vin_campaign_restore_unplanned_offensive = {
	if = {
		limit = { NOT = { has_global_flag = Indochina_War_Over } }
		every_state = {
			limit = {
				OR = {
					is_claimed_by = VIN
					is_claimed_by = VIE
					is_core_of = CAM
					is_core_of = LOS
					is_core_of = VIN
					is_core_of = VIE
					is_core_of = FRE
				}
			}
			set_state_flag = VIN_unplanned_offensive_flag
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = unplanned_offensive } } }
				add_dynamic_modifier = { modifier = unplanned_offensive }
			}
		}
	}
}

vin_campaign_restore_unplanned_state = { # state scope
	set_state_flag = VIN_unplanned_offensive_flag
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = unplanned_offensive } } }
		add_dynamic_modifier = { modifier = unplanned_offensive }
	}
}

		lao_join_the_battle_of_dien_bien_phu = yes

	FRA = { activate_mission = VIN_Campaign_Objective_Timer }	# FRA always exists; the mission is the hard backstop if the tick ever stalls

}

vin_campaign_objective_tick = {
	if = {
		limit = {
			NOT = { VIN_no_campaign_running_trigger = yes }	# a campaign is in the field
			NOT = { has_country_flag = VIN_Campaign_Resolving }
		}
		add_to_variable = { VIN_camp_days_elapsed = 1 }
		if = {
			limit = {
				NOT = { vin_indochina_ai_support_trigger = yes }
				has_idea = VIN_Campaign_AI_Support
			}
			remove_ideas = VIN_Campaign_AI_Support	# either principal combatant is now human
		}
		vin_campaign_sync_objective_modifiers = yes	# the objective can change hands in either direction on any day
		vin_campaign_update_overextension = yes
		if = {
			limit = { country_exists = FRE }
			FRE = {
				fre_lorraine_response_daily_tick = yes
				fre_hoa_binh_response_daily_tick = yes
			}
		}
		vin_campaign_check_resolution = yes
	}
}

# One classifier for every live exit. Ordering is contractual: an objective
# taken on the same day as a settlement or deadline still counts as a victory.
vin_campaign_check_resolution = {
	if = {
		limit = {
			vin_campaign_objective_met_trigger = yes
			NOT = { check_variable = { VIN_camp_days_elapsed > VIN_camp_deadline_y } }
		}
		if = {
			limit = { NOT = { check_variable = { VIN_camp_days_elapsed > VIN_camp_deadline_x } } }
			set_temp_variable = { vin_camp_outcome = 1 }	# clean success, inclusive boundary
		}
		else = {
			set_temp_variable = { vin_camp_outcome = 2 }	# costly success
		}
		vin_campaign_resolve = yes
	}
	else_if = {
		limit = { vin_campaign_superseded_trigger = yes }
		set_temp_variable = { vin_camp_outcome = 5 }
		vin_campaign_resolve = yes
	}
	else_if = {
		limit = {
			check_variable = { VIN_camp_current = 2 }
			has_country_flag = VIN_Hoa_Binh_Amarante_Withdrawal_Ready
		}
		clr_country_flag = VIN_Hoa_Binh_Amarante_Withdrawal_Ready
		set_temp_variable = { vin_camp_outcome = 1 }	# the parent owns Amarante's voluntary withdrawal result
		vin_campaign_resolve = yes
	}
	else_if = {
		limit = { check_variable = { VIN_camp_days_elapsed > VIN_camp_deadline_y } }
		set_temp_variable = { vin_camp_outcome = 4 }
		vin_campaign_resolve = yes
	}
	else_if = {
		limit = { vin_campaign_war_abandoned_trigger = yes }
		set_temp_variable = { vin_camp_outcome = 3 }
		vin_campaign_resolve = yes
	}
}

# --- The siege --------------------------------------------------------------

vin_dbp_siege_tick = {
	if = {
		limit = {
			has_country_flag = VIN_Campaign_Dien_Bien_Active
			NOT = { has_country_flag = VIN_Campaign_Resolving }
			NOT = { controls_province = 4529 }
			vin_dbp_valley_invested_trigger = yes
		}
		add_to_variable = { global.VIN_dbp_siege_days = 1 }

		if = {
			limit = {
				check_variable = { global.VIN_dbp_siege_days > 13 }
				check_variable = { global.VIN_dbp_fort_level > 3 }
			}
			set_variable = { global.VIN_dbp_fort_level = 3 }
			671 = { set_building_level = { type = bunker level = 3 province = { id = 4529 } } }
		}
		if = {
			limit = {
				check_variable = { global.VIN_dbp_siege_days > 27 }
				check_variable = { global.VIN_dbp_fort_level > 2 }
			}
			set_variable = { global.VIN_dbp_fort_level = 2 }
			671 = { set_building_level = { type = bunker level = 2 province = { id = 4529 } } }
		}
		if = {
			limit = {
				check_variable = { global.VIN_dbp_siege_days > 41 }
				check_variable = { global.VIN_dbp_fort_level > 1 }
			}
			set_variable = { global.VIN_dbp_fort_level = 1 }
			671 = { set_building_level = { type = bunker level = 1 province = { id = 4529 } } }
		}
		if = {
			limit = {
				check_variable = { global.VIN_dbp_siege_days > 55 }
				check_variable = { global.VIN_dbp_fort_level > 0 }
			}
			set_variable = { global.VIN_dbp_fort_level = 0 }
			671 = {	# the last works go, and the airstrip with them
				set_building_level = { type = bunker level = 0 province = { id = 4529 } }
				set_building_level = { type = anti_air_building level = 0 }
			}
		}
		if = {
			limit = {
				is_ai = yes
				check_variable = { global.VIN_dbp_siege_days > 55 }
				check_variable = { global.VIN_dbp_fort_level < 1 }
			}
			fre_dbp_disband_gono = yes	# the group is destroyed or taken prisoner where it stands
			set_province_controller = 4529
			log = "[GetDateText]: Dien Bien Phu overrun by an AI siege after [?global.VIN_dbp_siege_days|0] days"
		}
	}
}

# --- Guerilla networks ------------------------------------------------------

vin_sync_guerilla_networks = {
	every_state = {
		limit = {
			is_core_of = VIN
			NOT = { is_core_of = MEO }
			NOT = { is_owned_by = VIE }
			NOT = { is_owned_by = CCC }
			NOT = { is_owned_by = FUL }
			NOT = { is_owned_by = NLF }
			NOT = { is_controlled_by = VIN }
			NOT = { vin_aligned_state_trigger = yes }
			NOT = { has_dynamic_modifier = { modifier = VIN_guerilla_network } }
		}
		add_dynamic_modifier = { modifier = VIN_guerilla_network }
	}
	every_state = {
		limit = {
			has_dynamic_modifier = { modifier = VIN_guerilla_network }
			OR = {
				is_controlled_by = VIN
				is_owned_by = VIE
				is_owned_by = CCC
				is_owned_by = FUL
				is_owned_by = NLF
				vin_aligned_state_trigger = yes
			}
		}
		remove_dynamic_modifier = { modifier = VIN_guerilla_network }
	}
}
