
vin_campaign_initialize = {
	if = {
		limit = { NOT = { has_variable = VIN_Campaign_Supply } }
		set_variable = { VIN_Campaign_Supply = 100 }
		set_variable = { VIN_Supply_Accrual_Counter = 0 }
	}
}

vin_change_campaign_supply = {
	add_to_variable = { VIN_Campaign_Supply = vin_supply_amount }
	clamp_variable = {
		var = VIN_Campaign_Supply
		min = 0
		max = 1000
	}
	custom_effect_tooltip = VIN_Campaign_Supply_Change_tt
}

vin_accrue_monthly_supply = {
	set_temp_variable = { vin_supply_amount = 15 }
	if = {
		limit = { has_global_flag = PRC_Victory }
		set_temp_variable = { vin_supply_amount = 30 }	# the Chinese border is open
	}
	if = {
		limit = { has_country_flag = Enable_Larger_Raids }
		add_to_temp_variable = { vin_supply_amount = 10 }
	}
	hidden_effect = { vin_change_campaign_supply = yes }
}

vin_campaign_begin = {
	if = {
		limit = { country_exists = FRE FRE = { has_country_flag = FRE_Limited_Operation_Active } }
		log = "FRE_OP_TRACE: RESOLUTION caller=vin_campaign_begin reason=VIN_campaign_precedence vin_campaign_requested=[?vin_camp_id|0]"
		FRE = { set_temp_variable = { fre_op_outcome = 5 } fre_operation_resolve_outcome = yes }
	}
	vin_campaign_initialize = yes
	set_variable = { VIN_camp_current = vin_camp_id }
}

vin_campaign_restore_unplanned_offensive = {
	if = {
		limit = { NOT = { has_global_flag = Indochina_War_Over } }
		every_state = {
			limit = {
				OR = {
					is_claimed_by = VIN
					is_claimed_by = VIE
					is_core_of = CAM
					is_core_of = LOS
					is_core_of = VIN
					is_core_of = VIE
					is_core_of = FRE
				}
			}
			set_state_flag = VIN_unplanned_offensive_flag
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = unplanned_offensive } } }
				add_dynamic_modifier = { modifier = unplanned_offensive }
			}
		}
	}
}

vin_campaign_declare_war = {
	if = {
		limit = { has_country_flag = VIN_Limited_Campaign_Armistice_Pending }
		vin_limited_campaign_close_spillover_wars = yes
	}
	vin_campaign_open_corridors = yes
	vin_campaign_restore_unplanned_offensive = yes
	vin_campaign_sync_objective_modifiers = yes
	vin_campaign_apply_war_spirits = yes
	vin_campaign_update_overextension = yes
	set_variable = { VIN_camp_days_elapsed = 0 }
	set_variable = { VIN_camp_deadline_x = 120 }	# a clean campaign
	set_variable = { VIN_camp_deadline_y = 210 }	# past this the offensive has failed
	set_variable = { VIN_camp_objective_hold_days = 0 }
	if = {
		limit = { check_variable = { VIN_camp_current = 3 } }
		set_variable = { VIN_camp_deadline_x = 150 }	# the valley is a siege, not a march
		set_variable = { VIN_camp_deadline_y = 260 }
	}
	if = {
		limit = { check_variable = { VIN_camp_current = 5 } }
		set_variable = { VIN_camp_deadline_x = 20 }
		set_variable = { VIN_camp_deadline_y = 35 }
	}
	else_if = {
		limit = { check_variable = { VIN_camp_current = 6 } }
		set_variable = { VIN_camp_deadline_x = 20 }
		set_variable = { VIN_camp_deadline_y = 35 }
	}
	else_if = {
		limit = { check_variable = { VIN_camp_current = 7 } }
		set_variable = { VIN_camp_deadline_x = 30 }
		set_variable = { VIN_camp_deadline_y = 50 }
	}
	else_if = {
		limit = { check_variable = { VIN_camp_current = 8 } }
		set_variable = { VIN_camp_deadline_x = 30 }
		set_variable = { VIN_camp_deadline_y = 45 }
	}
	set_variable = { VIN_camp_timer_days = VIN_camp_deadline_y }
	add_to_variable = { VIN_camp_timer_days = 1 }
	vin_limited_campaign_add_objective_vps = yes
	if = {
		limit = {
			check_variable = { VIN_camp_current = 2 }
			country_exists = TAM
			NOT = { has_war_with = TAM }
		}
		hidden_effect = { declare_war_on = { target = TAM type = annex_everything } }
	}
	else_if = {
		limit = {
			check_variable = { VIN_camp_current = 4 }
			country_exists = THO
			NOT = { has_war_with = THO }
		}
		hidden_effect = { declare_war_on = { target = THO type = annex_everything } }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { VIN_camp_current = 5 }
				check_variable = { VIN_camp_current = 8 }
			}
			country_exists = TAI
		}
		set_country_flag = VIN_Limited_Campaign_Owns_TAI_War
		if = {
			limit = { NOT = { has_war_with = TAI } }
			hidden_effect = { declare_war_on = { target = TAI type = annex_everything } }
		}
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { VIN_camp_current = 6 }
				check_variable = { VIN_camp_current = 7 }
			}
			country_exists = FRE
		}
		set_country_flag = VIN_Limited_Campaign_Owns_FRE_War
		if = {
			limit = { NOT = { has_war_with = FRE } }
			hidden_effect = { declare_war_on = { target = FRE type = annex_everything } }
		}
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { VIN_camp_current = 1 }
				check_variable = { VIN_camp_current = 3 }
			}
			country_exists = TAI
			NOT = { has_war_with = TAI }
		}
		hidden_effect = { declare_war_on = { target = TAI type = annex_everything } }
	}
	indochina_call_all_vin_fre_theater_forces = yes
	if = {
		limit = { check_variable = { VIN_camp_current = 3 } }
		lao_join_the_battle_of_dien_bien_phu = yes
		if = {
			limit = { country_exists = FRE }
			FRE = {
				fre_castor_handoff_to_campaign = yes
				fre_pollux_handoff_to_campaign = yes
				fre_dbp_response_begin = yes
			}
		}
	}
	if = { limit = { check_variable = { VIN_camp_current = 5 } } activate_mission = VIN_Vinh_Yen_Objective }
	else_if = { limit = { check_variable = { VIN_camp_current = 6 } } activate_mission = VIN_Mao_Khe_Objective }
	else_if = { limit = { check_variable = { VIN_camp_current = 7 } } activate_mission = VIN_Day_River_Objective }
	else_if = { limit = { check_variable = { VIN_camp_current = 8 } } activate_mission = VIN_Na_San_Objective }
	else = { activate_mission = VIN_Campaign_Objective_Timer }
	if = {

			check_variable = { VIN_camp_current = 2 }

			country_exists = FRE
		}

	}

			NOT = { check_variable = { VIN_camp_days_elapsed > VIN_camp_deadline_y } }

			}
			set_variable = { global.VIN_dbp_fort_level = 0 }

				set_building_level = { type = bunker level = 0 province = { id = 4529 } }
				set_building_level = { type = anti_air_building level = 0 }
