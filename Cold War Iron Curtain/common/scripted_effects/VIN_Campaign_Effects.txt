# The southern Viet Minh has one state and no industry, so without help from
# the north it never replaces a division (14 to 6 by 1953 in the 2026-10-02d
# run, with its manpower unused). From 1950, while it is at war and the north
# can spare it, the north pays Campaign Supply for a monthly rifle shipment
# and, below 10 divisions, one new regiment. It stops during the post-DBP
# stand-down and once the post-DBP supply line takes over.
vin_nlf_northern_resupply_monthly = {
	if = {
		limit = {
			date > 1950.1.1
			NOT = { has_global_flag = Indochina_War_Over }
			NOT = { has_global_flag = geneva_conference_concluded }
			NOT = { has_country_flag = VIN_Post_DBP_NLF_Stand_Down_Ordered }
			NOT = { has_country_flag = VIN_Post_DBP_Funded_NLF }
			country_exists = NLF
			NLF = { has_war = yes }
			check_variable = { VIN_Campaign_Supply > 49 }
		}
		set_temp_variable = { vin_nlf_regiment = 0 }
		NLF = { add_equipment_to_stockpile = { type = infantry_equipment_1 amount = 300 producer = VIN } }
		set_temp_variable = { vin_supply_amount = -5 }
		hidden_effect = { vin_change_campaign_supply = yes }
		if = {
			limit = {
				NLF = {
					num_divisions < 10
					capital_scope = { is_controlled_by = NLF }
				}
			}
			NLF = {
				if = {
					limit = { NOT = { has_template = "Trung Doan Bo Binh Infantry" } }
					division_template = {
						name = "Trung Doan Bo Binh Infantry"
						division_names_group = VIN_INF_01
						regiments = {
							infantry = { x = 0 y = 0 }
							infantry = { x = 0 y = 1 }
							infantry = { x = 0 y = 2 }
							artillery = { x = 1 y = 0 }
						}
						support = {
							field_hospital = { x = 0 y = 0 }
							artillery_support = { x = 0 y = 1 }
							signal_company = { x = 0 y = 2 }
						}
					}
				}
				capital_scope = {
					create_unit = {
						division = "name = \"Trung Doan Bo Binh Infantry\" division_template = \"Trung Doan Bo Binh Infantry\" start_experience_factor = 0.3 start_equipment_factor = 0.8"
						owner = NLF
					}
				}
			}
			set_temp_variable = { vin_supply_amount = -10 }
			hidden_effect = { vin_change_campaign_supply = yes }
			set_temp_variable = { vin_nlf_regiment = 1 }
		}
		log = "IC_AFK|NLF_RESUPPLY|date=[GetDateText]|rifles=300|regiment=[?vin_nlf_regiment|0]|nlf_divisions=[?NLF.num_divisions|0]|vin_supply=[?VIN_Campaign_Supply|0]"
	}
	else_if = {
		limit = {
			date > 1950.1.1
			NOT = { has_global_flag = Indochina_War_Over }
		}
		ic_afk_validation_nlf_resupply_skip = yes
	}
}

vin_campaign_apply_command_inputs = {
	if = {
		limit = { has_country_flag = VIN_Command_Inputs_Applied }
		log = "IC_AFK|CAMPAIGN_INPUTS_SKIPPED|date=[GetDateText]|campaign=[?VIN_camp_current|0]|reason=already_applied"
	}
	else_if = {
		limit = { NOT = { VIN_indochina_campaign_window_trigger = yes } }
		log = "IC_AFK|CAMPAIGN_INPUTS_SKIPPED|date=[GetDateText]|campaign=[?VIN_camp_current|0]|reason=window_closed"
	}
	else = {
		set_country_flag = VIN_Command_Inputs_Applied
		set_temp_variable = { vin_cmd_supply = 0 }
		set_temp_variable = { vin_cmd_clean = 0 }
		set_temp_variable = { vin_cmd_sources = 0 }
		if = {
			limit = { has_country_flag = VIN_Cmd_Input_Mimic_French }
			add_to_temp_variable = { vin_cmd_supply = 5 }
			add_to_temp_variable = { vin_cmd_sources = 1 }
		}
		if = {
			limit = { has_country_flag = VIN_Cmd_Input_Domestic_Programs }
			add_to_temp_variable = { vin_cmd_clean = 2 }
			add_to_temp_variable = { vin_cmd_sources = 2 }
		}
		if = {
			limit = { has_country_flag = VIN_Cmd_Input_Formal_Squads }
			add_to_temp_variable = { vin_cmd_clean = 2 }
			add_to_temp_variable = { vin_cmd_sources = 4 }
		}
		if = {
			limit = { has_country_flag = VIN_Cmd_Input_Decree_17 }
			add_to_temp_variable = { vin_cmd_supply = 4 }
			add_to_temp_variable = { vin_cmd_sources = 8 }
		}
		if = {
			limit = { has_country_flag = VIN_Cmd_Input_Defector_Cadres }
			add_to_temp_variable = { vin_cmd_supply = 4 }
			add_to_temp_variable = { vin_cmd_sources = 16 }
		}
		if = {
			limit = { has_country_flag = VIN_Cmd_Input_Vo_Command }
			add_to_temp_variable = { vin_cmd_supply = 4 }
			add_to_temp_variable = { vin_cmd_clean = 3 }
			add_to_temp_variable = { vin_cmd_sources = 32 }
		}
		if = {
			limit = { has_country_flag = VIN_Cmd_Input_Rear_Services }
			add_to_temp_variable = { vin_cmd_supply = 4 }
			add_to_temp_variable = { vin_cmd_sources = 64 }
		}
		if = {
			limit = { has_country_flag = VIN_Patronage_Chinese }
			add_to_temp_variable = { vin_cmd_supply = 15 }
			add_to_temp_variable = { vin_cmd_sources = 128 }
		}
		if = {
			limit = { has_country_flag = VIN_Patronage_Soviet }
			add_to_temp_variable = { vin_cmd_supply = 5 }
			add_to_temp_variable = { vin_cmd_sources = 256 }
		}
		clamp_temp_variable = { var = vin_cmd_supply min = 0 max = 30 }
		clamp_temp_variable = { var = vin_cmd_clean min = 0 max = 7 }
		if = {
			limit = { check_variable = { VIN_camp_deadline_x < 1 } }
			set_temp_variable = { vin_cmd_clean = 0 }
		}
		if = {
			limit = { check_variable = { vin_cmd_clean > 0 } }
			add_to_variable = { VIN_camp_deadline_x = vin_cmd_clean }
			add_to_variable = { VIN_camp_deadline_y = vin_cmd_clean }
		}
		if = {
			limit = { check_variable = { vin_cmd_supply > 0 } }
			set_temp_variable = { vin_supply_amount = vin_cmd_supply }
			hidden_effect = { vin_change_campaign_supply = yes }
		}
		log = "IC_AFK|CAMPAIGN_INPUTS|date=[GetDateText]|campaign=[?VIN_camp_current|0]|supply=+[?vin_cmd_supply|0]|clean_days=+[?vin_cmd_clean|0]|sources=[?vin_cmd_sources|0]"
	}
}

vin_campaign_update_overextension = {
	if = {
		limit = {
			OR = {
				vin_campaign_severe_overextension_trigger = yes
				vin_all_in_corridor_overextended_trigger = yes
			}
		}
		vin_overextension_severe_apply = yes
	}
	else = {
		vin_overextension_severe_clear = yes
	}
	# Ordinary tier: only off-envelope ground held 5 days in a row counts, so a
	# border grab in the first days of a declaration does not penalise the
	# whole army. The severe tier above stays immediate.
	if = {
