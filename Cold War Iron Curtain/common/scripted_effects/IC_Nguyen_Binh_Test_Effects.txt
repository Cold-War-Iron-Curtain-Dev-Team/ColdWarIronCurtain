test_binh_status = {
	if = { limit = { NOT = { country_exists = VIN } } log = "BINH: FAIL - Hanoi does not exist" }
	if = { limit = { NOT = { country_exists = NLF } } log = "BINH: southern resistance is gone; the recall chain is inert" }
	if = { limit = { country_exists = VIN } log = "BINH|route=[?VIN.VIN_Binh_Route|0]|date=[GetDateText]" }
	if = { limit = { country_exists = VIN VIN = { has_country_flag = VIN_Binh_Censured } } log = "BINH: Hanoi censured him" }
	if = { limit = { country_exists = VIN VIN = { has_country_flag = VIN_Binh_Defended } } log = "BINH: Hanoi defended him" }
	if = { limit = { country_exists = VIN VIN = { has_country_flag = VIN_Binh_Escort_Committed } } log = "BINH: escort committed" }
	if = { limit = { country_exists = VIN VIN = { has_country_flag = Binh_Outcome } } log = "BINH: outcome recorded" }
	if = { limit = { any_country = { has_country_flag = FRE_Binh_Corridor_Interdiction } } log = "BINH: the Cambodian corridor is interdicted" }

	if = { limit = { country_exists = NLF NLF = { has_country_flag = NLF_Binh_Killed } } log = "BINH: killed in the ambush" }
	if = { limit = { country_exists = NLF NLF = { NOT = { NLF_southern_command_settled_trigger = yes } } } log = "BINH: FAIL - no southern command flag is set" }
	if = { limit = { country_exists = NLF NLF = { has_country_flag = Nguyen_Binh_Leader } } log = "BINH: command = Nguyen Binh" }
	if = { limit = { country_exists = NLF NLF = { has_country_flag = Le_Duan_Leader } } log = "BINH: command = Le Duan" }
	if = { limit = { country_exists = NLF NLF = { has_country_flag = Le_Duc_Tho_Leader } } log = "BINH: command = Le Duc Tho" }
	if = { limit = { country_exists = NLF NLF = { has_country_flag = Tran_quoc_Buu_Leader } } log = "BINH: command = Tran Quoc Buu" }
	if = { limit = { country_exists = NLF NLF = { has_country_flag = Kieu_Cong_Cung_Leader } } log = "BINH: command = Kieu Cong Cung" }

	if = { limit = { country_exists = NLF NLF = { has_idea = NLF_vietminh_disarray } } log = "BINH: NLF holds vietminh_disarray" }
	if = { limit = { country_exists = NLF NLF = { has_idea = NLF_post_binh_disorganization } } log = "BINH: NLF holds post_binh_disorganization" }
	if = { limit = { country_exists = NLF NLF = { has_idea = NLF_fractured_command_structure } } log = "BINH: NLF holds fractured_command_structure" }

	if = { limit = { country_exists = NLF NLF = { has_country_flag = NLF_Cung_Displaced } } log = "BINH: Cung displaced from the southern command" }
	if = { limit = { country_exists = NLF NLF = { has_country_flag = NLF_Buu_Displaced } } log = "BINH: Buu displaced from the southern command" }
	if = { limit = { country_exists = VIE VIE = { has_country_flag = Cung_Defection } } log = "BINH: Cung has rallied to Saigon" }
	if = { limit = { country_exists = VIE VIE = { has_character = NLF_Kieu_Cong_Cung } } log = "BINH: Cung is on Saigon's roster" }
}
