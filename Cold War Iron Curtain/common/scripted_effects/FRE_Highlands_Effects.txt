fre_highlands_mang_yang_risk = {
	set_temp_variable = { hl_risk = 45 }
	if = {
		limit = { has_country_flag = FRE_Highlands_Mang_Yang_Posture_Hold }
		add_to_temp_variable = { hl_risk = 10 }
	}
	if = {
		limit = { FRE_highlands_enemy_holds_dak_doa_trigger = yes }
		add_to_temp_variable = { hl_risk = 25 }
	}
	if = {
		limit = { FRE_highlands_enemy_holds_pleiku_trigger = yes }
		add_to_temp_variable = { hl_risk = 15 }
	}
	if = {
		limit = {
			NOT = { FRE_highlands_enemy_holds_dak_doa_trigger = yes }
			NOT = { FRE_highlands_enemy_holds_pleiku_trigger = yes }
		}
		add_to_temp_variable = { hl_risk = -20 }
	}
	if = {
		limit = { FRE_highlands_dak_doa_nlf_result_trigger = yes }
		add_to_temp_variable = { hl_risk = 15 }
	}
	clamp_temp_variable = { var = hl_risk min = 5 max = 90 }
}

fre_highlands_mang_yang_resolve = {
	if = {
		limit = { has_country_flag = FRE_Highlands_Mang_Yang_Posture_Airlift }
		set_temp_variable = { hl_outcome = 4 }
	}
	else = {
		fre_highlands_mang_yang_risk = yes
		fre_highlands_roll = yes
		log = "IC_AFK|HIGHLANDS_ROLL|date=[GetDateText]|package=Mang_Yang|risk=[?hl_risk|0]|roll=[?hl_roll|0]"
		if = { limit = { check_variable = { hl_roll < hl_half } } set_temp_variable = { hl_outcome = 1 } }
		else_if = { limit = { check_variable = { hl_roll < hl_risk } } set_temp_variable = { hl_outcome = 2 } }
		else_if = { limit = { has_country_flag = FRE_Highlands_Mang_Yang_Posture_Hold } set_temp_variable = { hl_outcome = 5 } }
		else = { set_temp_variable = { hl_outcome = 3 } }
	}
	fre_highlands_mang_yang_apply = yes
}

		indochina_struggle_award_communist = yes

		fre_metropole_patience_change = yes

		limit = { check_variable = { hl_outcome = 2 } }
		set_country_flag = FRE_Highlands_Chu_Dreh_Result_Escaped
		fre_highlands_remove_gm42 = yes
		set_temp_variable = { hl_place = 2 }
		set_temp_variable = { hl_strength = 1 }
		fre_highlands_place_gm42 = yes
		set_temp_variable = { ic_award = 5 }
		set_temp_variable = { ic_trace_writer = 20 }
		indochina_struggle_award_pro_france = yes
	}
	else = {
		set_country_flag = FRE_Highlands_Chu_Dreh_Result_Held
	}
	fre_highlands_clear_chu_dreh_postures = yes
	clr_country_flag = FRE_Highlands_Chu_Dreh_Active
	set_variable = { FRE_Highlands_Chu_Dreh_Days = 0 }
	country_event = { id = FRE_Highlands.4 hours = 4 }
}

fre_highlands_chu_dreh_tick = {
	if = {
		limit = { FRE_highlands_chu_dreh_due_trigger = yes }
		fre_highlands_chu_dreh_launch = yes
	}
	else_if = {
		limit = {
			OR = {
				has_country_flag = FRE_Highlands_Chu_Dreh_Choice_Pending
				has_country_flag = FRE_Highlands_Chu_Dreh_Active
			}
			NOT = { FRE_highlands_window_open_trigger = yes }
		}
		fre_highlands_chu_dreh_supersede = yes
		country_event = { id = FRE_Highlands.4 hours = 4 }
	}
	else_if = {
		limit = { has_country_flag = FRE_Highlands_Chu_Dreh_Active }
		add_to_variable = { FRE_Highlands_Chu_Dreh_Days = 1 }
		if = {
			limit = { check_variable = { FRE_Highlands_Chu_Dreh_Days > 4 } }
			fre_highlands_chu_dreh_resolve = yes
		}
	}
}

fre_highlands_cleanup = {
	if = {
		limit = {
			OR = {
				has_country_flag = FRE_Highlands_Mang_Yang_Choice_Pending
				has_country_flag = FRE_Highlands_Mang_Yang_Active
			}
		}
		fre_highlands_mang_yang_supersede = yes
	}
	if = {
		limit = {
			OR = {
				has_country_flag = FRE_Highlands_Chu_Dreh_Choice_Pending
				has_country_flag = FRE_Highlands_Chu_Dreh_Active
			}
		}
		fre_highlands_chu_dreh_supersede = yes
	}
	if = {
		limit = { has_country_flag = FRE_Highlands_GM100_Template_Loaded }
		delete_unit_template_and_units = { division_template = "Groupement Mobile 100" }
		clr_country_flag = FRE_Highlands_GM100_Template_Loaded
	}
	if = {
		limit = { has_country_flag = FRE_Highlands_GM42_Template_Loaded }
		delete_unit_template_and_units = { division_template = "Groupement Mobile 42" }
		clr_country_flag = FRE_Highlands_GM42_Template_Loaded
	}
	clr_country_flag = FRE_Highlands_GM100_Deployed
	clr_country_flag = FRE_Highlands_GM42_Deployed
}
