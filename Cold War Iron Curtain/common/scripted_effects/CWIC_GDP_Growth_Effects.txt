# ============================================================
# scripted_effects/CWIC_GDP_Growth_Effects.txt
# GDP storage + monthly annualized growth + yearly YoY growth
# Growth is measured only on states the country owned in both periods,
# so gaining or losing territory never reads as growth or contraction.
# ============================================================

cwic_init_gdp_growth_tracking = {
	set_variable = { CWIC_gdp_current = grossdomesticproduct }

	set_variable = { CWIC_gdp_growth_mom_annualized = 0 }
	set_variable = { CWIC_real_gdp_growth_yoy = 0 }
	set_variable = { CWIC_gdp_growth_yoy = 0 }
	set_variable = { CWIC_gdp_growth_yoy_prev1 = 0 }
	set_variable = { CWIC_gdp_growth_yoy_prev2 = 0 }
}

cwic_begin_gdp_growth_pass = {
	set_variable = { cwic_gdp_prev_pass = cwic_gdp_pass }
	add_to_variable = { cwic_gdp_pass = 1 }
	set_variable = { cwic_gdp_prev_state_count = cwic_gdp_state_count }
	set_variable = { cwic_gdp_state_count = 0 }
	set_variable = { cwic_gdp_kept_state_count = 0 }
	set_variable = { cwic_gdp_kept_now = 0 }
	set_variable = { cwic_gdp_kept_last_month = 0 }
	set_variable = { cwic_gdp_kept_now_year = 0 }
	set_variable = { cwic_gdp_kept_last_year = 0 }
}

cwic_track_state_gdp_for_growth = {
	set_temp_variable = { cwic_state_gdp_now = net_factory_earnings }
	add_to_temp_variable = { cwic_state_gdp_now = pop_lowerclass_gross_wage }
	add_to_temp_variable = { cwic_state_gdp_now = pop_middleclass_gross_wage }
	add_to_temp_variable = { cwic_state_gdp_now = pop_substinents_gross_wage }
	add_to_temp_variable = { cwic_state_gdp_now = pop_artisan_gross_wage }
	add_to_temp_variable = { cwic_state_gdp_now = pop_upperclass_gross_wage }

	if = {
		limit = {
			has_variable = cwic_gdp_owner_month
			check_variable = { cwic_gdp_owner_month = PREV.id }
			check_variable = { cwic_gdp_pass_seen = PREV.cwic_gdp_prev_pass }
		}
		add_to_variable = { PREV.cwic_gdp_kept_now = cwic_state_gdp_now }
		add_to_variable = { PREV.cwic_gdp_kept_last_month = cwic_gdp_state_last_month }
		add_to_variable = { PREV.cwic_gdp_kept_state_count = 1 }
	}

	if = {
		limit = { NOT = { has_variable = cwic_gdp_owner_year } }
		set_variable = { cwic_gdp_state_last_year = cwic_state_gdp_now }
		set_variable = { cwic_gdp_owner_year = PREV.id }
		set_variable = { cwic_gdp_year_seen = PREV.cwic_gdp_year_index }
	}
	if = {
		limit = {
			check_variable = { cwic_gdp_owner_year = PREV.id }
			check_variable = { cwic_gdp_year_seen = PREV.cwic_gdp_year_index }
		}
		add_to_variable = { PREV.cwic_gdp_kept_now_year = cwic_state_gdp_now }
		add_to_variable = { PREV.cwic_gdp_kept_last_year = cwic_gdp_state_last_year }
	}

	set_variable = { cwic_gdp_state_last_month = cwic_state_gdp_now }
	set_variable = { cwic_gdp_owner_month = PREV.id }
	set_variable = { cwic_gdp_pass_seen = PREV.cwic_gdp_pass }
	add_to_variable = { PREV.cwic_gdp_state_count = 1 }
}

cwic_store_gdp_and_monthly_growth = {
	set_variable = { CWIC_gdp_current = grossdomesticproduct }

	# Monthly growth annualized on retained states, percent*100: -250 = -2.50%
	set_variable = { CWIC_gdp_growth_mom_annualized = 0 }
	if = {
		limit = { check_variable = { cwic_gdp_kept_last_month > 0.1 } }
		set_temp_variable = { cwic_gdp_growth_tmp = cwic_gdp_kept_now }
		subtract_from_temp_variable = { cwic_gdp_growth_tmp = cwic_gdp_kept_last_month }
		divide_temp_variable = { cwic_gdp_growth_tmp = cwic_gdp_kept_last_month }
		multiply_temp_variable = { cwic_gdp_growth_tmp = 120000 }
		set_variable = { CWIC_gdp_growth_mom_annualized = cwic_gdp_growth_tmp }
	}

	if = {
		limit = { check_variable = { cwic_gdp_kept_state_count < cwic_gdp_prev_state_count } }
		set_country_flag = { flag = CWIC_CRISIS_GRACE days = 180 value = 1 }
	}
}

cwic_update_yearly_gdp_growth = {
	# YoY growth on states owned at the last yearly roll and still owned, percent*100: 375 = 3.75%
	set_variable = { CWIC_gdp_current = grossdomesticproduct }

	set_variable = { CWIC_real_gdp_growth_yoy = 0 }
	if = {
		limit = { check_variable = { cwic_gdp_kept_last_year > 0.1 } }
		set_temp_variable = { cwic_gdp_growth_tmp = cwic_gdp_kept_now_year }
		subtract_from_temp_variable = { cwic_gdp_growth_tmp = cwic_gdp_kept_last_year }
		divide_temp_variable = { cwic_gdp_growth_tmp = cwic_gdp_kept_last_year }
		multiply_temp_variable = { cwic_gdp_growth_tmp = 10000 }
		set_variable = { CWIC_real_gdp_growth_yoy = cwic_gdp_growth_tmp }
	}

	# 3-year smoothing buffer
	set_temp_variable = { cwic_gdp_growth_smooth = CWIC_real_gdp_growth_yoy }
	add_to_temp_variable = { cwic_gdp_growth_smooth = CWIC_gdp_growth_yoy_prev1 }
	add_to_temp_variable = { cwic_gdp_growth_smooth = CWIC_gdp_growth_yoy_prev2 }
	divide_temp_variable = { cwic_gdp_growth_smooth = 3 }
	set_variable = { CWIC_gdp_growth_yoy = cwic_gdp_growth_smooth }

	set_variable = { CWIC_gdp_growth_yoy_prev2 = CWIC_gdp_growth_yoy_prev1 }
	set_variable = { CWIC_gdp_growth_yoy_prev1 = CWIC_real_gdp_growth_yoy }

	# Roll the yearly baseline to the states measured in the latest monthly pass
	if = {
		limit = { has_variable = cwic_gdp_pass }
		add_to_variable = { cwic_gdp_year_index = 1 }
		every_owned_state = {
			limit = {
				check_variable = { cwic_gdp_owner_month = PREV.id }
				check_variable = { cwic_gdp_pass_seen = PREV.cwic_gdp_pass }
			}
			set_variable = { cwic_gdp_state_last_year = cwic_gdp_state_last_month }
			set_variable = { cwic_gdp_owner_year = PREV.id }
			set_variable = { cwic_gdp_year_seen = PREV.cwic_gdp_year_index }
		}
	}
}
