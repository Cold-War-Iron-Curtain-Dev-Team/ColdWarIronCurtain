# Archive findings only. This charter adds no idea, modifier, or operative.

SOV_psych_clamp_variables = {
	clamp_variable = {
		var = SOV_psych_record
		min = 0
		max = 100
	}
	clamp_variable = { var = SOV_psych_charter_state min = 0 max = 2 }
	clamp_variable = { var = SOV_psych_demo_stage min = 0 max = 7 }
	clamp_variable = { var = SOV_psych_slot min = 0 max = 1 }
	if = { limit = { has_variable = SOV_psych_integrated_stage } clamp_variable = { var = SOV_psych_integrated_stage min = 0 max = 6 } }
	if = { limit = { has_variable = SOV_project_x_stage } clamp_variable = { var = SOV_project_x_stage min = 0 max = 9 } }
	clamp_variable = {
		var = SOV_psych_scrutiny
		min = 0
		max = 100
	}
}

SOV_psych_initialize = {
	if = {
		limit = {
			SOV_psych_enabled_trigger = yes
			NOT = { has_country_flag = SOV_psych_initialized }
		}
		set_variable = { SOV_psych_record = 0 }
		set_variable = { SOV_psych_scrutiny = 0 }
		set_variable = { SOV_psych_charter_state = 0 }
		set_variable = { SOV_psych_demo_stage = 0 }
		set_variable = { SOV_psych_slot = 0 }
		set_variable = { SOV_psych_integrated_stage = 0 }
		if = {
			limit = { SOV_project_x_enabled_trigger = yes }
			set_variable = { SOV_project_x_stage = 0 }
		}
		set_country_flag = SOV_psych_initialized
		WP_psych_global_initialize = yes
	}
}

SOV_psych_complete_demonstration = {
	if = {
		limit = {
			SOV_psych_charter_active_trigger = yes
			check_variable = { SOV_psych_demo_stage = 2 }
			has_country_flag = SOV_psych_demo_paid
		}
		set_country_flag = SOV_psych_advancing_stage
		set_variable = { SOV_psych_demo_stage = 3 }
		if = {
			limit = { has_active_mission = SOV_psych_demo_research }
			remove_mission = SOV_psych_demo_research
		}
		clr_country_flag = SOV_psych_advancing_stage
		activate_mission = SOV_psych_demo_review
	}
}

# Removing a mission fires its cancel effect. The advancing flag keeps a finished
# study from being marked cancelled.
SOV_psych_cancel_open_study = {
	if = {
		limit = {
			NOT = { has_country_flag = SOV_psych_advancing_stage }
			NOT = { has_country_flag = SOV_psych_claim_rejected }
			OR = {
				check_variable = { SOV_psych_demo_stage = 2 }
				check_variable = { SOV_psych_demo_stage = 3 }
			}
		}
		set_variable = { SOV_psych_demo_stage = 6 }
		set_variable = { SOV_psych_slot = 0 }
	}
}

	}

SOV_psych_resolve_epilogue_flags = {
	if = {
		limit = {
			NOT = { has_country_flag = SOV_psych_methodological_archive }
			NOT = { has_country_flag = SOV_psych_fragmented_archive }
			NOT = { has_country_flag = SOV_psych_claims_without_program }
		}
		if = {
			limit = {
				OR = {
					has_country_flag = SOV_psych_claim_rejected
					has_country_flag = SOV_psych_integrated_complete
					AND = {
						check_variable = { SOV_psych_record > 19 }
						check_variable = { SOV_psych_scrutiny < 25 }
					}
				}
			}
			set_country_flag = SOV_psych_methodological_archive
		}
		else_if = {
			limit = { has_country_flag = SOV_psych_prague_reviewed }
			set_country_flag = SOV_psych_fragmented_archive
		}
		else = {
			set_country_flag = SOV_psych_claims_without_program
		}
	}
}
