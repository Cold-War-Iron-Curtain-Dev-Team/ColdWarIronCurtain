domestic_bond_calc = {
	if = {
		limit = {
			check_variable = {
				available_liquidity < 1
			}
		}
		set_variable = {
			available_liquidity = 1
		}
	}	#Failsafe to prevent crash

	set_variable = {
		outcrowding_limit = 10
	}	#Billions per office park
	multiply_variable = {
		outcrowding_limit = ROOT.office_park_total
	}
	#MULTIPLY OUTCROWDING LIMIT PER OFFICE PARK BY TECH#
	set_variable = {
		office_park_tech = modifier@office_park_income_bonus
	}
    add_to_variable = {
        office_park_tech = 1
    }
	multiply_variable = {
		outcrowding_limit = office_park_tech
	}
	set_variable = {
		outcrowding_temp_var = outcrowding_limit
	}
	#REPLENISHMENT RATES FOR OUTCROWDING LIMITS#
	if = {
		limit = {
			check_variable = {
				outcrowding_limit < 100
			}
		}
		multiply_variable = {
			outcrowding_temp_var = 0.065
		}		#Percent to restore difference per month
	}
	if = {
		limit = {
			check_variable = {
				outcrowding_limit < 500
			}
			check_variable = {
				outcrowding_limit > 100
			}
		}
		multiply_variable = {
			outcrowding_temp_var = 0.05
		}		#Percent to restore difference per month
	}
	if = {
		limit = {
			check_variable = {
				outcrowding_limit > 500
			}
		}
		multiply_variable = {
			outcrowding_temp_var = 0.035
		}		#Percent to restore difference per month
	}
	#REPLENISHMENT SCRIPT#
	if = {
		limit = {
			check_variable = {
				available_liquidity < outcrowding_limit
			}
		}		#Liquidity less than limit
		add_to_variable = {
			available_liquidity = outcrowding_temp_var
		}		#Replenish available liquidity
	}
}

		}
		else_if = {
			limit = {
				check_variable = { National_HDI > 0.360 }
			}
			set_variable = { Credit_Rating = 4 }
		}
		else = {
			set_variable = { Credit_Rating = 3 }
		}

		# Reserve currency privilege
		if = {
			limit = {
				has_variable = CWIC_reserve_currency
				check_variable = { CWIC_reserve_currency > 0 }
				check_variable = { Credit_Rating < 10 }
			}
			add_to_variable = { Credit_Rating = 1 }
		}

		# Postwar dollar shortage / reconstruction pressure
		if = {
			limit = {
				OR = {
					tag = FRA
					tag = FRE
					tag = ITA
					tag = WGR
					tag = AUS
					tag = JAP
					tag = GRE
					tag = PDG
				}
				check_variable = { Credit_Rating > 3 }
			}
			subtract_from_variable = { Credit_Rating = 1 }
		}

		# Command economy external credit limitations
		if = {
			limit = {
				OR = {
					tag = DDR
					tag = POL
					tag = CZE
					tag = HUN
					tag = ROM
					tag = BUL
					tag = ALB
					tag = PRC
					tag = KPA
					tag = MON
				}
				check_variable = { Credit_Rating > 3 }
			}
			subtract_from_variable = { Credit_Rating = 1 }
		}

		# Strong U.S.-aligned / advanced creditor states
		if = {
			limit = {
				OR = {
					tag = CAN
					tag = SWI
					tag = SWE
				}
				check_variable = { Credit_Rating < 10 }
			}
			add_to_variable = { Credit_Rating = 1 }
		}

		# Clamp
		clamp_variable = { var = Credit_Rating min = 1 max = 10 }
	}
	else = {
		set_variable = { Credit_Rating = 3 }
	}
}

		multiply_temp_variable = { infl_discount = 0.50 }      # max 50% dampening
		set_temp_variable = { infl_mult = 1 }

	if = {
		limit = { has_variable = CWIC_inflation_mom }

				check_variable = { CWIC_reserve_currency > 0 }
			}
			multiply_temp_variable = { infl_premium = 0.50 } # tuning knob
		}

		add_to_variable = { Interest_Rate = infl_premium }
	}
	if = {

	if = {
		limit = { has_global_flag = CWIC_oil_pricing_SOV }
		if = { limit = { original_tag = SOV } multiply_variable = { Interest_Rate = 0.95 } }
	}
}

cwic_default_monthly_check = {

	# Safety init
	if = { limit = { NOT = { has_variable = currentDebtPayments } } set_variable = { currentDebtPayments = 0 } }
	if = { limit = { NOT = { has_variable = currentMoney } } set_variable = { currentMoney = 0 } }
	if = { limit = { NOT = { has_variable = Credit_Rating } } set_variable = { Credit_Rating = 7 } }

	# Evaluate only if payments exist
	if = {
		limit = { check_variable = { currentDebtPayments > 0.01 } }

		# Liquidity default: cannot cover debt service this month
		if = {
			limit = { check_variable = { currentMoney < currentDebtPayments } }

			set_country_flag = CWIC_DEFAULTED
			set_country_flag = CWIC_MARKET_LOCKOUT

			# Immediate credibility hit (tunable)
			subtract_from_variable = { Credit_Rating = 2 }
			if = { limit = { check_variable = { Credit_Rating < 1 } } set_variable = { Credit_Rating = 1 } }

			# Macro shock (tunable)
			add_to_variable = { CWIC_inflation_mom = 0.010 }
			add_to_variable = { CWIC_black_market_pressure = 0.10 }

			# Clamp black market
			if = { limit = { check_variable = { CWIC_black_market_pressure > 1 } } set_variable = { CWIC_black_market_pressure = 1 } }
		}
	}
}
cwic_clear_default_status = {
	clr_country_flag = CWIC_DEFAULTED
	clr_country_flag = CWIC_MARKET_LOCKOUT
}
cwic_reserve_currency_update = {

	if = { limit = { NOT = { has_variable = CWIC_reserve_currency } } set_variable = { CWIC_reserve_currency = 0 } }
	if = { limit = { NOT = { has_variable = CWIC_reserve_trust } } set_variable = { CWIC_reserve_trust = 0 } }

	# Only evaluate for current reserve issuers
	if = {
		limit = { check_variable = { CWIC_reserve_currency > 0 } }

		# Inflation erosion
		if = { limit = { check_variable = { CWIC_inflation_mom > 0.01 } } subtract_from_variable = { CWIC_reserve_trust = 0.02 } }

		# Default destroys trust
		if = { limit = { has_country_flag = CWIC_DEFAULTED } subtract_from_variable = { CWIC_reserve_trust = 0.10 } }

		# Peg collapse reduces credibility
		if = { limit = { has_country_flag = CWIC_fx_pegged_currency_break } subtract_from_variable = { CWIC_reserve_trust = 0.05 } }

		# Stability helps rebuild trust slowly
		if = { limit = { check_variable = { CWIC_inflation_mom < 0.004 } } add_to_variable = { CWIC_reserve_trust = 0.01 } }

		# Clamp trust
		if = { limit = { check_variable = { CWIC_reserve_trust > 1 } } set_variable = { CWIC_reserve_trust = 1 } }
		if = { limit = { check_variable = { CWIC_reserve_trust < 0 } } set_variable = { CWIC_reserve_trust = 0 } }

		# Lose reserve status if trust collapses
		if = {
			limit = { check_variable = { CWIC_reserve_trust < 0.30 } }
			set_variable = { CWIC_reserve_currency = 0 }
		}
	}
	# Lose reserve if GDP collapses
	if = {
		limit = {
			check_variable = { CWIC_reserve_currency > 0 }
			check_variable = { grossdomesticproduct < 200 }
		}
		set_variable = { CWIC_reserve_currency = 0 }
	}
}
cwic_reserve_candidate_check = {

	if = { limit = { NOT = { has_variable = grossdomesticproduct } } set_variable = { grossdomesticproduct = 0 } }

	set_variable = { CWIC_reserve_candidate = 0 }

	if = {
		limit = {
			check_variable = { grossdomesticproduct > 300 }
			check_variable = { CWIC_inflation_mom < 0.004 }
			check_variable = { Credit_Rating > 8 }
			NOT = { has_country_flag = CWIC_DEFAULTED }
		}
		set_variable = { CWIC_reserve_candidate = 1 }
	}
}

	# Clear structural oil trust bonus
	if = { limit = { has_variable = CWIC_oil_structural_bonus } set_variable = { CWIC_oil_structural_bonus = 0 } }

	# USA oil pricing
	if = {
		limit = { has_global_flag = CWIC_oil_pricing_USA }
		if = { limit = { original_tag = USA } add_to_variable = { CWIC_reserve_trust = 0.01 } }
	}

	# USSR oil pricing
	if = {
		limit = { has_global_flag = CWIC_oil_pricing_USSR }
		if = { limit = { original_tag = SOV } add_to_variable = { CWIC_reserve_trust = 0.01 } }
	}

	# Clamp trust
	if = { limit = { check_variable = { CWIC_reserve_trust > 1 } } set_variable = { CWIC_reserve_trust = 1 } }
}
