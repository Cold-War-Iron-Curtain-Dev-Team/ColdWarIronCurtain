# Export variants used by the NSB branches of legacy country focuses.
# Each flag is stored on the producer, so every producer gets one stable design.

cwic_create_export_main_battle_tank_1942 = {
	hidden_effect = {
		if = {
			limit = { NOT = { has_country_flag = cwic_export_main_battle_tank_1942_created } }
			create_equipment_variant = {
				name = "CWIC Export Main Battle Tank 1942"
				type = medium_tank_chassis_1
				obsolete = yes # Export-only design; keep out of the producer's default production list.
				allow_without_tech = yes
				parent_version = 0
				modules = {
					main_armament_slot = tank_medium_cannon0
					turret_type_slot = conventional_turret
					suspension_type_slot = Bogie_0
					armor_type_slot = Armor_0_W
					engine_type_slot = tank_gasoline_engine
					tank_special_slot_1 = ap_0p
					tank_special_slot_2 = tank_he_0p
				}
			}
			set_country_flag = cwic_export_main_battle_tank_1942_created
		}
	}
}
