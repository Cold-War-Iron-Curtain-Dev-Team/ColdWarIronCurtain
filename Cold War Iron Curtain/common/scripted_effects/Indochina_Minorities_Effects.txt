# =============================================================================
# Highland Settlement (Viet Bac / Tai-Meo)
# =============================================================================
# Hanoi answers the autonomy question twice: the Tay and Nung marches (THO.1)
# and the Tai-Meo highlands (MEO.27).
#   direct rule - national spirit plus a resentment modifier on every state taken
#   autonomy    - national spirit plus an autonomous region modifier and state
#                 flag on the region's states
# Both routes keep the states owned by VIN. An autonomous region is never a
# released tag: GDP, tax, consumer goods, resources and factories are summed
# over owned states, so a subject would take all of them away from Hanoi.
# Both spirits have two tiers; tier 2 fires when both regions are settled the
# same way. Every effect here runs in VIN scope unless noted.
# =============================================================================

vin_viet_bac_question_check = {
	if = {
		limit = {
			NOT = { has_country_flag = VIN_viet_bac_question_asked }
			VIN_viet_bac_settlement_trigger = yes
			owns_state = 1280	# Cao Bang - no zone to offer without its seat
			owns_state = 1768	# Lang Son
			NOT = { country_exists = THO }
		}
		set_country_flag = VIN_viet_bac_question_asked
		country_event = { id = THO.1 days = 240 random_days = 60 }	# the OTL region was constituted in 1956, not at the ceasefire
	}
}

vin_highland_direct_rule_spirit = {
	add_to_variable = { VIN_highland_direct_rule_regions = 1 }
	if = {
		limit = {
			has_idea = VIN_highland_direct_rule
			check_variable = { VIN_highland_direct_rule_regions > 1 }
		}
		swap_ideas = { remove_idea = VIN_highland_direct_rule add_idea = VIN_highland_direct_rule_2 }
	}
	else_if = {
		limit = {
			NOT = { has_idea = VIN_highland_direct_rule }
			NOT = { has_idea = VIN_highland_direct_rule_2 }
		}
		add_ideas = VIN_highland_direct_rule
	}
}

vin_highland_autonomy_spirit = {
	add_to_variable = { VIN_highland_autonomy_regions = 1 }
	if = {
		limit = {
			has_idea = VIN_highland_autonomy_compact
			check_variable = { VIN_highland_autonomy_regions > 1 }
		}
		swap_ideas = { remove_idea = VIN_highland_autonomy_compact add_idea = VIN_highland_autonomy_compact_2 }
	}
	else_if = {
		limit = {
			NOT = { has_idea = VIN_highland_autonomy_compact }
			NOT = { has_idea = VIN_highland_autonomy_compact_2 }
		}
		add_ideas = VIN_highland_autonomy_compact
	}
}

vin_integrate_viet_bac_directly = {
	vin_highland_direct_rule_spirit = yes
	every_owned_state = {
		limit = {
			OR = {
				state = 1280	# Cao Bang
				state = 1768	# Lang Son
				state = 881	# Dong Bac Bo - the tungsten and steel
				state = 1767	# Ha Giang, if the Tai-Meo settlement did not take it
			}
			vin_highland_state_unsettled = yes
		}
		add_dynamic_modifier = { modifier = VIN_highland_direct_rule_state }
	}
}

vin_integrate_tai_meo_directly = {
	vin_highland_direct_rule_spirit = yes
	every_owned_state = {
		limit = {
			OR = {
				state = 671	# Tay Bac Bo - the chromium and tungsten
				state = 1767	# Ha Giang
			}
			vin_highland_state_unsettled = yes
		}
		add_dynamic_modifier = { modifier = VIN_highland_direct_rule_state }
	}
}

vin_establish_viet_bac_zone = {
	vin_highland_autonomy_spirit = yes
	every_owned_state = {
		limit = {
			OR = {
				state = 1280
				state = 1768
				state = 881
				state = 1767
			}
			vin_highland_state_unsettled = yes
		}
		set_state_flag = VIN_viet_bac_region
		add_dynamic_modifier = { modifier = VIN_viet_bac_region_state }
	}
	set_country_flag = VIN_viet_bac_zone_established
}

# Runs after VIN has annexed MEO, so Ha Giang is Hanoi's again.
vin_establish_tai_meo_region = {
	vin_highland_autonomy_spirit = yes
	every_owned_state = {
		limit = {
			OR = {
				state = 671
				state = 1767
			}
			vin_highland_state_unsettled = yes
		}
		set_state_flag = VIN_tai_meo_region
		add_dynamic_modifier = { modifier = VIN_tai_meo_region_state }
	}
	set_country_flag = VIN_tai_meo_region_established
}

# on_daily_VIN. The 1959 Duong Trung Nhan rising, against the region rather than
# a separate MEO zone. A region chaired by Vuong Chi Sinh never sees it.
vin_tai_meo_revolt_check = {
	if = {
		limit = {
			date > 1959.10.01
			date < 1959.11.01
			has_country_flag = VIN_tai_meo_region_established
			NOT = { has_country_flag = VIN_tai_meo_led_by_vuong_chi_sinh }
			NOT = { has_country_flag = VIN_tai_meo_revolt_opened }
			owns_state = 1767
			NOT = { country_exists = MEO }
		}
		set_country_flag = VIN_tai_meo_revolt_opened
		country_event = MEO.18
	}
}

vin_tai_meo_release_duong_trung_nhan = {
	1767 = {
		clr_state_flag = VIN_tai_meo_region
		remove_dynamic_modifier = { modifier = VIN_tai_meo_region_state }
	}
	release = MEO
	MEO = {
		if = {
			limit = { NOT = { has_country_flag = MEO_Templates_Loaded } }
			set_country_flag = MEO_Templates_Loaded
			load_oob = "MEO_templates"
		}
		capital_scope = {
			create_unit = {
				division = "name = \"Dan-quan Meo\" division_template = \"Dan-quan Meo\" start_experience_factor = 0.5 start_manpower_factor = 1 start_equipment_factor = 0.8"
				owner = MEO
			}
			create_unit = {
				division = "name = \"Dan-quan Meo\" division_template = \"Dan-quan Meo\" start_experience_factor = 0.5 start_manpower_factor = 1 start_equipment_factor = 0.8"
				owner = MEO
			}
		}
		set_country_flag = MEO_1959_uprising_joined
		complete_national_focus = MEO_the_duong_trung_nhan_revolt
		complete_national_focus = MEO_proclaim_duong_trung_nhan_as_king_of_the_meo
	}
}
