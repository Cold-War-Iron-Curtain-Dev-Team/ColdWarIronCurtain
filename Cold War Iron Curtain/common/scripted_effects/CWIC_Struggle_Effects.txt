					AND = {
						OR = {
							NOT = { country_exists = VIE }
							VIE = { has_capitulated = yes }
						}
						NOT = {
							USA = {
								has_completed_focus = USA_50s_Operation_Vulture

						has_war_with = VIE

				}
			}

	}
}

indochina_struggle_check_geneva_negotiations_complete = {
    set_temp_variable = { negotiations_complete_count = 0 }

    if = {
        limit = {
            OR = {
                AND = {
                    country_exists = FRA
                    FRA = { has_country_flag = FRA_Geneva_Negotiations_Complete }
                }
                NOT = { country_exists = FRA }
            }
        }
        add_to_temp_variable = { negotiations_complete_count = 1 }
    }
    if = {
        limit = {
            OR = {
                AND = {
                    country_exists = VIN
                    VIN = { has_country_flag = VIN_Geneva_Negotiations_Complete }
                }
                NOT = { country_exists = VIN }
            }
        }
        add_to_temp_variable = { negotiations_complete_count = 1 }
    }
    if = {
        limit = {
            OR = {
                AND = {
                    country_exists = USA
                    USA = { has_country_flag = USA_Geneva_Negotiations_Complete }
                }
                NOT = { country_exists = USA }
            }
        }
        add_to_temp_variable = { negotiations_complete_count = 1 }
    }
    if = {
        limit = {
            OR = {
                AND = {
                    country_exists = VIE
                    VIE = { has_country_flag = VIE_Geneva_Negotiations_Complete }
                }
                NOT = { country_exists = VIE }
            }
        }
        add_to_temp_variable = { negotiations_complete_count = 1 }
    }

    if = {
        limit = {
            check_variable = { temp.negotiations_complete_count = 4 }
            has_global_flag = Geneva_Conference_Preparations
        }
        set_global_flag = Geneva_Conference_Negotiations_Complete
    }
}

indochina_struggle_clamp_scores = {
    FRA = {
        if = {
            limit = { check_variable = { StruggleInvolvedNationsIndochinaCommunistScore < 0 } }
            set_variable = { StruggleInvolvedNationsIndochinaCommunistScore = 0 }
        }
        if = {
            limit = { check_variable = { StruggleInvolvedNationsIndochinaProFranceScore < 0 } }
            set_variable = { StruggleInvolvedNationsIndochinaProFranceScore = 0 }
        }
        if = {
            limit = { check_variable = { StruggleInvolvedNationsIndochinaProIndependenceScore < 0 } }
            set_variable = { StruggleInvolvedNationsIndochinaProIndependenceScore = 0 }
        }
        if = {
            limit = { check_variable = { StruggleInvolvedNationsIndochinaProEthnicScore < 0 } }
            set_variable = { StruggleInvolvedNationsIndochinaProEthnicScore = 0 }
        }
        if = {
            limit = { check_variable = { StruggleInvolvedNationsIndochinaKuomintangScore < 0 } }
            set_variable = { StruggleInvolvedNationsIndochinaKuomintangScore = 0 }
        }
        if = {
            limit = { check_variable = { StruggleInvolvedNationsIndochinaTotalAntiCommunistScore < 0 } }
            set_variable = { StruggleInvolvedNationsIndochinaTotalAntiCommunistScore = 0 }
        }
    }
    if = {
        limit = {
            check_variable = { global.Indochina_War_Active_Phase = 3 }
            check_variable = { global.Indochina_War_Next_Phase_A_Points > 499 }
        }
        set_variable = { global.Indochina_War_Next_Phase_A_Points = 499 }	# the bar reads full and stays there
    }
}

indochina_struggle_update_phase_transitions = {
    if = {
        limit = { check_variable = { global.Indochina_War_Active_Phase = 3 } }
        set_global_flag = Indochina_War_Escalated
    }
    else = {
        clr_global_flag = Indochina_War_Escalated
    }
    if = {
        limit = { check_variable = { global.Indochina_War_Active_Phase = 3 } }
        if = {
            # Held open for the test effects, which write the points directly;
            # in play the daily clamp pins escalation at 499.
            limit = {
                check_variable = { global.Indochina_War_Next_Phase_A_Points > 499 }
                NOT = { has_global_flag = Indochina_War_Over }
                NOT = { has_global_flag = Geneva_Conference }
            }
            set_variable = { global.Indochina_War_Next_Phase_A = 9 }
        }
        else = {
            set_variable = { global.Indochina_War_Next_Phase_A = 3 }
        }
        set_variable = { global.Indochina_War_Next_Phase_B = 4 }
    }
    else_if = {
        limit = { check_variable = { global.Indochina_War_Active_Phase = 4 } }
        set_variable = { global.Indochina_War_Next_Phase_A = 3 }
        set_variable = { global.Indochina_War_Next_Phase_B = 5 }
    }
    else_if = {
        limit = { check_variable = { global.Indochina_War_Active_Phase = 5 } }
        set_variable = { global.Indochina_War_Next_Phase_A = 4 }
        set_variable = { global.Indochina_War_Next_Phase_B = 6 }
    }
    else_if = {
        limit = { check_variable = { global.Indochina_War_Active_Phase = 6 } }
        set_variable = { global.Indochina_War_Next_Phase_A = 5 }
        set_variable = { global.Indochina_War_Next_Phase_B = 7 }
    }
    else_if = {
        limit = { check_variable = { global.Indochina_War_Active_Phase = 7 } }
        set_variable = { global.Indochina_War_Next_Phase_A = 6 }
        set_variable = { global.Indochina_War_Next_Phase_B = 8 }
    }
    else_if = {
        limit = { check_variable = { global.Indochina_War_Active_Phase = 8 } }
        set_variable = { global.Indochina_War_Next_Phase_A = 7 }
        if = {
            limit = {
                check_variable = { global.Indochina_War_Next_Phase_B_Points > 499 }
                NOT = { has_global_flag = Indochina_War_Over }
                NOT = { has_global_flag = Geneva_Conference }
            }
            set_variable = { global.Indochina_War_Next_Phase_B = 11 }
        }
        else = {
            set_variable = { global.Indochina_War_Next_Phase_B = 8 }
        }
    }
}

# The "resolve with whichever ending the war has already earned" ladder. It used
# to live inline in Indochina_Wrapup_Timer.timeout_effect; the failsafe router
# needs the same decision, so it lives here and both callers share it.
#
# The rungs test the ic_ending_*_ok mirrors rather than the indochina_struggle_
# ending_*_trigger originals: the mirrors read the global score copies, so this
# still decides correctly once France has been annexed and the FRA-scope scores
# are unreadable. A conference that is genuinely running still outranks every
# decisive rung, which is what the timer always did; only a dead Geneva latch -
# the softlock case - is ignored.
indochina_struggle_resolve_best_available_ending = {
	if = {
		limit = { NOT = { ic_failsafe_geneva_live_trigger = yes } }
		if = {
			limit = { has_global_flag = USA_VIN_Reunification_Complete }
			indochina_struggle_american_north_vietnam = yes
		}
		else_if = {
			limit = {
				country_exists = VIN
				VIN = { ic_ending_communist_ok = yes }
			}
			indochina_struggle_communist_victory = yes
		}
		else_if = {
			limit = {
				country_exists = VIE
				VIE = { ic_ending_southern_ok = yes }
			}
			indochina_struggle_southern_victory = yes
		}
		else_if = {
			limit = {
				country_exists = PQC
				PQC = { ic_ending_kmt_ok = yes }
			}
			indochina_struggle_kuomintang_victory = yes
		}
		else_if = {
			limit = {
				country_exists = VIE
				VIE = { ic_ending_dan_quoc_ok = yes }
			}
			indochina_struggle_dan_quoc_peace = yes
		}
		else_if = {
			limit = {
				country_exists = FRA
				FRA = { ic_ending_federal_ok = yes }
			}
			indochina_struggle_federal_vietnam = yes
		}
		else_if = {
			limit = {
				country_exists = FRA
				FRA = { ic_ending_balkanized_ok = yes }
			}
			indochina_struggle_balkanized_vietnam = yes
		}
	}
}

					limit = { NOT = { 671 = { is_owned_by = MEO } } }
					transfer_state = 671
				}
				transfer_state = 1761	# Hoang Lien Son
				transfer_state = 1766	# Hoa Binh
				transfer_state = 1281	# Van Don-Quang Ninh, the Nung marches
				transfer_state = 881
				# Cao Bang and Lang Son come north with the rest, unless the Viet Bac
				# region already governs them - the accords do not undo a
				# reorganisation Hanoi made on its own side of the line.
				if = {
					limit = { NOT = { country_exists = THO } }
					transfer_state = 1280
					transfer_state = 1768	# Lang Son, a VIN and THO core exactly like Cao Bang
				}
				if = {
					limit = { country_exists = MEO }
					if = {
						limit = {
							has_global_flag = geneva_conference_concluded
							check_variable = { global.geneva_minorities_term = 1 }
						}
						set_autonomy = {
							target = MEO
							autonomy_state = autonomy_easternbloc
							end_wars = no
							end_civil_wars = no
						}
					}
					else = {
						set_autonomy = {
							target = MEO
							autonomy_state = autonomy_integrated_puppet
							end_wars = no
							end_civil_wars = no
						}
					}
				}
				else = {
					transfer_state = 1767
				}
				# Only the line between the two Vietnams varies.
				if = {
					limit = { has_global_flag = geneva_unified_settlement }
					transfer_state = 838
				}
				else_if = {
					limit = { check_variable = { global.geneva_partition_line = 1 } }
					# 13th parallel: Hanoi's zone runs down past Hue and Da Nang.
					transfer_state = 838
					transfer_state = 1758
					transfer_state = 757
					transfer_state = 1287
					transfer_state = 1286
					transfer_state = 1759
				}
				else_if = {
					limit = { check_variable = { global.geneva_partition_line = 3 } }
					# 20th parallel: Quang Binh and Nghe-Tinh stay south.
				}
				else = {
					transfer_state = 838
				}
				set_capital = { state = 1760 }
				if = {
					limit = {
						OR = {
							NOT = { has_global_flag = geneva_conference_concluded }
							check_variable = { global.geneva_alignment_term = 1 }
						}
					}
					dismantle_faction = yes
				}
				drop_cosmetic_tag = yes
				set_cosmetic_tag = DRV	# Hanoi leaves Geneva as the DRV; the Socialist Republic name only comes with reunification in 1976
			}
		}
		if = { limit = { country_exists = NLF } NLF = { drop_cosmetic_tag = yes } }
		if = {
			limit = { country_exists = VIE }
			VIE = {
				if = {
					limit = { is_subject_of = FRA }
					remove_ideas = State_of_Vietnam_Faction
				}
			}
		}
		if = {
			limit = { country_exists = VIN }
			VIN = {
				if = {
					limit = { is_subject_of = FRA }
					remove_ideas = Viet_Minh_Faction
				}
			}
		}
		every_state = {
			limit = {
				OR = {
					has_state_flag = VIN_unplanned_offensive_flag
					is_core_of = VIN
				}
			}
			clr_state_flag = VIN_unplanned_offensive_flag
			remove_dynamic_modifier = { modifier = unplanned_offensive }
		}
		if = {
			limit = { country_exists = CAM }
			CAM = {
				remove_ideas = State_of_Vietnam_Faction
			}
		}
		if = {
			limit = { country_exists = LOS }
			LOS = {
				remove_ideas = State_of_Vietnam_Faction
			}
		}
		if = {
			limit = {
				country_exists = VIN
				VIN = {
					has_focus_tree = VIN_1950s
					NOT = { has_completed_focus = VIN_Accept_Temporary_Partitions }
				}
			}
			VIN = { complete_national_focus = VIN_Accept_Temporary_Partitions }	# opens Hanoi's reunification-election chain, whether or not the accords drew a line
		}
		indochina_struggle_geneva_ending_wrapup = yes
	}
}
