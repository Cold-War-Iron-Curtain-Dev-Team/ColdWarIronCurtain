
indochina_struggle_auto_trigger_communist_victory_preparation = {
	if = {
		limit = {
			NOT = { has_global_flag = Indochina_War_Over }
			country_exists = VIN
			VIN = {
				NOT = { has_country_flag = VIN_Communist_Victory_Preparation_Triggered }
				OR = {
					AND = {
						OR = {
							NOT = { country_exists = VIE }
							VIE = { has_capitulated = yes }
						}
						NOT = {
							USA = {
								has_completed_focus = USA_50s_Operation_Vulture
							}
						}
					}
					AND = {
						has_war_with = VIE
						owns_state = 286
					}
				}
			}
		}
		VIN = {
			set_country_flag = VIN_Communist_Victory_Preparation_Triggered
		}
		add_to_variable = { FRA.StruggleInvolvedNationsIndochinaCommunistScore = 1500 }
		add_to_variable = { global.Indochina_War_Next_Phase_A_Points = 200 }
	}
}

indochina_struggle_check_geneva_negotiations_complete = {
    set_temp_variable = { negotiations_complete_count = 0 }

    if = {
        limit = {
            OR = {
                AND = {
                    country_exists = FRA
                    FRA = { has_country_flag = FRA_Geneva_Negotiations_Complete }
                }
                NOT = { country_exists = FRA }
            }
        }
        add_to_temp_variable = { negotiations_complete_count = 1 }
    }
    if = {
        limit = {
            OR = {
                AND = {
                    country_exists = VIN
                    VIN = { has_country_flag = VIN_Geneva_Negotiations_Complete }
                }
                NOT = { country_exists = VIN }
            }
        }
        add_to_temp_variable = { negotiations_complete_count = 1 }
    }
    if = {
        limit = {
            OR = {
                AND = {
                    country_exists = USA
                    USA = { has_country_flag = USA_Geneva_Negotiations_Complete }
                }
                NOT = { country_exists = USA }
            }
        }
        add_to_temp_variable = { negotiations_complete_count = 1 }
    }
    if = {
        limit = {
            OR = {
                AND = {
                    country_exists = VIE
                    VIE = { has_country_flag = VIE_Geneva_Negotiations_Complete }
                }
                NOT = { country_exists = VIE }
            }
        }
        add_to_temp_variable = { negotiations_complete_count = 1 }
    }

    if = {
        limit = {
            check_variable = { temp.negotiations_complete_count = 4 }
            has_global_flag = Geneva_Conference_Preparations
        }
        set_global_flag = Geneva_Conference_Negotiations_Complete
    }
}

indochina_struggle_update_phase_transitions = {
    if = {
        limit = { check_variable = { global.Indochina_War_Active_Phase = 3 } }
        if = {
            limit = {
                check_variable = { global.Indochina_War_Next_Phase_A_Points > 499 }
                NOT = { has_global_flag = Indochina_War_Over }
                NOT = { has_global_flag = Geneva_Conference }
            }
            set_variable = { global.Indochina_War_Next_Phase_A = 9 }
        }
        else = {
            set_variable = { global.Indochina_War_Next_Phase_A = 3 }
        }
        set_variable = { global.Indochina_War_Next_Phase_B = 4 }
    }
    else_if = {
        limit = { check_variable = { global.Indochina_War_Active_Phase = 4 } }
        set_variable = { global.Indochina_War_Next_Phase_A = 3 }
        set_variable = { global.Indochina_War_Next_Phase_B = 5 }
    }
    else_if = {
        limit = { check_variable = { global.Indochina_War_Active_Phase = 5 } }
        set_variable = { global.Indochina_War_Next_Phase_A = 4 }
        set_variable = { global.Indochina_War_Next_Phase_B = 6 }
    }
    else_if = {
        limit = { check_variable = { global.Indochina_War_Active_Phase = 6 } }
        set_variable = { global.Indochina_War_Next_Phase_A = 5 }
        set_variable = { global.Indochina_War_Next_Phase_B = 7 }
    }
    else_if = {
        limit = { check_variable = { global.Indochina_War_Active_Phase = 7 } }
        set_variable = { global.Indochina_War_Next_Phase_A = 6 }
        set_variable = { global.Indochina_War_Next_Phase_B = 8 }
    }
    else_if = {
        limit = { check_variable = { global.Indochina_War_Active_Phase = 8 } }
        set_variable = { global.Indochina_War_Next_Phase_A = 7 }
        if = {
            limit = {
                check_variable = { global.Indochina_War_Next_Phase_B_Points > 499 }
                NOT = { has_global_flag = Indochina_War_Over }
                NOT = { has_global_flag = Geneva_Conference }
            }
            set_variable = { global.Indochina_War_Next_Phase_B = 11 }
        }
        else = {
            set_variable = { global.Indochina_War_Next_Phase_B = 8 }
        }
    }
}

indochina_struggle_communist_victory = {
	set_variable = { global.Indochina_War_Active_Phase = 100 }

	if = {
		limit = { country_exists = VIN }
		VIN = {
			if = {
				limit = { country_exists = VIE }
				VIE = {
					remove_ideas = State_of_Vietnam_Faction
				}
				transfer_state = 757
				transfer_state = 982
				transfer_state = 286
				transfer_state = 1287
				transfer_state = 983
				annex_country = { target = VIE }
			}
			if = {
				limit = { country_exists = NLF }
				annex_country = { target = NLF }
			}
			if = {
				limit = { 1760 = { is_capital = no } }
				set_capital = { state = 1760 }
			}
			drop_cosmetic_tag = yes
			dismantle_faction = yes
		}
	}

	# The CEFEO cannot outlive a communist victory: the corps evacuates and
	# the DRV takes its ground (previously FRE was left dangling here,
	# producing orphaned borders).
	if = {
		limit = { country_exists = FRE }
		fre_dissolve_cefeo_to_vin = yes
	}

	if = {
		limit = {
			country_exists = VIN
			country_exists = LAO
		}
		VIN = {
			every_controlled_state = {
				limit = { is_core_of = LAO }
				remove_core_of = LOS
				transfer_state_to = LAO
			}
		}
	}
	if = {
		limit = {
			country_exists = LOS
			country_exists = LAO
		}
		LOS = {
			every_controlled_state = {
				limit = { is_core_of = LAO }
				remove_core_of = LOS
				transfer_state_to = LAO
			}
		}
	}

	if = {
		limit = { country_exists = CAM }
		FRA = {
			set_autonomy = { target = CAM autonomy_state = autonomy_free }
		}
	}

	if = { limit = { country_exists = FRA country_exists = VIN } FRA = { if = { limit = { has_war_with = VIN } white_peace = VIN } } }
	if = { limit = { country_exists = FRA country_exists = LAO } FRA = { if = { limit = { has_war_with = LAO } white_peace = LAO } } }
	if = { limit = { country_exists = FRA country_exists = NLF } FRA = { if = { limit = { has_war_with = NLF } white_peace = NLF } } }
	if = { limit = { country_exists = VIN country_exists = FRA } VIN = { if = { limit = { has_war_with = FRA } white_peace = FRA } } }
	if = { limit = { country_exists = VIN country_exists = LAO } VIN = { if = { limit = { has_war_with = LAO } white_peace = LAO } } }
	if = { limit = { country_exists = VIN country_exists = NLF } VIN = { if = { limit = { has_war_with = NLF } white_peace = NLF } } }

	every_state = {
		limit = {
			OR = {
				has_state_flag = VIN_unplanned_offensive_flag
				is_core_of = VIN
				is_core_of = VIE
			}
		}
		clr_state_flag = VIN_unplanned_offensive_flag
		remove_dynamic_modifier = { modifier = unplanned_offensive }
	}

	if = {
		limit = { country_exists = VIN }
		VIN = {
			remove_ideas = Viet_Minh_Faction
		}
	}

	if = {
		limit = { country_exists = CAM }
		CAM = {
			remove_ideas = State_of_Vietnam_Faction
		}
	}
	if = {
		limit = { country_exists = LOS }
		LOS = {
			remove_ideas = State_of_Vietnam_Faction
		}
	}

    if = {
        limit = { country_exists = FRA }
        FRA = {
            indochina_struggle_ending_cleanup = yes
            news_event = Indochina_Struggle_Ending.1
        }
    }
    else = {
        indochina_struggle_ending_cleanup = yes
        news_event = Indochina_Struggle_Ending.1
    }
}

indochina_struggle_run_geneva_ending_effect = {
	if = {
		limit = { geneva_conference_complete_failure_trigger = yes }
		every_country = {
			limit = { is_ai = no }
			news_event = Geneva_Conference_Outcome.3
		}

		indochina_struggle_ending_cleanup = yes
		every_country = {
			limit = { is_ai = no }
			news_event = { id = Indochina_Struggle_Ending.10 days = 3 }
		}
	}
	else = {
		indochina_struggle_unlock_geneva_conference = yes
		set_variable = { global.Indochina_War_Active_Phase = 11 }
		# France's peace-outs and subject releases must run in FRA scope: this
		# effect can be executed by the wrap-up mission (FRA) or by any player
		# clicking the ending in the struggle GUI.
		if = {
			limit = { country_exists = FRA }
			FRA = {
				if = { limit = { country_exists = VIN has_war_with = VIN } white_peace = VIN }
				if = { limit = { country_exists = NLF has_war_with = NLF } white_peace = NLF }
				if = { limit = { country_exists = LAO has_war_with = LAO } white_peace = LAO }
				# Laos and Cambodia stay associated states of the French Union
				# if the accords kept them there (clause 5 = 0).
				if = {
					limit = {
						country_exists = CAM
						has_global_flag = geneva_conference_concluded
						check_variable = { global.geneva_laos_cambodia_term = 0 }
						CAM = { is_subject_of = FRA }
					}
					set_autonomy = { target = CAM autonomy_state = autonomy_associated_state }
				}
				else_if = {
					limit = { country_exists = CAM }
					set_autonomy = { target = CAM autonomy_state = autonomy_free }
				}
				if = {
					limit = {
						country_exists = LOS
						has_global_flag = geneva_conference_concluded
						check_variable = { global.geneva_laos_cambodia_term = 0 }
						LOS = { is_subject_of = FRA }
					}
					set_autonomy = { target = LOS autonomy_state = autonomy_associated_state }
				}
				else_if = {
					limit = { country_exists = LOS }
					set_autonomy = { target = LOS autonomy_state = autonomy_free }
				}
				# A retained French presence (clause 3 = 0) keeps the State of
				# Vietnam an associated state instead of freeing it outright.
				if = {
					limit = {
						country_exists = VIE
						has_global_flag = geneva_conference_concluded
						check_variable = { global.geneva_withdrawal_term = 0 }
						VIE = { is_subject_of = FRA }
					}
					set_autonomy = { target = VIE autonomy_state = autonomy_associated_state }
				}
				else_if = {
					limit = { country_exists = VIE }
					set_autonomy = { target = VIE autonomy_state = autonomy_free }
				}
			}
		}
		if = {
			limit = { country_exists = VIE }
			VIE = {
				if = { limit = { country_exists = VIN } white_peace = VIN }
				if = { limit = { country_exists = LAO } white_peace = LAO }
				if = { limit = { country_exists = NLF } white_peace = NLF }
				if = { limit = { country_exists = NLF } annex_country = { target = NLF } }
				if = { limit = { country_exists = CCC } annex_country = { target = CCC } }
				# Geneva's minority guarantees (clause 6 = 1): the highland
				# front survives as an autonomous state instead of being
				# absorbed into the South.
				if = {
					limit = {
						country_exists = FUL
						has_global_flag = geneva_conference_concluded
						check_variable = { global.geneva_minorities_term = 1 }
					}

						target = FUL
						autonomy_state = autonomy_associated_state
					}
				}
				else_if = {

					annex_country = { target = FUL }
				}

				# line (global.geneva_partition_line: 1 = 13th, 3 = 20th,
				# anything else = the historical 17th parallel). A settlement
				# with no partition line at all draws no border: both
				# governments stand where they are until the interim
				# arrangements decide who rules the single Vietnam.
				if = {
					limit = { has_global_flag = geneva_unified_settlement }
				}
				else_if = {
					limit = { check_variable = { global.geneva_partition_line = 1 } }
					# 13th parallel: Hue, Da Nang and Quang Ngai go north.
					transfer_state = 982
					transfer_state = 286
					transfer_state = 983
				}
				else_if = {
					limit = { check_variable = { global.geneva_partition_line = 3 } }
					# 20th parallel: the South also keeps Quang Binh and Nghe-Tinh.
					transfer_state = 757
					transfer_state = 982
					transfer_state = 286
					transfer_state = 1287
					transfer_state = 983
					transfer_state = 838
					transfer_state = 1763
					transfer_state = 1875
				}
				else = {
					transfer_state = 757
					transfer_state = 982
					transfer_state = 286
					transfer_state = 1287
					transfer_state = 983
				}
				# Enforced neutrality (clause 4 = 1) pulls the South out of
				# its bloc; a free-alignment settlement lets it keep its ties.
				if = {
					limit = {
						OR = {
							NOT = { has_global_flag = geneva_conference_concluded }
							check_variable = { global.geneva_alignment_term = 1 }
						}
					}
					leave_faction = yes
				}
			}
		}
		if = {
			limit = { country_exists = CAM }
			CAM = {
				drop_cosmetic_tag = yes
				if = { limit = { country_exists = VIN } white_peace = VIN }
				if = { limit = { country_exists = LAO } white_peace = LAO }
				if = { limit = { country_exists = NLF } white_peace = NLF }
				leave_faction = yes
			}
		}
		# Pathet Lao keep their OTL regroupment zones (Phongsaly & Sam Neua) as
		# a VIN satellite - but only if the accords granted communist
		# regroupment zones in Laos (clause 5 = 2; legacy endings keep the OTL
		# behaviour). This must run before LOS absorbs LAO, or the rump state
		# would be annexed away.
		if = {
			limit = {
				country_exists = VIN
				country_exists = LAO
				OR = {
					NOT = { has_global_flag = geneva_conference_concluded }
					check_variable = { global.geneva_laos_cambodia_term = 2 }
				}
			}

					end_wars = no
					end_civil_wars = no
				}
			}
		}
		if = {
			limit = { country_exists = LOS }
			LOS = {
				drop_cosmetic_tag = yes
				if = { limit = { country_exists = VIN } white_peace = VIN }
				if = { limit = { country_exists = LAO } white_peace = LAO }
				if = { limit = { country_exists = NLF } white_peace = NLF }
				leave_faction = yes
				if = {
					limit = {
						country_exists = LAO
						OR = {
							NOT = { country_exists = VIN }
							AND = {
								# No regroupment zones agreed at Geneva: the
								# Kingdom reabsorbs the Pathet Lao areas.
								has_global_flag = geneva_conference_concluded
								NOT = { check_variable = { global.geneva_laos_cambodia_term = 2 } }
							}
						}
					}
					annex_country = { target = LAO }
				}
			}
		}
		if = {
			limit = { country_exists = VIN }
			VIN = {
				if = {
					limit = { has_global_flag = geneva_unified_settlement }
					# No partition line: no transfers either way.
				}
				else_if = {
					limit = { check_variable = { global.geneva_partition_line = 1 } }
					# 13th parallel: Hanoi's zone runs down past Hue and Da Nang.
					transfer_state = 838
					transfer_state = 786
					transfer_state = 671
					transfer_state = 881
					transfer_state = 1280
					transfer_state = 1281
					transfer_state = 1760
					transfer_state = 1761
					transfer_state = 1766
					transfer_state = 1758
					transfer_state = 757
					transfer_state = 1287
					transfer_state = 1286
					transfer_state = 1759
				}
				else_if = {
					limit = { check_variable = { global.geneva_partition_line = 3 } }
					# 20th parallel: Quang Binh and Nghe-Tinh stay south.
					transfer_state = 786
					transfer_state = 671
					transfer_state = 881
					transfer_state = 1280
					transfer_state = 1281
					transfer_state = 1760
					transfer_state = 1761
					transfer_state = 1766
				}
				else = {
					transfer_state = 838
					transfer_state = 786
					transfer_state = 671
					transfer_state = 881
					transfer_state = 1280
					transfer_state = 1281
					transfer_state = 1760
					transfer_state = 1761
					transfer_state = 1766
				}
				set_capital = { state = 1760 }
				# Enforced neutrality (clause 4 = 1) dissolves the Viet Minh
				# war coalition; free alignment lets Hanoi keep its bloc.
				if = {
					limit = {
						OR = {
							NOT = { has_global_flag = geneva_conference_concluded }
							check_variable = { global.geneva_alignment_term = 1 }
						}
					}
					dismantle_faction = yes
				}
				drop_cosmetic_tag = yes
			}
		}
		if = { limit = { country_exists = NLF } NLF = { drop_cosmetic_tag = yes } }
		if = {
			limit = { country_exists = VIE }
			VIE = {
				if = {
					limit = { is_subject_of = FRA }
					remove_ideas = State_of_Vietnam_Faction
				}
			}
		}
		if = {
			limit = { country_exists = VIN }
			VIN = {
				if = {
					limit = { is_subject_of = FRA }
					remove_ideas = Viet_Minh_Faction
				}
			}
		}
		every_state = {
			limit = {
				OR = {
					has_state_flag = VIN_unplanned_offensive_flag
					is_core_of = VIN
				}
			}
			clr_state_flag = VIN_unplanned_offensive_flag
			remove_dynamic_modifier = { modifier = unplanned_offensive }
		}
		if = {
			limit = { country_exists = CAM }
			CAM = {
				remove_ideas = State_of_Vietnam_Faction
			}
		}
		if = {
			limit = { country_exists = LOS }
			LOS = {
				remove_ideas = State_of_Vietnam_Faction
			}
		}
		indochina_struggle_geneva_ending_wrapup = yes
	}
