stronghold_drift = { #Ensures that the minimum population (before other calculations) follows the stronghold's religion by a percent equal to the stronghold's value. IE THE RELIGION CAN NEVER FALL BELOW THE STRONGHOLD STRENGTH% IN THIS STATE
	set_temp_variable = { strongholdTrigger = state_population_k }
	
	set_temp_variable = { strongholdTriggerReligionZionistChristian = strongholdTrigger } #stronghold trigger = state pops
	multiply_temp_variable = { strongholdTriggerReligionZionistChristian = ReligionZionistChristianStronghold } #stronghold trigger = state pops x stronghold value (desired min)
	if = {
		limit = { check_variable = { ReligionZionistChristianStronghold > 0 } check_variable = { ReligionZionistChristian < strongholdTriggerReligionZionistChristian } } #Religious presence less than the stronghold religion's desired minimum
		set_temp_variable = { strongholdVar = state_population_k }
		multiply_temp_variable = { strongholdVar = ReligionZionistChristianStronghold }
		set_variable = { ReligionZionistChristianSafe = strongholdVar }

hdi_religious_drift_modifier_non_secular = {
	# Apply HDI multiplier to missionary effectiveness (default)
	if = {
		limit = { check_variable = { State_HDI < 0.3 } }
		multiply_temp_variable = { driftVar = 1.6 } # Very low HDI: +60%
	}
	else_if = {
		limit = { check_variable = { State_HDI < 0.5 } }
		multiply_temp_variable = { driftVar = 1.4 } # Low HDI: +40%
	}
	else_if = {
		limit = { check_variable = { State_HDI < 0.7 } }
		multiply_temp_variable = { driftVar = 1.15 } # Slightly below average HDI: +15%
	}
	else_if = {
		limit = { check_variable = { State_HDI > 0.9 } }
		multiply_temp_variable = { driftVar = 0.6 } # Very high HDI: -40%
	}
	else_if = {
		limit = { check_variable = { State_HDI > 0.7 } }
		multiply_temp_variable = { driftVar = 0.85 } # High HDI: -15%
	}
	else = { }
}
