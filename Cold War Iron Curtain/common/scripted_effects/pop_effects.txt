mission_drift = {
	prepare_religion_mission_drift_checks = yes
	if = {
		limit = { check_variable = { religion_any_mission_count > 0 } }
		set_temp_variable = {
			religion_base_mission_drift = {
				value = global.MissionStrength
				multiply = state_population_k
			}
		}
		if = {
			limit = { check_variable = { religion_non_secular_mission_count > 0 } }
			set_temp_variable = {
				religion_non_secular_hdi_multiplier = {
					value = 1
					add = { value = State_HDI less_than = 0.3 multiply = 0.6 }
					add = {
						value = State_HDI
						greater_than_or_equals = 0.3
						multiply = { value = State_HDI less_than = 0.5 }
						multiply = 0.4
					}
					add = {
						value = State_HDI
						greater_than_or_equals = 0.5
						multiply = { value = State_HDI less_than = 0.7 }
						multiply = 0.15
					}
					add = {
						value = State_HDI
						greater_than = 0.9
						multiply = -0.4
					}
					add = {
						value = State_HDI
						greater_than = 0.7
						multiply = { value = State_HDI less_than_or_equals = 0.9 }
						multiply = -0.15
					}
				}
			}
			set_temp_variable = {
				religion_non_secular_drift = {
					value = religion_base_mission_drift
					multiply = religion_non_secular_hdi_multiplier
				}
			}
		}
		if = {
			limit = { check_variable = { religion_secular_mission_count > 0 } }
			set_temp_variable = {
				religion_secular_hdi_multiplier = {
					value = 1
					add = { value = State_HDI less_than = 0.3 multiply = -0.4 }
					add = {
						value = State_HDI
						greater_than_or_equals = 0.3
						multiply = { value = State_HDI less_than = 0.5 }
						multiply = -0.25
					}
					add = {
						value = State_HDI
						greater_than_or_equals = 0.5
						multiply = { value = State_HDI less_than = 0.7 }
						multiply = -0.1
					}
					add = {
						value = State_HDI
						greater_than = 0.9
						multiply = 0.6
					}
					add = {
						value = State_HDI
						greater_than = 0.7
						multiply = { value = State_HDI less_than_or_equals = 0.9 }
						multiply = 0.25
					}
				}
			}
			set_temp_variable = {
				religion_secular_drift = {
					value = religion_base_mission_drift
					multiply = religion_secular_hdi_multiplier
				}
			}
		}
		if = { limit = { check_variable = { ReligionChristianityCatholicMission > 0 } } add_to_variable = { ReligionChristianityCatholic = religion_non_secular_drift } subtract_from_variable = { ReligionChristianityCatholicMission = 1 } }
		if = { limit = { check_variable = { ReligionChristianityProtestantMission > 0 } } add_to_variable = { ReligionChristianityProtestant = religion_non_secular_drift } subtract_from_variable = { ReligionChristianityProtestantMission = 1 } }
		if = { limit = { check_variable = { ReligionChristianityOrthodoxMission > 0 } } add_to_variable = { ReligionChristianityOrthodox = religion_non_secular_drift } subtract_from_variable = { ReligionChristianityOrthodoxMission = 1 } }
