

add_money_with_tooltip_effect = {
	if = {
		limit = {
			AND = {
				check_variable = { money_to_gain < 1 }
				check_variable = { money_to_gain > -1 }
			}
		}
		add_to_variable = {
			tooltip = adjust_money_mil2_tt
			currentMoney = money_to_gain
		}
	}
	else = {
		add_to_variable = {
			tooltip = adjust_money_tt
			currentMoney = money_to_gain
		}
	}
}

	multiply_variable = { pie_chart_test_g = 10 }
	add_to_variable = { pie_chart_test_g = pie_chart_test_r }

	set_variable = { pie_chart_test_y = influence_array_val^2 }
	multiply_variable = { pie_chart_test_y = 10 }
	add_to_variable = { pie_chart_test_y = pie_chart_test_g }

	# Blue: influence_array_val^3
	set_variable = { pie_chart_test_b = influence_array_val^3 }
	multiply_variable = { pie_chart_test_b = 10 }
	add_to_variable = { pie_chart_test_b = pie_chart_test_y }

	# Dark Blue: influence_array_val^4
	set_variable = { pie_chart_test_db = influence_array_val^4 }
	multiply_variable = { pie_chart_test_db = 10 }
	add_to_variable = { pie_chart_test_db = pie_chart_test_b }

	# Orange: influence_array_val^5
	set_variable = { pie_chart_test_o = influence_array_val^5 }
	multiply_variable = { pie_chart_test_o = 10 }
	add_to_variable = { pie_chart_test_o = pie_chart_test_db }

	# Pink: influence_array_val^6
	set_variable = { pie_chart_test_p = influence_array_val^6 }
	multiply_variable = { pie_chart_test_p = 10 }
	add_to_variable = { pie_chart_test_p = pie_chart_test_o }
	set_temp_variable = { show = 1 }
	subtract_from_temp_variable = { show = refresh_influence_chart }
	set_variable = { refresh_influence_chart = show }
	if = {
		limit = {
			check_variable = { show_chart = 15 }
		}
		set_variable = { show_chart = -80000 }
		set_variable = { show_chart_cover = -80000 }
	}
	else = {
		set_variable = { show_chart = 15 }
		set_variable = { show_chart_cover = 5 }
	}
}

increase_conscription = {
    if = {
        limit = {
            has_idea = disarmed_nation
        }   
        add_ideas = volunteer_only
    }			
    else_if = {
        limit = {
            has_idea = volunteer_only
        }   
        add_ideas = limited_conscription
    }
    else_if = {
        limit = {
            has_idea = limited_conscription
        }   
        add_ideas = extensive_conscription
    }
    else_if = {
        limit = {
            has_idea = extensive_conscription
        }   
        add_ideas = service_by_requirement
    }
    else_if = {
        limit = {
            has_idea = service_by_requirement
        }
        add_ideas = all_adults_serve
    }
    else_if = {
        limit = {
            has_idea = all_adults_serve
        }
        add_ideas = scraping_the_barrel
    }
    else = {
        add_political_power = 100
    }
}
decrease_conscription = {

    }
}
