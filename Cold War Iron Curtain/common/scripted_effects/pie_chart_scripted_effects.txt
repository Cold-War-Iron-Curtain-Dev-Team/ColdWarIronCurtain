# this creates a pie chart of popularities - can be easily adapted to show anything that adds up to 100%
create_pie_chart = {
	set_variable = { income_comparator = total_personal_income }
	add_to_variable = { income_comparator = total_corporate_tax }
	add_to_variable = { income_comparator = trade_income }
	add_to_variable = { income_comparator = tourism_infrastructure_total }

    clear_array = 4pie_chart
	resize_array = {
		array = 4pie_chart
		value = 1 # 1 is value for neutrality frame, so we don't have to make a loop for it
		size = 100 # 100 pieces, each percent is one piece
	}
	set_temp_variable = { tax_pop = total_personal_income }
	divide_temp_variable = {tax_pop = income_comparator }
	multiply_temp_variable = { tax_pop = 100 }
	round_temp_variable = tax_pop`

	set_temp_variable = { services_pop = total_corporate_tax }
	divide_temp_variable = {services_pop = income_comparator }
	multiply_temp_variable = { services_pop = 100 }
	round_temp_variable = services_pop 
	set_variable = { ROOT.last_idx = tax_pop }
	add_to_temp_variable = { services_pop = ROOT.last_idx }
	for_loop_effect = {
		start = ROOT.last_idx
		end = services_pop
		set_variable = { 4pie_chart^v = 2 }
		add_to_variable = { ROOT.last_idx = 1 }
	}

	set_temp_variable = { industry_pop = trade_income }
	divide_temp_variable = {industry_pop = income_comparator }
	multiply_temp_variable = { industry_pop = 100 }
	round_temp_variable = industry_pop
	add_to_temp_variable = { industry_pop = ROOT.last_idx }
	for_loop_effect = {
		start = ROOT.last_idx
		end = industry_pop
		set_variable = { 4pie_chart^v = 3 }
		add_to_variable = { ROOT.last_idx = 1 }
	}

	set_temp_variable = { farm_pop = tourism_infrastructure_total }
	divide_temp_variable = {farm_pop = income_comparator }
	multiply_temp_variable = { farm_pop = 100 }
	round_temp_variable = farm_pop 
	add_to_temp_variable = { farm_pop = ROOT.last_idx }
	for_loop_effect = {
		start = ROOT.last_idx
		end = farm_pop
		set_variable = { 4pie_chart^v = 4 }
	}

	clear_variable = ROOT.last_idx
	add_to_variable = { 4pie_chart_dirty = 1 }


	##military chart
	set_variable = { military_total_gui = mil_gui_total }
	add_to_variable = { military_total_gui = navy_gui_total }
	add_to_variable = { military_total_gui = af_gui_total }
	add_to_variable = { military_total_gui = sf_gui_total }

	clear_array = 2pie_chart
	resize_array = {
		array = 2pie_chart
		value = 1 # 1 is value for neutrality frame, so we don't have to make a loop for it
		size = 100 # 100 pieces, each percent is one piece
	}
	set_temp_variable = { army_pop = mil_gui_total }
	divide_temp_variable = {army_pop = military_total_gui }
	multiply_temp_variable = { army_pop = 100 }
	round_temp_variable = army_pop

	set_temp_variable = { navy_pop = navy_gui_total }
	divide_temp_variable = {navy_pop = military_total_gui }
	multiply_temp_variable = { navy_pop = 100 }
	round_temp_variable = navy_pop 
	set_variable = { ROOT.last_idx = army_pop }
	add_to_temp_variable = { navy_pop = ROOT.last_idx }
	for_loop_effect = {
		start = ROOT.last_idx
		end = navy_pop
		set_variable = { 2pie_chart^v = 2 }
		add_to_variable = { ROOT.last_idx = 1 }
	}

	set_temp_variable = { airforce_pop = af_gui_total }
	divide_temp_variable = {airforce_pop = military_total_gui }
	multiply_temp_variable = { airforce_pop = 100 }
	round_temp_variable = airforce_pop
	add_to_temp_variable = { airforce_pop = ROOT.last_idx }
	for_loop_effect = {
		start = ROOT.last_idx
		end = airforce_pop
		set_variable = { 2pie_chart^v = 3 }
		add_to_variable = { ROOT.last_idx = 1 }
	}

	set_temp_variable = { sf_pop = sf_gui_total }
	divide_temp_variable = {sf_pop = military_total_gui }
	multiply_temp_variable = { sf_pop = 100 }
	round_temp_variable = sf_pop 
	add_to_temp_variable = { sf_pop = ROOT.last_idx }

	# clamp
	if = {
		limit = { check_variable = { sf_pop > 100 } }
		set_temp_variable = { sf_pop = 100 }
	}

	for_loop_effect = {
		start = ROOT.last_idx
		end = sf_pop
		set_variable = { 2pie_chart^v = 4 }
	}

	clear_variable = ROOT.last_idx
	add_to_variable = { 2pie_chart_dirty = 1 }

	##expenditure pie chart
	
	clear_array = 3pie_chart
	resize_array = {
		array = 3pie_chart
		value = 1 # 1 is value for neutrality frame, so we don't have to make a loop for it
		size = 100 # 100 pieces, each percent is one piece
	}

	set_temp_variable = { military_expenditure = land_army_expenditure }
	divide_temp_variable = { military_expenditure = totalexp }
	multiply_temp_variable = { military_expenditure = 100 }
	round_temp_variable = military_expenditure

	set_temp_variable = { police_expenditure_gui = policing_expenditure }
	divide_temp_variable = { police_expenditure_gui = totalexp }
	multiply_temp_variable = { police_expenditure_gui = 100 }
	round_temp_variable = police_expenditure_gui 
	set_variable = { ROOT.last_idx = military_expenditure }
	add_to_temp_variable = { police_expenditure_gui = ROOT.last_idx }
	for_loop_effect = {
		start = ROOT.last_idx
		end = police_expenditure_gui
		set_variable = { 3pie_chart^v = 2 }
		add_to_variable = { ROOT.last_idx = 1 }
	}

	set_temp_variable = { edu_expenditure = education_expenditure }
	divide_temp_variable = { edu_expenditure = totalexp }
	multiply_temp_variable = { edu_expenditure = 100 }
	round_temp_variable = edu_expenditure
	add_to_temp_variable = { edu_expenditure = ROOT.last_idx }
	for_loop_effect = {
		start = ROOT.last_idx
		end = edu_expenditure
		set_variable = { 3pie_chart^v = 3 }
		add_to_variable = { ROOT.last_idx = 1 }
	}

	set_temp_variable = { public_expenditure = public_expenditure_total }
	add_to_temp_variable = { public_expenditure = welfare_cost }
	add_to_temp_variable = { public_expenditure = health_cost }
	set_variable = { public_gui_total = public_expenditure }
	divide_temp_variable = { public_expenditure = totalexp }
	multiply_temp_variable = { public_expenditure = 100 }
	round_temp_variable = public_expenditure 
	add_to_temp_variable = { public_expenditure = ROOT.last_idx }
	for_loop_effect = {
		start = ROOT.last_idx
		end = public_expenditure
		set_variable = { 3pie_chart^v = 4 }
	}

	clear_variable = ROOT.last_idx
	add_to_variable = { 3pie_chart_dirty = 1 }
}
