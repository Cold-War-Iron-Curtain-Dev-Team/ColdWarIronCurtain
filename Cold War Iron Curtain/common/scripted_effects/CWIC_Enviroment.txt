calc_state_pollution_effect = {
    # ===========================
    # STEP 1: Tech Factor
    # ===========================
    set_temp_variable = { tech_factor = PREV.pollution_tech_factor }
    set_temp_variable = { agri_organic_factor = PREV.pollution_agri_organic_factor }

    # ===========================
    # STEP 2: Population Factor
    # ===========================
    set_temp_variable = { population_factor = 1 }
    if = { limit = { pop_level_1_check = yes } set_temp_variable = { population_factor = 0.5 } }
    else_if = { limit = { pop_level_2_check = yes } set_temp_variable = { population_factor = 1 } }
    else_if = { limit = { pop_level_3_check = yes } set_temp_variable = { population_factor = 1.5 } }
    else_if = { limit = { pop_level_5_check = yes } set_temp_variable = { population_factor = 2 } }
    else_if = { limit = { pop_level_7_check = yes } set_temp_variable = { population_factor = 2.5 } }
    else_if = { limit = { pop_level_10_check = yes } set_temp_variable = { population_factor = 3 } }
    else_if = { limit = { pop_level_15_check = yes } set_temp_variable = { population_factor = 4 } }
    else_if = { limit = { pop_level_20_check = yes } set_temp_variable = { population_factor = 5 } }
    else = { }

    # ===========================
    # STEP 3: Development Factor
    # ===========================
    set_temp_variable = { development_factor = 1 }
    if = { limit = { development_level_zero_check = yes } set_temp_variable = { development_factor = 0.2 } }
    else_if = { limit = { development_level_one_check = yes } set_temp_variable = { development_factor = 0.4 } }
    else_if = { limit = { development_level_two_check = yes } set_temp_variable = { development_factor = 0.8 } }
    else_if = { limit = { development_level_three_check = yes } set_temp_variable = { development_factor = 1.2 } }
    else_if = { limit = { development_level_four_check = yes } set_temp_variable = { development_factor = 1.5 } }
    else_if = { limit = { development_level_five_check = yes } set_temp_variable = { development_factor = 1.8 } }
    else = { }

    # Agricultural scaling
    set_temp_variable = { agri_dev_factor = 0.25 }
    if = { limit = { development_level_one_check = yes } set_temp_variable = { agri_dev_factor = 0.5 } }
    else_if = { limit = { development_level_two_check = yes } set_temp_variable = { agri_dev_factor = 1 } }
    else_if = { limit = { development_level_three_check = yes } set_temp_variable = { agri_dev_factor = 1.5 } }
    else_if = { limit = { development_level_four_check = yes } set_temp_variable = { agri_dev_factor = 2 } }
    else_if = { limit = { development_level_five_check = yes } set_temp_variable = { agri_dev_factor = 3 } }
    else = { }

    # ===========================
    # STEP 4: Land Pollution
    # ===========================
    set_temp_variable = {
        land_base = {
            value = building_level@arms_factory
            multiply = 1.5
            add = { value = building_level@industrial_complex multiply = 1.2 }
            add = { value = building_level@dockyard multiply = 0.8 }
            add = { value = building_level@synthetic_refinery multiply = 2.0 }
            add = { value = building_level@steel_factory multiply = 4 }
            add = { value = building_level@tungsten_factory multiply = 1.5 }
            add = { value = building_level@fossil_fuel_powerplant multiply = 2.0 }
            add = { value = building_level@nuclear_reactor multiply = 0.8 }
            add = { value = building_level@office_park multiply = 0.5 }
            add = {
                value = building_level@agri_industrial_complex
                multiply = 0.6
                subtract = agri_organic_factor
                multiply = agri_dev_factor
                clamp = { min = 0 max = 100 }
            }
            subtract = { value = building_level@renewable_powerplant multiply = 1.5 }
        }
    }

    set_variable = { land_base_log = land_base }

    set_temp_variable = {
        land_base = {
            value = land_base
            multiply = population_factor
            multiply = development_factor
            multiply = tech_factor
        }
    }

    set_variable = { land_base_log_2 = land_base }

    # ===========================
    # RESOURCE EXTRACTION - LAND
    # ===========================
    set_temp_variable = {
        resource_pollution_land = {
            value = resource@oil
            multiply = 0.03
            add = { value = resource@aluminium multiply = 0.02 }
            add = { value = resource@tungsten multiply = 0.02 }
            add = { value = resource@steel multiply = 0.02 }
            add = { value = resource@chromium multiply = 0.05 }
        }
    }

    # Add to total land pollution
    set_temp_variable = {
        land_base = {
            value = land_base
            add = resource_pollution_land
            divide = 10
        }
    }

    if = {  #Urban Micro State Adjustment
        limit = {
            has_state_flag = micro
            has_state_flag = is_urban_state
        }
        divide_variable = { var = land_base value = 3 }
    }

    add_to_variable = { var = statelandpollution value = land_base }
    clamp_variable = { var = statelandpollution min = 0 max = 600 }

    # ===========================
    # STEP 5: Water Pollution
    # ===========================
    set_temp_variable = {
        water_base = {
            value = building_level@industrial_complex
            multiply = 0.5
            add = { value = building_level@arms_factory multiply = 0.35 }
            add = { value = building_level@dockyard multiply = 1 }
            add = {
                value = building_level@agri_industrial_complex
                multiply = 2
                subtract = agri_organic_factor
                multiply = agri_dev_factor
                clamp = { min = 0 max = 100 }
            }
            add = {
                value = building_level@water_infrastructure
                multiply = 0.4
                multiply = agri_dev_factor
            }
            add = { value = building_level@synthetic_refinery multiply = 2 }
            add = { value = building_level@steel_factory multiply = 4 }
            add = { value = building_level@tungsten_factory multiply = 1 }
            add = { value = building_level@fossil_fuel_powerplant multiply = 1.2 }
            add = { value = building_level@nuclear_reactor multiply = 3 }
            multiply = population_factor
            multiply = development_factor
            multiply = tech_factor
        }
    }

    #Resource Extraction
    # ===========================
    # RESOURCE EXTRACTION - WATER
    # ===========================
    set_temp_variable = {
        resource_pollution_water = {
            value = resource@oil
            multiply = 0.05
            add = { value = resource@chromium multiply = 0.1 }
            add = { value = resource@steel multiply = 0.05 }
            add = { value = resource@aluminium multiply = 0.05 }
        }
    }

    # Add to total water pollution
    set_temp_variable = {
        water_base = {
            value = water_base
            add = resource_pollution_water
        }
    }

    set_variable = { water_base_log_1 = water_base }

    # Scale down pollution to match game timeframe
    set_temp_variable = {
        water_base = {
            value = water_base
            divide = 10
        }
    }

    set_variable = { water_base_log_2 = water_base }

    if = {  #Urban Micro State Adjustment
        limit = {
            has_state_flag = micro
            has_state_flag = is_urban_state
        }
        divide_variable = { var = water_base value = 3 }
    }

    add_to_variable = { var = statewaterpollution value = water_base }
    clamp_variable = { var = statewaterpollution min = 0 max = 600 }
    apply_pollution_recovery_effect = yes
}

apply_pollution_recovery_effect = {
    # Base Recovery
    set_temp_variable = { land_recovery = -1 }
    set_temp_variable = { water_recovery = -2 }

    # Environmental Recovery Campaigns
    if = {
        limit = { check_variable = { local_water_recovery_initiative > 0 } }
        add_to_variable = { local_water_recovery_initiative = -1 }
        set_temp_variable = { campaign_water = -8 }
        # Scale by tech
        if = { limit = { owner = { has_tech = inframanagement_VI } } add_to_temp_variable = { campaign_water = -2 } }
        if = { limit = { owner = { has_tech = inframanagement_VII } } add_to_temp_variable = { campaign_water = -3 } }
        if = { limit = { owner = { has_tech = renewables_V } } add_to_temp_variable = { campaign_water = -2 } }
        if = { limit = { owner = { has_tech = renewables_conversion } } add_to_temp_variable = { campaign_water = -3 } }
        # Scale by global policies
        if = { limit = { has_country_flag = global_enviro_treaty } multiply_temp_variable = { campaign_water = 1.5 } }
        if = { limit = { has_country_flag = green_funding_program } multiply_temp_variable = { campaign_water = 1.5 } }
        add_to_temp_variable = { water_recovery = campaign_water }
    }
    if = {
        limit = { check_variable = { local_land_recovery_initiative > 0 } }
        add_to_variable = { local_land_recovery_initiative = -1 }
        set_temp_variable = { campaign_land = -10 }
        # Scale by tech
        if = { limit = { owner = { has_tech = inframanagement_VI } } add_to_temp_variable = { campaign_land = -2 } }
        if = { limit = { owner = { has_tech = inframanagement_VII } } add_to_temp_variable = { campaign_land = -3 } }
        if = { limit = { owner = { has_tech = renewables_IV } } add_to_temp_variable = { campaign_land = -3 } }
        if = { limit = { owner = { has_tech = renewables_conversion } } add_to_temp_variable = { campaign_land = -4 } }
        # Scale by global policies
        if = { limit = { has_country_flag = global_enviro_treaty } multiply_temp_variable = { campaign_land = 1.5 } }
        if = { limit = { has_country_flag = green_funding_program } multiply_temp_variable = { campaign_land = 1.5 } }
        add_to_temp_variable = { land_recovery = campaign_land }
    }

    # Building Factors
    if = { limit = { check_variable = { building_level@renewable_powerplant > 0 } } add_to_temp_variable = { land_recovery = -0.5 } }
    if = { limit = { check_variable = { building_level@industrial_complex < 3 } } add_to_temp_variable = { land_recovery = -0.5 } }
    if = { limit = { check_variable = { building_level@arms_factory < 2 } } add_to_temp_variable = { land_recovery = -0.5 } }
    if = { limit = { check_variable = { building_level@water_infrastructure > 1 } } add_to_temp_variable = { water_recovery = -1 } }
    if = { limit = { check_variable = { building_level@renewable_powerplant > 0 } } add_to_temp_variable = { water_recovery = -1 } }

    # Tech Factors - Land
    if = { limit = { owner = { has_tech = inframanagement_V } } add_to_temp_variable = { land_recovery = -0.5 } }
    if = { limit = { owner = { has_tech = inframanagement_VI } } add_to_temp_variable = { land_recovery = -1 } }
    if = { limit = { owner = { has_tech = inframanagement_VII } } add_to_temp_variable = { land_recovery = -1.5 } }
    if = { limit = { owner = { has_tech = renewables_V } } add_to_temp_variable = { land_recovery = -1 } }
    if = { limit = { owner = { has_tech = renewables_IV } } add_to_temp_variable = { land_recovery = -1 } }
    if = { limit = { owner = { has_tech = renewables_conversion } } add_to_temp_variable = { land_recovery = -1 } }

    # Tech Factors - Water
    if = { limit = { owner = { has_tech = inframanagement_V } } add_to_temp_variable = { water_recovery = -0.5 } }
    if = { limit = { owner = { has_tech = inframanagement_VI } } add_to_temp_variable = { water_recovery = -1 } }
    if = { limit = { owner = { has_tech = inframanagement_VII } } add_to_temp_variable = { water_recovery = -1.5 } }
    if = { limit = { owner = { has_tech = renewables_V } } add_to_temp_variable = { water_recovery = -1 } }
    if = { limit = { owner = { has_tech = renewables_conversion } } add_to_temp_variable = { water_recovery = -1 } }

    # Renewable Tech Boost (applies to both)
    if = { limit = { owner = { has_tech = renewables_I } } add_to_temp_variable = { land_recovery = -0.5 } }
    if = { limit = { owner = { has_tech = renewables_III } } add_to_temp_variable = { land_recovery = -0.5 } }
    if = { limit = { owner = { has_tech = renewables_IV } } add_to_temp_variable = { land_recovery = -0.5 } }
    if = { limit = { owner = { has_tech = renewables_I } } add_to_temp_variable = { water_recovery = -0.5 } }
    if = { limit = { owner = { has_tech = renewables_III } } add_to_temp_variable = { water_recovery = -0.5 } }
    if = { limit = { owner = { has_tech = renewables_IV } } add_to_temp_variable = { water_recovery = -0.5 } }

    # Apply Recovery
    add_to_variable = { var = statelandpollution value = land_recovery }
    add_to_variable = { var = statewaterpollution value = water_recovery }

    clamp_variable = { var = statelandpollution min = 0 max = 600 }
    clamp_variable = { var = statewaterpollution min = 0 max = 600 }

    update_pollution_flags_effect = yes
}

    if = { limit = { owner = { has_tech = organic_VII } } add_to_temp_variable = { agri_organic_factor = 0.10 } }

    # ===========================
    set_temp_variable = { land_base = 0 }       #Values Adjusted for Startup Rolling

    add_to_temp_variable = { land_base = temp_val }

    multiply_temp_variable = { temp_val = 1.5 }
    add_to_temp_variable = { land_base = temp_val }

    set_temp_variable = { temp_val = building_level@dockyard }                # 0.8 per level
    multiply_temp_variable = { temp_val = 1 }
    add_to_temp_variable = { land_base = temp_val }

    multiply_temp_variable = { temp_val = 2.0 }
    add_to_temp_variable = { land_base = temp_val }

    set_temp_variable = { temp_val = building_level@steel_factory }           # 1.5 per level
    multiply_temp_variable = { temp_val = 4 }
    add_to_temp_variable = { land_base = temp_val }

    set_temp_variable = { temp_val = building_level@tungsten_factory }        # 1.5 per level
    multiply_temp_variable = { temp_val = 1.5 }
    add_to_temp_variable = { land_base = temp_val }

    set_temp_variable = { temp_val = building_level@fossil_fuel_powerplant }  # 2.0 per level
    multiply_temp_variable = { temp_val = 2.5 }
    add_to_temp_variable = { land_base = temp_val }

    add_to_temp_variable = { land_base = temp_val }

    multiply_temp_variable = { temp_val = 0.1 }
    add_to_temp_variable = { land_base = temp_val }

    set_temp_variable = { temp_val = building_level@agri_industrial_complex } # 0.6 per level
    multiply_temp_variable = { temp_val = 0.6 }
    subtract_from_temp_variable = { temp_val = agri_organic_factor }
    multiply_temp_variable = { temp_val = agri_dev_factor }
    clamp_temp_variable = { var = temp_val min = 0 max = 100 }
    add_to_temp_variable = { land_base = temp_val }


    set_temp_variable = { temp_val = building_level@renewable_powerplant }    # -1.5 per level
    multiply_temp_variable = { temp_val = 1.5 }
    subtract_from_temp_variable = { land_base = temp_val }

    multiply_temp_variable = { land_base = population_factor }
    multiply_temp_variable = { land_base = development_factor }
    multiply_temp_variable = { land_base = tech_factor }

    # RESOURCE EXTRACTION - LAND
    # ===========================
    set_temp_variable = { resource_pollution_land = 0 }

    # Oil

    multiply_temp_variable = { temp_val = agri_dev_factor }
    clamp_temp_variable = { var = temp_val min = 0 max = 100 }
    add_to_temp_variable = { water_base = temp_val }

    set_temp_variable = { temp_val = building_level@water_infrastructure }    # 0.4 per level * agri_dev_factor
    multiply_temp_variable = { temp_val = 0.4 }
    multiply_temp_variable = { temp_val = agri_dev_factor }
    add_to_temp_variable = { water_base = temp_val }

    set_temp_variable = { temp_val = building_level@synthetic_refinery }      # 1.2 per level
    multiply_temp_variable = { temp_val = 2 }
    add_to_temp_variable = { water_base = temp_val }

    set_temp_variable = { temp_val = building_level@steel_factory }           # 0.8 per level
    multiply_temp_variable = { temp_val = 4 }
    add_to_temp_variable = { water_base = temp_val }

    set_temp_variable = { temp_val = building_level@tungsten_factory }        # 0.8 per level
    multiply_temp_variable = { temp_val = 1 }
    add_to_temp_variable = { water_base = temp_val }

    set_temp_variable = { temp_val = building_level@fossil_fuel_powerplant }  # 0.8 per level
    multiply_temp_variable = { temp_val = 1.2 }
    add_to_temp_variable = { water_base = temp_val }

    set_temp_variable = { temp_val = building_level@nuclear_reactor }         # 1.0 per level
    multiply_temp_variable = { temp_val = 3 }
    add_to_temp_variable = { water_base = temp_val }

    multiply_temp_variable = { water_base = population_factor }
    multiply_temp_variable = { water_base = development_factor }
    multiply_temp_variable = { water_base = tech_factor }

    #Resource Extraction
    # ===========================
    # RESOURCE EXTRACTION - WATER
    # ===========================
    set_temp_variable = { resource_pollution_water = 0 }

    # Oil
    set_temp_variable = { temp_res = resource@oil }
    multiply_temp_variable = { temp_res = 0.05 }
    add_to_temp_variable = { resource_pollution_water = temp_res }

    # Chromium (Uranium)    Reduced for Startup due to Less Mining History
    set_temp_variable = { temp_res = resource@chromium }
    multiply_temp_variable = { temp_res = 0.01 }
    add_to_temp_variable = { resource_pollution_water = temp_res }

    # Steel 
    set_temp_variable = { temp_res = resource@steel }
    multiply_temp_variable = { temp_res = 0.1 }
    add_to_temp_variable = { resource_pollution_water = temp_res }

    # Aluminum
    set_temp_variable = { temp_res = resource@aluminium }
    multiply_temp_variable = { temp_res = 0.1 }
    add_to_temp_variable = { resource_pollution_water = temp_res }

    # Add to total water pollution
    add_to_temp_variable = { water_base = resource_pollution_water }

    # Scale down pollution to match game timeframe
    multiply_temp_variable = { water_base = 10 }


    add_to_variable = { var = statewaterpollution value = water_base }
    clamp_variable = { var = statewaterpollution min = 0 max = 300 }

    update_pollution_flags_effect = yes
}
