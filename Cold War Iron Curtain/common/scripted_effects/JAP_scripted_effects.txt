JAP_Agricultural_Development = {
	custom_effect_tooltip = JAP_Agricultural_Development_tt
	if = {
		limit = { check_variable = { JAP_Agricultural_Development > 0.8 } }
		random_owned_controlled_state = {
			add_building_construction = {
				type = agri_industrial_complex
				level = 1
				instant_build = yes
			}
		}
	}
	else_if = {
		limit = { check_variable = { JAP_Agricultural_Development > 0.7 } }
		add_tech_bonus = {
			name = agri_tech_bonus
			bonus = 0.1
			uses = 1
			category = agriculture
		}
	}
	else_if = {
		limit = { check_variable = { JAP_Agricultural_Development < 0.5 } }
		add_to_variable = { food_bonus = 0.01 }
	}
	clamp_variable = {
		var = JAP_Agricultural_Development
		min = 0
		max = 1
	}
}

		limit = { 
			AND = {
				has_country_flag = JAP_BORYU_PREVIOUS_LEADER
				has_country_flag = JAP_HONRYU_NEW_LEADER
			}
		}
		add_popularity = {
			ideology = Liberal_Conservatism
			popularity = party_popularity@conservative
		}
		set_temp_variable = { boryu_pop = party_popularity@conservative }
		multiply_temp_variable = { boryu_pop = -1 }
		add_popularity = {
			ideology = conservative
			popularity = boryu_pop
		}
		clr_country_flag = JAP_BORYU_PREVIOUS_LEADER
		set_country_flag = JAP_HONRYU_PREVIOUS_LEADER
	}
	else_if = {
		limit = { 
			AND = {
				has_country_flag = JAP_HONRYU_PREVIOUS_LEADER
				has_country_flag = JAP_BORYU_NEW_LEADER
			}
		}
		add_popularity = {
			ideology = conservative
			popularity = party_popularity@Liberal_Conservatism
		}
		set_temp_variable = { honryu_pop = party_popularity@Liberal_Conservatism }
		multiply_temp_variable = { honryu_pop = -1 }
		add_popularity = {
			ideology = Liberal_Conservatism
			popularity = honryu_pop
		}
		clr_country_flag = JAP_HONRYU_PREVIOUS_LEADER
		set_country_flag = JAP_BORYU_PREVIOUS_LEADER
	}
}
