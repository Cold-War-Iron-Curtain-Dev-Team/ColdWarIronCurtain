# Fire-support designs created on research completion, 2026-09-17.
# Owner ruling: a fire-support technology must unlock the era's modules AND leave the
# player a design, because on an NSB profile its legacy vehicle is unbuildable and the
# research produced nothing visible. Each helper reuses the bookmark flag
# cwic_starting_<generation>_created, so a country that already received the design at a
# bookmark never gets a duplicate. Recipes are copied from the generic bookmark block
# where one exists, otherwise from the research-naming recipe for that generation.

cwic_firesupport_heavy_tank_artillery_chassis_4_design = {

		limit = {
			has_dlc = "No Step Back"
			has_tech = nsb_heavy_tanks3

		}
		create_equipment_variant = {
			name = "Standard Heavy SPG 1955"
			type = heavy_tank_artillery_chassis_4
