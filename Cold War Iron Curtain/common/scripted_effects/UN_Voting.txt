un_ai_vote_weights ={
    #var:UNS.un_nation_target is the UN Voting Target
    #influence_array^0 is the largest foreign influencer

    if = {
        limit = {
            check_variable = { UNS.un_action_set = 9 }
        }
        set_variable = {
            UNS.un_nation_target = UNS.un_state_target_owner
        }
    }
    set_variable = {
        ai_un_voting_weight = 0
    }
    # Infamy removed 2026-09-13. The AI used to subtract the target's badboy
    # score here, so a country nobody liked was easier to vote against. Nothing
    # generates or reads infamy any more - opinion, spheres and blocs carry
    # that weight instead.
    meta_effect = {
        text = {
            add_to_variable = { ai_un_voting_weight = THIS.opinion@[CHECKTAG] }
        }
        CHECKTAG = "[?var:UNS.un_nation_target.GetTag]"
    }       
    if = {
        limit = {
            has_war_with = UNS.un_nation_target
        }
        add_to_variable = { ai_un_voting_weight = -500 }
    }
    if = {
        limit = {
            tag = USA
            var:UNS.un_nation_target = {
                tag = ISR
            }
        }
        add_to_variable = { ai_un_voting_weight = 25 }
    }
    if = {
        limit = {
            OR = {
                tag = USA
                tag = ENG
            }
            NOT = {
                has_country_flag = Anti_Apartheid_FP
            }
            var:UNS.un_nation_target = {
                tag = SAF
            }
        }
        add_to_variable = { ai_un_voting_weight = 25 }
    }
    if = {
        limit = {
            var:UNS.un_nation_target = {
                is_subject_of = PREV
            }
        }
        add_to_variable = { ai_un_voting_weight = 400 }
    }
    # ---- BLOC COHESION ----
    # These two blocks used to read influence_array^0 ("our largest foreign
    # influencer") for 200 points each. The influence system is gone and that
    # array is never written any more, so 400 points of bloc behaviour had
    # silently stopped existing - countries were voting as individuals.
    #
    # Sphere_Leader replaces it. It is set by history and content, which is
    # where alignment belongs, so this reads the game's own alignment rather
    # than inventing a new one.
    if = {                                  # the target is our patron's client
        limit = {
            NOT = { check_variable = { Sphere_Leader = 0 } }
            var:UNS.un_nation_target = {
                check_variable = { Sphere_Leader = PREV.Sphere_Leader }
            }
        }
        add_to_variable = { ai_un_voting_weight = 200 }
    }
    if = {                                  # the target IS our patron
        limit = {
            NOT = { check_variable = { Sphere_Leader = 0 } }
            check_variable = { Sphere_Leader = UNS.un_nation_target }
        }
        add_to_variable = { ai_un_voting_weight = 200 }
    }
    if = {                                  # our patron's rival is the target
        limit = {
            NOT = { check_variable = { Sphere_Leader = 0 } }
            var:UNS.un_nation_target = {
                NOT = { check_variable = { Sphere_Leader = 0 } }
                NOT = { check_variable = { Sphere_Leader = PREV.Sphere_Leader } }
            }
        }
        subtract_from_variable = { ai_un_voting_weight = 120 }
    }
    if = {
        limit = {
            OR = {
                has_idea = SEATO_member
                has_idea = CENTO_member
                has_idea = nato_member
            }
        }
        if = {
            limit = {
                var:UNS.un_nation_target = {
                    OR = {
                        has_idea = SEATO_member
                        has_idea = CENTO_member
                        has_idea = nato_member
                    }
                }
            }
            add_to_variable = { ai_un_voting_weight = 200 }
        }
        #if = {
        #    limit = {
        #        var:UNS.un_nation_target = {
        #            OVERLORD = {
        #                OR = {
        #                    has_idea = SEATO_member
        #                    has_idea = CENTO_member
        #                    has_idea = nato_member
        #                }
        #            }
        #        }
        #    }
        #    add_to_variable = { ai_un_voting_weight = 200 }
        #} Didn't work either way, maybe OVERLORD scope doesn't work with variables (hoi4 is a great game)
    }
    if = {
        limit = {
            OR = {
                has_government = fascism
                has_government = nationalist
            }
        }
        if = {
            limit = {
                var:UNS.un_nation_target = {
                    OR = {
                        has_government = communism
                        has_government = maoism
                        has_government = trotskyism
                    }
                }
            }
            add_to_variable = { ai_un_voting_weight = -50 }
        }
        if = {
            limit = {
                var:UNS.un_nation_target = {
                    OR = {
                        has_government = fascism
                        has_government = nationalist
                    }
                }
            }
            add_to_variable = { ai_un_voting_weight = 50 }
        }
    }
    if = {
        limit = {
            OR = {
                has_government = communism
                has_government = maoism
                has_government = trotskyism
            }
        }
        if = {
            limit = {
                var:UNS.un_nation_target = {
                    OR = {
                        has_government = communism
                        has_government = maoism
                        has_government = trotskyism
                    }
                }
            }
            add_to_variable = { ai_un_voting_weight = 50 }
        }
        if = {
            limit = {
                var:UNS.un_nation_target = {
                    has_government = fascism
                }
            }
            add_to_variable = { ai_un_voting_weight = -50 }
        }
        if = {
            limit = {
                var:UNS.un_nation_target = {
                    OR = {
                        has_idea = SEATO_member
                        has_idea = CENTO_member
                        has_idea = nato_member
                    }
                }
            }
            add_to_variable = { ai_un_voting_weight = -50 }
        }
    }
    
    #if = {
       # limit = {
      #      is_subject = yes
     #   }
    #}   

    # ---- PLEDGED VOTES ----
    # Deliberately NOT handled here any more. This function runs when the
    # resolution STARTS; the patron votes during the ten days that follow, so
    # a term here could never see the patron's vote and every bought vote was
    # silently ignored. Pledges are applied from the vote buttons instead -
    # see cwic_un_apply_pledges in CWIC_un_pledges.txt.
}
UN_ai_vote_sorting = {
    un_ai_vote_weights = yes
	if = {
        limit = {
            OR = {
                check_variable = { UNS.un_action_set = 0 }
                check_variable = { UNS.un_action_set = 1 }
                check_variable = { UNS.un_action_set = 2 }
                check_variable = { UNS.un_action_set = 3 }
                check_variable = { UNS.un_action_set = 4 }
                check_variable = { UNS.un_action_set = 5 }
                check_variable = { UNS.un_action_set = 6 }
                check_variable = { UNS.un_action_set = 7 }
                check_variable = { UNS.un_action_set = 8 }
                check_variable = { UNS.un_action_set = 17 }
            }
        }
        
            if = {
                limit = {
                    has_idea = UN_Security_Council_Member_Perm
                    check_variable = { UNS.un_chamber = 1 }   # no veto in the Assembly
                    NOT = { has_country_flag = un_veto_spent }
                    has_political_power > 249
                    check_variable = {
                        ai_un_voting_weight > 24
                    }
                }
                add_to_array = { UNS.un_vote_veto_list = THIS.id }
                add_to_array = { UNS.un_vote_ = THIS.id }
                 add_to_variable = { UNS.un_no_count = 1 }
                add_to_variable = { UNS.un_veto_count = 1 }
                add_political_power = -250
                set_country_flag = { flag = un_veto_spent days = 180 value = 1 }
               set_country_flag = UN_VETO_ONGOING
            }
            else_if = {
                limit = {
                   check_variable = {
                        ai_un_voting_weight > 24
                    }
                }
                add_to_array = { UNS.un_vote_no_list = THIS.id }
                add_to_variable = { UNS.un_no_count = 1 }
                set_country_flag = UN_NO_ONGOING
            }
            else_if = {
                limit = {
                   check_variable = {
                        ai_un_voting_weight < -24
                    }

                }
                add_to_array = { UNS.un_vote_yes_list = THIS.id }
                add_to_variable = { UNS.un_yes_count = 1 }
                set_country_flag = UN_YES_ONGOING
            }
            else = {
                 add_to_array = { UNS.un_vote_abstain_list = THIS.id }
                add_to_variable = { UNS.un_abstain_count = 1 }  
                set_country_flag = UN_ABSTAIN_ONGOING                 
            }
        }
    
	if = {
        limit = {
            check_variable = { UNS.un_action_set = 9 }
        }

                    check_variable = { UNS.un_chamber = 1 }   # no veto in the Assembly

                        ai_un_voting_weight > 49
                    }
                }

                    }
                }
                add_to_array = { UNS.un_vote_no_list = THIS.id }
                add_to_variable = { UNS.un_no_count = 1 }
                set_country_flag = UN_NO_ONGOING
            }
            else_if = {
                limit = {
                  check_variable = { ai_un_voting_weight < 0 }
                }
                add_to_array = { UNS.un_vote_yes_list = THIS.id }
                add_to_variable = { UNS.un_yes_count = 1 }
                set_country_flag = UN_YES_ONGOING
            }
            else = {
                 add_to_array = { UNS.un_vote_abstain_list = THIS.id }
                add_to_variable = { UNS.un_abstain_count = 1 }  
                set_country_flag = UN_ABSTAIN_ONGOING                 
            }
    }
	if = {
        limit = {
            OR = {
                check_variable = { UNS.un_action_set = 10 }
                check_variable = { UNS.un_action_set = 11 }
                check_variable = { UNS.un_action_set = 12 }
            }
        }
            if = {
                limit = {
                    has_idea = UN_Security_Council_Member_Perm
                    check_variable = { UNS.un_chamber = 1 }   # no veto in the Assembly
                    NOT = { has_country_flag = un_veto_spent }
                    has_political_power > 249
                    check_variable = { ai_un_voting_weight < 0 }
                }
