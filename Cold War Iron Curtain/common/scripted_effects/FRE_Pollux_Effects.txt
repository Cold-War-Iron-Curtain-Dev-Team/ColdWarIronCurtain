# Operation Pollux evacuates Lai Chau into an already established Dien Bien Phu
# camp. It owns no territory, peace, primary-campaign result, or Geneva record.
# Results: force preserved, column lost, or theatre-superseded.

fre_pollux_focus_complete = {
	set_country_flag = FRE_Operation_Pollux_Launched
	if = {
		limit = {
			fre_pollux_live_window_trigger = yes
			fre_pollux_french_aligned_controls_lai_chau_trigger = yes
		}
		country_event = { id = FRE_Pollux.1 days = 1 }
	}
	else_if = {
		limit = {
			fre_pollux_live_window_trigger = yes
			NOT = { fre_pollux_french_aligned_controls_lai_chau_trigger = yes }
		}
		fre_pollux_resolve_origin_lost = yes
	}
	else = { fre_pollux_resolve_superseded = yes }
}

# Historical Pollux preserves the regular detachments flown out under Operation
# Leda, but the exposed partisan column is lost. The safer alternatives must
# earn seven consecutive checks before the campaign begins.
fre_pollux_daily_tick = {
	if = {
		limit = { has_country_flag = FRE_Pollux_Evacuation_Active }
		add_to_variable = { FRE_Pollux_Days_Elapsed = 1 }
		if = {
			limit = { has_country_flag = FRE_Pollux_Posture_Escort }
			if = {
				limit = { fre_pollux_corridor_open_trigger = yes }
				add_to_variable = { FRE_Pollux_Hold_Days = 1 }
			}
			else = { set_variable = { FRE_Pollux_Hold_Days = 0 } }
		}
		else_if = {
			limit = { has_country_flag = FRE_Pollux_Posture_Airlift }
			if = {
				limit = { fre_dbp_french_aligned_controls_camp_trigger = yes }
				add_to_variable = { FRE_Pollux_Hold_Days = 1 }
			}
			else = { set_variable = { FRE_Pollux_Hold_Days = 0 } }
		}

		# A valid seventh day outranks a same-day campaign launch or settlement.
		if = {
			limit = {
				OR = {
					AND = { has_country_flag = FRE_Pollux_Posture_Escort  NOT = { check_variable = { FRE_Pollux_Hold_Days < 7 } } }
					AND = { has_country_flag = FRE_Pollux_Posture_Airlift  NOT = { check_variable = { FRE_Pollux_Hold_Days < 7 } } }
				}
			}
			fre_pollux_resolve_preserved = yes
		}
		else_if = {
			limit = {
				OR = {
					has_global_flag = Indochina_War_Over
					has_global_flag = geneva_conference_concluded
					check_variable = { global.Indochina_War_Active_Phase > 8 }
					NOT = { country_exists = VIN }
					NOT = { country_exists = TAI }
					VIN = { has_country_flag = VIN_Campaign_Dien_Bien_Active }
				}
			}
			fre_pollux_resolve_superseded = yes
		}
		else_if = {
			limit = { has_country_flag = FRE_Pollux_Posture_Historical  check_variable = { FRE_Pollux_Days_Elapsed > 9 } }
			set_country_flag = FRE_Pollux_Result_Column_Loss_Historical
			fre_pollux_resolve_column_loss = yes
		}
		else_if = {
			limit = { check_variable = { FRE_Pollux_Days_Elapsed > 20 } }
			fre_pollux_resolve_column_loss = yes
		}
	}
}

fre_pollux_resolve_preserved = {
	if = {
		limit = { has_country_flag = FRE_Pollux_Evacuation_Active }
		set_country_flag = FRE_Operation_Pollux_Result_Preserved
		if = {
			limit = { has_country_flag = FRE_Pollux_Posture_Escort }
			set_country_flag = FRE_Pollux_Result_Preserved_Escort
			add_manpower = 2000
		}
		else = { set_country_flag = FRE_Pollux_Result_Preserved_Airlift }
		set_country_flag = FRE_Mission_Operation_Pollux_Completed
		set_country_flag = FRE_Operation_Cooldown_Pollux
		set_country_flag = FRE_Pollux_Garrison_Reinforced
		fre_pollux_supply_garrison_full = yes
		fre_pollux_deploy_survivor_group = yes
		army_experience = 15
		if = {
			limit = { NOT = { has_global_flag = Indochina_War_Over } }
			FRA = {
				add_to_variable = { StruggleInvolvedNationsIndochinaProFranceScore = 10  tooltip = Add_to_French_Victory_Points_TT }
				indochina_struggle_recalculate_total_anti_communist_score = yes
			}
		}
		fre_pollux_cleanup = yes
		country_event = { id = FRE_Pollux.2 hours = 6 }
	}
}

fre_pollux_resolve_column_loss = {
	if = {
		limit = {
			OR = {
				has_country_flag = FRE_Pollux_Evacuation_Active
				has_country_flag = FRE_Pollux_Result_Column_Loss_Historical
				has_country_flag = FRE_Pollux_Result_Column_Loss_Origin
				has_country_flag = FRE_Pollux_Result_Column_Loss_Unfunded
			}
			NOT = { has_country_flag = FRE_Operation_Pollux_Result_Column_Loss }
		}
		if = {
			limit = { has_country_flag = FRE_Pollux_Posture_Escort }
			set_country_flag = FRE_Pollux_Result_Column_Loss_Escort
			add_manpower = -3000
			set_temp_variable = { fre_patience_delta = -4 }
			fre_metropole_patience_change = yes
		}
		else_if = {
			limit = { has_country_flag = FRE_Pollux_Posture_Airlift }
			set_country_flag = FRE_Pollux_Result_Column_Loss_Airlift
			add_manpower = -1500
		}
		else_if = {
			limit = { has_country_flag = FRE_Pollux_Result_Column_Loss_Unfunded }
			add_manpower = -2000
		}
		else = {
			if = {
				limit = { NOT = { has_country_flag = FRE_Pollux_Result_Column_Loss_Origin } }
				set_country_flag = FRE_Pollux_Result_Column_Loss_Historical
			}
			add_manpower = -2000
			if = {
				limit = { has_country_flag = FRE_Pollux_Result_Column_Loss_Historical }
				fre_pollux_supply_garrison_leda = yes
			}
		}
		set_country_flag = FRE_Operation_Pollux_Result_Column_Loss
		set_country_flag = FRE_Mission_Operation_Pollux_Failed
		set_country_flag = FRE_Operation_Cooldown_Pollux
		add_stability = -0.01
		add_war_support = -0.02
		if = {
			limit = { NOT = { has_global_flag = Indochina_War_Over } }
			FRA = {
				add_to_variable = { StruggleInvolvedNationsIndochinaCommunistScore = 15  tooltip = Add_to_Communist_Victory_Points_TT }
				add_to_variable = { StruggleInvolvedNationsIndochinaProFranceScore = -15  tooltip = Add_to_French_Victory_Points_TT }
				indochina_struggle_recalculate_total_anti_communist_score = yes
			}
		}
		fre_pollux_cleanup = yes
		country_event = { id = FRE_Pollux.3 hours = 6 }
	}
}

fre_pollux_resolve_origin_lost = {
	fre_pollux_prepare_new_operation = yes
	set_country_flag = FRE_Pollux_Result_Column_Loss_Origin
	fre_pollux_resolve_column_loss = yes
}

fre_pollux_resolve_superseded = {
	if = {
		limit = {
			NOT = {
				OR = {
					has_country_flag = FRE_Operation_Pollux_Result_Preserved
					has_country_flag = FRE_Operation_Pollux_Result_Column_Loss
					has_country_flag = FRE_Operation_Pollux_Result_Superseded
				}
			}
		}
		set_country_flag = FRE_Operation_Pollux_Result_Superseded
		fre_pollux_cleanup = yes
		country_event = { id = FRE_Pollux.4 hours = 6 }
	}
}

# Called immediately from the parent campaign's begin effect. The daily tick's
# objective-first priority is reproduced here so a completed seventh day is not
# discarded merely because Hanoi launches on the same date.
fre_pollux_handoff_to_campaign = {
	if = {
		limit = { has_country_flag = FRE_Pollux_Evacuation_Active }
		if = {
			limit = {
				OR = {
					AND = { has_country_flag = FRE_Pollux_Posture_Escort  fre_pollux_corridor_open_trigger = yes  NOT = { check_variable = { FRE_Pollux_Hold_Days < 6 } } }
					AND = { has_country_flag = FRE_Pollux_Posture_Airlift  fre_dbp_french_aligned_controls_camp_trigger = yes  NOT = { check_variable = { FRE_Pollux_Hold_Days < 6 } } }
				}
			}
			fre_pollux_resolve_preserved = yes
		}
		else = { fre_pollux_resolve_superseded = yes }
	}
}

fre_pollux_orphan_check = {
	if = {
		limit = {
			has_country_flag = FRE_Pollux_Evacuation_Active
			OR = { NOT = { country_exists = VIN }  NOT = { country_exists = TAI } }
		}
		fre_pollux_resolve_superseded = yes
	}
}

fre_pollux_force_superseded_cleanup = {
	if = {
		limit = { has_country_flag = FRE_Pollux_Evacuation_Active }
		set_country_flag = FRE_Operation_Pollux_Result_Superseded
	}
	fre_pollux_cleanup = yes
}

	response_package_validation_log = yes
	clr_country_flag = FRE_Pollux_Evacuation_Active
	clr_country_flag = FRE_Mission_Operation_Pollux_Activated
	clr_country_flag = FRE_Pollux_Posture_Historical
	clr_country_flag = FRE_Pollux_Posture_Escort
	clr_country_flag = FRE_Pollux_Posture_Airlift

	set_variable = { FRE_Pollux_Hold_Days = 0 }

}
