# Adds one generic category's operations for generic_operations_selected_target.

# the ON click and the OFF refresh build the same list.

# Runs from the slot mission's timeout_effect, right after the closeout. The
# closeout sets CWIC_espionage_aar_pending, so a stray timeout cannot repeat it.
# Member status: 0 returned, 1 wounded, 2 killed, 3 captured. Captured members
# keep their record (the closeout no longer wipes them) until exchanged or
# disavowed; the launch gate keeps a squad with a missing member at home.
CWIC_espionage_after_action = {
	if = {
		limit = { check_variable = { CWIC_espionage_aar_pending = 1 } }
		set_variable = { CWIC_espionage_aar_pending = 0 }
		set_temp_variable = { CWIC_esp_page = selected_squad_slot_1_Current_Page_Ref }
		if = {
			limit = { var:history_target = { exists = yes } }
			var:history_target = { save_global_event_target_as = CWIC_esp_mia_target }
		}
		set_variable = { CWIC_esp_aar_1 = 0 }
		if = {
			limit = { check_variable = { paramilitary_operative_squad_leader_name_type^CWIC_esp_page = -1 } }
			set_variable = { CWIC_esp_aar_1 = 2 }
		}
		else_if = {
			limit = { check_variable = { operative_1_mia^CWIC_esp_page = 1 } }
			set_variable = { CWIC_esp_aar_1 = 3 }
			set_variable = { CWIC_esp_mia_target_1 = history_target }
			set_variable = { CWIC_esp_mia_page_1 = CWIC_esp_page }
			set_variable = { CWIC_esp_mia_pending_1 = 1 }
			country_event = { id = cwic_esp.1 }
		}
		else_if = {
			limit = { check_variable = { operative_1_wia^CWIC_esp_page > 0 } }
			set_variable = { CWIC_esp_aar_1 = 1 }
		}
		set_variable = { CWIC_esp_aar_2 = 0 }
		if = {
			limit = { check_variable = { paramilitary_operative_squad_unit_2_name_type^CWIC_esp_page = -1 } }

			set_variable = { CWIC_esp_aar_3 = 1 }
		}
		set_variable = { CWIC_esp_aar_4 = 0 }

			set_variable = { CWIC_esp_aar_5 = 2 }

		# political capital and puts the target's counter-intelligence on alert.
		set_variable = { CWIC_esp_aar_consequence = 0 }

			limit = {
				check_variable = { slot_one_successful = 0 }

				limit = { check_variable = { slot_one_detected = 1 } }

			}
			clamp_variable = { var = network_strength^history_target min = 0 max = 1 }
		}
		CWIC_espionage_mark_gui_dirty = yes
	}
}

# Target counter-intelligence: 1.5% detection per agency upgrade (up to 25%),
# plus 10% while a CI alert from an earlier detected failure is live. It is
# computed on the GUI path, before the probability maths, so previews and the
# launch read the same CWIC_espionage_ci_penalty.
CWIC_espionage_update_counter_intelligence = {
	set_variable = { CWIC_espionage_ci_penalty = 0 }
	# Operation slots: one, plus one with at least two squads, plus one with four agency upgrades.
	set_variable = { CWIC_espionage_slot_cap = 1 }
	if = {
		limit = { check_variable = { num_espionage_paramilitary_squads > 1 } }
		add_to_variable = { CWIC_espionage_slot_cap = 1 }
	}
	if = {
		limit = { agency_upgrade_number > 3 }
		add_to_variable = { CWIC_espionage_slot_cap = 1 }
	}
	if = {
		limit = { var:selected_operation_index_target = { exists = yes } }
		var:selected_operation_index_target = {
			if = { limit = { agency_upgrade_number > 16 } set_variable = { ROOT.CWIC_espionage_ci_penalty = 0.25 } }
			else_if = { limit = { agency_upgrade_number > 15 } set_variable = { ROOT.CWIC_espionage_ci_penalty = 0.225 } }
			else_if = { limit = { agency_upgrade_number > 14 } set_variable = { ROOT.CWIC_espionage_ci_penalty = 0.21 } }
			else_if = { limit = { agency_upgrade_number > 13 } set_variable = { ROOT.CWIC_espionage_ci_penalty = 0.195 } }
			else_if = { limit = { agency_upgrade_number > 12 } set_variable = { ROOT.CWIC_espionage_ci_penalty = 0.18 } }
			else_if = { limit = { agency_upgrade_number > 11 } set_variable = { ROOT.CWIC_espionage_ci_penalty = 0.165 } }
			else_if = { limit = { agency_upgrade_number > 10 } set_variable = { ROOT.CWIC_espionage_ci_penalty = 0.15 } }
			else_if = { limit = { agency_upgrade_number > 9 } set_variable = { ROOT.CWIC_espionage_ci_penalty = 0.135 } }
			else_if = { limit = { agency_upgrade_number > 8 } set_variable = { ROOT.CWIC_espionage_ci_penalty = 0.12 } }
			else_if = { limit = { agency_upgrade_number > 7 } set_variable = { ROOT.CWIC_espionage_ci_penalty = 0.105 } }
			else_if = { limit = { agency_upgrade_number > 6 } set_variable = { ROOT.CWIC_espionage_ci_penalty = 0.09 } }
			else_if = { limit = { agency_upgrade_number > 5 } set_variable = { ROOT.CWIC_espionage_ci_penalty = 0.075 } }
			else_if = { limit = { agency_upgrade_number > 4 } set_variable = { ROOT.CWIC_espionage_ci_penalty = 0.06 } }
			else_if = { limit = { agency_upgrade_number > 3 } set_variable = { ROOT.CWIC_espionage_ci_penalty = 0.045 } }
			else_if = { limit = { agency_upgrade_number > 2 } set_variable = { ROOT.CWIC_espionage_ci_penalty = 0.03 } }
			else_if = { limit = { agency_upgrade_number > 1 } set_variable = { ROOT.CWIC_espionage_ci_penalty = 0.015 } }
			else_if = { limit = { agency_upgrade_number > 0 } set_variable = { ROOT.CWIC_espionage_ci_penalty = 0.015 } }
			if = {
				limit = { has_country_flag = CWIC_esp_ci_alert_@ROOT }
				add_to_variable = { ROOT.CWIC_espionage_ci_penalty = 0.10 }
			}
			# A counter-espionage embed inside the target's service (token 19) offsets it.
			if = {
				limit = { has_country_flag = CWIC_esp_ci_embed_@ROOT }
				subtract_from_variable = { ROOT.CWIC_espionage_ci_penalty = 0.10 }
			}
		}
	}
}

# Monthly upkeep: $2M per squad (currentMoney is in billions). A month that
# cannot be paid marks the service underfunded (-10% success for 35 days)
# instead of running the treasury into debt.
CWIC_espionage_monthly_upkeep = {
	set_variable = { CWIC_espionage_upkeep = num_espionage_paramilitary_squads }
	multiply_variable = { CWIC_espionage_upkeep = 0.002 }
	if = {
		limit = { check_variable = { currentMoney < CWIC_espionage_upkeep } }
		set_country_flag = { flag = CWIC_espionage_underfunded value = 1 days = 35 }
	}
	else = {
		subtract_from_variable = { currentMoney = CWIC_espionage_upkeep }
	}
}

# Stage 3: trait window cache and the national-service trait slots.

# The trait window used ~230 meta_triggers that rebuilt the selected member's
# array name on every frame. The selection is cached here instead, on every
# preview refresh (every click in the window goes through it).
CWIC_espionage_cache_selected_operative = {
	clear_array = CWIC_esp_sel_traits
	set_variable = { CWIC_esp_sel_level = 0 }
	if = {
		limit = {
			check_variable = { squad_position > 0 }
			check_variable = { squad_position < 6 }
			check_variable = { current_page > 0 }
		}
		meta_effect = {
			text = {
				for_each_loop = {
					array = paramilitary_operative_squad_unit_[SQUAD_POSITION]_skill_trait_[CURRENT_PAGE]
					add_to_array = { CWIC_esp_sel_traits = v }
				}
				set_variable = { CWIC_esp_sel_level = paramilitary_operative_squad_unit_[SQUAD_POSITION]_level^[CURRENT_PAGE] }
			}
			CURRENT_PAGE = "[?current_page]"
			SQUAD_POSITION = "[?squad_position]"
		}
	}
}

# ---------------------------------------------------------------------------
# Stage 4: operation slots 2 and 3.
#
# slot_one_* stay the working registers that launch, closeout and the bulletin
# use. Each launch copies them into esp_sN_* for the slot it took, and that
# slot's mission timeout loads them back before the closeout runs. The slot cap
# is computed on the GUI path (CWIC_espionage_update_counter_intelligence).

CWIC_espionage_pick_free_slot = {
	set_variable = { CWIC_esp_launch_slot = 0 }
	if = {
		limit = {
			NOT = { has_active_mission = Espionage_Operation_Slot_1 }
			check_variable = { esp_s1_active = 0 }
		}
		set_variable = { CWIC_esp_launch_slot = 1 }
	}
	else_if = {
		limit = {
			check_variable = { CWIC_espionage_slot_cap > 1 }
			NOT = { has_active_mission = Espionage_Operation_Slot_2 }
			check_variable = { esp_s2_active = 0 }
		}
		set_variable = { CWIC_esp_launch_slot = 2 }
	}
	else_if = {
		limit = {
			check_variable = { CWIC_espionage_slot_cap > 2 }
			NOT = { has_active_mission = Espionage_Operation_Slot_3 }
			check_variable = { esp_s3_active = 0 }
		}
		set_variable = { CWIC_esp_launch_slot = 3 }
	}
}

# Called by launch_operation_effect in place of a fixed activate_mission.
CWIC_espionage_activate_slot_mission = {
	CWIC_espionage_pick_free_slot = yes
	if = {
		limit = { check_variable = { CWIC_esp_launch_slot = 2 } }
		activate_mission = Espionage_Operation_Slot_2
		CWIC_espionage_store_slot_2 = yes
	}
	else_if = {
		limit = { check_variable = { CWIC_esp_launch_slot = 3 } }
		activate_mission = Espionage_Operation_Slot_3
		CWIC_espionage_store_slot_3 = yes
	}
	else = {
		set_variable = { CWIC_esp_launch_slot = 1 }
		activate_mission = Espionage_Operation_Slot_1
		CWIC_espionage_store_slot_1 = yes
	}
}
