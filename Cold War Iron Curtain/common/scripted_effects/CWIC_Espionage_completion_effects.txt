# Espionage completion, stage 1: shared Black Operations helpers. The window's
# project panes and text selectors all key off show_* flags, so exactly one may
# be set at a time; otherwise another project's labels bleed into the list.

CWIC_espionage_reset_project_views = {
	set_variable = { show_generic_operations = 0 }
	set_variable = { show_bay_of_pigs = 0 }
	set_variable = { show_gladio = 0 }
	set_variable = { show_tibet = 0 }
	set_variable = { show_albania = 0 }
	set_variable = { show_uia = 0 }
	set_variable = { show_fbk = 0 }
	set_variable = { show_indochina = 0 }
	set_variable = { show_espionage_details_menu = 0 }
	set_variable = { generic_operations_selected_target = 0 }
	clear_array = current_operations_potential_toggle
	clear_array = current_operations_potential_toggle_enabled
	clear_array = gladio_toggle_enabled
	clear_array = local_gladio_possible_operations_operation_type
	CWIC_espionage_mark_gui_dirty = yes
}

# Adds one generic category's operations for generic_operations_selected_target.
# Temp CWIC_espionage_toggle_id is the category id stored in the toggle grid, so
# the ON click and the OFF refresh build the same list.
CWIC_espionage_add_generic_category = {
	if = {
		limit = { check_variable = { CWIC_espionage_toggle_id = 1 } }
		var:generic_operations_selected_target = {
			add_assassinate_generals_operations = yes
			add_assassinate_admirals_operations = yes
		}
	}
	else_if = {
		limit = { check_variable = { CWIC_espionage_toggle_id = 2 } }
		var:generic_operations_selected_target = {
			add_nuclear_site_sabotage_operations = yes
			add_sabotage_nuclear_programme_operations = yes
		}
	}
	else_if = {
		limit = { check_variable = { CWIC_espionage_toggle_id = 3 } }
		var:generic_operations_selected_target = {
			add_arms_factory_sabotage_operations = yes
			add_factory_sabotage_operations = yes
			add_airfield_sabotage_operations = yes
			add_dockyard_sabotage_operations = yes
			add_port_sabotage_operations = yes
			add_infrastructure_sabotage_operations = yes
		}
	}
	else_if = {
		limit = { check_variable = { CWIC_espionage_toggle_id = 4 } }
		var:generic_operations_selected_target = {
			add_terrorist_attack_operations = yes
		}
	}
	else_if = {
		limit = { check_variable = { CWIC_espionage_toggle_id = 5 } }
		var:generic_operations_selected_target = {
			add_local_uprising_operations = yes
			add_local_resistance_contacts_operations = yes
			add_arm_local_resistance_operations = yes
		}
	}
	else_if = {
		limit = { check_variable = { CWIC_espionage_toggle_id = 6 } }
		var:generic_operations_selected_target = {
			Make_Embed_into_Insurgency_List = yes
			Make_Into_Counter_Espionage_Embed = yes
		}
	}
	else_if = {
		limit = { check_variable = { CWIC_espionage_toggle_id = 7 } }
		var:generic_operations_selected_target = {
			add_tech_stealing_operations = yes
		}
	}
}
