Start_Total_War_Between_Factions = {
	ROOT = {
		for_each_scope_loop = {
			array = FROM.allies
			declare_war_on = {
				target = ROOT
				type = annex_everything
			}
		}
		for_each_scope_loop = {
			array = allies
			for_each_scope_loop = {
				array = FROM.allies
				declare_war_on = {
					target = PREV
					type = annex_everything
				}
			}
		}
	}
	FROM = {
		for_each_scope_loop = {
			array = ROOT.allies
			declare_war_on = {
				target = FROM
				type = annex_everything
			}
		}
		for_each_scope_loop = {
			array = allies
			for_each_scope_loop = {
				array = ROOT.allies
				declare_war_on = {
					target = PREV
					type = annex_everything
				}
			}
		}
	}
}

WW3_1950_war_exhaustion_change = {
	multiply_temp_variable = { exhaustion_change = -1 }	
	set_temp_variable = { Soviet_War_Exhaustion_speed_building_change = exhaustion_change }
	multiply_temp_variable = { Soviet_War_Exhaustion_speed_building_change = 1.5 }
	add_to_variable = { Soviet_War_Exhaustion_speed_building = Soviet_War_Exhaustion_speed_building_change }
	set_temp_variable = { Soviet_War_Exhaustion_attack_change = exhaustion_change }
	multiply_temp_variable = { Soviet_War_Exhaustion_attack_change = 0.5 }
	add_to_variable = { Soviet_War_Exhaustion_attack = Soviet_War_Exhaustion_attack_change }
	set_temp_variable = { Soviet_War_Exhaustion_defence_change = exhaustion_change }
	multiply_temp_variable = { Soviet_War_Exhaustion_defence_change = 0.5 }
	add_to_variable = { Soviet_War_Exhaustion_defence = Soviet_War_Exhaustion_defence_change }
	set_temp_variable = { Soviet_War_Exhaustion_drift_change = exhaustion_change }
	add_to_variable = { Soviet_War_Exhaustion_drift = Soviet_War_Exhaustion_drift_change }
	
	subtract_from_variable = {
		Soviet_War_Exhaustion_variable = exhaustion_change
	}
	multiply_temp_variable = { exhaustion_change = -1 }	
	subtract_from_variable = {
		Soviet_War_Exhaustion_variable_show = exhaustion_change
		tooltip = remove_war_exhaustion
	}
}
