# ============================================================================
# Indochina QA / testing effects (Geneva Conference, GUIs, routes)
# ----------------------------------------------------------------------------
# Run from the console with:   e <effect_name>
# The effect executes in the scope of the country you are playing, so switch
# tag first (`tag VIE`, `tag FRA`, ...) when a test is tag-specific.
#
# Companion file: IC_Struggle_Test_Effects.txt covers the struggle core
# (phases, faction scores, legacy/decisive endings, struggle GUI toggles).
# This file covers everything built since: the round-based Geneva Conference,
# its GUI and table actions, the interim election chain, and the VIE
# sectarian GUI.
#
# Quick reference:
#   -- Geneva lifecycle --
#   e test_geneva_reset               tear down all conference state (rerunnable)
#   e test_geneva_announce            player-facing "Road to Geneva" entry (6-day delay)
#   e test_geneva_start               convene instantly, seat delegations, open GUI
#   e test_geneva_tick                advance one negotiation round
#   e test_geneva_tick_5              advance five rounds
#   e test_geneva_finish              force the settlement compiler now
#   e test_geneva_auto_resolve        full headless run (wrap-up timer path)
#   -- Geneva GUI / player helpers --
#   e test_geneva_show_gui            open the conference window for your country
#   e test_geneva_hide_gui            close it
#   e test_geneva_seat_me             seat your country as a delegation
#   e test_geneva_buff_me             +500 weight, full patience, actions + PP refill
#   e test_geneva_refresh_actions     reset per-round action count and cooldowns
#   e test_geneva_max_rapprochement   VIE-VIN rapprochement to 120 + pref refresh
#   e test_geneva_lev_all             +100 to every leverage pool
#   -- Geneva routes / outcomes --
#   e test_geneva_settle_otl          force the historical 17th-parallel accords
#   e test_geneva_settle_unified      force a no-partition settlement + election chain
#   e test_geneva_settle_13th         force a 13th-parallel partition
#   e test_geneva_settle_20th         force a 20th-parallel partition
#   e test_geneva_collapse_no_ceasefire   collapse: ceasefire clause voted down
#   e test_geneva_collapse_walkouts   collapse: VIE + VIN storm out
#   e test_geneva_election_now        fire the interim all-Vietnam election arbiter
#   e test_geneva_election_favor_ho   stack the election snapshot for Hanoi
#   e test_geneva_election_favor_saigon   stack it for Saigon
#   e test_geneva_debug               branch-check the conference flags
#   -- Geneva hardening regressions --
#   e test_geneva_runoff              13th wins a split vote -> must resolve to the 17th
#   e test_geneva_runoff_20th         20th wins a split vote -> must resolve to the 17th
#   e test_geneva_fre_pursuit         de-escalation pursuit: defers below the gate, convenes above it
#   e test_geneva_stale_hollow        in-session save with no delegations + stale round
#   e test_geneva_panic               force the Dien Bien Phu collapse path to the table
#   e test_geneva_panic_ballot        Washington's delegation must refuse the vote
#   e test_geneva_campaign_window     ballot focuses shut until the run-up opens
#   e test_geneva_unified_liquidation a no-line settlement must still end the French presence
#   -- VIE sectarian GUI --
#   e test_vie_sect_show_gui          open the sect window (as VIE)
#   e test_vie_sect_hide_gui          close it
#   e test_vie_sect_reset             restore start-of-game sect values
#   e test_vie_sect_ready_integration all loyalty tracks to integration threshold
#   e test_vie_sect_ready_dismantle   all annihilation tracks to dismantle threshold
#   e test_vie_sect_max_pity          guarantee the next attack/seize rolls
#   e test_vie_sect_prompt_attack_success / _attack_failure /
#     _seize_success / _seize_failure   show the conflict result prompt
#   -- Struggle gameplay extras --
#   e test_indochina_diplomacy_phase  jump to tension phase 6 + 300 PP for the
#                                     de-escalation decision layer
#   -- War Outcomes panel --
#   e test_outcomes_show_gui / _hide_gui   toggle the panel for your country
#   e test_outcomes_reset             re-arm all scenario detectors + recorders
#   e test_outcomes_pulse             run the daily world detector once
#   e test_outcomes_nlf_kill          annex NLF (then pulse to record the tier)
#   e test_outcomes_dbp_victory / _fall / _breakout   plant the DBP result flag
#   e test_outcomes_sects_all         integrate all five sects on VIE
#   e test_outcomes_highlands_all     VIN annexes TAI, MEO and NUN
#   e test_outcomes_laos_victory / _stalemate / _failure   plant raid flags
#   -- Negotiation pipeline --
#   e test_geneva_neg_debug           dump the negotiation arrays to game.log
#   e test_geneva_neg_tick            run the daily maturation sweep once
#   e test_geneva_neg_flush           bank every in-flight stake now
# ============================================================================

# Full teardown of the conference so it can be convened again in the same
# session. Also clears Indochina_War_Over so a concluded run does not block
# re-initialization -- territorial changes from a previous ending are NOT
# undone (reload for that). Struggle phase/scores are untouched; use
# test_indochina_reset (IC_Struggle_Test_Effects.txt) for those.
test_geneva_reset = {
	clr_global_flag = geneva_conference_announced
	clr_global_flag = geneva_conference_in_session
	clr_global_flag = geneva_conference_concluded
	clr_global_flag = geneva_unified_settlement
	clr_global_flag = geneva_interim_government_period
	clr_global_flag = geneva_election_resolved
	clr_global_flag = Geneva_Conference_Preparations
	clr_global_flag = Geneva_Conference_Negotiations_Complete
	clr_global_flag = Geneva_Conference
	clr_global_flag = geneva_conf_attendance_autogranted	# otherwise a re-convened conference silently pays no delegation payload
	clr_global_flag = Indochina_Geneva_Pursuit	# a pending de-escalation pursuit would re-announce the conference behind the test
	clr_global_flag = Indochina_Geneva_Panic	# else the collapse path is one-shot across the whole session
	clr_global_flag = geneva_usa_panic_no_ballot
	clr_global_flag = geneva_election_campaign_open
	clr_global_flag = Indochina_War_Over
	set_variable = { global.geneva_round = 0 }
	set_variable = { global.geneva_tick_days = 0 }
	set_variable = { global.geneva_outcome = 0 }
	set_variable = { global.geneva_wrecker = 0 }
	set_variable = { global.geneva_partition_line = 0 }
	set_variable = { global.geneva_elections_term = 0 }
	set_variable = { global.geneva_withdrawal_term = 0 }
	set_variable = { global.geneva_alignment_term = 0 }
	set_variable = { global.geneva_laos_cambodia_term = 0 }
	set_variable = { global.geneva_minorities_term = 0 }
	set_variable = { global.geneva_final_comm_score = 0 }
	set_variable = { global.geneva_final_anticomm_score = 0 }
	clear_array = global.geneva_delegations
	clear_array = global.geneva_clause_current
	clear_array = global.geneva_push
	clear_array = global.geneva_severity
	clear_array = global.geneva_neg_country
	clear_array = global.geneva_neg_clause
	clear_array = global.geneva_neg_option
	clear_array = global.geneva_neg_amount
	clear_array = global.geneva_neg_days
	clear_array = global.geneva_neg_clause_count
	every_country = {
		limit = {
			OR = {
				tag = FRA
				tag = VIE
				tag = VIN
				tag = USA
				tag = SOV
				tag = PRC
				tag = ENG
				tag = LOS
				tag = CAM
				tag = LAO
				tag = PQC
			}
		}
		clr_country_flag = geneva_walked_out
		clr_country_flag = geneva_core_delegation
		set_variable = { View_Geneva_Conference = 0 }
		set_variable = { geneva_grievance = 0 }
		set_variable = { geneva_neg_display_clause = 0 }
		set_variable = { geneva_neg_display_days = 0 }
		clear_array = geneva_neg_mine
	}
	every_country = {	# reseating re-adds it; a mid-session reset must not leave the button behind
		limit = { is_in_array = { array = country_politics_list value = 29 } }
		remove_from_array = { array = country_politics_list value = 29 }
	}
	# Legacy outcome vocabulary from a previous compile.
	if = {
		limit = { country_exists = FRA }
		FRA = {
			clr_country_flag = FRA_Accepted_Ceasefire
			clr_country_flag = FRA_Rejected_Ceasefire
			clr_country_flag = FRA_Accepted_Reunification
			clr_country_flag = FRA_Rejected_Reunification
		}
	}
	if = {
		limit = { country_exists = VIE }
		VIE = {
			clr_country_flag = VIE_Accepted_Ceasefire
			clr_country_flag = VIE_Rejected_Ceasefire
			clr_country_flag = VIE_Accepted_Reunification
			clr_country_flag = VIE_Rejected_Reunification
			clr_country_flag = VIE_Rejected_Reunification_USA_Intervention
		}
	}
	if = {
		limit = { country_exists = VIN }
		VIN = {
			clr_country_flag = VIN_Accepted_Ceasefire
			clr_country_flag = VIN_Rejected_Ceasefire
			clr_country_flag = VIN_Accepted_Reunification
			clr_country_flag = VIN_Rejected_Reunification
		}
	}
	if = {
		limit = { country_exists = USA }
		USA = {
			clr_country_flag = USA_Tampered_With_VIE
			clr_country_flag = USA_geneva_leaned_on_saigon
		}
	}
}

# Shared helper: make sure a live session exists. A concluded (or otherwise
# blocked) conference is torn down and reconvened.
test_geneva_force_session = {
	if = {
		limit = { NOT = { has_global_flag = geneva_conference_in_session } }
		test_geneva_reset = yes
		geneva_conf_initialize = yes
	}
}

# Seat the country you are playing as a delegation (e.g. to test the table
# actions from a tag that is not normally invited).

	if = {
		limit = {
			has_global_flag = geneva_conference_in_session
			NOT = { is_in_array = { global.geneva_delegations = THIS.id } }
		}
		geneva_conf_seat_delegation = yes
	}
}

# Collapse route B: two core belligerents storm out.
test_geneva_collapse_walkouts = {
	test_geneva_force_session = yes
	if = {
		limit = { country_exists = VIE }
		VIE = {
			set_country_flag = geneva_walked_out
			set_variable = { geneva_patience = 0 }
		}
	}
	if = {
		limit = { country_exists = VIN }
		VIN = {
			set_country_flag = geneva_walked_out
			set_variable = { geneva_patience = 0 }
			save_global_event_target_as = Geneva_Wrecker
			set_variable = { global.geneva_wrecker = THIS.id }
		}
	}
	geneva_conf_compile_settlement = yes
}

# Fire the interim all-Vietnam election arbiter (Session.21) now instead of
# waiting out the 150-540 day timer. The arbiter re-checks its own guards,
# so it still stands down if the VIN focus-tree crisis resolved things or
# the war resumed.
test_geneva_election_now = {
	if = {
		limit = {
			country_exists = VIE
			country_exists = VIN
		}
		set_global_flag = geneva_interim_government_period
		clr_global_flag = geneva_election_resolved
		VIN = { country_event = { id = Geneva_Conference_Session.21 days = 1 } }
	}
}

# Clause 1 is the only ordinal clause: options 1/2/3 are the 13th, 17th and 20th
# parallels. A bare plurality let a concentrated bloc take an extreme line, so the
# vote falls back to the 17th whenever the majority sits on the far side of the
# winner. Both tests zero every delegation's clause-1 priority so only the staked
# pushes count and the arithmetic is deterministic.

test_geneva_runoff = {
	test_geneva_force_session = yes
	every_country = {
		limit = { is_in_array = { global.geneva_delegations = THIS.id } }
		set_variable = { geneva_prio^1 = 0 }
	}
	set_variable = { global.geneva_round = 0 }	# keep the tick from tripping the round-10 compile
	set_variable = { global.geneva_clause_current^1 = 2 }
	set_variable = { global.geneva_push^5 = 1000 }	# clause 1, option 1 (13th parallel): the plurality
	set_variable = { global.geneva_push^6 = 600 }	# clause 1, option 2 (17th parallel)
	set_variable = { global.geneva_push^7 = 500 }	# clause 1, option 3 (20th parallel)
	geneva_conf_round_tick = yes
	log = "GENEVA RUNOFF: 13th=1000 vs 17th+20th=1100 -> line=[?global.geneva_clause_current^1|0] (expect 2)"
}
