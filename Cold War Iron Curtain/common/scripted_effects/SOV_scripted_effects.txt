terraforming_or_something_idk = {
	if = {
		limit = {
			OR = {
				has_state_flag = lakes
				has_state_flag = basic
			}
		}
		clr_state_flag = basic
		clr_state_flag = lakes
		set_state_flag = river
		custom_effect_tooltip = State_Farm_150
	}
	else_if = {
		limit = {
			has_state_flag = river
		}
		clr_state_flag = river
		set_state_flag = riverlarge
		custom_effect_tooltip = State_Farm_175
	}
	else_if = {
		limit = {
			has_state_flag = riverlarge
		}
		clr_state_flag = riverlarge
		set_state_flag = freshwatersea
		custom_effect_tooltip = State_Farm_200
	}
	else = {
		add_extra_state_shared_building_slots = 2
	}
}

	every_owned_state = {
		limit = {
			NOT = {
				is_core_of = FIN
				is_core_of = LAT
				is_core_of = EST
				is_core_of = LIT
			}
			check_variable = {
				ReligionChristianityProtestant > 0
			}
		}
		add_to_array = {
			SOV.soviet_german_state = THIS.id
		}
		add_to_variable = {
			SOV.Soviet_German_Pop = ReligionChristianityProtestant
		}
		subtract_from_variable = {
			ReligionChristianityProtestant = ReligionChristianityProtestant
		}
		normalize_religion = yes

	}
	for_each_loop = {
		array = soviet_german_state
		var:soviet_german_state^i = {
			remove_from_array = {
				array = culture
				value = 386 
			}		
		}
	}
}
