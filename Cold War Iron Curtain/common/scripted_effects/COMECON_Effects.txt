COMECON_init_member_state = {
	if = {
		limit = { NOT = { has_variable = COMECON_satisfaction } }
		set_variable = { COMECON_satisfaction = 55 }
		set_variable = { COMECON_dependency = 80 }
		if = {
			limit = { date > 1970.1.1 }
			if = { limit = { tag = POL } set_variable = { COMECON_satisfaction = 28 } }
			else_if = { limit = { tag = HUN } set_variable = { COMECON_satisfaction = 60 } }
			else_if = { limit = { tag = ROM } set_variable = { COMECON_satisfaction = 40 } }
			else_if = { limit = { tag = CZE } set_variable = { COMECON_satisfaction = 45 } }
			else_if = { limit = { tag = DDR } set_variable = { COMECON_satisfaction = 55 } }
			else_if = { limit = { tag = BUL } set_variable = { COMECON_satisfaction = 65 } }
			else_if = { limit = { tag = MON } set_variable = { COMECON_satisfaction = 65 } }
			else_if = { limit = { tag = CUB } set_variable = { COMECON_satisfaction = 60 } }
			else_if = { limit = { tag = VIN } set_variable = { COMECON_satisfaction = 50 } }
		}
	}
}

COMECON_member_drift = {
	COMECON_init_member_state = yes
	set_temp_variable = { cd_sat = ECO_COMECON_RATE }
	add_to_temp_variable = { cd_sat = TRADE_COMECON_RATE }
	add_to_temp_variable = { cd_sat = RES_COMECON_RATE }
	multiply_temp_variable = { cd_sat = 0.1 }
	set_temp_variable = { cd_mil = MIL_COMECON_RATE }
	multiply_temp_variable = { cd_mil = 0.4 }
	subtract_from_temp_variable = { cd_sat = cd_mil }
	if = {
		limit = {
			OR = {
				tag = HUN
				tag = POL
				tag = ROM
				tag = ALB
			}
		}
		subtract_from_temp_variable = { cd_sat = 0.2 }
	}
	if = {
		limit = { tag = HUN }
		subtract_from_temp_variable = { cd_sat = 0.3 }
	}
	if = {
		limit = { tag = CZE }
		subtract_from_temp_variable = { cd_sat = 0.4 }
	}
	if = {
		limit = { tag = DDR }
		subtract_from_temp_variable = { cd_sat = 0.5 }
	}
	if = {
		limit = {
			tag = DDR
			date > 1951.1.1
			date < 1953.7.1
		}
		subtract_from_temp_variable = { cd_sat = 0.6 }
	}
	set_temp_variable = { cd_dep = ECO_COMECON_RATE }
	add_to_temp_variable = { cd_dep = CON_COMECON_RATE }
	add_to_temp_variable = { cd_dep = PRO_COMECON_RATE }
	multiply_temp_variable = { cd_dep = 0.0833333 }
	subtract_from_temp_variable = { cd_dep = 0.3 }
	add_to_variable = { COMECON_satisfaction = cd_sat }
	add_to_variable = { COMECON_dependency = cd_dep }
	if = {
		limit = {
			has_country_flag = COMECON_MIL_CAPPED
			check_variable = { MIL_COMECON_RATE > 1 }
		}
		set_variable = { MIL_COMECON_RATE = 1 }
	}
	clamp_variable = { var = COMECON_satisfaction min = 0 max = 100 }
	clamp_variable = { var = COMECON_dependency min = 0 max = 100 }
}

COMECON_reduce_highest_slider = {
	set_temp_variable = { cr_max = ECO_COMECON_RATE }
	if = { limit = { check_variable = { CON_COMECON_RATE > cr_max } } set_temp_variable = { cr_max = CON_COMECON_RATE } }
	if = { limit = { check_variable = { PRO_COMECON_RATE > cr_max } } set_temp_variable = { cr_max = PRO_COMECON_RATE } }
	if = { limit = { check_variable = { MIL_COMECON_RATE > cr_max } } set_temp_variable = { cr_max = MIL_COMECON_RATE } }
	if = {
		limit = { check_variable = { ECO_COMECON_RATE = cr_max } }
		add_to_variable = { ECO_COMECON_RATE = -1 }
	}
	else_if = {
		limit = { check_variable = { CON_COMECON_RATE = cr_max } }
		add_to_variable = { CON_COMECON_RATE = -1 }
	}
	else_if = {
		limit = { check_variable = { PRO_COMECON_RATE = cr_max } }
		add_to_variable = { PRO_COMECON_RATE = -1 }
	}
	else = {
		add_to_variable = { MIL_COMECON_RATE = -1 }
	}
}

COMECON_monthly_upkeep = {
	set_variable = { COMECON_cost = 0 }
	set_variable = { COMECON_res_total = 0 }
	for_each_scope_loop = {
		array = COMECON_MEMBERSHIP
		COMECON_member_upkeep = yes
	}
	subtract_from_variable = { currentMoney = COMECON_cost }
	set_variable = { COMECON_RES_MALUS = COMECON_res_total }
	multiply_variable = { COMECON_RES_MALUS = -0.0015 }
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = comecon_sov_burden } } }
		add_dynamic_modifier = { modifier = comecon_sov_burden }
	}
	force_update_dynamic_modifier = yes
	if = {
		limit = { check_variable = { currentMoney < 0 } }
		random_scope_in_array = {
			array = COMECON_MEMBERSHIP
			limit = {
				OR = {
					check_variable = { ECO_COMECON_RATE > 0 }
					check_variable = { CON_COMECON_RATE > 0 }
					check_variable = { PRO_COMECON_RATE > 0 }
					check_variable = { MIL_COMECON_RATE > 0 }
				}
			}
			COMECON_reduce_highest_slider = yes
			SOV = {
				if = {
					limit = { is_ai = no }
					country_event = { id = COMECON.400 }
				}
			}
		}
	}
}

							set_country_flag = { flag = COMECON_req_year days = 365 }

	if = {
		limit = { SOV = { is_ai = yes } }
		SOV = { COMECON_ai_resolve_request = yes }
	}
}

COMECON_ai_resolve_request = {
	if = {
		limit = { event_target:COMECON_REQ_MEMBER = { check_variable = { COMECON_req_kind = 1 } } }
		if = {
			limit = { check_variable = { currentMoney > 50 } event_target:COMECON_REQ_MEMBER = { check_variable = { COMECON_satisfaction < 40 } } }
			COMECON_grant_request = yes
		}
		else = { COMECON_refuse_request = yes }
	}
	else_if = {
		limit = { event_target:COMECON_REQ_MEMBER = { check_variable = { COMECON_req_kind = 2 } } }
		if = {
			limit = { check_variable = { currentMoney > 0 } }
			COMECON_grant_request = yes
		}
		else = { COMECON_refuse_request = yes }
	}
	else_if = {
		limit = { event_target:COMECON_REQ_MEMBER = { check_variable = { COMECON_req_kind = 3 } } }
		if = {
			limit = { event_target:COMECON_REQ_MEMBER = { check_variable = { COMECON_satisfaction < 40 } } }
			COMECON_grant_request = yes
		}
		else = { COMECON_refuse_request = yes }
	}
	else = {
		if = {
			limit = {
				event_target:COMECON_REQ_MEMBER = {
					check_variable = { COMECON_satisfaction < 10 }
					check_variable = { COMECON_dependency < 30 }
				}
			}
			COMECON_grant_request = yes
		}
		else = { COMECON_refuse_request = yes }
	}
}

COMECON_grant_request = {
	event_target:COMECON_REQ_MEMBER = {
		if = {
			limit = { check_variable = { COMECON_req_kind = 1 } }
			add_to_variable = { ECO_COMECON_RATE = 1 }
			clamp_variable = { var = ECO_COMECON_RATE min = 0 max = 6 }
		}
		if = {
			limit = { check_variable = { COMECON_req_kind = 2 } }
			add_to_variable = { TRADE_COMECON_RATE = 1 }
			clamp_variable = { var = TRADE_COMECON_RATE min = 0 max = 6 }
		}
		if = {
			limit = { check_variable = { COMECON_req_kind = 3 } }
			add_to_variable = { MIL_COMECON_RATE = -1 }
			clamp_variable = { var = MIL_COMECON_RATE min = 0 max = 6 }
		}
		if = {
			limit = { check_variable = { COMECON_req_kind = 4 } }
			COMECON_member_leave = yes
		}
		if = {
			limit = { check_variable = { COMECON_req_kind < 4 } }
			COMECON_apply_member_modifiers = yes
		}
	}
}

COMECON_refuse_request = {
	event_target:COMECON_REQ_MEMBER = {
		if = {
			limit = { check_variable = { COMECON_req_kind < 3 } }
			add_to_variable = { COMECON_satisfaction = -5 }
		}
		if = {
			limit = { check_variable = { COMECON_req_kind = 3 } }
			add_to_variable = { COMECON_satisfaction = -8 }
		}
		if = {
			limit = { check_variable = { COMECON_req_kind = 4 } }
			add_to_variable = { COMECON_satisfaction = -10 }
			add_stability = -0.05
			if = {
				limit = { check_variable = { COMECON_crisis_id = 0 } }
				set_variable = { COMECON_crisis_id = 9 }
				set_variable = { COMECON_crisis_days = 60 }
			}
		}
		clamp_variable = { var = COMECON_satisfaction min = 0 max = 100 }
	}
}
