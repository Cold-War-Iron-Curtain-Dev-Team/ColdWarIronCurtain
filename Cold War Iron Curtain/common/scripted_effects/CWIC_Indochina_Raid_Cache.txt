# State scope. Call after changing VIN/CAM/LOS cores (including generic coring).
# Rebuilt on startup/save load; only eligible states retain one flag.
cwic_refresh_indochina_raid_state_cache = {
	if = {
		limit = {
			OR = {
				is_core_of = VIN
				is_core_of = CAM
				is_core_of = LOS
			}
		}
		set_state_flag = cwic_indochina_raid_target
	}
	else = {
		clr_state_flag = cwic_indochina_raid_target
	}
}
