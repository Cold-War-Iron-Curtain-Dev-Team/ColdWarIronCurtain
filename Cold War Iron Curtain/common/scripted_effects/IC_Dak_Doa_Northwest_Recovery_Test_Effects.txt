ic_dak_doa_test_inspect = {
	if = {
		limit = { country_exists = NLF }
		NLF = {
			set_temp_variable = { ic_dak_war_ful = 0 }
			if = { limit = { country_exists = FUL has_war_with = FUL } set_temp_variable = { ic_dak_war_ful = 1 } }
			log = "IC_TEST|DAK_DOA|INSPECT|active=[?NLF_Dak_Doa_Days_Elapsed|0]|hold=[?NLF_Dak_Doa_Hold_Days|0]|outcome=[?NLF_Dak_Doa_Last_Outcome|0]|war_ful=[?ic_dak_war_ful|0]"
		}
	}
	if = {
		limit = { country_exists = FRE }
		FRE = { set_temp_variable = { response_validation_package = 7 } response_package_validation_log = yes }
	}
	if = { limit = { country_exists = VIE country_exists = NLF VIE = { has_war_with = NLF } } log = "IC_TEST|DAK_DOA|WIDER_WAR|VIE_NLF=preserved" }
	if = { limit = { country_exists = FRE country_exists = NLF FRE = { has_war_with = NLF } } log = "IC_TEST|DAK_DOA|WIDER_WAR|FRE_NLF=preserved" }
	if = { limit = { country_exists = FRA country_exists = NLF FRA = { has_war_with = NLF } } log = "IC_TEST|DAK_DOA|WIDER_WAR|FRA_NLF=preserved" }
}
