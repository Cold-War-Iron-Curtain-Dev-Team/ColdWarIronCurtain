cwic_update_shipping_cost_effect = {
	if = {
		limit = { any_owned_state = { is_coastal = yes } }
		set_variable = { shipping_cost_effect = 0 }
		if = {
			limit = { has_navy_size = { size > 25 type = convoy } }
			set_variable = { shipping_cost_effect = -0.05 }
		}
		if = {
			limit = { has_navy_size = { size > 75 type = convoy } }
			set_variable = { shipping_cost_effect = -0.10 }
		}
		if = {
			limit = { has_navy_size = { size > 150 type = convoy } }
			set_variable = { shipping_cost_effect = -0.15 }
		}
	}
	else = {
		set_variable = { shipping_cost_effect = 0.25 }
	}
}

cwic_trade_manual_default = {
	if = {
		limit = {
			is_ai = no
			NOT = { has_variable = trade_automation }
		}
		set_variable = { trade_automation = 1 }
	}
}
