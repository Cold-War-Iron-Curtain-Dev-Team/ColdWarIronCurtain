# =====================================================================
# CWIC Super Event System - runtime.
#
# Super events are defined in ONE place: _tools/super_events/super_events.csv.
# Run  python _tools/super_events/build_super_events.py  to regenerate
#   CWIC_Super_Event_Generated_Effects.txt  (the <name>_Play fire effects + CWIC_Open_SE_Window)
#   CWIC_Super_Event_Generated_Loc.txt      (picture / title / option / quote / news-list lookups)
#
# Fire one from any event, focus or decision:   Super_Event_My_Event_Play = yes
#
# Variables (on each player country):
#   CWIC_se_shown    id of the super event currently on screen (0 = none)
#   CWIC_se_current  id of the last super event received (logged even when filtered)
#   CWIC_se_log      last 10 ids, newest first (Latest News window)
# =====================================================================

# Called by every generated fire effect, inside every_country (player countries).
# Temp CWIC_se_fire = id, ROOT = the country that fired it. Sets temp CWIC_se_opened = 1 when the popup opens.
CWIC_Show_Super_Event = {
	set_variable = { CWIC_se_current = CWIC_se_fire }
	CWIC_Log_Current_SE = yes
	set_temp_variable = { CWIC_se_opened = 0 }
	if = {
		limit = { CWIC_should_show_super_event = yes }
		set_variable = { CWIC_se_shown = CWIC_se_fire }
		CWIC_Open_SE_Window = yes
		set_temp_variable = { CWIC_se_opened = 1 }
	}
}

# Append current event id to the news log (newest first), keeping only the last 10.
# Skips if this event is already the newest entry.
CWIC_Log_Current_SE = {
	if = {
		limit = {
			check_variable = { CWIC_se_current > 0 }
			OR = {
				check_variable = { CWIC_se_log^num = 0 }
				NOT = { check_variable = { CWIC_se_log^0 = CWIC_se_current } }
			}
		}
		add_to_array = { array = CWIC_se_log value = CWIC_se_current index = 0 }
		set_variable = { CWIC_news_frame = 2 }
		if = {
			limit = { check_variable = { CWIC_se_log^num > 10 } }
			remove_from_array = { array = CWIC_se_log index = 10 }
		}
	}
}

# Close button of every super event window.
CWIC_Clear_All_SE_Flags = {
	clr_country_flag = CWIC_Super_Event_Visible
	clr_country_flag = KOR_Big_Event_Flag
	clr_country_flag = JAP_Big_Event_Flag
	set_variable = { CWIC_se_shown = 0 }
}

# Reopen a past super event from the Latest News list (bypasses filters; the player clicked it).
CWIC_Reopen_SE_By_Id = {
	set_variable = { CWIC_se_shown = CWIC_se_reopen }
	CWIC_Open_SE_Window = yes
}

# Sets the checkbox-button frames for the opening-screen popup settings tab.
# frame 1 = empty, frame 2 = ticked. Only the active mode is ticked (radio).
CWIC_Refresh_Mode_Frames = {
	set_variable = { CWIC_cb_all = 1 }
	set_variable = { CWIC_cb_own = 1 }
	set_variable = { CWIC_cb_majors = 1 }
	set_variable = { CWIC_cb_off = 1 }
	if = { limit = { has_country_flag = CWIC_se_pref_off } set_variable = { CWIC_cb_off = 2 } }
	else_if = { limit = { has_country_flag = CWIC_se_pref_majors } set_variable = { CWIC_cb_majors = 2 } }
	else_if = { limit = { has_country_flag = CWIC_se_pref_own_majors } set_variable = { CWIC_cb_own = 2 } }
	else = { set_variable = { CWIC_cb_all = 2 } }
}
