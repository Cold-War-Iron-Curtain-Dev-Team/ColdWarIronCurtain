cwic_inflation_monthly_pulse = {
	set_variable = { CWIC_cpi_index_last = CWIC_cpi_index }
	# ============================================================
	# 0) Defaults / init safety
	# ============================================================
	# price level index starts at 1.0
	if = {
		limit = { check_variable = { CWIC_cpi_index < 0.01 } }
		set_variable = { CWIC_cpi_index = 1 }
	}
	# stored monthly inflation starts at 0.2% if unset/zeroed
	if = {
		limit = { check_variable = { CWIC_inflation_mom = 0 } }
		set_variable = { CWIC_inflation_mom = 0.002 }
	}
	# Monetary policy multiplier default
	if = {
		limit = { NOT = { has_variable = CWIC_monetary_policy_mult } }
		set_variable = { CWIC_monetary_policy_mult = 1 }
	}

	# ============================================================
	# 1) Clear temps
	# ============================================================
	clear_variable = gdp_tmp
	clear_variable = cg_pressure
	clear_variable = sp
	clear_variable = def_ratio
	clear_variable = fp
	clear_variable = ds_ratio
	clear_variable = dp
	clear_variable = tight
	clear_variable = td
	clear_variable = cred
	clear_variable = expect_mult
	clear_variable = raw_infl
	clear_variable = prev_infl
	clear_variable = smooth_infl
	clear_variable = tmp

	# ============================================================
	# 2) Tuning knobs
	#    Keep these here so you can rebalance later
	# ============================================================
	set_temp_variable = { k_sp = 0.020 }   # shortage pressure weight (dominant)
	set_temp_variable = { k_fp = 0.030 }   # deficit pressure weight
	set_temp_variable = { k_dp = 0.020 }   # debt-service pressure weight
	set_temp_variable = { k_td = 0.025 }   # interest tightening drag
	set_temp_variable = { k_exp = 0.75 }   # expectation multiplier strength

	set_temp_variable = { base_infl = 0.001 }   # baseline inflation (0.1% per month)

	set_temp_variable = { def_cap = 0.20 }  # deficit ratio clamp
	set_temp_variable = { ds_cap  = 0.20 }  # debt service clamp
	set_temp_variable = { t_cap   = 0.50 }  # tightness clamp (Interest_Rate - 1)

	set_temp_variable = { w_new = 0.25 }    # smoothing: how much new info affects inflation each month

	# ============================================================
	# 3) GDP guard
	# ============================================================
	set_temp_variable = { gdp_tmp = grossdomesticproduct }
	if = { limit = { check_variable = { gdp_tmp < 0.1 } } set_temp_variable = { gdp_tmp = 0.1 } }

	# ============================================================
	# 4) Shortage pressure (SP)
	#    Reconstruct consumer goods pressure from CWIC_DM_CG_STABILITY_FACTOR
	#    since it equals (master_scalar * -0.18) in your CG script
	# ============================================================
	set_temp_variable = { cg_pressure = CWIC_DM_CG_STABILITY_FACTOR }
	divide_temp_variable = { cg_pressure = -0.18 } # -> master_scalar (0..1)

	if = { limit = { check_variable = { cg_pressure < 0 } } set_temp_variable = { cg_pressure = 0 } }
	if = { limit = { check_variable = { cg_pressure > 1 } } set_temp_variable = { cg_pressure = 1 } }

	# Nonlinear: square it (p = 2)
	set_temp_variable = { sp = cg_pressure }
	multiply_temp_variable = { sp = cg_pressure }
	multiply_temp_variable = { sp = k_sp }

	# ============================================================
	# 5) Fiscal pressure (FP) from pulse_deficit / GDP
	# ============================================================
	set_temp_variable = { def_ratio = 0 }
	if = {
		limit = { check_variable = { pulse_deficit > 0 } }
		set_temp_variable = { def_ratio = pulse_deficit }
		divide_temp_variable = { def_ratio = gdp_tmp }
		if = { limit = { check_variable = { def_ratio > def_cap } } set_temp_variable = { def_ratio = def_cap } }
	}

	set_temp_variable = { fp = def_ratio }
	multiply_temp_variable = { fp = k_fp }

	# ============================================================
	# 6) Debt stress pressure (DP) from debt service / GDP
	# ============================================================
	set_temp_variable = { ds_ratio = 0 }
	if = {
		limit = { check_variable = { currentDebtPayments > 0 } }
		set_temp_variable = { ds_ratio = currentDebtPayments }
		divide_temp_variable = { ds_ratio = gdp_tmp }
		if = { limit = { check_variable = { ds_ratio > ds_cap } } set_temp_variable = { ds_ratio = ds_cap } }
	}

	set_temp_variable = { dp = ds_ratio }
	multiply_temp_variable = { dp = k_dp }

	# ============================================================
	# 7) Policy drag (TD) from interest tightening
	# ============================================================
	set_temp_variable = { tight = Interest_Rate }
	subtract_from_temp_variable = { tight = 1 } # Interest_Rate - 1

	if = { limit = { check_variable = { tight < 0 } } set_temp_variable = { tight = 0 } }
	if = { limit = { check_variable = { tight > t_cap } } set_temp_variable = { tight = t_cap } }

	set_temp_variable = { td = tight }
	multiply_temp_variable = { td = k_td }

	# ============================================================
	# 7b) Monetary transmission channel (second-order effect)
	#     Higher rates suppress inflation further.
	#     Lower-than-neutral rates mildly stimulate.
	# ============================================================

	set_temp_variable = { rate_trans = Interest_Rate }
	subtract_from_temp_variable = { rate_trans = 1 }     # deviation from neutral

	# clamp deviation (avoid extreme spiral)
	if = { limit = { check_variable = { rate_trans > 0.50 } } set_temp_variable = { rate_trans = 0.50 } }
	if = { limit = { check_variable = { rate_trans < -0.50 } } set_temp_variable = { rate_trans = -0.50 } }

	# transmission strength tuning knob
	set_temp_variable = { k_trans = 0.50 }
	multiply_temp_variable = { rate_trans = k_trans }

	# ============================================================
	# 8) Expectations multiplier from Credit_Rating
	#    credibility = Credit_Rating / 10 clamped to 0.1..1.0
	#    expect_mult = 1 + k_exp * (1 - credibility)
	# ============================================================
	set_temp_variable = { cred = Credit_Rating }
	divide_temp_variable = { cred = 10 }

	if = { limit = { check_variable = { cred < 0.1 } } set_temp_variable = { cred = 0.1 } }
	if = { limit = { check_variable = { cred > 1 } } set_temp_variable = { cred = 1 } }

	set_temp_variable = { expect_mult = 1 }
	set_temp_variable = { tmp = 1 }
	subtract_from_temp_variable = { tmp = cred }        # (1 - cred)
	multiply_temp_variable = { tmp = k_exp }            # k_exp*(1-cred)
	add_to_temp_variable = { expect_mult = tmp }        # 1 + k_exp*(1-cred)

	# ============================================================
	# 9) Raw inflation this month
	#    raw = base + expect_mult*(SP + FP + DP) - TD
	# ============================================================
	set_temp_variable = { raw_infl = sp }
	add_to_temp_variable = { raw_infl = fp }
	add_to_temp_variable = { raw_infl = dp }

	multiply_temp_variable = { raw_infl = expect_mult }
	add_to_temp_variable = { raw_infl = base_infl }
	subtract_from_temp_variable = { raw_infl = td }
	subtract_from_temp_variable = { raw_infl = rate_trans }


   # Tag bias (can be negative for deflation-prone economies)
    add_to_temp_variable = { raw_infl = CWIC_inflation_bias_mom }
 
	# clamp monthly inflation (gameplay safety)
	if = { limit = { check_variable = { raw_infl < -0.02 } } set_temp_variable = { raw_infl = -0.02 } } # -2% / month floor
	if = { limit = { check_variable = { raw_infl > 0.20 } } set_temp_variable = { raw_infl = 0.20 } }   # 20% / month cap

	# ============================================================
	# 10) Smooth inflation (stickiness)
	#     smooth = prev*(1-w_new) + raw*w_new
	# ============================================================
	set_temp_variable = { prev_infl = CWIC_inflation_mom }

	set_temp_variable = { smooth_infl = prev_infl }
	set_temp_variable = { tmp = 1 }
	subtract_from_temp_variable = { tmp = w_new }           # (1 - w_new)
	multiply_temp_variable = { smooth_infl = tmp }          # prev*(1-w_new)

	set_temp_variable = { tmp = raw_infl }
	multiply_temp_variable = { tmp = w_new }                # raw*w_new
	add_to_temp_variable = { smooth_infl = tmp }

	set_variable = { CWIC_inflation_mom = smooth_infl }

	# ============================================================
	# 11) Update price level
	#     price_level *= (1 + inflation_mom)
	# ============================================================
	set_temp_variable = { tmp = 1 }
	add_to_temp_variable = { tmp = CWIC_inflation_mom }
	multiply_variable = { CWIC_cpi_index = tmp }

	# ============================================================
	# 12) Convenience display variables (optional)
	# ============================================================
	set_variable = { CWIC_inflation_yoy_display = CWIC_inflation_mom }
	multiply_variable = { CWIC_inflation_yoy_display = 12 }     # rough YoY
	multiply_variable = { CWIC_inflation_yoy_display = 100 }    # percent for GUI/loc
}

cwic_inflation_tier_monthly_assign = {

	# ------------------------------------------------------------
	# 0) Compute YoY inflation (rough)
	# ------------------------------------------------------------
	clear_variable = infl_yoy
	set_temp_variable = { infl_yoy = CWIC_inflation_mom }

	# optional display percent (for GUI/loc later)
	set_variable = { CWIC_infl_yoy = infl_yoy }

	# ------------------------------------------------------------
	# 1) Determine new tier id from YoY inflation
	# ------------------------------------------------------------
	clear_variable = new_tier
	set_temp_variable = { new_tier = 0 }

	# Deep deflation (< -3%)
	if = {
		limit = { check_variable = { infl_yoy < -0.03 } }
		set_temp_variable = { new_tier = 3 }
	}
	# Strong deflation (-3% to -2%)
	else_if = {
		limit = { check_variable = { infl_yoy < -0.02 } }
		set_temp_variable = { new_tier = 2 }
	}
	# Mild deflation (-2% to 0%)
	else_if = {
		limit = { check_variable = { infl_yoy < 0 } }
		set_temp_variable = { new_tier = 1 }
	}

	# Severe hyperinflation (> 200%)
	else_if = {
		limit = { check_variable = { infl_yoy > 2.00 } }
		set_temp_variable = { new_tier = 20 }
	}
	# Hyperinflation (> 100%)
	else_if = {
		limit = { check_variable = { infl_yoy > 1.00 } }
		set_temp_variable = { new_tier = 19 }
	}
	# Crisis inflation (> 50%)
	else_if = {
		limit = { check_variable = { infl_yoy > 0.50 } }
		set_temp_variable = { new_tier = 18 }
	}
	# Serious inflation (> 25%)
	else_if = {
		limit = { check_variable = { infl_yoy > 0.25 } }
		set_temp_variable = { new_tier = 17 }
	}
	# High inflation (> 15%)
	else_if = {
		limit = { check_variable = { infl_yoy > 0.15 } }
		set_temp_variable = { new_tier = 16 }
	}
	# Moderate inflation (> 10%)
	else_if = {
		limit = { check_variable = { infl_yoy > 0.10 } }
		set_temp_variable = { new_tier = 15 }
	}
	# Noticeable inflation (> 8%)
	else_if = {
		limit = { check_variable = { infl_yoy > 0.08 } }
		set_temp_variable = { new_tier = 14 }
	}
	# Mild inflation (> 4%)
	else_if = {
		limit = { check_variable = { infl_yoy > 0.04 } }
		set_temp_variable = { new_tier = 13 }
	}
	# Stable inflation (> 2%)
	else_if = {
		limit = { check_variable = { infl_yoy > 0.02 } }
		set_temp_variable = { new_tier = 12 }
	}
	# Very stable (0–2%)
	else = {
		set_temp_variable = { new_tier = 11 }
	}

	# ------------------------------------------------------------
	# 2) Hysteresis: require 2 consecutive months to change tier
	# ------------------------------------------------------------
	# init
	if = { limit = { check_variable = { CWIC_infl_tier_id = 0 } } set_variable = { CWIC_infl_tier_id = new_tier } }
	if = { limit = { check_variable = { CWIC_infl_tier_pending = 0 } } set_variable = { CWIC_infl_tier_pending = new_tier } }
	if = { limit = { check_variable = { CWIC_infl_tier_pending_count = 0 } } set_variable = { CWIC_infl_tier_pending_count = 0 } }

	# if new_tier equals current, reset pending
	if = {
		limit = { check_variable = { new_tier = CWIC_infl_tier_id } }
		set_variable = { CWIC_infl_tier_pending = new_tier }
		set_variable = { CWIC_infl_tier_pending_count = 0 }
	}
	else = {
		# if pending matches new_tier, increment count, else reset pending
		if = {
			limit = { check_variable = { new_tier = CWIC_infl_tier_pending } }
			add_to_variable = { CWIC_infl_tier_pending_count = 1 }
		}
		else = {
			set_variable = { CWIC_infl_tier_pending = new_tier }
			set_variable = { CWIC_infl_tier_pending_count = 1 }
		}

		# commit after 2 consecutive months
		if = {
			limit = { check_variable = { CWIC_infl_tier_pending_count >= 2 } }
			set_variable = { CWIC_infl_tier_id = new_tier }
			set_variable = { CWIC_infl_tier_pending_count = 0 }
		}
	}

	# ------------------------------------------------------------
	# 3) Apply tier idea (remove all, then add the one for CWIC_infl_tier_id)
	# ------------------------------------------------------------
	remove_ideas = CWIC_infl_env_deflation_1
	remove_ideas = CWIC_infl_env_deflation_2
	remove_ideas = CWIC_infl_env_deflation_3
	remove_ideas = CWIC_infl_env_very_stable_1
	remove_ideas = CWIC_infl_env_stable_2
	remove_ideas = CWIC_infl_env_mild_4
	remove_ideas = CWIC_infl_env_noticeable_8
	remove_ideas = CWIC_infl_env_moderate_10
	remove_ideas = CWIC_infl_env_high_15
	remove_ideas = CWIC_infl_env_serious_25
	remove_ideas = CWIC_infl_env_crisis_50
	remove_ideas = CWIC_infl_env_hyper_light_100
	remove_ideas = CWIC_infl_env_hyper_hard_200

	# Add the active one
	if = { limit = { check_variable = { CWIC_infl_tier_id = 3 } } add_ideas = CWIC_infl_env_deflation_3 }
	else_if = { limit = { check_variable = { CWIC_infl_tier_id = 2 } } add_ideas = CWIC_infl_env_deflation_2 }
	else_if = { limit = { check_variable = { CWIC_infl_tier_id = 1 } } add_ideas = CWIC_infl_env_deflation_1 }

	else_if = { limit = { check_variable = { CWIC_infl_tier_id = 20 } } add_ideas = CWIC_infl_env_hyper_hard_200 }
	else_if = { limit = { check_variable = { CWIC_infl_tier_id = 19 } } add_ideas = CWIC_infl_env_hyper_light_100 }
	else_if = { limit = { check_variable = { CWIC_infl_tier_id = 18 } } add_ideas = CWIC_infl_env_crisis_50 }
	else_if = { limit = { check_variable = { CWIC_infl_tier_id = 17 } } add_ideas = CWIC_infl_env_serious_25 }
	else_if = { limit = { check_variable = { CWIC_infl_tier_id = 16 } } add_ideas = CWIC_infl_env_high_15 }
	else_if = { limit = { check_variable = { CWIC_infl_tier_id = 15 } } add_ideas = CWIC_infl_env_moderate_10 }
	else_if = { limit = { check_variable = { CWIC_infl_tier_id = 14 } } add_ideas = CWIC_infl_env_noticeable_8 }
	else_if = { limit = { check_variable = { CWIC_infl_tier_id = 13 } } add_ideas = CWIC_infl_env_mild_4 }
	else_if = { limit = { check_variable = { CWIC_infl_tier_id = 12 } } add_ideas = CWIC_infl_env_stable_2 }
	else = { add_ideas = CWIC_infl_env_very_stable_1 }
}
cwic_capital_controls_monthly_effects = {

	# --------------------------------------------
	# Init safe defaults if missing (optional)
	# --------------------------------------------
	if = { limit = { NOT = { has_variable = CWIC_black_market_pressure } } set_variable = { CWIC_black_market_pressure = 0 } }
	if = { limit = { NOT = { has_variable = CWIC_productivity_mult } } set_variable = { CWIC_productivity_mult = 0 } }

	# --------------------------------------------
	# Ongoing inflation suppression + distortion costs
	# --------------------------------------------

	# Light capital controls
	if = {
		limit = { has_country_flag = CWIC_capital_controls_light }

		# modest ongoing inflation suppression
		subtract_from_variable = { CWIC_inflation_mom = 0.0005 }

		# distortions accumulate
		add_to_variable = { CWIC_black_market_pressure = 0.01 }
		add_to_variable = { CWIC_productivity_mult = -0.002 }
	}

	# Heavy capital controls
	if = {
		limit = { has_country_flag = CWIC_capital_controls_heavy }

		# stronger ongoing inflation suppression
		subtract_from_variable = { CWIC_inflation_mom = 0.0015 }

		# distortions accumulate faster
		add_to_variable = { CWIC_black_market_pressure = 0.03 }
		add_to_variable = { CWIC_productivity_mult = -0.006 }
	}

	# Price controls (essentials)
	if = {
		limit = { has_country_flag = CWIC_price_controls_basic }

		# suppress inflation mechanically short-term
		subtract_from_variable = { CWIC_inflation_mom = 0.0007 }

		# black market grows (shortages / informal trade)
		add_to_variable = { CWIC_black_market_pressure = 0.02 }
		add_to_variable = { CWIC_productivity_mult = -0.003 }
	}

	# Dual exchange rate system
	if = {
		limit = { has_country_flag = CWIC_fx_controls_dual }

		# suppress inflation via forced FX allocation
		subtract_from_variable = { CWIC_inflation_mom = 0.0010 }

		# very distortionary
		add_to_variable = { CWIC_black_market_pressure = 0.03 }
		add_to_variable = { CWIC_productivity_mult = -0.008 }
	}

	# --------------------------------------------
	# Natural recovery / decay if no controls
	# --------------------------------------------
	if = {
		limit = {
			NOT = { has_country_flag = CWIC_capital_controls_light }
			NOT = { has_country_flag = CWIC_capital_controls_heavy }
			NOT = { has_country_flag = CWIC_price_controls_basic }
			NOT = { has_country_flag = CWIC_fx_controls_dual }
		}

		# black market pressure slowly decays
		subtract_from_variable = { CWIC_black_market_pressure = 0.01 }

		# productivity drag slowly recovers toward 0
		add_to_variable = { CWIC_productivity_mult = 0.003 }
	}

	# --------------------------------------------
	# Clamp safety (avoid negative black market or positive runaway productivity mult)
	# --------------------------------------------
	if = { limit = { check_variable = { CWIC_black_market_pressure < 0 } } set_variable = { CWIC_black_market_pressure = 0 } }
	if = { limit = { check_variable = { CWIC_black_market_pressure > 1 } } set_variable = { CWIC_black_market_pressure = 1 } }

	if = { limit = { check_variable = { CWIC_productivity_mult > 0 } } set_variable = { CWIC_productivity_mult = 0 } }
	if = { limit = { check_variable = { CWIC_productivity_mult < -0.25 } } set_variable = { CWIC_productivity_mult = -0.25 } }

	# --------------------------------------------
	# Black market feedback loop
	# High black market pressure makes inflation harder to control
	# and slows productivity recovery even if controls are lifted.
	# --------------------------------------------

	# Mild black market feedback
	if = {
		limit = { check_variable = { CWIC_black_market_pressure > 0.25 } }
		add_to_variable = { CWIC_inflation_mom = 0.0003 }
		add_to_variable = { CWIC_productivity_mult = -0.002 }
	}

	# Strong feedback
	if = {
		limit = { check_variable = { CWIC_black_market_pressure > 0.50 } }
		add_to_variable = { CWIC_inflation_mom = 0.0007 }
		add_to_variable = { CWIC_productivity_mult = -0.004 }
	}

	# Severe feedback
	if = {
		limit = { check_variable = { CWIC_black_market_pressure > 0.75 } }
		add_to_variable = { CWIC_inflation_mom = 0.0012 }
		add_to_variable = { CWIC_productivity_mult = -0.006 }
	}

}

cwic_black_market_monthly_update = {

	# --------------------------------------------
	# Safe init
	# --------------------------------------------
	if = { limit = { NOT = { has_variable = CWIC_black_market_pressure } } set_variable = { CWIC_black_market_pressure = 0 } }
	if = { limit = { NOT = { has_variable = CWIC_black_market_tier_id } } set_variable = { CWIC_black_market_tier_id = 0 } }

	# Optional inputs from other systems (if absent, defaults)
	# CWIC_policing_eff should be 0..1 where 1 = strong policing capacity
	if = { limit = { NOT = { has_variable = CWIC_policing_eff } } set_variable = { CWIC_policing_eff = 0.5 } }

	# Optional shortage driver (0..1) if you have it
	if = { limit = { NOT = { has_variable = CWIC_shortage_pressure } } set_variable = { CWIC_shortage_pressure = 0 } }

	# --------------------------------------------
	# Compute monthly delta = base + controls + shortages - policing - no-controls decay
	# --------------------------------------------
	set_temp_variable = { bm_delta = 0.002 } # base drift (tunable)

	# Controls increase pressure
	if = { limit = { has_country_flag = CWIC_capital_controls_light } add_to_temp_variable = { bm_delta = 0.010 } }
	if = { limit = { has_country_flag = CWIC_capital_controls_heavy } add_to_temp_variable = { bm_delta = 0.030 } }
	if = { limit = { has_country_flag = CWIC_price_controls_basic } add_to_temp_variable = { bm_delta = 0.020 } }
	if = { limit = { has_country_flag = CWIC_fx_controls_dual } add_to_temp_variable = { bm_delta = 0.030 } }

	# Shortages increase pressure (scaled by CWIC_shortage_pressure 0..1)
	# Adds up to +0.020 at max shortage (tunable)
	set_temp_variable = { bm_short = CWIC_shortage_pressure }
	multiply_temp_variable = { bm_short = 0.020 }
	add_to_temp_variable = { bm_delta = bm_short }

	# High inflation tends to accelerate informalization (optional, lightweight)
	if = { limit = { has_variable = CWIC_inflation_mom } 
		if = { limit = { check_variable = { CWIC_inflation_mom > 0.010 } } add_to_temp_variable = { bm_delta = 0.005 } }  # ~12%+
		if = { limit = { check_variable = { CWIC_inflation_mom > 0.020 } } add_to_temp_variable = { bm_delta = 0.010 } }  # ~24%+
	}

	# Policing reduces pressure: subtract (policing_eff * 0.020)
	set_temp_variable = { bm_pol = CWIC_policing_eff }
	multiply_temp_variable = { bm_pol = 0.020 }
	subtract_from_temp_variable = { bm_delta = bm_pol }

	# If no controls are active, apply extra decay (market normalizes)
	if = {
		limit = {
			NOT = { has_country_flag = CWIC_capital_controls_light }
			NOT = { has_country_flag = CWIC_capital_controls_heavy }
			NOT = { has_country_flag = CWIC_price_controls_basic }
			NOT = { has_country_flag = CWIC_fx_controls_dual }
		}
		subtract_from_temp_variable = { bm_delta = 0.010 } # extra decay (tunable)
	}

	# Apply delta
	add_to_variable = { CWIC_black_market_pressure = bm_delta }

	# --------------------------------------------
	# Clamp 0..1
	# --------------------------------------------
	if = { limit = { check_variable = { CWIC_black_market_pressure < 0 } } set_variable = { CWIC_black_market_pressure = 0 } }
	if = { limit = { check_variable = { CWIC_black_market_pressure > 1 } } set_variable = { CWIC_black_market_pressure = 1 } }
}

		subtract_from_variable = { CWIC_inflation_mom = 0.0010 }
		add_to_variable = { CWIC_productivity_mult = -0.002 }
	}

	if = {
		limit = { has_country_flag = CWIC_currency_reform_recent }
		add_to_variable = { CWIC_inflation_mom = 0.0005 }
	}

	# Clear reform recent flag after 6 months
	if = { limit = { NOT = { has_variable = CWIC_reform_recent_months } } set_variable = { CWIC_reform_recent_months = 0 } }

	if = {
		limit = { has_country_flag = CWIC_currency_reform_recent }
		add_to_variable = { CWIC_reform_recent_months = 1 }
	}

	if = {
		limit = {
			has_country_flag = CWIC_currency_reform_recent
			check_variable = { CWIC_reform_recent_months > 6 }
		}
		clr_country_flag = CWIC_currency_reform_recent
		set_variable = { CWIC_reform_recent_months = 0 }
	}


	# safety clamps
	if = { limit = { check_variable = { CWIC_black_market_pressure < 0 } } set_variable = { CWIC_black_market_pressure = 0 } }
	if = { limit = { check_variable = { CWIC_black_market_pressure > 1 } } set_variable = { CWIC_black_market_pressure = 1 } }
}

cwic_calc_inflation_pressure_index = {

	# =========================
	# CPI-BASED INFLATION PRESSURE (0..6)
	# Requires:
	#   CWIC_cpi_index (current month)
	#   CWIC_cpi_index_last (previous month)
	# Writes:
	#   CWIC_inflation_pressure_index (0..6)
	# Optional:
	#   CWIC_inflation_monthly_rate_pct (debug, percent)
	# =========================

	# Safety: if last is missing or <= 0, set it to current
	if = {
		limit = {
			OR = {
				NOT = { has_variable = CWIC_cpi_index_last }
				check_variable = { CWIC_cpi_index_last <= 0 }
			}
		}
		set_variable = { CWIC_cpi_index_last = CWIC_cpi_index }
	}

	# rate = (now - last) / last
	set_temp_variable = { cpi_now = CWIC_cpi_index }
	set_temp_variable = { cpi_last = CWIC_cpi_index_last }

	set_temp_variable = { delta = cpi_now }
	subtract_from_temp_variable = { delta = cpi_last }

	set_temp_variable = { rate = delta }
	divide_temp_variable = { rate = cpi_last }

	# Convert to percent for easier thresholds: rate_pct = rate * 100
	set_temp_variable = { rate_pct = rate }
	multiply_temp_variable = { rate_pct = 100 }

	# Optional clamp to avoid extreme spikes breaking tuning
	# (does not affect sign, only bounds)
	clamp_temp_variable = { var = rate_pct min = -5 max = 25 }

	# Store debug percent change (optional)
	set_variable = { CWIC_inflation_monthly_rate_pct = rate_pct }

	# =========================
	# Map monthly CPI % change to pressure tier 0..6
	# Thresholds (percent per month):
	# <= 0.0  -> 0
	# 0.0-0.3 -> 1
	# 0.3-0.7 -> 2
	# 0.7-1.5 -> 3
	# 1.5-3.0 -> 4
	# 3.0-6.0 -> 5
	# > 6.0   -> 6
	# =========================

	set_variable = { CWIC_inflation_pressure_index = 0 }

	if = {
		limit = { check_variable = { rate_pct > 0 } }

		# default at least 1 if positive
		set_variable = { CWIC_inflation_pressure_index = 1 }

		if = { limit = { check_variable = { rate_pct >= 0.3 } } set_variable = { CWIC_inflation_pressure_index = 2 } }
		if = { limit = { check_variable = { rate_pct >= 0.7 } } set_variable = { CWIC_inflation_pressure_index = 3 } }
		if = { limit = { check_variable = { rate_pct >= 1.5 } } set_variable = { CWIC_inflation_pressure_index = 4 } }
		if = { limit = { check_variable = { rate_pct >= 3.0 } } set_variable = { CWIC_inflation_pressure_index = 5 } }
		if = { limit = { check_variable = { rate_pct > 6.0 } } set_variable = { CWIC_inflation_pressure_index = 6 } }
	}

	# Hard clamp just in case
	clamp_variable = { var = CWIC_inflation_pressure_index min = 0 max = 6 }
}
