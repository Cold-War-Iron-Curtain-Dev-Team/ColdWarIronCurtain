# Hooks the MKUltra expansion content plugs into. Each is called by a single
# line in the base files, so new dossiers never need edits inside them.

# Called from USA_mkultra_gui_reconcile after cards 1-8 are added.
USA_mkultra_expansion_gui_cards = {
	if = {
		limit = { USA_mkultra_card_9_revealed_trigger = yes }
		add_to_array = { USA_mkultra_dossier_index_array = 9 }
	}
	if = {
		limit = { USA_mkultra_card_10_revealed_trigger = yes }
		add_to_array = { USA_mkultra_dossier_index_array = 10 }
	}
	if = {
		limit = { USA_mkultra_card_11_revealed_trigger = yes }
		add_to_array = { USA_mkultra_dossier_index_array = 11 }
	}
	if = {
		limit = { USA_mkultra_card_12_revealed_trigger = yes }
		add_to_array = { USA_mkultra_dossier_index_array = 12 }
	}
	if = {
		limit = { USA_mkultra_card_13_revealed_trigger = yes }
		add_to_array = { USA_mkultra_dossier_index_array = 13 }
	}
	if = {
		limit = { USA_mkultra_card_14_revealed_trigger = yes }
		add_to_array = { USA_mkultra_dossier_index_array = 14 }
	}
	if = {

}

# Called at the end of USA_mkultra_gui_reconcile: an expansion record that is
# not revealed can never stay selected behind a closed card.
USA_mkultra_expansion_gui_orphans = {
	if = {
		limit = {
			check_variable = { USA_mkultra_selected_record > 8 }
			NOT = { USA_mkultra_expansion_record_revealed_trigger = yes }
		}
		set_variable = { USA_mkultra_selected_record = 0 }
		set_variable = { USA_mkultra_record_file_open = 0 }
	}
}
