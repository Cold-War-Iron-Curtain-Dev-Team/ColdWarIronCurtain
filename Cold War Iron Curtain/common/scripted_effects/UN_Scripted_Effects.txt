UN_CLEAR_CURRENT_VOTE_COUNTS = {
    set_variable = { un_veto_count = 0 }
    set_variable = { un_yes_count = 0 }
    set_variable = { un_no_count = 0 }
    set_variable = { un_abstain_count = 0 }

    clear_array = UNS.un_vote_abstain_list
    clear_array = UNS.un_vote_no_list
    clear_array = UNS.un_vote_veto_list
    clear_array = UNS.un_vote_yes_list
    clear_array = UNS.un_vote_

    clear_array = un_vote_abstain_list
    clear_array = un_vote_no_list
    clear_array = un_vote_veto_list
    clear_array = un_vote_yes_list
    clear_array = un_vote_

    every_country = {
        limit = {
            OR = {
                has_country_flag = UN_YES_ONGOING
                has_country_flag = UN_VETO_ONGOING
                has_country_flag = UN_ABSTAIN_ONGOING
                has_country_flag = UN_NO_ONGOING
            }
        }
        clr_country_flag = UN_YES_ONGOING
        clr_country_flag = UN_VETO_ONGOING
        clr_country_flag = UN_ABSTAIN_ONGOING
        clr_country_flag = UN_NO_ONGOING
    }
}

UN_RESET_VOTE_STATE = {
    UN_CLEAR_CURRENT_VOTE_COUNTS = yes

    set_variable = { UNS.un_state_target_owner = 0 }
    set_variable = { UNS.un_action_set = 0 }
    set_variable = { UNS.un_nation_target = 0 }
    set_variable = { UNS.un_nation_proposal = 0 }
    set_variable = { UNS.un_state_target = 0 }
    set_variable = { UNS.un_historical_resolution_id = 0 }
    set_variable = { UNS.coalition_tag = 0 }

    set_variable = { root.un_state_target_owner = 0 }
    set_variable = { root.un_action_set = 0 }
    set_variable = { root.un_nation_target = 0 }
    set_variable = { root.un_state_target = 0 }
    set_variable = { root.un_historical_resolution_id = 0 }

    clr_country_flag = Current_UN_Vote
    clr_country_flag = UN_Vote_Cleanup

    # Drop pledges whose two years are up. The flag expires on its own; the
    # variable does not, and un_ai_vote_weights only honours a pledge while
    # both are present - so without this the variable would block the action
    # from ever being offered to that country again.
    every_country = {
        limit = {
            has_variable = un_pledged_to
            NOT = { has_country_flag = un_pledged }
        }
        clear_variable = un_pledged_to
    }
}

UN_START_SCRIPTED_RESOLUTION = {
    if = {
        limit = {
            UNS = {
                has_country_flag = Current_UN_Vote
            }
            has_variable = un_force_current_vote_end
            check_variable = { un_force_current_vote_end = 1 }
        }
        UN_VOTE_FORCE_END = yes
    }
    if = {
        limit = {
            UNS = {
                NOT = { has_country_flag = Current_UN_Vote }
            }
        }
        cwic_un_set_chamber = yes
        UNS = {
            UN_CLEAR_CURRENT_VOTE_COUNTS = yes
            activate_mission = UN_Vote
            activate_mission = UN_Vote_Cleanuo
            set_country_flag = Current_UN_Vote
            set_country_flag = UN_Vote_Ongoing
        }
        every_country = {
            limit = {
                OR = {
                    has_idea = un_member
                    has_idea = UN_Security_Council_Member_Perm
                }
            }
            set_variable = { un_check = 1 }
        }
        set_variable = { UNS.un_nation_proposal = root.id }
        set_variable = { UNS.un_nation_target = root.un_nation_target }
        set_variable = { UNS.un_action_set = root.un_action_set }
        set_variable = { UNS.un_state_target = root.un_state_target }
        if = {
            limit = {
                has_variable = un_historical_resolution_id
            }
            set_variable = { UNS.un_historical_resolution_id = root.un_historical_resolution_id }
        }
        else = {
            set_variable = { UNS.un_historical_resolution_id = 0 }
        }
        if = {
            limit = {
                check_variable = { root.un_nation_target > 0 }
            }
            var:root.un_nation_target = {
                save_global_event_target_as = un_nation_global_target
            }
        }
        if = {
            limit = {
                check_variable = { root.un_state_target > 0 }
            }
            var:root.un_state_target = {
                save_global_event_target_as = un_state_global_target
                owner = {
                    set_variable = { UNS.un_state_target_owner = THIS.id }
                }
            }
            var:UNS.un_state_target_owner = {
                save_global_event_target_as = un_state_owner_global_target
            }
        }
        # WHO IS IN THE ROOM. In the Assembly, every member. In the Council,
        # the five permanent members and the ten elected ones - nobody else is
        # entitled to a vote, which is the whole point of having two chambers.
        every_country = {
            limit = {
                is_ai = yes
                OR = {
                    has_idea = un_member
                    has_idea = UN_Security_Council_Member_Perm
                }
                OR = {
                    check_variable = { UNS.un_chamber = 0 }
                    has_idea = UN_Security_Council_Member_Perm
                    has_country_flag = un_sc_elected
                }
            }
            UN_ai_vote_sorting = yes
        }
        set_variable = { root.un_nation_target = 0 }
        set_variable = { root.un_action_set = 0 }
        set_variable = { root.un_state_target = 0 }
        set_variable = { root.un_historical_resolution_id = 0 }
    }
}

UN_RESOLVE_CURRENT_VOTE = {
    # Bank the result before anything clears the vote flags.
    cwic_un_award_points = yes

    # WHAT CARRIES A RESOLUTION depends on which chamber heard it.
    #   Assembly - a simple majority of the votes cast, and no veto exists.
    #   Council  - nine of the fifteen, and any permanent member can stop it.
    set_variable = { un_resolution_carried = 0 }
    if = {
        limit = {
            check_variable = { un_chamber = 1 }
            check_variable = { un_veto_count < 1 }
            check_variable = { un_yes_count > 8 }
        }
        set_variable = { un_resolution_carried = 1 }
    }
    if = {
        limit = {
            check_variable = { un_chamber = 0 }
            check_variable = { un_yes_count > un_no_count }
        }
        set_variable = { un_resolution_carried = 1 }
    }

    if = {
        limit = {
            check_variable = { un_resolution_carried = 1 }
        }
        add_to_array = { UNS.un_nation_proposal_historical = UNS.un_nation_proposal }
        add_to_array = { UNS.un_state_target_owner_historical = UNS.un_state_target_owner }
        add_to_array = { UNS.un_action_set_historical = UNS.un_action_set }
        add_to_array = { UNS.un_nation_target_historical = UNS.un_nation_target }
        if = {
            limit = {
                check_variable = { UNS.un_historical_resolution_id > 0 }
            }
            add_to_array = { UNS.un_historical_resolution_id_historical = UNS.un_historical_resolution_id }
        }
        UN_APPLY_RESOLUTION_EFFECT = yes
    }
    if = {
        limit = {
            check_variable = { un_resolution_carried = 0 }
            check_variable = { un_veto_count < 1 }
        }
        news_event = UN_RESOLUTION.29
    }
    if = {
        limit = {
            check_variable = { un_veto_count > 0 }
        }
        news_event = UN_RESOLUTION.30
    }
    set_country_flag = UN_Vote_Cleanup
}

balkanise_country = {
    every_owned_state = {
        for_each_loop = {
            array = core_countries
            value = country_temp
            if = {
                limit = {
                    NOT = { country_exists = var:country_temp }
                    # One entry per tag, not one per (state x core country).
                    # Without this a tag is added once for every state that
                    # cores it, and the transfer loop below re-runs for each
                    # duplicate.
                    NOT = { is_in_array = { PREV.releasable_tags_array = country_temp } }
                }
                add_to_array = {
                    array = PREV.releasable_tags_array
                    value = country_temp
                }
            }
        }
    }
    for_each_loop = {
        array = releasable_tags_array
        value = country_temp
        every_owned_state = {
            if = {
                limit = {
                    is_core_of = var:country_temp
                }
                transfer_state_to = var:country_temp
            }
        }
    }
    clear_array = releasable_tags_array
}
