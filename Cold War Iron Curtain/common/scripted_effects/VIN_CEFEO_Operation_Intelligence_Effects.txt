# Observer-only CEFEO operation intelligence for a VIN player. These helpers
# never resolve an operation, alter its result, declare peace or award score.

vin_cefeo_intel_launch_lorraine = {
	if = { limit = { country_exists = VIN } VIN = {
		clr_country_flag = VIN_CEFEO_Intel_Lorraine_Concluded
		set_country_flag = VIN_CEFEO_Intel_Lorraine_Active
		if = { limit = { OR = { FRE = { has_country_flag = FRE_Lorraine_Response_Deep } FRE = { has_country_flag = FRE_Lorraine_Response_Limited } } } activate_mission = VIN_Intel_Operation_Lorraine }
		country_event = { id = VIN_CEFEO.1 hours = 1 }
		log = "CEFEO_INTEL|LAUNCH|date=[GetDateText]|operation=Lorraine"
	} }
}

vin_cefeo_intel_finish_lorraine = {
	if = { limit = { country_exists = VIN } VIN = {
		if = { limit = { has_country_flag = VIN_CEFEO_Intel_Lorraine_Active } set_country_flag = VIN_CEFEO_Intel_Lorraine_Concluded }
		clr_country_flag = VIN_CEFEO_Intel_Lorraine_Active
		remove_mission = VIN_Intel_Operation_Lorraine
		log = "CEFEO_INTEL|CLEANUP|date=[GetDateText]|operation=Lorraine"
	} }
}

vin_cefeo_intel_launch_limited_operation = {
	if = { limit = { country_exists = VIN } VIN = {
		if = {
			limit = { FRE = { check_variable = { FRE_op_current = 2 } } }
			clr_country_flag = VIN_CEFEO_Intel_Hirondelle_Concluded
			set_country_flag = VIN_CEFEO_Intel_Hirondelle_Active
			activate_mission = VIN_Intel_Operation_Hirondelle
			country_event = { id = VIN_CEFEO.2 hours = 1 }
			log = "CEFEO_INTEL|LAUNCH|date=[GetDateText]|operation=Hirondelle"
		}
		else_if = {
			limit = { FRE = { check_variable = { FRE_op_current = 3 } } }
			clr_country_flag = VIN_CEFEO_Intel_Mouette_Concluded
			set_country_flag = VIN_CEFEO_Intel_Mouette_Active
			activate_mission = VIN_Intel_Operation_Mouette
			country_event = { id = VIN_CEFEO.4 hours = 1 }
			log = "CEFEO_INTEL|LAUNCH|date=[GetDateText]|operation=Mouette"
		}
		else_if = {
			limit = { FRE = { check_variable = { FRE_op_current = 4 } } }
			clr_country_flag = VIN_CEFEO_Intel_Brochet_Concluded
			set_country_flag = VIN_CEFEO_Intel_Brochet_Active
			activate_mission = VIN_Intel_Operation_Brochet
			country_event = { id = VIN_CEFEO.6 hours = 1 }
			log = "CEFEO_INTEL|LAUNCH|date=[GetDateText]|operation=Brochet"
		}
	} }
}

vin_cefeo_intel_finish_limited_operation = {
	if = { limit = { country_exists = VIN } VIN = {
		if = {
			limit = { FRE = { check_variable = { FRE_op_current = 2 } } has_country_flag = VIN_CEFEO_Intel_Hirondelle_Active }

			set_country_flag = VIN_CEFEO_Intel_Hirondelle_Concluded

			clr_country_flag = VIN_CEFEO_Intel_Mouette_Active
			set_country_flag = VIN_CEFEO_Intel_Mouette_Concluded
			remove_mission = VIN_Intel_Operation_Mouette

			log = "CEFEO_INTEL|CONCLUSION|date=[GetDateText]|operation=Mouette"
		}
		else_if = {
			limit = { FRE = { check_variable = { FRE_op_current = 4 } } has_country_flag = VIN_CEFEO_Intel_Brochet_Active }
			clr_country_flag = VIN_CEFEO_Intel_Brochet_Active
			set_country_flag = VIN_CEFEO_Intel_Brochet_Concluded

	} }
}
