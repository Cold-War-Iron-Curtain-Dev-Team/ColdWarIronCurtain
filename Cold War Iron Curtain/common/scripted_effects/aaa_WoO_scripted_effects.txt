###########################################################################
# Regime Change Opportunity - dispatch
# Scope: THIS = target, ROOT = sender
# Uses the same triggers as visible, so the branch that fires is always
# the one that made the button appear. Setting the done flag is all the
# cleanup needed.
###########################################################################
RCO_fire = {
	if = {
		limit = { RCO_SOV_greece = yes }
		ROOT = {
			country_event = sov_percentage_agreement.1
			set_country_flag = SOV_Percentage_agreement
		}
	}
	else_if = {
		limit = { RCO_SOV_china = yes }
		ROOT = {
			PRC_Lose_CCW = yes
			set_country_flag = SOV_Manchurian_Interventon
		}
	}
	else_if = {
		limit = { RCO_SOV_wgr49 = yes }
		ROOT = {
			country_event = SOV_WGR.1
			set_country_flag = SOV_WGR49_interfered
		}
	}
	else_if = {
		limit = { RCO_SOV_stalin_note = yes }
		ROOT = {
			country_event = StalinNotes.1
			set_country_flag = SOV_Stalin_Note
		}
	}
	else_if = {
		limit = { RCO_SOV_Yug = yes }
		ROOT = {
			YUG = { country_event = country_event = TUR_SOV.11 }
			set_country_flag = SOV_YUG_Invasion
		}
	}
}
