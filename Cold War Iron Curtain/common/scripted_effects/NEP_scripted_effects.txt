### =========================================================
### DETERMINE TOP TWO BLOCS
### 1 = Monarchist (Shah + Gorkha)
### 2 = Democratic (Congress + Singh)
### 3 = Rana
### 4 = Communist
### =========================================================

NEP_determine_top_two_factions = {
	set_temp_variable = { nep_top_1 = 0 }
	set_temp_variable = { nep_top_2 = 0 }
	set_temp_variable = { nep_top_1_value = -1 }
	set_temp_variable = { nep_top_2_value = -1 }

	# Monarchist
	set_temp_variable = { tmp_val = NEP_shah_monarchy_percent }
	add_to_temp_variable = { tmp_val = NEP_gorkha_nationalist_percent }

	if = {
		limit = { check_variable = { tmp_val > nep_top_1_value } }
		set_temp_variable = { nep_top_2 = nep_top_1 }
		set_temp_variable = { nep_top_2_value = nep_top_1_value }
		set_temp_variable = { nep_top_1 = 1 }
		set_temp_variable = { nep_top_1_value = tmp_val }
	}
	else_if = {
		limit = { check_variable = { tmp_val > nep_top_2_value } }
		set_temp_variable = { nep_top_2 = 1 }
		set_temp_variable = { nep_top_2_value = tmp_val }
	}

	# Democratic
	set_temp_variable = { tmp_val = NEP_nepali_congress_percent }
	add_to_temp_variable = { tmp_val = NEP_tanka_prasad_singh_faction_percent }

	if = {
		limit = { check_variable = { tmp_val > nep_top_1_value } }
		set_temp_variable = { nep_top_2 = nep_top_1 }
		set_temp_variable = { nep_top_2_value = nep_top_1_value }
		set_temp_variable = { nep_top_1 = 2 }
		set_temp_variable = { nep_top_1_value = tmp_val }
	}
	else_if = {
		limit = { check_variable = { tmp_val > nep_top_2_value } }
		set_temp_variable = { nep_top_2 = 2 }
		set_temp_variable = { nep_top_2_value = tmp_val }
	}

	# Rana
	if = {
		limit = { check_variable = { NEP_rana_percent > nep_top_1_value } }
		set_temp_variable = { nep_top_2 = nep_top_1 }
		set_temp_variable = { nep_top_2_value = nep_top_1_value }
		set_temp_variable = { nep_top_1 = 3 }
		set_temp_variable = { nep_top_1_value = NEP_rana_percent }
	}
	else_if = {
		limit = { check_variable = { NEP_rana_percent > nep_top_2_value } }
		set_temp_variable = { nep_top_2 = 3 }
		set_temp_variable = { nep_top_2_value = NEP_rana_percent }
	}

	# Communist
	if = {
		limit = { check_variable = { NEP_communist_party_nepal_percent > nep_top_1_value } }
		set_temp_variable = { nep_top_2 = nep_top_1 }
		set_temp_variable = { nep_top_2_value = nep_top_1_value }
		set_temp_variable = { nep_top_1 = 4 }
		set_temp_variable = { nep_top_1_value = NEP_communist_party_nepal_percent }
	}
	else_if = {
		limit = { check_variable = { NEP_communist_party_nepal_percent > nep_top_2_value } }
		set_temp_variable = { nep_top_2 = 4 }
		set_temp_variable = { nep_top_2_value = NEP_communist_party_nepal_percent }
	}
}

### =========================================================
### APPLY CURRENT BOP MATCHUP
### =========================================================

nep_bop_apply_current_pairing = {
	NEP_determine_top_two_factions = yes

	if = {
		limit = {
			OR = {
				AND = { check_variable = { nep_top_1 = 3 } check_variable = { nep_top_2 = 2 } }
				AND = { check_variable = { nep_top_1 = 2 } check_variable = { nep_top_2 = 3 } }
			}
		}
		nep_bop_init_rana_vs_dem = yes
	}
	else_if = {
		limit = {
			OR = {
				AND = { check_variable = { nep_top_1 = 3 } check_variable = { nep_top_2 = 1 } }
				AND = { check_variable = { nep_top_1 = 1 } check_variable = { nep_top_2 = 3 } }
			}
		}
		nep_bop_init_rana_vs_shah = yes
	}
	else_if = {
		limit = {
			OR = {
				AND = { check_variable = { nep_top_1 = 3 } check_variable = { nep_top_2 = 4 } }
				AND = { check_variable = { nep_top_1 = 4 } check_variable = { nep_top_2 = 3 } }
			}
		}
		nep_bop_init_rana_vs_cpn = yes
	}
	else_if = {
		limit = {
			OR = {
				AND = { check_variable = { nep_top_1 = 2 } check_variable = { nep_top_2 = 1 } }
				AND = { check_variable = { nep_top_1 = 1 } check_variable = { nep_top_2 = 2 } }
			}
		}
		nep_bop_init_dem_vs_shah = yes
	}
	else_if = {
		limit = {
			OR = {
				AND = { check_variable = { nep_top_1 = 2 } check_variable = { nep_top_2 = 4 } }
				AND = { check_variable = { nep_top_1 = 4 } check_variable = { nep_top_2 = 2 } }
			}
		}
		nep_bop_init_dem_vs_cpn = yes
	}
	else_if = {
		limit = {
			OR = {
				AND = { check_variable = { nep_top_1 = 1 } check_variable = { nep_top_2 = 4 } }
				AND = { check_variable = { nep_top_1 = 4 } check_variable = { nep_top_2 = 1 } }
			}
		}
		nep_bop_init_shah_vs_cpn = yes
	}
	else = {
		nep_bop_init_rana_vs_shah = yes
	}
}

### =========================================================
### SYNC BOP VALUE FROM CURRENT PAIRING
### Formula:
### (right_strength - left_strength) / (right_strength + left_strength)
### =========================================================

nep_bop_sync_value_from_current_pairing = {
	set_temp_variable = { nep_bop_left_strength = 0 }
	set_temp_variable = { nep_bop_right_strength = 0 }
	set_temp_variable = { nep_bop_total_strength = 0 }
	set_temp_variable = { nep_bop_value = 0 }

	if = {
		limit = {
			has_country_flag = nep_bop_left_rana
			has_country_flag = nep_bop_right_dem
		}
		set_temp_variable = { nep_bop_left_strength = NEP_rana_percent }
		set_temp_variable = { nep_bop_right_strength = NEP_nepali_congress_percent }
		add_to_temp_variable = { nep_bop_right_strength = NEP_tanka_prasad_singh_faction_percent }
	}
	else_if = {
		limit = {
			has_country_flag = nep_bop_left_rana
			has_country_flag = nep_bop_right_shah
		}
		set_temp_variable = { nep_bop_left_strength = NEP_rana_percent }
		set_temp_variable = { nep_bop_right_strength = NEP_shah_monarchy_percent }
		add_to_temp_variable = { nep_bop_right_strength = NEP_gorkha_nationalist_percent }
	}
	else_if = {
		limit = {
			has_country_flag = nep_bop_left_rana
			has_country_flag = nep_bop_right_cpn
		}
		set_temp_variable = { nep_bop_left_strength = NEP_rana_percent }
		set_temp_variable = { nep_bop_right_strength = NEP_communist_party_nepal_percent }
	}
	else_if = {
		limit = {
			has_country_flag = nep_bop_left_dem
			has_country_flag = nep_bop_right_shah
		}
		set_temp_variable = { nep_bop_left_strength = NEP_nepali_congress_percent }
		add_to_temp_variable = { nep_bop_left_strength = NEP_tanka_prasad_singh_faction_percent }
		set_temp_variable = { nep_bop_right_strength = NEP_shah_monarchy_percent }
		add_to_temp_variable = { nep_bop_right_strength = NEP_gorkha_nationalist_percent }
	}
	else_if = {
		limit = {
			has_country_flag = nep_bop_left_dem
			has_country_flag = nep_bop_right_cpn
		}
		set_temp_variable = { nep_bop_left_strength = NEP_nepali_congress_percent }
		add_to_temp_variable = { nep_bop_left_strength = NEP_tanka_prasad_singh_faction_percent }
		set_temp_variable = { nep_bop_right_strength = NEP_communist_party_nepal_percent }
	}
	else_if = {
		limit = {
			has_country_flag = nep_bop_left_shah
			has_country_flag = nep_bop_right_cpn
		}
		set_temp_variable = { nep_bop_left_strength = NEP_shah_monarchy_percent }
		add_to_temp_variable = { nep_bop_left_strength = NEP_gorkha_nationalist_percent }
		set_temp_variable = { nep_bop_right_strength = NEP_communist_party_nepal_percent }
	}

	set_temp_variable = { nep_bop_total_strength = nep_bop_left_strength }
	add_to_temp_variable = { nep_bop_total_strength = nep_bop_right_strength }

	if = {
		limit = { 
            OR = { 
                check_variable = { nep_bop_total_strength < 0 } 
                check_variable = { nep_bop_total_strength > 0 }
            } 
        }
		set_temp_variable = { nep_bop_value = nep_bop_right_strength }

		subtract_from_temp_variable = { nep_bop_value = nep_bop_left_strength }

		divide_temp_variable = { nep_bop_value = nep_bop_total_strength }

		clamp_temp_variable = { var = nep_bop_value min = -1 max = 1 }



		set_power_balance = {
			id = NEP_political_balance
			set_value = nep_bop_value
		}
	}
	else = {
		set_power_balance = {
			id = NEP_political_balance
			set_value = 0
		}
	}
}

# --- Apply the dynamic modifier if missing (values come from variables) ---
nep_pajani_apply_modifier = {
    if = { limit = { NOT = { has_dynamic_modifier = NEP_Pajani_System } }
        add_dynamic_modifier = { modifier = NEP_Pajani_System }
    }
}

# --- Initialize: baseline Pajani (light-but-real elite capture) ---
# --- Initialize: baseline Pajani (revised starting values) ---
#   • Fractional factors use small decimals
#   • Percentage-based fields (taxes/wages/workforce) use whole % points
nep_pajani_init_baseline = {
    nep_pajani_zero_vars = yes

    set_variable = { NEP_PAJANI_consumer_goods_var        = 0.05 }  # +3% CG burden
    set_variable = { NEP_PAJANI_education_spend_var       = 0.05 }  # +1% edu spend
    set_variable = { NEP_PAJANI_public_services_spend_var = 0.05 }  # +2% admin spend
    set_variable = { NEP_PAJANI_welfare_spend_var         = 0.00 }  # neutral welfare
    set_variable = { NEP_PAJANI_finance_income_var        = 0.05 }  # +3% finance rents
    set_variable = { NEP_PAJANI_industry_patronage_var    = 0.03 }  # +1% industry capture

    # percentage points (pp):
    set_variable = { NEP_PAJANI_corporate_tax_var         = -0.10 }    # -2 pp from corp tax
    set_variable = { NEP_PAJANI_upper_tax_var             = -0.10 }    # -2 pp from upper-class tax
    set_variable = { NEP_PAJANI_wage_factor_var           = -0.10 }    # -2 pp wage pressure

    set_country_flag = nep_pajani_active
}
