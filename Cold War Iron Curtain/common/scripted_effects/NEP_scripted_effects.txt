#Panjani System
# --- Hard reset (zero out all Pajani variables) ---
nep_pajani_zero_vars = {
    set_variable = { NEP_PAJANI_consumer_goods_var        = 0 }
    set_variable = { NEP_PAJANI_education_spend_var       = 0 }
    set_variable = { NEP_PAJANI_public_services_spend_var = 0 }
    set_variable = { NEP_PAJANI_welfare_spend_var         = 0 }
    set_variable = { NEP_PAJANI_finance_income_var        = 0 }
    set_variable = { NEP_PAJANI_industry_patronage_var    = 0 }
    set_variable = { NEP_PAJANI_corporate_tax_var         = 0 }
    set_variable = { NEP_PAJANI_upper_tax_var             = 0 }
    set_variable = { NEP_PAJANI_wage_factor_var           = 0 }
    set_variable = { NEP_PAJANI_workforce_ratio_var       = 0 }
}

# --- Apply the dynamic modifier if missing (values come from variables) ---
nep_pajani_apply_modifier = {
    if = { limit = { NOT = { has_dynamic_modifier = NEP_Pajani_System } }
        add_dynamic_modifier = { modifier = NEP_Pajani_System }
    }
}

# --- Initialize: baseline Pajani (light-but-real elite capture) ---
# --- Initialize: baseline Pajani (revised starting values) ---
#   • Fractional factors use small decimals
#   • Percentage-based fields (taxes/wages/workforce) use whole % points
nep_pajani_init_baseline = {
    nep_pajani_zero_vars = yes

    set_variable = { NEP_PAJANI_consumer_goods_var        = 0.05 }  # +3% CG burden
    set_variable = { NEP_PAJANI_education_spend_var       = 0.05 }  # +1% edu spend
    set_variable = { NEP_PAJANI_public_services_spend_var = 0.05 }  # +2% admin spend
    set_variable = { NEP_PAJANI_welfare_spend_var         = 0.00 }  # neutral welfare
    set_variable = { NEP_PAJANI_finance_income_var        = 0.05 }  # +3% finance rents
    set_variable = { NEP_PAJANI_industry_patronage_var    = 0.03 }  # +1% industry capture

    # percentage points (pp):
    set_variable = { NEP_PAJANI_corporate_tax_var         = -0.10 }    # -2 pp from corp tax
    set_variable = { NEP_PAJANI_upper_tax_var             = -0.10 }    # -2 pp from upper-class tax
    set_variable = { NEP_PAJANI_wage_factor_var           = -0.10 }    # -2 pp wage pressure
    set_variable = { NEP_PAJANI_workforce_ratio_var       = -0.10 }    # -2 pp allocative efficiency

    nep_pajani_apply_modifier = yes
    set_country_flag = nep_pajani_active
}

# --- One step tighter Pajani control (increase capture) ---
# Fractionals go UP; percentage-based go DOWN by 1 pp each step.
nep_pajani_increase_one_step = {
    custom_effect_tooltip = nep_pajani_increase_one_step_tt
    add_to_variable = { NEP_PAJANI_consumer_goods_var        = 0.01 }
    add_to_variable = { NEP_PAJANI_education_spend_var       = 0.005 }
    add_to_variable = { NEP_PAJANI_public_services_spend_var = 0.01 }
    add_to_variable = { NEP_PAJANI_finance_income_var        = 0.01 }
    add_to_variable = { NEP_PAJANI_industry_patronage_var    = 0.005 }

    # percentage points (tighten = lower taxes/wages/efficiency by 1 pp)
    add_to_variable = { NEP_PAJANI_corporate_tax_var         = -0.01 }
    add_to_variable = { NEP_PAJANI_upper_tax_var             = -0.01 }
    add_to_variable = { NEP_PAJANI_wage_factor_var           = -0.01 }
    add_to_variable = { NEP_PAJANI_workforce_ratio_var       = -0.01 }

    nep_pajani_apply_modifier = yes
}

# --- One step looser Pajani control (reform/unwind) ---
# Fractionals go DOWN; percentage-based go UP by 1 pp each step.
nep_pajani_reduce_one_step = {
    custom_effect_tooltip = nep_pajani_reduce_one_step_tt
    add_to_variable = { NEP_PAJANI_consumer_goods_var        = -0.01 }
    add_to_variable = { NEP_PAJANI_education_spend_var       = -0.005 }
    add_to_variable = { NEP_PAJANI_public_services_spend_var = -0.01 }
    add_to_variable = { NEP_PAJANI_finance_income_var        = -0.01 }
    add_to_variable = { NEP_PAJANI_industry_patronage_var    = -0.005 }

    # percentage points (loosen = raise taxes/wages/efficiency by 1 pp)
    add_to_variable = { NEP_PAJANI_corporate_tax_var         = 0.01 }
    add_to_variable = { NEP_PAJANI_upper_tax_var             = 0.01 }
    add_to_variable = { NEP_PAJANI_wage_factor_var           = 0.01 }
    add_to_variable = { NEP_PAJANI_workforce_ratio_var       = 0.01 }

    nep_pajani_apply_modifier = yes
}

# --- Fully dismantle Pajani (zeros values + remove) ---
nep_pajani_remove = {
    custom_effect_tooltip = nep_pajani_remove_tt
    nep_pajani_zero_vars = yes
    remove_dynamic_modifier = {
        modifier = NEP_Pajani_System
    }
    clr_country_flag = nep_pajani_active
}
