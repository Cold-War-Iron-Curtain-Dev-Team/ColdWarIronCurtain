# ============================================================================
# VIN/FRE First Indochina War theatre participation
# ----------------------------------------------------------------------------
# The French Indochina and Viet Minh factions intentionally use a hidden
# no-call rule so ordinary faction diplomacy cannot merge the northern and
# southern wars. Primary campaigns and limited CEFEO operations are the
# controlled exception: once their real war exists, attach every northern
# CEFEO command to that same war explicitly.
#
# VIN's own divisions are already in the war that VIN declares. The explicit
# French-side list covers the CEFEO leader and its northern crown domains. The
# State of Vietnam, Cochinchina, the Montagnard southern command, NLF, and Laos
# remain outside this bridge; their separate war/raid lifecycles own their own
# participation.
# ============================================================================

indochina_call_all_vin_fre_theater_forces = {
	# VIN campaign IDs 1, 3, 5, and 8 open against the Tai Federation.
	if = {
		limit = {
			country_exists = VIN
			country_exists = FRE
			VIN = {
				has_country_flag = VIN_Campaign_Tay_Bac_Active
				has_war_with = TAI
			}
			FRE = { NOT = { has_war_with = VIN } }
		}
		FRE = {
			add_to_war = {
				targeted_alliance = TAI
				enemy = VIN
				hostility_reason = asked_to_join
			}
		}
	}
	if = {
		limit = {
			country_exists = VIN
			country_exists = FRE
			VIN = {
				has_country_flag = VIN_Campaign_Dien_Bien_Active
				has_war_with = TAI
			}
			FRE = { NOT = { has_war_with = VIN } }
		}
		FRE = {
			add_to_war = {
				targeted_alliance = TAI
				enemy = VIN
				hostility_reason = asked_to_join
			}
		}
	}
	if = {
		limit = {
			country_exists = VIN
			country_exists = FRE
			VIN = {
				has_country_flag = VIN_Campaign_Vinh_Yen_Active
				has_war_with = TAI
			}
			FRE = { NOT = { has_war_with = VIN } }
		}
		FRE = {
			add_to_war = {
				targeted_alliance = TAI
				enemy = VIN
				hostility_reason = asked_to_join
			}
		}
	}
	if = {
		limit = {
			country_exists = VIN
			country_exists = FRE
			VIN = {
				has_country_flag = VIN_Campaign_Na_San_Active
				has_war_with = TAI
			}
			FRE = { NOT = { has_war_with = VIN } }
		}
		FRE = {
			add_to_war = {
				targeted_alliance = TAI
				enemy = VIN
				hostility_reason = asked_to_join
			}
		}
	}

	# Hoa Binh opens against the Muong Federation.
	if = {
		limit = {
			country_exists = VIN
			country_exists = FRE
			VIN = {
				has_country_flag = VIN_Campaign_Hoa_Binh_Active
				has_war_with = TAM
			}
			FRE = { NOT = { has_war_with = VIN } }
		}
		FRE = {
			add_to_war = {
				targeted_alliance = TAM
				enemy = VIN
				hostility_reason = asked_to_join
			}
		}
	}

	# Cao-Bac/Cao Bang opens against Tho territory.
	if = {
		limit = {
			country_exists = VIN
			country_exists = FRE
			VIN = {
				has_country_flag = VIN_Campaign_Cao_Bac_Active
				has_war_with = THO
			}
			FRE = { NOT = { has_war_with = VIN } }
		}
		FRE = {
			add_to_war = {
				targeted_alliance = THO
				enemy = VIN
				hostility_reason = asked_to_join
			}
		}
	}

	# Attach every other northern CEFEO command to FRE's side. The war guards keep
	# repeated calls idempotent when subject propagation already pulled someone in.
	if = {
		limit = {
			country_exists = VIN
			country_exists = FRE
			FRE = { has_war_with = VIN }
		}
		if = {
			limit = { country_exists = TAI TAI = { NOT = { has_war_with = VIN } } }
			TAI = {
				add_to_war = {
					targeted_alliance = FRE
					enemy = VIN
					hostility_reason = asked_to_join
				}
			}
		}
		if = {
			limit = { country_exists = TAM TAM = { NOT = { has_war_with = VIN } } }
			TAM = {
				add_to_war = {
					targeted_alliance = FRE
					enemy = VIN
					hostility_reason = asked_to_join
				}
			}
		}
		if = {
			limit = { country_exists = NUN NUN = { NOT = { has_war_with = VIN } } }
			NUN = {
				add_to_war = {
					targeted_alliance = FRE
					enemy = VIN
					hostility_reason = asked_to_join
				}
			}
		}
		if = {
			limit = { country_exists = THO THO = { NOT = { has_war_with = VIN } } }
			THO = {
				add_to_war = {
					targeted_alliance = FRE
					enemy = VIN
					hostility_reason = asked_to_join
				}
			}
		}
	}
}
