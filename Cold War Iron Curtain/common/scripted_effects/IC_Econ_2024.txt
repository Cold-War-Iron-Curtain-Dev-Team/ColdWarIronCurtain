calculate_free_building_slots = {
			set_variable = { THIS.building_slots_var = modifier@local_building_slots }
			add_to_variable = { THIS.building_slots_var = owner:modifier@global_building_slots }
			add_to_variable = { THIS.building_slots_var = extra_building_slots }
			#log = "[?THIS.building_slots_var] compared to [?THIS.buildings^8]"

			subtract_from_variable = { THIS.building_slots_var = THIS.buildings^8}

		    set_temp_variable = { slot_factor = 1 }
		    add_to_temp_variable = {
		        slot_factor = modifier@local_building_slots_factor
		    }
		    add_to_temp_variable = {
		        slot_factor = owner:modifier@global_building_slots_factor
		    }
		    multiply_variable = {
		        THIS.building_slots_var = slot_factor
		    }			

		    #Always will round the variable down, if it is an int, it stays at the value

		    add_to_variable = {
		    	THIS.building_slots_var = -0.5
		    }
		    round_variable = THIS.building_slots_var

   		#log = "Slot factor [?slot_factor] at [THIS.GetID]"
   		#log = "Free building slots: [?THIS.building_slots_var] at [THIS.GetID]"

		
}

calculate_employable_population = {
			set_temp_variable = { state_population = state_population_k }

  	 		set_temp_variable = { country_recruitablePopulation = owner:modifier@conscription }

  	 		set_temp_variable = { conscription_factor_fix = 1}
  	 		add_to_temp_variable = { conscription_factor_fix = owner:modifier@conscription_factor }

    		multiply_temp_variable = { country_recruitablePopulation = conscription_factor_fix } 
    		multiply_temp_variable = { country_recruitablePopulation = state_population_k}

    		set_variable = { theoretical_soldier_sum = country_recruitablePopulation }

			#log = "Soldier Deduction: [?country_recruitablePopulation] at [THIS.GetID]"

    		subtract_from_temp_variable = { state_population = country_recruitablePopulation }

			multiply_temp_variable = {
				state_population = owner:modifier@workforce_ratio
			}
			multiply_temp_variable = {
				state_population = 0.75
			}

			set_variable = { employable_population = state_population }

			#log = "[?employable_population] at [THIS.GetID]"

	
}

calculate_theoretical_population = {
		set_variable = {
			theoretical_pop_sum = {
				value = THIS.buildings^0
				multiply = THIS.State_Employment^0
				add = { value = THIS.buildings^1 multiply = THIS.State_Employment^2 }
				add = { value = THIS.buildings^2 multiply = THIS.State_Employment^1 }
				add = { value = THIS.buildings^3 multiply = THIS.State_Employment^2 }
				add = { value = THIS.buildings^5 multiply = THIS.State_Employment^3 }
				add = { value = THIS.buildings^6 multiply = THIS.State_Employment^4 }
				add = { value = THIS.resources^7 multiply = THIS.State_Employment^4 }
			}
		}

		# ^- Labourer Pops only

		set_variable = { theoretical_labourer_sum = theoretical_pop_sum }

		# Below only farmer pops

		#Food		
		set_temp_variable = { theoretical_pop_sum_food = THIS.resources^2 }
		add_to_temp_variable = { theoretical_pop_sum_food = THIS.farm_output }

		set_variable = { Test_Farmer_Value_Local = theoretical_pop_sum_food }

		set_variable = {
			farmer_y_axis_shift = {
				value = PREV.economic_y_axis
				subtract = 18
				multiply = -0.1
				add = 0.5
			}
		}

		set_temp_variable = {
			theoretical_pop_sum_food = {
				value = theoretical_pop_sum_food
				multiply = farmer_y_axis_shift
				multiply = THIS.State_Employment^5
			}
		}
		add_to_variable = { theoretical_pop_sum = theoretical_pop_sum_food }

		set_variable = { theoretical_farmer_sum = theoretical_pop_sum_food }

		#log = "[?theoretical_pop_sum] compared to [?employable_population] at [THIS.GetID]"		

	
}

calculate_unemployment_factor = {
		set_temp_variable = {
			theoretical_unemployed_raw = {
				value = employable_population
				subtract = theoretical_pop_sum
			}
		}
		set_variable = {
			overemployment_factor = {
				value = 1
				if = {
					limit = {
						value = theoretical_unemployed_raw
						less_than = 0
					}
					multiply = employable_population
					divide = theoretical_pop_sum
				}
			}
		}
		set_variable = {
			theoretical_unemployed = {
				value = theoretical_unemployed_raw
				max = 0
			}
		}
		calculate_conversion_farmer_artisan = yes
	
}

calculate_conversion_farmer_artisan = {

		set_variable = {
			artisan_rate = {
				value = college_educated
				multiply = 0.5
				add = {
					value = high_school_educated
					multiply = 0.25
				}
				add = {
					value = primary_educated
					multiply = 0.25
				}
				multiply = 0.01
			}
		}
		set_temp_variable = {
			building_slots_worker_sum = {
				value = THIS.State_Employment^6
				multiply = THIS.building_slots_var
			}
		}
		set_temp_variable = {
			artisan_conversion_base = {
				value = building_slots_worker_sum
				max = 0
				min = theoretical_unemployed
			}
		}
		set_variable = {
			theoretical_artisan_sum = {
				value = artisan_conversion_base
				multiply = artisan_rate
			}
		}
		set_variable = {
			theoretical_substinents_sum = {
				value = artisan_conversion_base
				subtract = theoretical_artisan_sum
			}
		}
		set_variable = {
			theoretical_unemployed = {
				value = theoretical_unemployed
				subtract = artisan_conversion_base
				max = 0
			}
		}
		set_variable = {
			unemployment_factor = {
				value = 0
				if = {
					limit = {
						value = theoretical_unemployed
						greater_than = 0
					}
					add = {
						value = theoretical_unemployed
						divide = employable_population
					}
				}
			}
		}
}

calculate_conversion_labourer_worker = {
	multiply_variable = { theoretical_labourer_sum = overemployment_factor }
	set_variable = { theoretical_worker_sum = theoretical_labourer_sum }
	multiply_variable = { theoretical_worker_sum = high_school_educated }
	multiply_variable = { theoretical_worker_sum = 0.01 }
	subtract_from_variable = { theoretical_labourer_sum = theoretical_worker_sum}
	
}

calculate_conversion_lwa_capitalist_and_bureaucrat = {
	set_temp_variable = { economic_x_axis_minus_1 = { value = PREV.economic_x_axis subtract = 1 max = 0 } }

	set_temp_variable = {
		capitalist_office_park_adjustment = {
			value = 1
			add = { value = building_level@office_park multiply = 0.33 }
		}
	}

	set_temp_variable = {
		worker_to_capitalist_sum = {
			value = theoretical_worker_sum
			multiply = 0.001
			multiply = capitalist_office_park_adjustment
			multiply = economic_x_axis_minus_1
		}
	}
	set_temp_variable = {
		artisan_to_capitalist_sum = {
			value = theoretical_artisan_sum
			multiply = 0.0015
			multiply = capitalist_office_park_adjustment
			multiply = economic_x_axis_minus_1
		}
	}
	set_temp_variable = {
		labourer_to_capitalist_sum = {
			value = theoretical_labourer_sum
			multiply = 0.0005
			multiply = capitalist_office_park_adjustment
			multiply = economic_x_axis_minus_1
		}
	}
	set_variable = {
		theoretical_capitalist_sum = {
			value = worker_to_capitalist_sum
			add = artisan_to_capitalist_sum
			add = labourer_to_capitalist_sum
		}

calculate_national_productivity_factor_var = {
		set_variable = {
			productivity_factor_var = {
				value = literate_educated
				add = { value = primary_educated multiply = 1.25 }
				add = { value = high_school_educated multiply = 1.75 }
				add = { value = college_educated multiply = 3.5 }
				divide = 100
				multiply = state_development
				add = modifier@productivity_factor
				multiply = overemployment_factor
				multiply = { value = infrastructure_level multiply = 0.05 add = 1 }
				multiply = PREV.electricity_debuff_factor
			}
		}
		# log = "[?THIS.productivity_factor_var] at [THIS.GetId.GetName]"

		#Black Market and Inflation
		# --------------------------------------------
		# Black market + controls productivity impacts
		#  - CWIC_productivity_mult: accumulative drag (typically negative, ex: -0.02)
		#  - CWIC_bm_productivity_penalty: tiered penalty (positive, ex: 0.03)
		# Final multiplier = 1 + CWIC_productivity_mult - CWIC_bm_productivity_penalty
		# --------------------------------------------

		# Safe init if missing
		if = { limit = { NOT = { has_variable = CWIC_productivity_mult } } set_variable = { CWIC_productivity_mult = 0 } }
		if = { limit = { NOT = { has_variable = CWIC_bm_productivity_penalty } } set_variable = { CWIC_bm_productivity_penalty = 0 } }

		set_temp_variable = {
			cwic_prod_final_mult = {
				value = 1
				add = CWIC_productivity_mult
				subtract = CWIC_bm_productivity_penalty
				clamp = { min = 0.10 max = 1.10 }
			}
		}

		multiply_variable = { productivity_factor_var = cwic_prod_final_mult }


}

		if = {

		else_if = {
			limit = { has_state_flag = subtropical_monsoon }

			add_to_temp_variable = { cropBoost = global.CashCrop_Spice }

			limit = { has_state_flag = wetlands }
			add_to_temp_variable = { cropBoost = global.CashCrop_Rice }
			add_to_temp_variable = { cropBoost = global.CashCrop_Sugar }
		}
		else = { }
		set_temp_variable = {

				value = cropBoost
				multiply = {
					value = 1
					add = {
						value = PREV.economic_x_axis

						multiply = 0.5
					}
					add = {
						value = PREV.economic_x_axis
						equals = 24

					}
				}
			}
		}

			food_gross_factory_earnings = {

				divide = 100
			}
		}

calculate_take_taxes_from_people = {

	# -------------------------------------------------------
	# Personal income tax (mass-based, per class)
	# FIX 2026-07-05: divisor was 12, but the old wage-based system
	# carried a built-in /100 on every wage (see calculate_net_poptype_earnings).
	# The pay masses from Employment_Effects have no such scaling, so taxes
	# came out ~8.3x too high (USSR ~200B/month, debts cleared in month one).
	# /100 here + the /100 on IncomeTotal after the state loop restores the
	# old overall scale.
	# -------------------------------------------------------
	set_temp_variable = { lower_tax = { value = total_lower_class_pay multiply = owner:pop_lowerclass_tax_rate divide = 100 } }
	set_temp_variable = { middle_tax = { value = total_middle_class_pay multiply = owner:pop_middleclass_tax_rate divide = 100 } }
	set_temp_variable = { upper_middle_tax = { value = total_upper_middle_class_pay multiply = owner:pop_upperclass_tax_rate divide = 100 } }
	set_temp_variable = { upper_tax = { value = total_upper_class_pay multiply = owner:pop_upperclass_tax_rate divide = 100 } }
	set_variable = {
		state_personal_income_tax = {
			value = lower_tax
			add = middle_tax
			add = upper_middle_tax
			add = upper_tax
		}
	}

	# Add to national totals (country scope = PREV in your every_owned_state loop)
	add_to_variable = { PREV.IncomeTotal = state_personal_income_tax }
	add_to_variable = { PREV.total_personal_income_tax = state_personal_income_tax }
	# FIX 2026-07-05: removed 'divide_variable = { PREV.total_personal_income_tax = 10 }'
	# — it ran once per state, dividing the accumulating national total by 10
	# every iteration and corrupting the ui_income_tax display.
	# The total is now scaled once after the loop in gdp_calculation_effect.

	# -------------------------------------------------------
	# Net pay (optional, for GUI/national income contribution)
	# -------------------------------------------------------
	set_variable = { total_lower_class_pay_net = { value = total_lower_class_pay subtract = lower_tax } }
	set_variable = { total_middle_class_pay_net = { value = total_middle_class_pay subtract = middle_tax } }
	set_variable = { total_upper_middle_class_pay_net = { value = total_upper_middle_class_pay subtract = upper_middle_tax } }
	set_variable = { total_upper_class_pay_net = { value = total_upper_class_pay subtract = upper_tax } }

	calculate_state_hdi = yes


	set_temp_variable = {
		net_factory_taxed_earnings = {
			value = net_factory_earnings
			min = 0	# FIX 2026-07-05: was 'max = 0' (upper clamp), which made corporate tax always <= 0. min = 0 means: no tax on loss-making states, normal tax otherwise.
			multiply = owner:pop_corporateclass_tax_rate
		}
	}
	add_to_variable = { PREV.total_corporate_tax = net_factory_taxed_earnings }
	add_to_variable = { PREV.IncomeTotal = net_factory_taxed_earnings }
	# log = "[?owner:pop_middleclass_tax_rate]"
}

biome_life_expectancy_adjustment = {
	# Reset temporary modifier
	set_temp_variable = { life_expectancy_biome_pollution_mod = 0 }

	### — Biome Effects —
	if = { limit = { has_state_flag = temperate }                   add_to_temp_variable = { life_expectancy_biome_pollution_mod = 0 } }
	else_if = { limit = { has_state_flag = tropical }                   add_to_temp_variable = { life_expectancy_biome_pollution_mod = -1.0 } }
	else_if = { limit = { has_state_flag = savanna }                    add_to_temp_variable = { life_expectancy_biome_pollution_mod = -0.5 } }
	else_if = { limit = { has_state_flag = subtropical_monsoon }        add_to_temp_variable = { life_expectancy_biome_pollution_mod = -0.2 } }
	else_if = { limit = { has_state_flag = mediterranean }              add_to_temp_variable = { life_expectancy_biome_pollution_mod = 0.5 } }
	else_if = { limit = { has_state_flag = wetlands }                   add_to_temp_variable = { life_expectancy_biome_pollution_mod = -1.5 } }
	else_if = { limit = { has_state_flag = delta }                      add_to_temp_variable = { life_expectancy_biome_pollution_mod = -0.8 } }
	else_if = { limit = { has_state_flag = steppe }                     add_to_temp_variable = { life_expectancy_biome_pollution_mod = -0.3 } }
	else_if = { limit = { has_state_flag = desert }               add_to_temp_variable = { life_expectancy_biome_pollution_mod = -1.5 } }
	else_if = { limit = { has_state_flag = desertlush }           add_to_temp_variable = { life_expectancy_biome_pollution_mod = -0.5 } }
	else_if = { limit = { has_state_flag = mountainshot }         add_to_temp_variable = { life_expectancy_biome_pollution_mod = -1 } }
	else_if = { limit = { has_state_flag = mountains }            add_to_temp_variable = { life_expectancy_biome_pollution_mod = -0.5 } }
	else_if = { limit = { has_state_flag = cold }                 add_to_temp_variable = { life_expectancy_biome_pollution_mod = -2 } }
	else_if = { limit = { has_state_flag = boreal }               add_to_temp_variable = { life_expectancy_biome_pollution_mod = 0 } }

	else = { }

	### — Water Pollution Effects —
	if = { limit = { has_state_flag = water_pollution_disturbed }     add_to_temp_variable = { life_expectancy_biome_pollution_mod = -0.3 } }
	else_if = { limit = { has_state_flag = water_pollution_degraded }      add_to_temp_variable = { life_expectancy_biome_pollution_mod = -0.7 } }
	else_if = { limit = { has_state_flag = water_pollution_severe }        add_to_temp_variable = { life_expectancy_biome_pollution_mod = -1.5 } }
	else_if = { limit = { has_state_flag = water_pollution_catastrophic }  add_to_temp_variable = { life_expectancy_biome_pollution_mod = -3.0 } }
	else_if = { limit = { has_state_flag = water_pollution_deadzone }      add_to_temp_variable = { life_expectancy_biome_pollution_mod = -5.0 } }
	else = { }
	### — Land Pollution Effects —
	if = { limit = { has_state_flag = land_pollution_disturbed }      add_to_temp_variable = { life_expectancy_biome_pollution_mod = -0.2 } }
	else_if = { limit = { has_state_flag = land_pollution_degraded }       add_to_temp_variable = { life_expectancy_biome_pollution_mod = -0.5 } }
	else_if = { limit = { has_state_flag = land_pollution_severe }         add_to_temp_variable = { life_expectancy_biome_pollution_mod = -1.0 } }
	else_if = { limit = { has_state_flag = land_pollution_catastrophic }   add_to_temp_variable = { life_expectancy_biome_pollution_mod = -2.0 } }
	else_if = { limit = { has_state_flag = land_pollution_deadzone }       add_to_temp_variable = { life_expectancy_biome_pollution_mod = -3.0 } }
	else = { }
	set_variable = { life_expectancy_biome_pollution_mod_stored = life_expectancy_biome_pollution_mod }
	# Final application
	add_to_variable = { state_life_expectancy = life_expectancy_biome_pollution_mod }

	set_temp_variable = { alu_val = { value = resource@aluminium add = aluminium_exports subtract = aluminium_imports clamp = { min = 0 max = 1000000 } multiply = 0.100 } }
	set_temp_variable = { tng_val = { value = resource@tungsten add = tungsten_exports subtract = tungsten_imports clamp = { min = 0 max = 1000000 } multiply = 0.250 } }
	set_temp_variable = { stl_val = { value = resource@steel add = steel_exports subtract = steel_imports clamp = { min = 0 max = 1000000 } multiply = 0.100 } }
	set_temp_variable = { chr_val = { value = resource@chromium add = chromium_exports subtract = chromium_imports clamp = { min = 0 max = 1000000 } multiply = 0.500 } }

	set_temp_variable = { metal_val = { value = alu_val add = tng_val add = stl_val add = chr_val } }
	set_temp_variable = { total_val = { value = oil_val add = metal_val } }

	# ---- If we have any extractive value, split rgo_sector into oil/mining earnings ----
	set_temp_variable = {
		oil_share = {
			value = 0

			}
		}
	}
	set_temp_variable = {
		mining_share = {
			value = 0
			if = {
				limit = { value = total_val greater_than = 0 }
				add = { value = metal_val divide = total_val clamp = { min = 0 max = 1 } }
			}
		}
	}

	set_variable = { CWIC_oil_output_index = { value = rgo_sector multiply = oil_share divide = 250 clamp = { min = 0.5 max = 8.0 } } }
	set_variable = { CWIC_mining_output_index = { value = rgo_sector multiply = mining_share divide = 250 clamp = { min = 0.5 max = 6.0 } } }

cwic_calc_nationalization_pressure_index = {

	# =========================
	# DEFAULT
	# =========================
	set_variable = { CWIC_nationalization_pressure_index = 0 }

	# =========================
	# COMPONENT 1: ECONOMIC AXIS BASE (0..4)
	# =========================
	# Higher economic_x_axis = more state control / more nationalization pressure
	# Tune the thresholds to your axis scale.

	# Example mapping for a 0..30-ish axis
	set_temp_variable = {
		base_axis = {
			value = 0
			add = { value = economic_x_axis greater_than = 13 }
			add = { value = economic_x_axis greater_than = 17 }
			add = { value = economic_x_axis greater_than = 21 }
			add = { value = economic_x_axis greater_than = 25 }
		}
	}

	# =========================
	# COMPONENT 2: INSTABILITY ADD (0..1.5)
	# =========================
	# Instability makes nationalization more likely (scapegoating, emergency measures).
	# CWIC_instability_index is 0..3
	set_temp_variable = { inst_add = CWIC_instability_index }
	multiply_temp_variable = { inst_add = 0.5 }  # 0..1.5

	# =========================
	# COMPONENT 3: STABILITY RELIEF (0..1.0)
	# =========================
	# Higher stability reduces pressure slightly.

	# If using vanilla stability (0..1)
	set_temp_variable = { stab_relief = { value = stability multiply = 1.0 } }  # 0..1

	# If you prefer CWIC_stability_percent instead, use this block and comment out the 3 lines above:
	# set_temp_variable = { stab_relief = CWIC_stability_percent }
	# divide_temp_variable = { stab_relief = 100 }
	# clamp_temp_variable = { var = stab_relief min = 0 max = 1 }

	# =========================
	# COMPONENT 4: FREE MARKET BONUSES RELIEF (0..1.5)
	# =========================
	set_temp_variable = {
		market_relief = {
			value = 0
			if = {
				limit = {
					value = economic_x_axis
					greater_than = 12
					and = { value = economic_x_axis less_than = 19 }
					and = { value = economic_y_axis less_than = 19 }
				}
				add = 1.5
			}
			else_if = {
				limit = {
					value = economic_x_axis
					greater_than = 6
					and = { value = economic_x_axis less_than = 19 }
					and = { value = economic_y_axis less_than = 19 }
				}
				add = 1.0
			}
			else_if = {
				limit = {
					value = economic_x_axis
					less_than = 7
					and = { value = economic_y_axis less_than = 19 }
				}
				add = 0.5
			}
		}
	}

	# =========================
	# FINAL COMBINE
	# pressure = base_axis + inst_add - stab_relief - market_relief
	# clamp to 0..5
	# =========================
	set_temp_variable = { press = { value = base_axis add = inst_add subtract = stab_relief subtract = market_relief clamp = { min = 0 max = 5 } } }
	set_variable = { CWIC_nationalization_pressure_index = press }
}

cwic_calc_corporate_risk_index = {

	# =========================
	# DEFAULT
	# =========================
	set_variable = { CWIC_corporate_risk_index = 0 }

	# =========================
	# COMPONENT WEIGHTS (TUNING KNOBS)
	# =========================
	set_temp_variable = { W_INST = 15 }   # instability weight
	set_temp_variable = { W_NAT  = 12 }   # nationalization weight
	set_temp_variable = { W_INF  = 6 }    # inflation weight
	set_temp_variable = { W_STAB = 0.10 } # stability dampener

	# =========================
	# BASE RISK BUILD
	# =========================

	set_temp_variable = { risk = 0 }

	# Instability (0..3 → 0..45)
	set_temp_variable = { tmp = CWIC_instability_index }
	multiply_temp_variable = { tmp = W_INST }
	add_to_temp_variable = { risk = tmp }

	# Nationalization pressure (0..5 → 0..60)
	set_temp_variable = { tmp = CWIC_nationalization_pressure_index }
	multiply_temp_variable = { tmp = W_NAT }
	add_to_temp_variable = { risk = tmp }

	# Inflation pressure (0..6 → 0..36)
	set_temp_variable = { tmp = CWIC_inflation_pressure_index }
	multiply_temp_variable = { tmp = W_INF }
	add_to_temp_variable = { risk = tmp }

	# =========================
	# WAR SHOCK ADD
	# =========================
	if = {
		limit = { has_war = yes }
		add_to_temp_variable = { risk = 20 }
	}

	# =========================
	# STABILITY DAMPENER
	# =========================
	set_temp_variable = { tmp = CWIC_stability_percent }
	multiply_temp_variable = { tmp = W_STAB }
	subtract_from_temp_variable = { risk = tmp }

	# =========================
	# FINAL CLAMP
	# =========================
	clamp_temp_variable = { var = risk min = 0 max = 100 }
	set_variable = { CWIC_corporate_risk_index = risk }
}
