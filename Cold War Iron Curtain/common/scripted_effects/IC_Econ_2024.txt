calculate_free_building_slots = {
			set_variable = { THIS.building_slots_var = modifier@local_building_slots }
			add_to_variable = { THIS.building_slots_var = owner:modifier@global_building_slots }
			add_to_variable = { THIS.building_slots_var = extra_building_slots }
			#log = "[?THIS.building_slots_var] compared to [?THIS.buildings^8]"

			subtract_from_variable = { THIS.building_slots_var = THIS.buildings^8}

		    set_temp_variable = { slot_factor = 1 }
		    add_to_temp_variable = {
		        slot_factor = modifier@local_building_slots_factor
		    }
		    add_to_temp_variable = {
		        slot_factor = owner:modifier@global_building_slots_factor
		    }
		    multiply_variable = {
		        THIS.building_slots_var = slot_factor
		    }			

		    #Always will round the variable down, if it is an int, it stays at the value

		    add_to_variable = {
		    	THIS.building_slots_var = -0.5
		    }
		    round_variable = THIS.building_slots_var

   		#log = "Slot factor [?slot_factor] at [THIS.GetID]"
   		#log = "Free building slots: [?THIS.building_slots_var] at [THIS.GetID]"

		
}

cwic_calc_state_output = {
	set_variable = { cwic_ru = { value = resource@oil add = resource@aluminium add = resource@tungsten add = resource@steel add = resource@chromium add = resource@coal } }
	set_variable = { cwic_wf = { value = state_population_k multiply = 0.7275 } }
	set_variable = { cwic_jobs = { value = building_level@industrial_complex multiply = 27 add = { value = building_level@arms_factory multiply = 9.45 } add = { value = building_level@dockyard multiply = 9.45 } add = { value = building_level@office_park multiply = 56.7 } add = { value = building_level@steel_factory multiply = 5.67 } add = { value = building_level@tungsten_factory multiply = 4.05 } add = { value = cwic_ru multiply = 4.05 } multiply = state_development add = { value = building_level@agri_industrial_complex multiply = 14.175 } add = { value = resource@rubber multiply = 14.175 } } }
	set_variable = { cwic_E = 1 }
	if = {
		limit = { check_variable = { cwic_jobs > 0 } }
		set_variable = { cwic_E = { value = cwic_wf divide = cwic_jobs min = 1 } }
	}
	set_variable = { cwic_idle = { value = cwic_wf subtract = cwic_jobs max = 0 } }
	if = {
		limit = { check_variable = { state_development < 3 } }
		set_variable = { cwic_base = 1.2524744 }
	}
	else_if = {
		limit = { check_variable = { state_development < 5 } }
		set_variable = { cwic_base = 3.1739341 }
	}
	else_if = {
		limit = { check_variable = { state_development < 6 } }
		set_variable = { cwic_base = 4.2 }
	}
	else = {
		set_variable = { cwic_base = 5.5 }
	}
	set_variable = { cwic_Y = { value = literate_educated multiply = 3 add = { value = primary_educated multiply = 3 } add = { value = high_school_educated multiply = 6 } add = { value = college_educated multiply = 4 } divide = 100 } }
	set_variable = { cwic_pi = { value = 1.1 pow = cwic_Y multiply = cwic_base multiply = { value = building_level@state_infrastructure add = building_level@infrastructure add = building_level@water_infrastructure multiply = 0.05 add = 1 } } }
	set_variable = { cwic_bld = { value = building_level@industrial_complex multiply = 2 multiply = ROOT.cwic_mciv add = { value = building_level@office_park multiply = 7 multiply = ROOT.cwic_mofp } add = { value = building_level@arms_factory multiply = 1.25 } add = { value = building_level@dockyard multiply = 2.5 } add = { value = building_level@steel_factory multiply = 1.75 } add = { value = building_level@tungsten_factory multiply = 1.75 } add = { value = building_level@agri_industrial_complex multiply = 3 } add = { value = resource@rubber multiply = 3 } add = { value = cwic_wf multiply = 0.02005722 } add = { value = cwic_ru multiply = 0.03 } } }
	set_variable = { cwic_state_output = { value = cwic_pi multiply = cwic_E multiply = cwic_bld multiply = ROOT.cwic_start_productivity add = { value = cwic_idle multiply = 0.002 } } }
	set_variable = { cwic_state_gdp_pc = { value = cwic_state_output multiply = 100 divide = { value = state_population_k max = 0.000001 } } }
	add_to_variable = { PREV.grossdomesticproduct = cwic_state_output }
	set_temp_variable = { cwic_pe = { value = cwic_pi multiply = cwic_E multiply = ROOT.cwic_start_productivity } }
	set_variable = { cwic_s_civ = { value = building_level@industrial_complex multiply = 2 multiply = ROOT.cwic_mciv multiply = cwic_pe } }
	set_variable = { cwic_s_srv = { value = building_level@office_park multiply = 7 multiply = ROOT.cwic_mofp multiply = cwic_pe } }
	set_variable = { cwic_s_mic = { value = building_level@arms_factory multiply = 1.25 add = { value = building_level@dockyard multiply = 2.5 } multiply = cwic_pe } }
	set_variable = { cwic_s_mill = { value = building_level@steel_factory multiply = 1.75 add = { value = building_level@tungsten_factory multiply = 1.75 } multiply = cwic_pe } }
	set_variable = { cwic_s_rgo = { value = cwic_ru multiply = 0.03 multiply = cwic_pe } }
	set_variable = { cwic_s_food = { value = building_level@agri_industrial_complex multiply = 3 add = { value = resource@rubber multiply = 3 } multiply = cwic_pe } }
	cwic_add_state_sectors = yes
}

	set_variable = { cwic_mofp = 1.9 }

		set_variable = { cwic_mciv = 1.5 }

	}

	set_temp_variable = { cwic_econ_steps = 1 }
	if = {
		limit = {
			is_ai = yes
			date > 1949.7.1
		}
		if = {
			limit = { NOT = { has_variable = cwic_econ_tick } }
			set_variable = { cwic_econ_tick = THIS.id }
			modulo_variable = { cwic_econ_tick = 3 }

		}
		else = {
			set_variable = { cwic_econ_tick = 0 }
			set_temp_variable = { cwic_econ_steps = 3 }
		}
	}
	if = {
		limit = { check_variable = { cwic_econ_skip = 0 } }
		cwic_begin_separatism_rollup = yes
	}
	CWIC_econ_drift_vars = yes
	cwic_begin_gdp_growth_pass = yes
	every_owned_state = {
		income_effect_every_owned_scope = yes
		get_total_infrastructure = yes
		if = {
			limit = { NOT = { has_variable = state_development } }
			cwic_cache_state_category = yes

cwic_calc_nationalization_pressure_index = {

	# =========================
	# DEFAULT
	# =========================
	set_variable = { CWIC_nationalization_pressure_index = 0 }

	# =========================
	# COMPONENT 1: ECONOMIC AXIS BASE (0..4)
	# =========================
	# Higher economic_x_axis = more state control / more nationalization pressure
	# Tune the thresholds to your axis scale.

	# Example mapping for a 0..30-ish axis
	set_temp_variable = {
		base_axis = {
			value = 0
			add = { value = economic_x_axis greater_than = 13 }
			add = { value = economic_x_axis greater_than = 17 }
			add = { value = economic_x_axis greater_than = 21 }
			add = { value = economic_x_axis greater_than = 25 }
		}
	}

	# =========================
	# COMPONENT 2: INSTABILITY ADD (0..1.5)
	# =========================
	# Instability makes nationalization more likely (scapegoating, emergency measures).
	# CWIC_instability_index is 0..3
	set_temp_variable = { inst_add = CWIC_instability_index }
	multiply_temp_variable = { inst_add = 0.5 }  # 0..1.5

	# =========================
	# COMPONENT 3: STABILITY RELIEF (0..1.0)
	# =========================
	# Higher stability reduces pressure slightly.

	# If using vanilla stability (0..1)
	set_temp_variable = { stab_relief = { value = stability multiply = 1.0 } }  # 0..1

	# If you prefer CWIC_stability_percent instead, use this block and comment out the 3 lines above:
	# set_temp_variable = { stab_relief = CWIC_stability_percent }
	# divide_temp_variable = { stab_relief = 100 }
	# clamp_temp_variable = { var = stab_relief min = 0 max = 1 }

	# =========================
	# COMPONENT 4: FREE MARKET BONUSES RELIEF (0..1.5)
	# =========================
	set_temp_variable = {
		market_relief = {
			value = 0
			if = {
				limit = {
					value = economic_x_axis
					greater_than = 12
					and = { value = economic_x_axis less_than = 19 }
					and = { value = economic_y_axis less_than = 19 }
				}
				add = 1.5
			}
			else_if = {
				limit = {
					value = economic_x_axis
					greater_than = 6
					and = { value = economic_x_axis less_than = 19 }
					and = { value = economic_y_axis less_than = 19 }
				}
				add = 1.0
			}
			else_if = {
				limit = {
					value = economic_x_axis
					less_than = 7
					and = { value = economic_y_axis less_than = 19 }
				}
				add = 0.5
			}
		}
	}

	# =========================
	# FINAL COMBINE
	# pressure = base_axis + inst_add - stab_relief - market_relief
	# clamp to 0..5
	# =========================
	set_temp_variable = { press = { value = base_axis add = inst_add subtract = stab_relief subtract = market_relief clamp = { min = 0 max = 5 } } }
	set_variable = { CWIC_nationalization_pressure_index = press }
}

cwic_calc_corporate_risk_index = {

	# =========================
	# DEFAULT
	# =========================
	set_variable = { CWIC_corporate_risk_index = 0 }

	# =========================
	# COMPONENT WEIGHTS (TUNING KNOBS)
	# =========================
	set_temp_variable = { W_INST = 15 }   # instability weight
	set_temp_variable = { W_NAT  = 12 }   # nationalization weight
	set_temp_variable = { W_INF  = 6 }    # inflation weight
	set_temp_variable = { W_STAB = 0.10 } # stability dampener

	# =========================
	# BASE RISK BUILD
	# =========================

	set_temp_variable = { risk = 0 }

	# Instability (0..3 → 0..45)
	set_temp_variable = { tmp = CWIC_instability_index }
	multiply_temp_variable = { tmp = W_INST }
	add_to_temp_variable = { risk = tmp }

	# Nationalization pressure (0..5 → 0..60)
	set_temp_variable = { tmp = CWIC_nationalization_pressure_index }
	multiply_temp_variable = { tmp = W_NAT }
	add_to_temp_variable = { risk = tmp }

	# Inflation pressure (0..6 → 0..36)
	set_temp_variable = { tmp = CWIC_inflation_pressure_index }
	multiply_temp_variable = { tmp = W_INF }
	add_to_temp_variable = { risk = tmp }

	# =========================
	# WAR SHOCK ADD
	# =========================
	if = {
		limit = { has_war = yes }
		add_to_temp_variable = { risk = 20 }
	}

	# =========================
	# STABILITY DAMPENER
	# =========================
	set_temp_variable = { tmp = CWIC_stability_percent }
	multiply_temp_variable = { tmp = W_STAB }
	subtract_from_temp_variable = { risk = tmp }

	# =========================
	# FINAL CLAMP
	# =========================
	clamp_temp_variable = { var = risk min = 0 max = 100 }
	set_variable = { CWIC_corporate_risk_index = risk }
}
