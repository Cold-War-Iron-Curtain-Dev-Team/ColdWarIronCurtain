fre_atlante_daily_tick = {
	if = {
		limit = { has_country_flag = FRE_Atlante_Offensive_Active }
		add_to_variable = { FRE_Atlante_Days_Elapsed = 1 }
		if = {
			limit = {
				OR = {
					AND = { has_country_flag = FRE_Atlante_Posture_Full  fre_atlante_full_objective_trigger = yes }
					AND = { has_country_flag = FRE_Atlante_Posture_Vietnamese  fre_atlante_vietnamese_objective_trigger = yes }
				}
			}
			add_to_variable = { FRE_Atlante_Hold_Days = 1 }
		}
		else = { set_variable = { FRE_Atlante_Hold_Days = 0 } }

		set_temp_variable = { fre_atlante_hold_gate = 14 }
		if = {
			limit = { has_variable = FRE_Atlante_Hold_Extension }
			add_to_temp_variable = { fre_atlante_hold_gate = FRE_Atlante_Hold_Extension }
		}
		if = {
			limit = { NOT = { check_variable = { FRE_Atlante_Hold_Days < fre_atlante_hold_gate } } }
			if = {
				limit = { has_country_flag = FRE_Atlante_Posture_Full }
				fre_atlante_resolve_central_success = yes
			}
			else = { fre_atlante_resolve_vietnamese_foothold = yes }
		}
		else_if = {
			limit = { NOT = { fre_atlante_live_window_trigger = yes } }
			fre_atlante_resolve_superseded = yes
		}
	}
}

fre_atlante_deadline_resolve = {

		}

# =============================================================================
# CEFEO (FRE) core systems: Metropole Patience
# =============================================================================
# FRE_Metropole_Patience (0-100, country-scoped on FRE) represents Paris's
# political will to keep bankrolling the Indochina war. It drives:
#   - the subsidy tier ideas (FRE_Metropole_Subsidies_Full/Reduced/Cut)
#   - FRA's generosity when answering lifeline requests (FRE_Metropole events)
#   - the forced wind-down of the corps when it hits 0
# All patience changes must go through fre_metropole_patience_change so tier
# swaps and milestone events can never be skipped.
# =============================================================================

# Swap the subsidy tier idea to match the current patience value.
fre_update_metropole_tier = {
	hidden_effect = {
		if = {
			limit = { check_variable = { FRE_Metropole_Patience > 65.99 } }
			if = {
				limit = { NOT = { has_idea = FRE_Metropole_Subsidies_Full } }
				remove_ideas = FRE_Metropole_Subsidies_Reduced
				remove_ideas = FRE_Metropole_Subsidies_Cut
				add_ideas = FRE_Metropole_Subsidies_Full
			}
		}
		else_if = {
			limit = { check_variable = { FRE_Metropole_Patience > 32.99 } }
			if = {
				limit = { NOT = { has_idea = FRE_Metropole_Subsidies_Reduced } }
				remove_ideas = FRE_Metropole_Subsidies_Full
				remove_ideas = FRE_Metropole_Subsidies_Cut
				add_ideas = FRE_Metropole_Subsidies_Reduced
			}
		}
		else_if = {
			# FRE_The_Expeditionary_Budget focus floors the tier at Reduced
			limit = { has_country_flag = FRE_Budget_Secured }
			if = {
				limit = { NOT = { has_idea = FRE_Metropole_Subsidies_Reduced } }
				remove_ideas = FRE_Metropole_Subsidies_Full
				remove_ideas = FRE_Metropole_Subsidies_Cut
				add_ideas = FRE_Metropole_Subsidies_Reduced
			}
		}
		else = {
			if = {
				limit = { NOT = { has_idea = FRE_Metropole_Subsidies_Cut } }
				remove_ideas = FRE_Metropole_Subsidies_Full
				remove_ideas = FRE_Metropole_Subsidies_Reduced
				add_ideas = FRE_Metropole_Subsidies_Cut
			}
		}
	}
}

# Change patience by temp.fre_patience_delta (set it before calling).
# Clamps to [0, 100], swaps the tier idea, and fires milestone events when a
# threshold is crossed.
fre_metropole_patience_change = {
	custom_effect_tooltip = FRE_Patience_Change_tt
	hidden_effect = {
		set_temp_variable = { fre_patience_old = FRE_Metropole_Patience }
		add_to_variable = { FRE_Metropole_Patience = fre_patience_delta }
		clamp_variable = {
			var = FRE_Metropole_Patience
			min = 0
			max = 100
		}
		fre_update_metropole_tier = yes
		# Milestones (each fires once per crossing direction, events re-armable)
		if = {
			limit = {
				check_variable = { fre_patience_old > 65.99 }
				check_variable = { FRE_Metropole_Patience < 66 }
			}
			country_event = { id = FRE_Metropole.10 days = 1 }
		}
		if = {
			limit = {
				check_variable = { fre_patience_old > 32.99 }
				check_variable = { FRE_Metropole_Patience < 33 }
			}
			country_event = { id = FRE_Metropole.11 days = 1 }
		}
		if = {
			limit = {
				check_variable = { fre_patience_old > 0.01 }
				check_variable = { FRE_Metropole_Patience < 0.01 }
			}
			country_event = { id = FRE_Metropole.12 days = 1 }
		}
		if = {
			limit = {
				check_variable = { fre_patience_old < 66 }
				check_variable = { FRE_Metropole_Patience > 65.99 }
			}
			country_event = { id = FRE_Metropole.13 days = 1 }
		}
	}
}

# One-time initialisation; idempotent so save-loads can't double-apply.

# --- War Credits ------------------------------------------------------------
# The corps pays for its own war in Credits; political power is left for
# politics. Credits accrue monthly at a rate set by the subsidy tier, and can
# be topped up out of turn by FRE_Request_Extraordinary_Credits at the cost of
# Metropole Patience. Callers set temp.fre_credit_amount (signed) first.
fre_change_war_credits = {
	add_to_variable = { FRE_War_Credits = fre_credit_amount }
	clamp_variable = {
		var = FRE_War_Credits
		min = 0
		max = 2000
	}
	custom_effect_tooltip = FRE_War_Credits_Change_tt
}

# Monthly accrual, called from fre_weekly_pulse every fourth week.
fre_accrue_monthly_credits = {
	set_temp_variable = { fre_credit_amount = 10 }	# Cut tier
	if = {
		limit = { check_variable = { FRE_Metropole_Patience > 65.99 } }
		set_temp_variable = { fre_credit_amount = 40 }
	}
	else_if = {
		limit = { check_variable = { FRE_Metropole_Patience > 32.99 } }
		set_temp_variable = { fre_credit_amount = 25 }
	}
	hidden_effect = { fre_change_war_credits = yes }
}

# Weekly pulse (called from on_weekly_FRE):
#   - monthly patience decay while the war runs (faster in intense phases;
#     struggle phase numbers RISE as tension falls, so 3-4 = intense)
#   - starts the wind-down chain once the Indochina War is over
fre_weekly_pulse = {
	fre_castor_orphan_check = yes
	fre_pollux_orphan_check = yes
	fre_atlante_orphan_check = yes
	fre_lorraine_response_orphan_check = yes
	fre_hoa_binh_response_orphan_check = yes
	fre_dbp_response_orphan_check = yes
	if = {
		limit = { has_country_flag = FRE_CEFEO_Dissolved }
	}
	else_if = {
		limit = {
			has_global_flag = Indochina_War_Over
			NOT = { has_country_flag = FRE_Winddown_Started }
		}
		add_to_variable = { FRE_Winddown_Grace_Weeks = 1 }
		if = {
			limit = { check_variable = { FRE_Winddown_Grace_Weeks > 3.9 } }
			set_country_flag = FRE_Winddown_Started
			country_event = { id = FRE_Winddown.1 days = 2 }
		}
	}
	else_if = {
		limit = { NOT = { has_global_flag = Indochina_War_Over } }
		fre_operation_watchdog = yes	# nothing may leave an operation flag standing forever
		add_to_variable = { FRE_Credit_Accrual_Counter = 1 }
		if = {
			limit = { check_variable = { FRE_Credit_Accrual_Counter > 3.9 } }	# monthly
			set_variable = { FRE_Credit_Accrual_Counter = 0 }
			fre_accrue_monthly_credits = yes
		}
		add_to_variable = { FRE_Patience_Decay_Counter = 1 }
		# FRE_Manage_the_Press_Corps stretches the decay interval 4 -> 6 weeks
		set_temp_variable = { fre_decay_interval = 3.9 }
		if = {
			limit = { has_country_flag = FRE_Press_Managed }
			set_temp_variable = { fre_decay_interval = 5.9 }
		}
		if = {
			limit = { check_variable = { FRE_Patience_Decay_Counter > fre_decay_interval } }
			set_variable = { FRE_Patience_Decay_Counter = 0 }
			set_temp_variable = { fre_patience_delta = -1 }
			if = {
				limit = {
					check_variable = { global.Indochina_War_Active_Phase > 2.9 }
					check_variable = { global.Indochina_War_Active_Phase < 4.1 }
				}
				set_temp_variable = { fre_patience_delta = -2 }
			}
			fre_metropole_patience_change = yes
		}
	}
}

# French Indochina command model: CEFEO leads the northern domains while the
# State of Vietnam retains its southern subjects. No-call policy keeps the
# French Indochina and Viet Minh factions from widening scripted wars.

# FRE scope. This is intentionally one-shot: later political events may release
# a southern domain, and reloading a save must not undo that choice.
fre_indochina_command_initialize = {
	if = {
		limit = {
			NOT = { has_country_flag = FRE_French_Indochina_Command_Established }
			NOT = { has_country_flag = FRE_CEFEO_Dissolved }
			NOT = { has_global_flag = Indochina_War_Over }
			country_exists = VIE
			VIE = { NOT = { has_country_flag = VIE_generic_faction_created } }
		}

		# Migration for saves made before this model: retire Saigon's old faction.
		if = {
			limit = { VIE = { is_faction_leader = yes } }
			VIE = { dismantle_faction = yes }
		}
		if = {
			limit = { is_faction_leader = yes }
			dismantle_faction = yes
		}
		if = {
			limit = {
				is_in_faction = yes
				NOT = { is_faction_leader = yes }
			}
			leave_faction = yes
		}
		if = {
			limit = { NOT = { is_in_faction = yes } }
			create_faction_from_template = faction_template_french_indochina
		}

		set_country_flag = FRE_French_Indochina_Command_Established
		add_ideas = French_Indochina_Faction

		if = {
			limit = { VIE = { NOT = { is_in_faction_with = FRE } } }
			add_to_faction = VIE
		}
		if = {
			limit = { country_exists = CCC CCC = { NOT = { is_in_faction_with = FRE } } }
			add_to_faction = CCC
		}
		if = {
			limit = { country_exists = FUL FUL = { NOT = { is_in_faction_with = FRE } } }
			add_to_faction = FUL
		}
		if = {
			limit = { country_exists = TAI TAI = { NOT = { is_in_faction_with = FRE } } }
			add_to_faction = TAI
		}
		if = {
			limit = { country_exists = NUN NUN = { NOT = { is_in_faction_with = FRE } } }
			add_to_faction = NUN
		}
		if = {
			limit = { country_exists = TAM TAM = { NOT = { is_in_faction_with = FRE } } }
			add_to_faction = TAM
		}
		if = {
			limit = { country_exists = THO THO = { NOT = { is_in_faction_with = FRE } } }
			add_to_faction = THO
		}

		# These are UI identities, not extra combat modifiers.
		VIE = { if = { limit = { NOT = { has_idea = State_of_Vietnam_Faction } } add_ideas = State_of_Vietnam_Faction } }
		if = { limit = { country_exists = CCC } CCC = { if = { limit = { NOT = { has_idea = State_of_Vietnam_Faction } } add_ideas = State_of_Vietnam_Faction } } }
		if = { limit = { country_exists = FUL } FUL = { if = { limit = { NOT = { has_idea = State_of_Vietnam_Faction } } add_ideas = State_of_Vietnam_Faction } } }
		if = { limit = { country_exists = TAI } TAI = { if = { limit = { NOT = { has_idea = French_Indochina_Faction } } add_ideas = French_Indochina_Faction } } }
		if = { limit = { country_exists = NUN } NUN = { if = { limit = { NOT = { has_idea = French_Indochina_Faction } } add_ideas = French_Indochina_Faction } } }
		if = { limit = { country_exists = TAM } TAM = { if = { limit = { NOT = { has_idea = French_Indochina_Faction } } add_ideas = French_Indochina_Faction } } }
		if = { limit = { country_exists = THO } THO = { if = { limit = { NOT = { has_idea = French_Indochina_Faction } } add_ideas = French_Indochina_Faction } } }
	}

	# Outside the one-shot block so saves from the temporary separation model heal.
	if = {
		limit = {
			NOT = { has_global_flag = Indochina_War_Over }
			country_exists = VIN
			country_exists = NLF
			VIN = {
				is_faction_leader = yes
				has_country_flag = VIN_NLF_Coordination_Channel
			}
			NLF = {
				has_country_flag = NLF_Hanoi_Coordination
				NOT = { is_in_faction = yes }
			}
		}
		VIN = { add_to_faction = NLF }
	}
}

# FRE Mission Success Calculation Effects

# Shared bonuses applied to every mission roll:
#   FRE_Commander_Mission_Bonus - set by the FRE_Command rotation events
#     (Carpentier 0, de Lattre 10, Salan 7, Navarre 8)
#   FRE_GCMA_Network (0-4)      - +3 per auxiliary network level
#   FRE_VNA_Coordination        - +10 once the Joint General Staff stands up
fre_mission_apply_modifier_bonuses = {
	add_to_temp_variable = { mission_success_chance = FRE_Commander_Mission_Bonus }
	set_temp_variable = { fre_gcma_bonus = FRE_GCMA_Network }
	multiply_temp_variable = { fre_gcma_bonus = 3 }
	add_to_temp_variable = { mission_success_chance = fre_gcma_bonus }
	add_to_temp_variable = { mission_success_chance = FRE_VNA_Coordination }
}

fre_mission_calculate_rc4_success = {
	# Base chance: 50% (reasonable starting point for FRE)
	set_temp_variable = { mission_success_chance = 50 }
	
	# War Credits committed to preparation help (up to +15%)
	if = {
		limit = { check_variable = { FRE_War_Credits > 60 } }
		add_to_temp_variable = { mission_success_chance = 5 }
	}
	if = {
		limit = { check_variable = { FRE_War_Credits > 120 } }
		add_to_temp_variable = { mission_success_chance = 5 }
	}
	if = {
		limit = { check_variable = { FRE_War_Credits > 200 } }
		add_to_temp_variable = { mission_success_chance = 5 }
	}
	
	# Stability helps (up to +10%) - lowered thresholds for FRE
	if = {
		limit = { has_stability > 0.25 }
		add_to_temp_variable = { mission_success_chance = 3 }
	}
	if = {
		limit = { has_stability > 0.40 }
		add_to_temp_variable = { mission_success_chance = 4 }
	}
	if = {
		limit = { has_stability > 0.55 }
		add_to_temp_variable = { mission_success_chance = 3 }
	}
	
	# War support helps (up to +8%) - lowered thresholds
	if = {
		limit = { has_war_support > 0.20 }
		add_to_temp_variable = { mission_success_chance = 3 }
	}
	if = {
		limit = { has_war_support > 0.35 }
		add_to_temp_variable = { mission_success_chance = 3 }
	}
	if = {
		limit = { has_war_support > 0.50 }
		add_to_temp_variable = { mission_success_chance = 2 }
	}
	
	# Equipment stockpile helps (up to +12%) - lowered thresholds
	if = {
		limit = { has_equipment = { infantry_equipment_1 > 5000 } }
		add_to_temp_variable = { mission_success_chance = 4 }
	}
	if = {
		limit = { has_equipment = { infantry_equipment_1 > 10000 } }
		add_to_temp_variable = { mission_success_chance = 4 }
	}
	if = {
		limit = { has_equipment = { infantry_equipment_1 > 20000 } }
		add_to_temp_variable = { mission_success_chance = 4 }
	}
	
	fre_mission_apply_modifier_bonuses = yes

	# Clamp to 0-100
	if = {
		limit = { check_variable = { temp.mission_success_chance > 100 } }
		set_temp_variable = { mission_success_chance = 100 }
	}
	if = {
		limit = { check_variable = { temp.mission_success_chance < 0 } }
		set_temp_variable = { mission_success_chance = 0 }
	}
}

fre_operation_commit = {
	log = "FRE_OP_TRACE: COMMIT requested id=[?FRE_op_current|0] tier=[?fre_op_tier|0]"
	if = {
		limit = {
			OR = {
				fre_operation_theatre_closed = yes
				has_global_flag = Laos_Raid_Active
				AND = { country_exists = VIN VIN = { NOT = { VIN_no_campaign_running_trigger = yes } } }
			}
		}
		log = "FRE_OP_TRACE: COMMIT rejected/superseded id=[?FRE_op_current|0] phase=[?global.Indochina_War_Active_Phase|0]"
		if = {
			limit = { has_global_flag = Laos_Raid_Active }
			log = "IC_AFK|PASS|check=FRE_operation_commit_guarded_during_Laos_raid|operation=[?FRE_op_current|0]"
		}
		set_temp_variable = { fre_op_outcome = 5 }
		fre_operation_resolve_outcome = yes
	}
	else = {
		set_variable = { FRE_op_commitment_tier = fre_op_tier }
		# Hirondelle is an airborne cache raid, not a territorial offensive. Higher
		# commitments shorten the time needed to locate and demolish the depots.
		if = {
			limit = { check_variable = { FRE_op_current = 2 } }
			set_variable = { FRE_op_objective_required = 13 }
			if = { limit = { check_variable = { FRE_op_commitment_tier = 1 } } set_variable = { FRE_op_objective_required = 9 } }
			else_if = { limit = { check_variable = { FRE_op_commitment_tier = 2 } } set_variable = { FRE_op_objective_required = 6 } }
		}
		set_temp_variable = { fre_credit_amount = FRE_op_price_light }
		if = { limit = { check_variable = { FRE_op_commitment_tier = 1 } } set_temp_variable = { fre_credit_amount = FRE_op_price_standard } }
		else_if = { limit = { check_variable = { FRE_op_commitment_tier = 2 } } set_temp_variable = { fre_credit_amount = FRE_op_price_maximum } }
	multiply_temp_variable = { fre_credit_amount = -1 }
	fre_change_war_credits = yes
	if = {
		limit = { check_variable = { FRE_op_commitment_tier = 2 } }
			set_temp_variable = { fre_patience_delta = -8 }
		fre_metropole_patience_change = yes
	}
	fre_operation_launch = yes
}
}

# Historical Pollux preserves the regular detachments flown out under Operation
# Leda, but the exposed partisan column is lost. The safer alternatives must
# earn seven consecutive checks before the campaign begins.
fre_pollux_daily_tick = {
	if = {
		limit = { has_country_flag = FRE_Pollux_Evacuation_Active }
		add_to_variable = { FRE_Pollux_Days_Elapsed = 1 }
		if = {
			limit = { has_country_flag = FRE_Pollux_Posture_Escort }
			if = {
				limit = { fre_pollux_corridor_open_trigger = yes }
				add_to_variable = { FRE_Pollux_Hold_Days = 1 }
			}
			else = { set_variable = { FRE_Pollux_Hold_Days = 0 } }
		}
		else_if = {
			limit = { has_country_flag = FRE_Pollux_Posture_Airlift }
			if = {
				limit = { fre_dbp_french_aligned_controls_camp_trigger = yes }
				add_to_variable = { FRE_Pollux_Hold_Days = 1 }
			}
			else = { set_variable = { FRE_Pollux_Hold_Days = 0 } }
		}

		# A valid seventh day outranks a same-day campaign launch or settlement.
		if = {
			limit = {
				OR = {
					AND = { has_country_flag = FRE_Pollux_Posture_Escort  NOT = { check_variable = { FRE_Pollux_Hold_Days < 7 } } }
					AND = { has_country_flag = FRE_Pollux_Posture_Airlift  NOT = { check_variable = { FRE_Pollux_Hold_Days < 7 } } }
				}
			}
			fre_pollux_resolve_preserved = yes
		}
		else_if = {
			limit = {
				OR = {
					has_global_flag = Indochina_War_Over
					has_global_flag = geneva_conference_concluded
					check_variable = { global.Indochina_War_Active_Phase > 8 }
					NOT = { country_exists = VIN }
					NOT = { country_exists = TAI }
					VIN = { has_country_flag = VIN_Campaign_Dien_Bien_Active }
				}
			}
			fre_pollux_resolve_superseded = yes
		}
		else_if = {
			limit = { has_country_flag = FRE_Pollux_Posture_Historical  check_variable = { FRE_Pollux_Days_Elapsed > 9 } }
			set_country_flag = FRE_Pollux_Result_Column_Loss_Historical
			fre_pollux_resolve_column_loss = yes
		}
		else_if = {
			limit = { check_variable = { FRE_Pollux_Days_Elapsed > 20 } }
			fre_pollux_resolve_column_loss = yes
		}
	}
}
