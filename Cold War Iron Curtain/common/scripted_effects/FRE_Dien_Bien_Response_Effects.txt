fre_dbp_response_apply_state_modifier = {
	671 = {
		if = { limit = { has_dynamic_modifier = { modifier = FRE_DBP_Standard_Holdout } } remove_dynamic_modifier = { modifier = FRE_DBP_Standard_Holdout } }
		if = { limit = { has_dynamic_modifier = { modifier = FRE_DBP_Maximum_Holdout } } remove_dynamic_modifier = { modifier = FRE_DBP_Maximum_Holdout } }
		if = { limit = { has_dynamic_modifier = { modifier = FRE_DBP_Condor_Staging } } remove_dynamic_modifier = { modifier = FRE_DBP_Condor_Staging } }
		if = { limit = { has_dynamic_modifier = { modifier = FRE_DBP_Vulture_Air_Support } } remove_dynamic_modifier = { modifier = FRE_DBP_Vulture_Air_Support } }
		if = {
			limit = { FRE = { has_country_flag = FRE_DBP_Posture_Standard } }
			add_dynamic_modifier = { modifier = FRE_DBP_Standard_Holdout }
		}
		else_if = {
			limit = { FRE = { has_country_flag = FRE_DBP_Posture_Maximum } }
			add_dynamic_modifier = { modifier = FRE_DBP_Maximum_Holdout }
		}
		else_if = {
			limit = { FRE = { has_country_flag = FRE_DBP_Posture_Condor } }
			add_dynamic_modifier = { modifier = FRE_DBP_Condor_Staging }
		}
		if = {
			limit = { FRE = { has_country_flag = FRE_DBP_Vulture_Support_Approved } }
			add_dynamic_modifier = { modifier = FRE_DBP_Vulture_Air_Support }
		}
	}
}

fre_dbp_response_daily_tick = {
	if = {
		limit = {
			has_country_flag = FRE_DBP_Response_Active
			NOT = { has_country_flag = FRE_DBP_Vulture_Support_Approved }
			country_exists = VIN
			VIN = { has_country_flag = Operation_Vulture }
		}
		set_country_flag = FRE_DBP_Vulture_Direct_Intervention
		fre_dbp_vulture_approved = yes
	}
	if = {
		limit = {
			has_country_flag = FRE_DBP_Response_Active
			has_country_flag = FRE_DBP_Posture_Condor
			NOT = { has_country_flag = FRE_DBP_Condor_Objective_Achieved }
			country_exists = VIN
			VIN = {
				has_country_flag = VIN_Campaign_Dien_Bien_Active
				vin_dbp_valley_invested_trigger = yes
			}
		}
		set_country_flag = FRE_DBP_Condor_Window_Open
		if = {
			limit = { fre_dbp_condor_corridor_open_trigger = yes }
			add_to_variable = { FRE_DBP_Condor_Hold_Days = 1 }
			if = {
				limit = { check_variable = { FRE_DBP_Condor_Hold_Days > 6.9 } }
				set_country_flag = FRE_DBP_Condor_Objective_Achieved
				set_country_flag = FRE_DBP_Garrison_Preserved
				subtract_from_variable = { global.VIN_dbp_siege_days = 14 }
				country_event = { id = FRE_Dien_Bien.4 hours = 6 }
			}
		}
		else = { set_variable = { FRE_DBP_Condor_Hold_Days = 0 } }
	}
}

fre_dbp_resolve_from_campaign = {
	if = {
		limit = { has_country_flag = FRE_DBP_Response_Active }
		if = {
			limit = { VIN = { check_variable = { VIN_camp_last_outcome = 5 } } }
			set_temp_variable = { fre_dbp_response_result = 7 }
		}
		else_if = {
			limit = {
				VIN = { check_variable = { VIN_camp_last_outcome = 4 } }
				has_country_flag = FRE_DBP_Posture_Maximum
			}
			set_temp_variable = { fre_dbp_response_result = 2 }
		}
		else_if = {
			limit = {
				VIN = { check_variable = { VIN_camp_last_outcome = 4 } }
				has_country_flag = FRE_DBP_Posture_Condor
				has_country_flag = FRE_DBP_Condor_Objective_Achieved
			}
			set_temp_variable = { fre_dbp_response_result = 3 }
		}
		else_if = {
			limit = {
				VIN = { check_variable = { VIN_camp_last_outcome = 4 } }
				has_country_flag = FRE_DBP_Posture_Vulture
				has_country_flag = FRE_DBP_Vulture_Support_Approved
			}
			set_temp_variable = { fre_dbp_response_result = 4 }
		}
		else_if = {
			limit = { VIN = { check_variable = { VIN_camp_last_outcome = 4 } } }
			if = {
				limit = {
					has_country_flag = FRE_DBP_Posture_Condor
					NOT = { has_country_flag = FRE_DBP_Condor_Objective_Achieved }
				}
				set_country_flag = FRE_DBP_Condor_Result_Failed
			}
			if = {
				limit = {
					has_country_flag = FRE_DBP_Posture_Vulture
					NOT = { has_country_flag = FRE_DBP_Vulture_Support_Approved }
				}
				set_country_flag = FRE_DBP_Vulture_Result_Refused
			}
			set_temp_variable = { fre_dbp_response_result = 1 }
		}
		else_if = {
			limit = {
				VIN = {
					OR = {
						check_variable = { VIN_camp_last_outcome = 1 }
						check_variable = { VIN_camp_last_outcome = 2 }
					}
				}
			}
			set_temp_variable = { fre_dbp_response_result = 5 }
		}
		else = { set_temp_variable = { fre_dbp_response_result = 6 } }
		fre_dbp_response_finish = yes
	}
}

fre_dbp_penalty_camp_lost = {

		add_equipment_to_stockpile = { type = support_equipment amount = -250 }
		add_equipment_to_stockpile = { type = artillery_equipment_1 amount = -125 }
		set_temp_variable = { fre_patience_delta = -10 }
		fre_metropole_patience_change = yes
		add_war_support = -0.05
	}
	else_if = {
		limit = { has_country_flag = FRE_DBP_Posture_Condor }
		set_country_flag = FRE_DBP_Response_Loss_Condor
		if = {
			limit = { has_country_flag = FRE_DBP_Condor_Objective_Achieved }
			add_manpower = -2000
			add_war_support = -0.02
		}
		else = {
			add_manpower = -5000
			set_temp_variable = { fre_patience_delta = -5 }
			fre_metropole_patience_change = yes
			add_war_support = -0.04
		}
	}
	else_if = {
		limit = { has_country_flag = FRE_DBP_Posture_Vulture }
		set_country_flag = FRE_DBP_Response_Loss_Vulture
		add_manpower = -4000
		set_temp_variable = { fre_patience_delta = -6 }
		fre_metropole_patience_change = yes

		add_war_support = -0.03
	}
}

fre_dbp_response_finish = {
	set_country_flag = FRE_DBP_Response_Resolved
	if = {
		limit = { check_variable = { fre_dbp_response_result = 1 } }
		set_country_flag = FRE_DBP_Response_Result_Held
		fre_dbp_reward_standard_hold = yes
	}
	else_if = {
		limit = { check_variable = { fre_dbp_response_result = 2 } }
		set_country_flag = FRE_DBP_Response_Result_Reinforced
		fre_dbp_reward_reinforced_hold = yes
	}
	else_if = {
		limit = { check_variable = { fre_dbp_response_result = 3 } }
		set_country_flag = FRE_DBP_Response_Result_Condor
		fre_dbp_reward_condor = yes
	}
	else_if = {
		limit = { check_variable = { fre_dbp_response_result = 4 } }
		set_country_flag = FRE_DBP_Response_Result_Vulture
		fre_dbp_reward_vulture_hold = yes
	}
	else_if = {
		limit = { check_variable = { fre_dbp_response_result = 5 } }
		set_country_flag = FRE_DBP_Response_Result_Fall
		fre_dbp_penalty_camp_lost = yes
	}
	else_if = {
		limit = { check_variable = { fre_dbp_response_result = 6 } }
		set_country_flag = FRE_DBP_Response_Result_Inconclusive
	}
	else = { set_country_flag = FRE_DBP_Response_Result_Superseded }
	FRA = { indochina_struggle_recalculate_total_anti_communist_score = yes }
	fre_dbp_response_cleanup = yes
	country_event = { id = FRE_Dien_Bien.5 hours = 6 }
	if = {
		limit = { country_exists = VIN }
		VIN = { country_event = { id = VIN_Dien_Bien.1 hours = 8 } }
	}
}
