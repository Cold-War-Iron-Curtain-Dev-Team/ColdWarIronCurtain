# The active latch is cleared before outcomes, so repeated callbacks cannot pay,
# resolve, release or award twice. Saved selections never participate in closeout.
operation_closeout_effect = {
    if = {
        limit = { check_variable = { slot_one_operation_active = 1 } }
        set_variable = { slot_one_operation_active = 0 }
        set_variable = { slot_one_current_failure_status = 0 }
        set_variable = { CWIC_espionage_aar_pending = 1 }
        set_variable = { current_espionage_slot_calc = 1 }
        set_variable = { current_page = selected_squad_slot_1_Current_Page_Ref }
        set_variable = { history_target = slot_one_target }
        set_variable = { history_personnel_target = slot_one_personnel_target }
        set_temp_variable = { current_token = slot_one_mission_token }
        set_variable_to_random = success_chance
        set_variable_to_random = slot_one_detection_roll
        set_variable_to_random = slot_one_quality_roll
        set_variable = { slot_one_successful = 0 }
        set_variable = { slot_one_detected = 0 }
        set_variable = { paramilitary_slot_one_bulletion_choices = 0 }
        if = {
            limit = { check_variable = { success_chance < slot_one_success_chance } }
            set_variable = { slot_one_successful = 1 }
        }
        if = {
            limit = { check_variable = { slot_one_detection_roll < slot_one_detection_chance } var:history_target = { exists = yes } }
            set_variable = { slot_one_detected = 1 }
            var:history_target = {
                add_opinion_modifier = { target = ROOT modifier = CWIC_espionage_detected_operation }
            }
        }
        if = {
            limit = { var:history_target = { exists = yes } }
            CWIC_espionage_dispatch_operation_outcome = yes
        }
        else = { slot_1_operation_failure_effect = yes }
        CWIC_espionage_level_up_checks = yes
        CWIC_espionage_resolve_casualties = yes
        subtract_from_variable = { current_operations = 1 }
        clamp_variable = { var = current_operations min = 0 max = 3 }
        set_variable = { paramilitary_squad_id_status^selected_squad_slot_1 = 1 }
        remove_from_array = { current_operation_list = slot_one_mission_token }
        set_variable = { paramilitary_slot_one_bulletin = 1 }
        set_variable = { slot_one_conclusion_window_toggle = 1 }
        CWIC_espionage_mark_gui_dirty = yes
    }
}
