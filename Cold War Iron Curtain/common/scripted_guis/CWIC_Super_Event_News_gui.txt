scripted_gui = {

	# Topbar buttons: toggle the News window and the Filter settings window.
	CWIC_se_topbar_buttons = {
		context_type = player_context
		parent_window_token = top_bar
		window_name = "CWIC_se_topbar_buttons"
		dirty = refresh_topbar

		visible = {
			is_ai = no
		}

		effects = {
			CWIC_news_button_click = {
				if = {
					limit = { has_country_flag = CWIC_News_Open }
					clr_country_flag = CWIC_News_Open
				}
				else = {
					set_country_flag = CWIC_News_Open
				}
			}
			CWIC_settings_button_click = {
				if = {
					limit = { has_country_flag = CWIC_SE_Settings_Open }
					clr_country_flag = CWIC_SE_Settings_Open
				}
				else = {
					set_country_flag = CWIC_SE_Settings_Open
				}
			}
		}
	}

	# Latest News window: dynamic list of the last 10 super events.
	CWIC_News_Window = {
		context_type = player_context
		window_name = "CWIC_News_Window"

		visible = {
			has_country_flag = CWIC_News_Open
		}

		effects = {
			CWIC_news_close_click = {
				clr_country_flag = CWIC_News_Open
			}
			CWIC_news_entry_button_click = {
				set_variable = { CWIC_se_reopen = CWIC_se_view }
				CWIC_Reopen_SE_By_Id = yes
			}
		}

		properties = {
			CWIC_news_entry_img = {
				image = "[get_se_list_texture]"
			}
		}

		dynamic_lists = {
			CWIC_news_gridbox = {
				array = CWIC_se_log
				change_scope = no
				value = CWIC_se_view
				index = CWIC_se_view_index
				entry_container = "CWIC_news_entry"
			}
		}
	}

	# Filter settings window: radio-style popup mode selection.
	CWIC_SE_Settings_Window = {
		context_type = player_context
		window_name = "CWIC_SE_Settings_Window"

		visible = {
			has_country_flag = CWIC_SE_Settings_Open
		}

		effects = {
			CWIC_settings_close_click = {
				clr_country_flag = CWIC_SE_Settings_Open
			}
			CWIC_mode_all_click = {
				clr_country_flag = CWIC_se_pref_off
				clr_country_flag = CWIC_se_pref_majors
				clr_country_flag = CWIC_se_pref_own_majors
			}
			CWIC_mode_own_click = {
				clr_country_flag = CWIC_se_pref_off
				clr_country_flag = CWIC_se_pref_majors
				set_country_flag = CWIC_se_pref_own_majors
			}
			CWIC_mode_majors_click = {
				clr_country_flag = CWIC_se_pref_off
				clr_country_flag = CWIC_se_pref_own_majors
				set_country_flag = CWIC_se_pref_majors
			}
			CWIC_mode_off_click = {
				clr_country_flag = CWIC_se_pref_majors
				clr_country_flag = CWIC_se_pref_own_majors
				set_country_flag = CWIC_se_pref_off
			}
		}
	}
}
