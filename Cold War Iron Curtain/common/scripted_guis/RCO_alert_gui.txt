scripted_gui = {
	# Topbar alert for Regime Change Opportunities.
	# Shown while the player has at least one open opportunity
	# (RCO_any_open, kept by RCO_sync). Right-click dismisses it until
	# a NEW opportunity opens.
	RCO_alert_gui = {
		context_type = player_context
		parent_window_token = top_bar
		window_name = "RCO_alert_window"

		visible = {
			has_country_flag = RCO_any_open
			NOT = { has_country_flag = RCO_alert_dismissed }
		}

		effects = {
			RCO_alert_button_right_click = {
				set_country_flag = RCO_alert_dismissed
			}
		}
	}
}
