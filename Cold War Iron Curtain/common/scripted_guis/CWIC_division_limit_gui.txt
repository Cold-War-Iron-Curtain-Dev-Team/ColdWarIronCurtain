# ============================================================
# CWIC Division Limit System - Scripted GUIs
# Based on Kaiserreich's unit limit GUI system (used with credit)
# ============================================================

scripted_gui = {

	# --------------------------------------------------
	# Deploy view (Recruit & Deploy screen)
	# Hooks into unit_limits_anchor in countrydeploymentview.gui
	# No visible block = always shown (mirrors KR's approach)
	# properties block forces re-evaluation each daily tick
	# --------------------------------------------------
	CWIC_unit_limits_deployment = {
		context_type = player_context
		window_name = "CWIC_unit_limit_deploy_view"
		parent_window_name = unit_limits_anchor

		ai_enabled = { always = no }

		properties = {
			status_icon = {
				frame = CWIC_unit_limit_status_frame
			}
		}
	}

	# --------------------------------------------------
	# Intel Ledger - Army tab
	# Hooks into army_ledger token (army tab of the intel ledger)
	# Only shows for countries that have GDP tracked
	# properties block forces re-evaluation when selected country changes
	# --------------------------------------------------
	CWIC_unit_limits_intel = {
		context_type = selected_country_context
		window_name = "CWIC_unit_limit_intel_view"
		parent_window_token = army_ledger

		ai_enabled = { always = no }

		visible = {
			has_variable = CWIC_max_divisions
		}

		properties = {
			status_icon = {
				frame = CWIC_unit_limit_status_frame
			}
		}
	}

}
