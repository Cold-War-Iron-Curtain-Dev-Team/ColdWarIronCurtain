					clr_country_flag = CWIC_Opening_Gui

					scoped_play_song = "National Anthem of the Republic of Korea"

				}

					limit = {
						tag = USA

				}

					scoped_play_song = "March of the Volunteers"

			}
			Cold_War_Intro_Screen_Credit_Description_visible = {

				}
