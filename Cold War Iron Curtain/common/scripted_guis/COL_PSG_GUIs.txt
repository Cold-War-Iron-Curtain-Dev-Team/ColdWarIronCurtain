scripted_gui = {
	
	#政治タブのボタン
	COL_PSG_GUI_Open = {	#動きを設定する名前
		context_type = player_context
		parent_window_token = politics_tab
		window_name = COL_PSG_GUI_Open_Window	#.guiのnameにするべき名前
		
        ai_enabled = { always = no 
        }
		
        visible = {
			is_ai = no
			OR = {
                tag = COL
				tag = CLL
				tag = CLM
            }
			OR = {
				has_country_flag = COL_PSG_GUI_First_Open
				has_country_flag = COL_PSG_GUI_ALR_OPENED
				}
		}

		effects = {
			COL_GUI_Open_Button_click = {
				set_country_flag = COL_PSG_GUI
				set_country_flag = COL_PSG_GUI_ALR_OPENED
				
				if = {
					limit = {
						has_country_flag = COL_PSG_GUI_First_Open
					}
					set_country_flag = COL_Dog_Button_1
					clr_country_flag = COL_PSG_GUI_First_Open
				}
			}
		}
	}
	


	#ミニゲーム画面
	COL_PSG_GUI_Main = {
		context_type = player_context
		window_name = COL_PSG_GUI_Main_Window
		
		ai_enabled = { always = no }
		
		visible = {
			has_country_flag = COL_PSG_GUI
		}

		effects = {

			#閉じるボタン
			COL_Buttons_Close_click = {
				clr_country_flag = COL_PSG_GUI
				clr_country_flag = COL_Dog_Button_1_Explain_1
				clr_country_flag = COL_Dog_Button_1
			}

		}
	}





	#犬ボタン１
	COL_Dog_Button_1 = {
		context_type = player_context
		parent_window_token = COL_PSG_GUI_Main
		window_name = COL_Dog_Button_1_Window

		visible = {
			is_ai = no
			OR = {
                tag = COL
				tag = CLL
				tag = CLM
            }
				has_country_flag = COL_Dog_Button_1
				}

		effects = {	
				#犬ボタン
			COL_Dog_Button_1_Window_click = {
					set_country_flag = COL_Dog_Button_1_Explain_1
				}
		}
		}




		#犬ボタン１によって表示される説明ウィンドウ
		COL_Dog_Button_1_Explain_1 = {
		context_type = player_context
		parent_window_token = COL_PSG_GUI_Main
		window_name = COL_Dog_Button_1_Explain_1_Window

		visible = {
			is_ai = no
			OR = {
                tag = COL
				tag = CLL
				tag = CLM
            }
				has_country_flag = COL_Dog_Button_1_Explain_1
				}

		effects = {	
		}
		}










	}
