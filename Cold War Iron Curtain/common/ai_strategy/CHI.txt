    allowed = { original_tag = CHI }
    enable = { 
		has_global_flag = CCW_Started
	}
    abort_when_not_enabled = yes
	
	ai_strategy = {
		type = theatre_distribution_demand_increase
		id = 613  # State ID for Alexandria, so will target the theatre where Alexandria belongs
		value = 20  # Increase desired unit demand by 10
	}
	ai_strategy = {
		type = front_control
		strategic_region = 323				# Target a strategic region. Can specify multiple.
		strategic_region = 165				# Target a strategic region. Can specify multiple.
		strategic_region = 322				# Target a strategic region. Can specify multiple.
		ratio = 0.50			# At least this ratio of the frontline provinces need to be targeted for the AI strat to apply
		priority = 100			# Higher prio strats will override lower
		ordertype = front		# Can be either {front, invasion}
		execution_type = rush	# Can be either {careful, balanced, rush, rush_weak}
		execute_order = yes		# If set, will force activate/deactivate execution
		manual_attack = yes		# If set the AI will try manual poke attacks (for front orders)
	}
	
	ai_strategy = {
		# use one of them
		type = front_unit_request

		# invasions will check invasion target, fronts will check provinces at front
		strategic_region = 200				# Target a strategic region. Can specify multiple.
		value = -50							# Will be added as a factor over regular requests.
	}
}
