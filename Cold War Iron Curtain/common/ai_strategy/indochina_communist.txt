### The Pathet Lao ###
# The Pathet Lao field their own army (lao_deploy_field_force /
# lao_deploy_frontier_garrison in IC_Laos_Raid_Effects.txt). The northern
# frontier - Phongsaly and Sam Neua, facing Dien Bien Phu and the Tai highlands
# - is a border and not a gap.

LAO_hold_the_kingdom = {
	allowed = { tag = LAO }
	enable = {
		OR = {
			has_war_with = LOS
			has_war_with = FRE
			has_war_with = FRA
			has_war_with = TAI
			has_war_with = VIE
		}
	}
	abort_when_not_enabled = yes
	abort = {
		NOT = {
			OR = {
				has_war_with = LOS
				has_war_with = FRE
				has_war_with = FRA
				has_war_with = TAI
				has_war_with = VIE
			}
		}
	}

	ai_strategy = {
		type = front_control
		tag = FRE
		ordertype = front
		execution_type = careful
		priority = 200
	}
	ai_strategy = {
		type = front_control
		tag = TAI
		ordertype = front
		execution_type = careful
		priority = 200
	}
	ai_strategy = {
		type = front_control
		tag = LOS
		ordertype = front
		execution_type = rush
		priority = 250
	}

	ai_strategy = {
		type = front_unit_request
		tag = FRE
		value = 150
	}
	ai_strategy = {
		type = front_unit_request
		tag = TAI
		value = 150
	}

	ai_strategy = {
		type = garrison
		value = -30
	}
	ai_strategy = {
		type = dont_defend_ally_borders
		value = 100
	}
}

# The Kingdom is theirs and the war moved on. The frontier is still a frontier.
LAO_hold_the_frontier = {
	allowed = { tag = LAO }
	enable = {
		NOT = { country_exists = LOS }
		NOT = { has_global_flag = Indochina_War_Over }
	}
	abort_when_not_enabled = yes
	abort = {
		OR = {
			country_exists = LOS
			has_global_flag = Indochina_War_Over
		}
	}

	ai_strategy = {
		type = front_control
		tag = FRE
		ordertype = front
		execution_type = careful
		priority = 300
	}
	ai_strategy = {
		type = front_control
		tag = TAI
		ordertype = front
		execution_type = careful
		priority = 300
	}
	ai_strategy = {
		type = front_unit_request
		tag = FRE
		value = 250
	}
	ai_strategy = {
		type = front_unit_request
		tag = TAI
		value = 250
	}
	ai_strategy = {
		type = garrison
		value = -20
	}

	ai_strategy = {
		type = build_army
		id = infantry
		value = 400
	}
	ai_strategy = {
		type = force_build_armies
		value = 150
	}
}
