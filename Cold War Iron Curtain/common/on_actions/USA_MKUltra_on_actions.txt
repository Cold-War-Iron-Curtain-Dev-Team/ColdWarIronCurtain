on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { country_exists = USA }
				USA = {
					if = {
						limit = {
							OR = {
								has_variable = USA_mkultra_schema_version
								USA_mkultra_has_entry_authority_trigger = yes
							}
						}
						USA_mkultra_initialize = yes
					}
				}
			}
		}
	}

	# The monthly hook is only a date gate and lightweight recovery pass. Exact
	# dossier durations are owned by the activated 90- and 30-day missions.
	on_monthly_USA = {
		effect = {
			if = {
				limit = {
					OR = {
						has_variable = USA_mkultra_schema_version
						USA_mkultra_has_entry_authority_trigger = yes
					}
				}
				USA_mkultra_initialize = yes
			}
		}
	}

	on_annex = {
		effect = {
			if = {
				limit = {
					ROOT = {
						tag = USA
						USA_mkultra_charter_active_trigger = yes
					}
				}
				ROOT = {
					USA_mkultra_terminate_charter = yes
				}
			}
		}
	}

	on_capitulation = {
		effect = {
			if = {
				limit = {
					ROOT = {
						tag = USA
						USA_mkultra_charter_active_trigger = yes
					}
				}
				ROOT = {
					USA_mkultra_terminate_charter = yes
				}
			}
		}
	}
}
