on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { country_exists = USA }
				USA = {
					if = {
						limit = {
							date > 1979.12.30
							NOT = { has_country_flag = USA_mkultra_initialized }
						}
						USA_mkultra_seed_1980_archive = yes
					}
					else_if = {
						limit = {
							OR = {
								has_variable = USA_mkultra_schema_version
								USA_mkultra_has_entry_authority_trigger = yes
							}
						}
						USA_mkultra_initialize = yes
					}
				}
			}
		}
	}

	# The monthly hook handles date gates, historical lifecycle milestones,
	# and lightweight recovery pass.
	on_monthly_USA = {
		effect = {
			if = {
				limit = {
					OR = {
						has_variable = USA_mkultra_schema_version
						USA_mkultra_has_entry_authority_trigger = yes
					}
				}
				USA_mkultra_initialize = yes

				# 1963-1964 Inspector General Audit
				if = {
					limit = {
						date > 1963.6.1
						USA_mkultra_charter_active_trigger = yes
						NOT = { has_country_flag = USA_mkultra_ig_review_fired }
						OR = {
							check_variable = { USA_mkultra_harm > 7 }
							check_variable = { USA_mkultra_exposure > 24 }
							has_country_flag = USA_mkultra_midnight_climax_paid
						}
					}
					set_country_flag = USA_mkultra_ig_review_fired
					country_event = { id = cwic_mkultra.300 }
				}

				# FY1966 (from 1 July 1965): MKSEARCH continuation, only if the
				# IG review chose reorganization and the CIA charter survives.
				if = {
					limit = {
						date > 1965.6.30
						USA_mkultra_charter_active_trigger = yes
						has_country_flag = USA_mkultra_mksearch_reorganization
						NOT = { has_country_flag = USA_mkultra_mksearch_remit_offered }
					}
					set_country_flag = USA_mkultra_mksearch_remit_offered
					country_event = { id = cwic_mkultra.340 }
				}

				# FY1972 ends 30 June 1972: MKSEARCH funding stops and open work settles.
				if = {
					limit = {
						date > 1972.6.30
						has_country_flag = USA_mkultra_mksearch_remit
						NOT = { has_country_flag = USA_mkultra_mksearch_funding_ended }
					}
					USA_mkultra_end_mksearch_funding = yes
					country_event = { id = cwic_mkultra.345 }
				}

				# 1973 Records Disposition
				if = {
					limit = {
						date > 1973.1.1
						has_country_flag = USA_mkultra_initialized
						NOT = { has_country_flag = USA_mkultra_records_disposition_decided }
					}
					country_event = { id = cwic_mkultra.320 }
				}

				# 1975 Church Committee
				if = {
					limit = {
						date > 1975.5.1
						has_country_flag = USA_mkultra_initialized
						NOT = { has_country_flag = USA_mkultra_church_committee_resolved }
						OR = {
							check_variable = { USA_mkultra_harm > 0 }
							check_variable = { USA_mkultra_exposure > 20 }
							has_country_flag = USA_mkultra_records_disposition_decided
						}
					}
					country_event = { id = cwic_mkultra.325 }
				}

				# 1977 Senate Hearings
				if = {
					limit = {
						date > 1977.8.1
						has_country_flag = USA_mkultra_initialized
						has_country_flag = USA_mkultra_church_committee_resolved
						NOT = { has_country_flag = USA_mkultra_senate_hearing_resolved }
					}
					country_event = { id = cwic_mkultra.330 }
				}

				# 1983 Army Gateway Assessment
				if = {
					limit = {
						date > 1983.5.1
						USA_mkultra_enabled_trigger = yes
						has_country_flag = USA_mkultra_initialized
						NOT = { has_country_flag = USA_mkultra_gateway_1983_fired }
					}
					set_country_flag = USA_mkultra_gateway_1983_fired
					country_event = { id = cwic_mkultra.400 }
				}

				# Gateway Expanded Fictional Episode: The Assessment Arrived Early
				if = {
					limit = {
						date > 1984.6.1
						has_game_rule = {
							rule = USA_MKULTRA_MODE
							option = USA_MKULTRA_GATEWAY_EXPANDED
						}
						has_country_flag = USA_mkultra_gateway_authorized
						NOT = { has_country_flag = USA_mkultra_early_assessment_fired }
					}
					set_country_flag = USA_mkultra_early_assessment_fired
					country_event = { id = cwic_mkultra.420 }
				}

				# Cultural Afterlife: A File Outside Its Setting
				if = {
					limit = {
						date > 1991.1.1
						USA_mkultra_enabled_trigger = yes
						has_country_flag = USA_mkultra_initialized
						NOT = { has_country_flag = USA_mkultra_cultural_afterlife_fired }
					}
					set_country_flag = USA_mkultra_cultural_afterlife_fired
					country_event = { id = cwic_mkultra.460 }
				}

				# 1995 AIR Utility Review
				if = {
					limit = {
						date > 1995.9.1
						USA_mkultra_enabled_trigger = yes
						has_country_flag = USA_mkultra_initialized
						NOT = { has_country_flag = USA_mkultra_air_review_fired }
					}
					set_country_flag = USA_mkultra_air_review_fired
					# Design 11.1: the review is a mechanical decision only for an
					# actually authorized remote-viewing remit; otherwise a brief.
					if = {
						limit = { has_country_flag = USA_mkultra_rv_paid }
						country_event = { id = cwic_mkultra.440 }
					}
					else = {
						country_event = { id = cwic_mkultra.441 }
					}
				}

				# 1999 Y2K Epilogue: The Last Machine in the Records Room
				if = {
					limit = {
						date > 1999.1.1
						USA_mkultra_enabled_trigger = yes
						has_country_flag = USA_mkultra_initialized
						NOT = { has_country_flag = USA_mkultra_y2k_epilogue_fired }
					}
					set_country_flag = USA_mkultra_y2k_epilogue_fired
					country_event = { id = cwic_mkultra.500 }
				}
			}
		}
	}

	# ROOT is the annexer and FROM the annexed country; checking ROOT closed the
	# program whenever the USA annexed someone else.
	on_annex = {
		effect = {
			if = {
				limit = {
					FROM = {
						tag = USA
						OR = {
							USA_mkultra_charter_active_trigger = yes
							USA_mkultra_gateway_charter_active_trigger = yes
						}
					}
				}
				FROM = {
					USA_mkultra_terminate_charter = yes
					USA_mkultra_close_gateway_charter = yes
				}
			}
		}
	}

	on_capitulation = {
		effect = {
			if = {
				limit = {
					ROOT = {
						tag = USA
						OR = {
							USA_mkultra_charter_active_trigger = yes
							USA_mkultra_gateway_charter_active_trigger = yes
						}
					}
				}
				ROOT = {
					USA_mkultra_terminate_charter = yes
					USA_mkultra_close_gateway_charter = yes
				}
			}
		}
	}
}
