on_actions = {
    on_daily_SOV = {
        effect = { cwic_ww3_campaign_tick = yes }
    }
    # ROOT is the capitulating country, FROM is the victor. Record the original
    # participant before/alongside the legacy scripts that replace its government.
    on_capitulation = {
        effect = {
            if = {
                limit = {
                    has_global_flag = WW3_Started
                    FROM = { cwic_ww3_soviet_aligned = yes }
                }
                ROOT = { set_country_flag = cwic_ww3_opponent }
                if = {
                    limit = { ROOT = { original_tag = USA } }
                    SOV = { set_country_flag = cwic_ww3_us_defeated }
                }
            }
        }
    }
}
