on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { country_exists = HUN }
				HUN = {
					if = { limit = { HUN_psych_enabled_trigger = yes has_government = communism date > 1971.12.31 } HUN_psych_initialize = yes }
					if = { limit = { has_global_flag = cwic_bookmark_1980_start } WP_psych_seed_1980_member = yes }
				}
			}
			if = {
				limit = { country_exists = ROM }
				ROM = {
					if = { limit = { ROM_psych_enabled_trigger = yes has_government = communism date > 1971.12.31 } ROM_psych_initialize = yes }
					if = { limit = { has_global_flag = cwic_bookmark_1980_start } WP_psych_seed_1980_member = yes }
				}
			}
			if = { limit = { country_exists = CZE } CZE = { if = { limit = { has_global_flag = cwic_bookmark_1980_start } WP_psych_seed_1980_member = yes } } }
			if = { limit = { country_exists = BUL } BUL = { if = { limit = { has_global_flag = cwic_bookmark_1980_start } WP_psych_seed_1980_member = yes } } }
			if = { limit = { country_exists = POL } POL = { if = { limit = { has_global_flag = cwic_bookmark_1980_start } WP_psych_seed_1980_member = yes } } }
			if = { limit = { country_exists = DDR } DDR = { if = { limit = { has_global_flag = cwic_bookmark_1980_start } WP_psych_seed_1980_member = yes } } }
		}
	}

	on_monthly_HUN = {
		effect = {
			if = { limit = { HUN_psych_enabled_trigger = yes has_government = communism date > 1971.12.31 NOT = { has_country_flag = HUN_psych_initialized } } HUN_psych_initialize = yes }
			if = { limit = { HUN_psych_enabled_trigger = yes has_country_flag = HUN_psych_initialized } WP_psych_reconcile_country = yes }
		}
	}

	on_monthly_ROM = {
		effect = {
			if = { limit = { ROM_psych_enabled_trigger = yes has_government = communism date > 1971.12.31 NOT = { has_country_flag = ROM_psych_initialized } } ROM_psych_initialize = yes }
			if = { limit = { ROM_psych_enabled_trigger = yes has_country_flag = ROM_psych_initialized } WP_psych_reconcile_country = yes }
		}
	}

	# ROOT is the annexer and FROM is the annexed country in this on-action. The
	# same scope convention is used by CWIC's MKUltra annexation cleanup.
	on_annex = {
		effect = {
			if = {
				limit = {
					FROM = {
						OR = {
							original_tag = SOV
							WP_psych_is_member_tag_trigger = yes
						}
					}
				}
				FROM = { WP_psych_handle_annexed_country = yes }
			}
		}
	}
}
