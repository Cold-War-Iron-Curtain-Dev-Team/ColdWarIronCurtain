on_actions = {
	# ROOT = the country ticking -> the sender when RCO_sync runs.
	# Weekly, and only the SOV and USA tags. An alert can open up to a week later.
	on_weekly_SOV = {
		effect = {
			if = {
				limit = { NOT = { has_global_flag = cwic_debug_skip_rco } }
			RCO_SOV_openers = yes
			RCO_sync = yes
			}
		}
	}
	on_weekly_USA = {
		effect = {
			if = {
				limit = { NOT = { has_global_flag = cwic_debug_skip_rco } }
			RCO_sync = yes
			}
		}
	}
}
