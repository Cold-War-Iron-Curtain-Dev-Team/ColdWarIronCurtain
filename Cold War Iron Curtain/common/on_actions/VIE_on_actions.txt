on_actions = {
	on_daily_VIE = {
		effect = {
			VIE_state_dirty_variable = yes
			VIE_ai_french_coup_gamerule = yes
			VIE_border_ladder_cleanup = yes
			#Keep the readouts honest: the sect *_number mirrors are derived, and the referendum
			#score only existed inside Diem.30 until now. Both are pure recomputes.
			vietnam_faction_ui_update_number = yes
			if = {
				limit = {
					has_country_flag = VIE_PM_Ngo_Dinh_Diem
					NOT = { has_global_flag = VIE_Republic_Proclaimed }	# the vote is over; stop rescoring it
				}
				VIE_calculate_referendum_score = yes
			}
		}
	}
	#Dai-Viet conspiracy pressure only accrues while Diem holds the premiership.
	on_monthly_VIE = {
		effect = {
			VIE_daiviet_threat_tick = yes
		}
	}
	#End of the Phuong Mutiny (started by VIE_Vinh.3) - VIE_Vinh.4 wraps up via End_Phuong_Mutiny
	on_capitulation = {
		effect = {
			#ROOT is capitulated country, FROM is winner.
			if = {
				limit = {
					ROOT = {
						original_tag = CCC
						has_country_flag = CCC_Phuong_Mutiny
					}
					FROM = { original_tag = VIE }
				}
				VIE = {
					country_event = { id = VIE_Vinh.4 hours = 6 }
				}
			}
			#Scripted end of a VIE-VIN unification war: resolve through the
			#Southern Victory struggle ending instead of leaving the outcome
			#(puppeting etc.) to a generic peace conference.
			if = {
				limit = {
					ROOT = { original_tag = VIN }
					NOT = { has_global_flag = Indochina_War_Over }
					NOT = { has_global_flag = ic_failsafe_routing }	# the failsafe router is mid-route and will pick the ending itself
					NOT = { has_global_flag = Geneva_Conference }
					country_exists = VIE
					VIE = {
						has_war_with = VIN
						NOT = { has_government = communism }
					}
				}
				indochina_struggle_southern_victory = yes
			}
		}
	}
	on_annex = {
		effect = {
			#ROOT is annexed country, FROM is annexer.
			if = {
				limit = {
					ROOT = {
						original_tag = CCC
						has_country_flag = CCC_Phuong_Mutiny
					}
					FROM = { original_tag = VIE }
				}
				VIE = {
					country_event = { id = VIE_Vinh.4 hours = 6 }
				}
			}
		}
	}
}
