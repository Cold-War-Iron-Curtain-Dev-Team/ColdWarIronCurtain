on_actions = {
    on_declare_war = {
        effect = {
            if = {
                limit = {
                    NOT = {
                        has_global_flag = FLAG_SINO_SOVIET_WAR
                    }
                    FROM = {
                        OR = {
                            original_tag = SOV
                            original_tag = PRC
                        }
                    }
                    ROOT = {
                        OR = {
                            original_tag = PRC
                            original_tag = SOV
                        }
                    }
                }
                set_global_flag = FLAG_SINO_SOVIET_WAR
                MON = {
                    add_to_war = {
                        targeted_alliance = SOV
                        enemy = PRC
                        hostility_reason = asked_to_join
                    }
                }
            }
        }
    }
}
