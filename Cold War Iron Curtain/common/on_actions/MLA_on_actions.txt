on_actions = {
	#Scripted end of the Malayan General Uprising: resolve through the Emergency's own
	#victory effects instead of leaving the outcome to a generic peace conference.
	on_capitulation = {
		effect = {
			#ROOT is capitulated country, FROM is winner.
			if = {
				limit = {
					ROOT = { original_tag = MAL }
					FROM = { original_tag = MLA }
					has_global_flag = MAL_EMERGENCY_GENERAL_UPRISING
					NOT = { has_global_flag = MAL_EMERGENCY_MLA_Victory }
					NOT = { has_global_flag = MAL_EMERGENCY_MAL_Victory }
				}
				Fall_Of_Malaya = yes
			}
			if = {
				limit = {
					ROOT = { original_tag = MLA }
					FROM = { original_tag = MAL }
					has_global_flag = MAL_EMERGENCY_GENERAL_UPRISING
					NOT = { has_global_flag = MAL_EMERGENCY_MLA_Victory }
					NOT = { has_global_flag = MAL_EMERGENCY_MAL_Victory }
				}
				Exile_Of_The_MNLA = yes
			}
		}
	}
}
