# Clocks for the Warsaw Pact consequence events (WP_Psychotronics_extra_effects.txt).
# The engine merges these blocks with the national ones in the base files; each
# exits on a single flag check until its source choice has been made.
on_actions = {
	on_monthly_HUN = {
		effect = {
			if = {
				limit = { has_country_flag = HUN_psych_negotiation_refused }
				WP_psych_refusal_clock = yes
			}
		}
	}

	on_monthly_ROM = {
		effect = {
			if = {
				limit = { has_country_flag = ROM_psych_negotiation_refused }
				WP_psych_refusal_clock = yes
			}
		}
	}

	on_monthly_DDR = {
		effect = {
			if = {
				limit = { has_country_flag = DDR_psych_aggressive_surveillance }
				DDR_psych_annex_clock = yes
			}
		}
	}

	on_monthly_POL = {
		effect = {
			if = {
				limit = {
					OR = {
						has_country_flag = POL_psych_rigorous_disposition
						has_country_flag = POL_psych_sponsor_disposition
					}
				}
				POL_psych_audit_followup_clock = yes
			}
		}
	}
}
