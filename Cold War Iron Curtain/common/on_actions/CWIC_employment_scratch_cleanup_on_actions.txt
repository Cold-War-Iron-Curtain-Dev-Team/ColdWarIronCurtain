on_actions = {
    # A daily hook also runs in existing campaigns; the global flag makes this a one-time sweep.
    on_daily = {
        effect = {
            if = {
                limit = {
                    NOT = { has_global_flag = cwic_employment_scratch_cleanup_v1 }
                }
                # Include unowned states and countries skipped by the normal economy pulses.
                every_state = {
                    cwic_clear_state_employment_pay_scratch = yes
                    cwic_clear_state_education_income_scratch = yes
                }
                set_global_flag = cwic_employment_scratch_cleanup_v1
            }
        }
    }
}
