on_actions = {
	#ROOT is attacker, FROM is defender
	on_declare_war = {
		effect = {
			if = {
				limit = {
					OR = {
						AND = {
							ROOT = { is_in_faction_with = SOV }
							FROM = { is_in_faction_with = NAT }
						}
						AND = {
							ROOT = { is_in_faction_with = NAT }
							FROM = { is_in_faction_with = SOV }
						}
					}
					NOT = {
						has_global_flag = WW3_Started
					}
				}
				WW3_1950_starting_effect = yes
				set_global_flag = WW3_Started
				Start_Total_War_Between_Factions = yes
			}
		}
	}

	####SOVIET WAR SPOILS###

	on_capitulation_immediate = {
		effect = {
			if = {
				limit = {
					has_global_flag = WW3_Started
					ROOT = {
						OR = {
							original_tag = SOV
							original_tag = DDR
							original_tag = POL
							original_tag = HUN
							original_tag = ROM
							original_tag = BUL
							original_tag = ALB
						}
					}
					FROM = {
						original_tag = ITA
					}
				}
				ROOT = {
					add_to_faction = FROM
				}
				FROM = {
					set_country_flag = SOV_ITA_CAPTURED
					set_cosmetic_tag = ITA_COM
					add_ideas = Warsaw_Treaty_Organization_member
					set_politics = {
						ruling_party = communism
					}
					set_popularities = {
						communism = 64
						neutrality = 6
						maoism = 10
						trotskyism = 10
						Social_Democratic = 10
					}
				}
				sardinia_independent = yes
				sicily_independent = yes
			}
			if = {
				limit = {
					has_global_flag = WW3_Started
					ROOT = {
						OR = {
							original_tag = SOV
							original_tag = DDR
							original_tag = POL
							original_tag = HUN
							original_tag = ROM
							original_tag = BUL
							original_tag = ALB
						}
					}
					FROM = {
						original_tag = FRA
					}
				}
			}
		}
	}
}
