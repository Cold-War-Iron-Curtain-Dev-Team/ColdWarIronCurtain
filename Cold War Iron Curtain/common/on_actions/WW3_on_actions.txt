on_actions = {
	#ROOT is attacker, FROM is defender
	on_declare_war = {
		effect = {
			if = {
				limit = {
					OR = {
						AND = {
							ROOT = { is_in_faction_with = SOV }
							FROM = { is_in_faction_with = NAT }
						}
						AND = {
							ROOT = { is_in_faction_with = NAT }
							FROM = { is_in_faction_with = SOV }
						}
					}
					NOT = {
						has_global_flag = WW3_Started
					}
				}
				WW3_1950_starting_effect = yes
				set_global_flag = WW3_Started
				Start_Total_War_Between_Factions = yes
			}
			USA = {
				country_event = {
					id = UN.3
					days = 5
				}
			}
			SOV = {
				country_event = {
					id = UN.3
					days = 5
				}
			}
		}
	}

	####SOVIET ON STATE CHANGE###
	on_state_control_changed = {
        effect = {
            if = {
                limit = {
				    has_global_flag = WW3_Started
				    NOT = {
					    country_exists = AUS
					    country_exists = ADR
				    }
                    FROM.FROM = {
					    AND = {
						    state = 1119
						    state = 1253
						    state = 758
						    state = 999
						    state = 4
						    state = 1192
						    state = 153
						    state = 1875
							state = 1075
					    }
				    }
					FROM = {
						original_tag = SOV
						original_tag = USA
						original_tag = ENG
						original_tag = FRA
					}
					ROOT = {
						OR = {
							original_tag = AUS
							original_tag = ADR
						}
					}
                }
                if =  {
                    limit = {
						OR = {
							original_tag = SOV
					        is_ally_with = SOV
						}
                    }
                    FROM.FROM = {
                        transfer_state_to = ROOT
                    }
					ROOT = {
						load_oob = ADR_1949
						add_ideas = Warsaw_Treaty_Organization_member
					}
					SOV = {
						add_to_faction = ADR
					}
                }
				else_if = {
					limit = {
						OR = {
							original_tag = USA
					        is_ally_with = USA
						}
					}
					FROM.FROM = {
                        transfer_state_to = AUS
                    }
					ROOT = {
						add_ideas = NATO_Member
					}
					USA = {
						add_to_faction = AUS
					}
				}
		    }
	    }
	}
