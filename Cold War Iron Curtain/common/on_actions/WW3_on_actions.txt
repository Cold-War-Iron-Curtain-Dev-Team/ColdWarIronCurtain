on_actions = {
	#ROOT is attacker, FROM is defender
	on_declare_war = {
		effect = {
			if = {
				limit = {
					OR = {
						AND = {
							ROOT = { is_in_faction_with = SOV }
							FROM = { is_in_faction_with = NAT }
						}
						AND = {
							ROOT = { is_in_faction_with = NAT }
							FROM = { is_in_faction_with = SOV }
						}
					}
					NOT = {
						has_global_flag = WW3_Started
					}
				}
				WW3_1950_starting_effect = yes
				set_global_flag = WW3_Started
				every_country = {
					limit = {
						is_in_faction_with = NAT
						NOT = { tag = NAT }
					}
					add_to_war = {
						targeted_alliance = SOV
						enemy  = NAT
						hostility_reason = asked_to_join
					}
				}
				every_country = {
					limit = {
						is_in_faction_with = SOV
						NOT = { tag = SOV }
					}
					add_to_war = {
						targeted_alliance = NAT
						enemy  = SOV
						hostility_reason = asked_to_join
					}
				}

			}
		}
	}
}
