# CEFEO (FRE) hooks: patience initialisation and the weekly pulse.
# Core logic lives in common/scripted_effects/FRE_CEFEO_Effects.txt.
on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { country_exists = FRE }
				FRE = { fre_cefeo_initialize = yes }
			}
		}
	}
	on_weekly_FRE = {
		effect = {
			fre_weekly_pulse = yes
		}
	}
}
