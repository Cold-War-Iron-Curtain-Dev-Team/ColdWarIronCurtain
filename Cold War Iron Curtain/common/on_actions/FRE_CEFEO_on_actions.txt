on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { country_exists = FRE }
				FRE = {
					fre_cefeo_initialize = yes
					fre_indochina_command_initialize = yes
				}
			}
			if = {

			}
		}
	}
	on_weekly_FRE = {
		effect = {
			fre_weekly_pulse = yes
		}
	}
}
