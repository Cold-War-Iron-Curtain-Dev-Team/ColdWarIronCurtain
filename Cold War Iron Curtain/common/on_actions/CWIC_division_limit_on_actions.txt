on_actions = {

	# --------------------------------------------------
	# Daily: keep the player-facing penalty modifier and GUI fresh.
	# --------------------------------------------------
	on_daily = {
		effect = {
			# Safety: only run if the GDP variable has been initialised
			# (countries using CWIC economy). Skips vanilla/untracked countries.
			if = {
				limit = {
					is_ai = no
					has_variable = grossdomesticproduct
					check_variable = { grossdomesticproduct > 0 }
				}
				CWIC_unit_limit_calculation_effect = yes
			}
		}
	}

	# --------------------------------------------------
	# Weekly: AI only needs the stored limit for demob checks.
	# --------------------------------------------------
	on_weekly = {
		effect = {
			if = {
				limit = {
					is_ai = yes
					has_variable = grossdomesticproduct
					check_variable = { grossdomesticproduct > 0 }
				}
				CWIC_unit_limit_calculation_effect = yes
			}
			if = {
				limit = {
					tag = UNS
				}
				UN_HISTORICAL_RESOLUTION_WEEKLY_CHECK = yes
			}

			if = {
				limit = {
					is_ai = yes
					has_war = no
					CWIC_indochina_demob_pilot = yes
					has_variable = grossdomesticproduct
					check_variable = { grossdomesticproduct > 0 }
					CWIC_has_exceeded_maximum_divisions = yes
					CWIC_has_unattained_important_goals = no
					NOT = { has_country_flag = CWIC_has_reduce_units_check }
				}
				set_country_flag = CWIC_has_reduce_units_check
				country_event = { id = cwic_demob.2 days = 7 random_hours = 48 }
			}
		}
	}

	# --------------------------------------------------
	# On peace: trigger the AI demobilisation chain
	# Fires for the country that just made peace (ROOT)
	# --------------------------------------------------
	on_peace = {
		effect = {
			if = {
				limit = {
					is_ai = yes
					has_war = no
					has_variable = grossdomesticproduct
				}
				CWIC_unit_limit_calculation_effect = yes
				# Law downgrade check (fires once, chains internally)
				if = {
					limit = { NOT = { has_country_flag = CWIC_demob_law_check } }
					set_country_flag = { flag = CWIC_demob_law_check days = 30 value = 1 }
					country_event = { id = cwic_demob.1 days = 7 }
				}
				# Division disband loop (fires every 7 days until under limit)
				if = {
					limit = { NOT = { has_country_flag = CWIC_has_reduce_units_check } }
					set_country_flag = CWIC_has_reduce_units_check
					country_event = { id = cwic_demob.2 days = 7 random_hours = 48 }
				}
			}
		}
	}
}
