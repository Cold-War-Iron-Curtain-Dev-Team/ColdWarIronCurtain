on_actions = {

	# --------------------------------------------------
	# Daily: keep the player-facing penalty modifier and GUI fresh.
	# --------------------------------------------------
	on_daily = {
		effect = {
			# Safety: only run if the GDP variable has been initialised
			# (countries using CWIC economy). Skips vanilla/untracked countries.
			if = {
				limit = {
					is_ai = no
					has_variable = grossdomesticproduct
					check_variable = { grossdomesticproduct > 0 }
				}
				CWIC_unit_limit_calculation_effect = yes
			}
		}
	}

	# --------------------------------------------------
	# Weekly: AI only needs the stored limit for demob checks.
	# --------------------------------------------------
	on_weekly = {
		effect = {
			if = {
				limit = {
					is_ai = yes
					has_variable = grossdomesticproduct
					check_variable = { grossdomesticproduct > 0 }
				}
				CWIC_unit_limit_calculation_effect = yes
			}
			if = {
				limit = {
					tag = UNS
				}
				# The 1965 enlargement: six elected seats become ten. The panel has
				# had the extra two seats all along, gated on a flag nothing set.
				if = {
					limit = {
						date > 1966.1.1
						NOT = { has_global_flag = un_sec_reform }
					}
					set_global_flag = un_sec_reform
					cwic_un_elect_council = yes
				}

				# Elections every year (classes overlap, so the council changes by
				# halves), and a weekly rebuild of the list the panel reads.
				if = {
					limit = { NOT = { has_country_flag = un_council_elected_recently } }
					set_country_flag = { flag = un_council_elected_recently days = 365 value = 1 }
					cwic_un_elect_council = yes
				}
				cwic_un_council_refresh = yes

				# HISTORY IS NOT ON A TIMER. An authored resolution is a landmark:
				# it convenes its own sitting the week the moment arrives, which is
				# what an emergency special session is for (Uniting for Peace, 1950).
				UN_HISTORICAL_RESOLUTION_WEEKLY_CHECK = yes

				# THE ORDINARY FLOOR SITS EVERY FOUR MONTHS - three times a year.
				# Fast enough that the UN is a place where things happen, slow enough
				# that a player has 120 days to actually do what a bloc asked of them
				# before the next vote. It was weekly, which meant the tab nagged and
				# one purchase covered two dozen votes.
				if = {
					limit = { NOT = { has_country_flag = un_sitting_recently } }
					set_country_flag = { flag = un_sitting_recently days = 120 value = 1 }
					cwic_un_auto_resolution = yes
				}
			}

			if = {
				limit = {
					is_ai = yes
					has_war = no
					CWIC_indochina_demob_pilot = yes
					has_variable = grossdomesticproduct
					check_variable = { grossdomesticproduct > 0 }
					CWIC_has_exceeded_maximum_divisions = yes
					CWIC_has_unattained_important_goals = no
					NOT = { has_country_flag = CWIC_has_reduce_units_check }
				}
				set_country_flag = CWIC_has_reduce_units_check
				country_event = { id = cwic_demob.2 days = 7 random_hours = 48 }
			}
		}
	}

	# --------------------------------------------------
	# On peace: trigger the AI demobilisation chain
	# Fires for the country that just made peace (ROOT)
	# --------------------------------------------------
	on_peace = {
		effect = {
			if = {
				limit = {
					is_ai = yes
					has_war = no
					has_variable = grossdomesticproduct
				}
				CWIC_unit_limit_calculation_effect = yes
				# Law downgrade check (fires once, chains internally)
				if = {
					limit = { NOT = { has_country_flag = CWIC_demob_law_check } }
					set_country_flag = { flag = CWIC_demob_law_check days = 30 value = 1 }
					country_event = { id = cwic_demob.1 days = 7 }
				}
				# Division disband loop (fires every 7 days until under limit)
				if = {
					limit = { NOT = { has_country_flag = CWIC_has_reduce_units_check } }
					set_country_flag = CWIC_has_reduce_units_check
					country_event = { id = cwic_demob.2 days = 7 random_hours = 48 }
				}
			}
		}
	}
}
