on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { country_exists = SOV }
				SOV = {
					if = {
						limit = {
							SOV_psych_enabled_trigger = yes
							NOT = { has_country_flag = SOV_psych_initialized }
							has_global_flag = cwic_bookmark_1980_start
						}
						SOV_psych_seed_1980 = yes
					}
					else_if = {
						limit = {
							SOV_psych_enabled_trigger = yes
							NOT = { has_country_flag = SOV_psych_initialized }
							SOV_psych_has_service_trigger = yes
							date > 1964.12.31
							SOV_psych_prague_file_ready_trigger = yes
						}
						SOV_psych_initialize = yes
					}
					if = {
						limit = { has_global_flag = cwic_bookmark_1980_start SOV_psych_enabled_trigger = yes }
						WP_psych_seed_1980_global = yes
					}
				}
			}
		}
	}

	on_monthly_SOV = {
		effect = {
			if = {
				limit = {
					SOV_psych_enabled_trigger = yes
					OR = {
						has_country_flag = SOV_psych_initialized
						has_global_flag = WP_psych_global_initialized
					}
				}
				WP_psych_reconcile_SOV = yes
				if = {
					limit = { check_variable = { global.WP_psych_directorate_state > 0 } check_variable = { global.WP_psych_directorate_state < 4 } }
					WP_psych_invite_eligible_members = yes
				}
			}
			if = {
				limit = {
					SOV_psych_enabled_trigger = yes
					SOV_psych_has_service_trigger = yes
					date > 1964.12.31
					SOV_psych_prague_file_ready_trigger = yes
					NOT = { has_country_flag = SOV_psych_initialized }
				}
				if = {
					limit = { has_global_flag = cwic_bookmark_1980_start }
					SOV_psych_seed_1980 = yes
				}
				else = {
					SOV_psych_initialize = yes
				}
			}
			if = {
				limit = {
					SOV_psych_charter_active_trigger = yes
					date > 1986.1.1
					NOT = { has_country_flag = SOV_psych_academy_asked }
				}
				set_country_flag = SOV_psych_academy_asked
				country_event = { id = cwic_sov_psych.3 }
			}
			if = {
				limit = {
					SOV_psych_enabled_trigger = yes
					# Any Soviet involvement ends here, not only a reviewed Prague file;
					# a directorate convened without the review was never archived.
					OR = {
						has_country_flag = SOV_psych_prague_reviewed
						has_country_flag = SOV_psych_initialized
						check_variable = { global.WP_psych_directorate_state > 0 }
					}
					date > 1991.8.1
					NOT = { has_country_flag = SOV_psych_epilogue_fired }
				}
				set_country_flag = SOV_psych_epilogue_fired
				if = {
					limit = {
						check_variable = { global.WP_psych_directorate_state > 0 }
						check_variable = { global.WP_psych_directorate_state < 4 }
					}
					WP_psych_archive_directorate = yes
					country_event = { id = cwic_wp_directorate.22 }
				}
				if = {
					limit = { SOV_psych_charter_active_trigger = yes }
					SOV_psych_close_charter = yes
				}
				country_event = { id = cwic_sov_psych.4 }
			}
		}
	}
}
