on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = { tag = USA is_ai = no }
				USA_open_congress_by_default = yes
			}
		}
	}
	on_daily = {
		effect = {
			# Fallback if on_startup ran before the player was assigned.
			if = {
				limit = { tag = USA is_ai = no NOT = { has_country_flag = usa_congress_opened_by_default } }
				USA_open_congress_by_default = yes
			}
			# Bills unlocked by focuses go before Congress automatically, one at a time.
			if = {
				limit = { tag = USA NOT = { has_country_flag = law_passing } }
				USA_start_next_congress_bill = yes
			}
			if = {
				limit = {
					tag = USA
					is_ai = no
					check_variable = { show_usa_congress = 1 }
					check_variable = { show_usa_congress_alt = 0 }
				}
				USA_refresh_congress_overview = yes
			}
		}
	}
}
