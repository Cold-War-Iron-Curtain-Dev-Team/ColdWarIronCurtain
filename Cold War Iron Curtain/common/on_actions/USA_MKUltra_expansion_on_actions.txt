# Expansion monthly pulse. The engine merges this on_monthly_USA with the one
# in USA_MKUltra_on_actions.txt. Everything called here is idempotent, so the
# reconcile hook can run the same effects without double-firing.
on_actions = {
	on_monthly_USA = {
		effect = {
			if = {
				limit = {
					USA_mkultra_enabled_trigger = yes
					has_country_flag = USA_mkultra_initialized
				}
				USA_mkultra_people_maintain = yes
				if = {
					limit = { date < 1976.1.1 }
					USA_mkultra_people_incidents = yes
				}
				if = {
					# Runs past the windows so the last open files are foreclosed.
					limit = { date < 1955.3.1 }
					USA_mkultra_subprojects_reconcile = yes
				}
			}
		}
	}
}
