on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { country_exists = CZE }
				CZE = {
					if = {
						limit = {
							CZE_psych_enabled_trigger = yes
							NOT = { has_country_flag = CZE_psych_initialized }
							has_global_flag = cwic_bookmark_1980_start
						}
						CZE_psych_seed_1980 = yes
					}
					else_if = {
						limit = {
							CZE_psych_enabled_trigger = yes
							NOT = { has_country_flag = CZE_psych_initialized }
							has_government = communism
							date > 1966.12.31
						}
						CZE_psych_initialize = yes
					}
				}
			}
			if = {
				limit = { country_exists = DDR }
				DDR = {
					if = {
						limit = {
							DDR_psych_enabled_trigger = yes
							NOT = { has_country_flag = DDR_psych_initialized }
							has_global_flag = cwic_bookmark_1980_start
						}
						DDR_psych_seed_1980 = yes
					}
					else_if = {
						limit = {
							DDR_psych_enabled_trigger = yes
							NOT = { has_country_flag = DDR_psych_initialized }
							has_government = communism
							DDR_psych_file_ready_trigger = yes
						}
						DDR_psych_initialize = yes
					}
				}
			}
			if = {
				limit = { country_exists = BUL }
				BUL = {
					if = {
						limit = {
							BUL_psych_enabled_trigger = yes
							NOT = { has_country_flag = BUL_psych_initialized }
							has_global_flag = cwic_bookmark_1980_start
						}
						BUL_psych_seed_1980 = yes
					}
					else_if = {
						limit = {
							BUL_psych_enabled_trigger = yes
							NOT = { has_country_flag = BUL_psych_initialized }
							has_government = communism
							date > 1965.12.31
						}
						BUL_psych_initialize = yes
					}
				}
			}
			if = {
				limit = { country_exists = POL }
				POL = {
					if = {
						limit = {
							POL_psych_enabled_trigger = yes
							NOT = { has_country_flag = POL_psych_initialized }
							has_global_flag = cwic_bookmark_1980_start
						}
						POL_psych_seed_1980 = yes
					}
					else_if = {
						limit = {
							POL_psych_enabled_trigger = yes
							NOT = { has_country_flag = POL_psych_initialized }
							has_government = communism
							POL_psych_file_ready_trigger = yes
						}
						POL_psych_initialize = yes
					}
				}
			}
		}
	}

	on_monthly_CZE = {
		effect = {
			if = { limit = { CZE_psych_enabled_trigger = yes has_country_flag = CZE_psych_initialized } WP_psych_reconcile_country = yes }
			if = {
				limit = {
					CZE_psych_enabled_trigger = yes
					has_government = communism
					date > 1966.12.31
					NOT = { has_country_flag = CZE_psych_initialized }
				}
				if = {
					limit = { has_global_flag = cwic_bookmark_1980_start }
					CZE_psych_seed_1980 = yes
				}
				else = {
					CZE_psych_initialize = yes
				}
			}
			if = {
				limit = {
					CZE_psych_charter_active_trigger = yes
					has_country_flag = prague_spring_begun
					NOT = { has_country_flag = CZE_psych_spring_asked }
				}
				set_country_flag = CZE_psych_spring_asked
				country_event = { id = cwic_wp_psych.3 }
			}
		}
	}

	on_monthly_DDR = {
		effect = {
			if = { limit = { DDR_psych_enabled_trigger = yes has_country_flag = DDR_psych_initialized } WP_psych_reconcile_country = yes }
			if = {
				limit = {
					DDR_psych_enabled_trigger = yes
					has_government = communism
					DDR_psych_file_ready_trigger = yes
					NOT = { has_country_flag = DDR_psych_initialized }
				}
				if = {
					limit = { has_global_flag = cwic_bookmark_1980_start }
					DDR_psych_seed_1980 = yes
				}
				else = {
					DDR_psych_initialize = yes
				}
			}
		}
	}

	on_monthly_BUL = {
		effect = {
			if = { limit = { BUL_psych_enabled_trigger = yes has_country_flag = BUL_psych_initialized } WP_psych_reconcile_country = yes }
			if = {
				limit = {
					BUL_psych_enabled_trigger = yes
					has_government = communism
					date > 1965.12.31
					NOT = { has_country_flag = BUL_psych_initialized }
				}
				if = {
					limit = { has_global_flag = cwic_bookmark_1980_start }
					BUL_psych_seed_1980 = yes
				}
				else = {
					BUL_psych_initialize = yes
				}
			}
			if = {
				limit = {
					BUL_psych_charter_active_trigger = yes
					date > 1981.7.1
					NOT = { has_country_flag = BUL_psych_audit_asked }
				}
				set_country_flag = BUL_psych_audit_asked
				country_event = { id = cwic_wp_psych.7 }
			}
		}
	}

	on_monthly_POL = {
		effect = {
			if = { limit = { POL_psych_enabled_trigger = yes has_country_flag = POL_psych_initialized } WP_psych_reconcile_country = yes }
			if = {
				limit = {
					POL_psych_enabled_trigger = yes
					has_government = communism
					POL_psych_file_ready_trigger = yes
					NOT = { has_country_flag = POL_psych_initialized }
				}
				if = {
					limit = { has_global_flag = cwic_bookmark_1980_start }
					POL_psych_seed_1980 = yes
				}
				else = {
					POL_psych_initialize = yes
				}
			}
		}
	}
}
