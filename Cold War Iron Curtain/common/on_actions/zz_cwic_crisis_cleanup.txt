# One-time cleanup for saves poisoned by the pre-2026-07 separatism drift bug.
# Resets passive-separatism state; safe on new games (everything is 0 anyway).
on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { NOT = { has_global_flag = cwic_sep_bugfix_cleanup_done } }
				set_global_flag = cwic_sep_bugfix_cleanup_done
				every_country = {
					set_variable = { CWIC_instability_index_corp_factor = 0 }
					set_variable = { CWIC_nationalization_pressure_index = 0 }
					set_variable = { CWIC_avg_separatism = 0 }
					set_variable = { CWIC_instability_index = 0 }
				}
				every_state = {
					if = {
						limit = { has_variable = separatism }
						set_variable = { separatism = 0 }
					}
				}
			}
		}
	}
}
