on_actions = {
	# Authored starting blocs, one pass at game start. UNS is the UN's own
	# country tag and always exists, so it is a safe scope to open from -
	# on_startup itself has none, which is what "Invalid Scope ... provided:
	# None" was complaining about.
	on_startup = {
		effect = {
			UNS = { cwic_un_blocs_startup = yes }
			UNS = { cwic_cold_war_embargo_startup = yes }
			# Seat a Security Council at once - the panel is on screen from day one.
			UNS = { cwic_un_council_startup = yes }
		}
	}
}
