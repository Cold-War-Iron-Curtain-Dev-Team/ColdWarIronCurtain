# ROOT is faction leader, FROM is faction member who opens UI and views goals, completed/cancel/complete_effect do not have FROM scopes
faction_goal_unlock_research = {
	name = faction_goal_unlock_research_name
	description = faction_goal_unlock_research_desc
	category = short_term
	group = FOCUS_FILTER_INTERNAL_AFFAIRS
	available = {
		FROM = { num_faction_members > 2 }
	}
	visible = {
		FROM = { has_faction_research_unlocked = no }
	}
	select_effect = {
		scope:faction = {
			set_temp_variable = {
				total_techs = {
					value = 0
					every_collection = {
						named_collection = faction_members
						add = {
							value = num_researched_technologies
							multiply = {
								value = 0
								if = { limit = { value = num_researched_technologies  less_than = 50 }  add = 1.25 }
								else_if = { limit = { value = num_researched_technologies  less_than = 75 }  add = 1.2 }
								else_if = { limit = { value = num_researched_technologies  less_than = 100 }  add = 1.15 }
								else_if = { limit = { value = num_researched_technologies  less_than = 150 }  add = 1.1 }
								else_if = { limit = { value = num_researched_technologies  less_than = 200 }  add = 1.05 }
							}
						}
					}
					round = yes
				}
			}
			set_variable = { unlock_military_techs_researched_target = total_techs }
		}
	}
	completed = {
		scope:faction = {
			hidden_trigger = {
				tooltip_evaluation = eval
				set_temp_variable = {
					unlock_military_techs_researched_compare_temp_target = {
						value = 0
						every_collection = {
							named_collection = faction_members
							add = num_researched_technologies
						}
					}
				}
				set_temp_variable = {
					unlock_military_techs_researched_compare = {
						value = 0
						every_collection = {
							named_collection = faction_members
							add = {
								value = num_researched_technologies
								multiply = {
									value = 0
									if = { limit = { value = num_researched_technologies  less_than = 50 }  add = 1.25 }
									else_if = { limit = { value = num_researched_technologies  less_than = 75 }  add = 1.2 }
									else_if = { limit = { value = num_researched_technologies  less_than = 100 }  add = 1.15 }
									else_if = { limit = { value = num_researched_technologies  less_than = 150 }  add = 1.1 }
									else_if = { limit = { value = num_researched_technologies  less_than = 200 }  add = 1.05 }
								}
							}
						}
						round = yes
					}
				}
			}

			if = {
				limit = {
					has_variable = unlock_military_techs_researched_target
				}
				check_variable = {
					var = unlock_military_techs_researched_compare_temp_target
					value = unlock_military_techs_researched_target
					compare = greater_than
					tooltip = "GOAL_T_TRIGGER_A"
				}
			}
			else = {
				check_variable = {
					value = unlock_military_techs_researched_compare
					var = unlock_military_techs_researched_compare_temp_target
					compare = greater_than
					tooltip = "GOAL_T_TRIGGER_A"
				}
			}
		}
	}
	complete_effect = {
		set_faction_research_unlocked = yes
		add_faction_initiative = 1
	}
	ai_will_do = {
		base = 1000 # They should always do this if they can
		modifier = {
			add = 1000
			original_tag = ENG
		}
	}
}

		}
