faction_goal_win_the_cold_war_1950s = {
	name = win_the_cold_war_1950s 
	description = win_the_cold_war_1950s_desc
	category = long_term
	group = FOCUS_FILTER_ANNEXATION
	allowed = {
		OR = {
			original_tag = SOV
			original_tag = USA
			is_subject_of = SOV
		}
	}
	visible = {
		date < 1960.1.1
	}
	available = {

	}
	completed = {
		if = {
			limit = { original_tag = USA }
			OR = {
				NOT = { country_exists = SOV }
				SOV = { has_capitulated = yes }
				SOV = { is_faction_leader = no }
			}
		}
		else = {
			OR = {
				NOT = { country_exists = USA }
				USA = { has_capitulated = yes }
				USA = { is_faction_leader = no }
			}
		}
	}
	complete_effect = {
		add_faction_power_projection = 150
		custom_effect_tooltip = generic_skip_one_line_tt
		custom_effect_tooltip = THE_FACTION_LEADER
		add_faction_initiative = 5
	}
	ai_will_do = {
		base = 0
	}
}
