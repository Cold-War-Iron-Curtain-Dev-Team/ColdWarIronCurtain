scripted_map_modes = {
	fallout_core_map_mode = {
		top = {
			type = state

			color = {
				FROM = {
					set_temp_variable = { alpha = 1.0 }
					if = {
						limit = {
							is_core_of = ROOT
						}
						if = {
							limit = {
								is_controlled_by = ROOT #is a core and controlled state of us.
							}
							set_temp_variable = { red = 0.0 }
							set_temp_variable = { green = 1 }
							set_temp_variable = { blue = 0.0 }
						}
						else = {
							set_temp_variable = { red = 0 } #is a core state of us.
							set_temp_variable = { green = 1 }
							set_temp_variable = { blue = 1 }
						}
					}
					else_if = {
						limit = {
							is_claimed_by = ROOT
						}
						if = {
							limit = {
								is_controlled_by = ROOT #is a claimed and controlled state of us.
							}
							set_temp_variable = { red = 0 }
							set_temp_variable = { green = 0.5 }
							set_temp_variable = { blue = 0.5 }
						}
						else = {
							set_temp_variable = { red = 1.0 } #is a claimed state of us.
							set_temp_variable = { green = 1.0 }
							set_temp_variable = { blue = 0 }
						}
					}
					else_if = {
						limit = {
							is_owned_by = ROOT
						}
						if = {
							limit = {
								is_controlled_by = ROOT #is a non core owned and controlled state of us.
							}
							set_temp_variable = { red = 0.5 }
							set_temp_variable = { green = 0 }
							set_temp_variable = { blue = 0.5 }
						}
						else = {
							set_temp_variable = { red = 1.0 }  #is a non core owned and not controlled state of us.
							set_temp_variable = { green = 0.0 }
							set_temp_variable = { blue = 0.0 }
						}
					}
					else = {
						set_temp_variable = { red = 1.0 } 
						set_temp_variable = { green = 1.0 }
						set_temp_variable = { blue = 0.0 }
					}
				}
			}

			thickness = no

			targets = {
				target_array = global.states
				target_trigger = {
					FROM = {
						OR = {
							is_owned_by = ROOT
							is_controlled_by = ROOT
							is_core_of = ROOT
							is_claimed_by = ROOT
						}
					}
				}
			}
		}

		bottom = {
			type = country

			color = {
				set_temp_variable = { alpha = 0.4 }
				set_temp_variable = { thickness = 10.0 }
				set_temp_variable = { red = 0.0 }
				set_temp_variable = { green = 0.0 }
				set_temp_variable = { blue = 0.0 }
			}

			thickness = yes

			targets = {
				targets = { ROOT }
			}
		}

		far_text = country
		near_text = state

		update_daily = no
	}

	fallout_stability_map_mode = {
		top = {
			type = country

			color = {
				FROM = {
					set_temp_variable = { alpha = 1.0 }

						set_temp_variable = { red = 0.0 }
						set_temp_variable = { green = 0.5 }
						set_temp_variable = { blue = 0.0 }

					else_if = {
						limit = {

						set_temp_variable = { green = 1.0 }
						set_temp_variable = { blue = 0.0 }

						}

					}
					else = {
						set_temp_variable = { red = 1.0 }
						set_temp_variable = { green = 0.0 }
						set_temp_variable = { blue = 0.0 }
					}
				}
			}

			thickness = yes
		}

			type = none
		}
