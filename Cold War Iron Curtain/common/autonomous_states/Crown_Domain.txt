# The northern crown domains (Tai Federation, Muong, Nung, Tho) answer to the
# CEFEO rather than to Paris. Modelled on autonomy_indon_union, with two
# deliberate differences:
#   - allowed on subjects of FRE, which is itself no longer a French puppet, so
#     a war against a domain drags in the corps and stops there instead of
#     walking the chain up to FRA and handing NATO a call to arms
#   - can_not_declare_war is dropped so the domains can be belligerents in
#     their own right in the Indochina campaigns
#   - they may join CEFEO's closed, no-call French Indochina faction
autonomy_state = {
	id = autonomy_crown_domain

	min_freedom_level = 0.4
	is_puppet = yes

	manpower_influence = 1.0

	rule = {
		can_boost_other_ideologies = no
		can_create_factions = no
		can_declare_war_on_same_ideology = no
		can_declare_war_without_wargoal_when_in_war = yes
		can_decline_call_to_war = yes
		can_force_government = no
		can_generate_female_aces = yes
		can_guarantee_other_ideologies = yes
		can_join_factions = yes	# only the CEFEO-led command admits them during the live theatre
		can_join_opposite_factions = no
		can_lower_tension = yes
		can_not_declare_war = no
		can_occupy_non_war = no
		can_only_justify_war_on_threat_country = no
		units_deployed_to_overlord = no
	}

	modifier = {
		autonomy_manpower_share = 0.35	# the domains raise the partisan auxiliaries the CEFEO fights alongside
		can_master_build_for_us = 1
		extra_trade_to_overlord_factor = 1.0
		overlord_trade_cost_factor = -0.75
	}

	use_overlord_color = yes

	ai_subject_wants_higher = {
		factor = 0.0
	}

	ai_overlord_wants_lower = {
		factor = 1.0
	}

	ai_overlord_wants_garrison = {
		always = yes
	}

	allowed = {
		has_dlc = "Together for Victory"
		NOT = { has_country_flag = UN_Mandate }
		is_subject_of = FRE
	}

	can_take_level = {
		always = no
	}

	can_lose_level = {
		always = no
	}
	allowed_levels_filter = {
	}
}
