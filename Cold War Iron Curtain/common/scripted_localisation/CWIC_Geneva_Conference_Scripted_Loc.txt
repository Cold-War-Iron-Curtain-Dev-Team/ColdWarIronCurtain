# Scripted localisation for the Geneva Conference data panel.
# Clause option names for each agenda item, conference status, and
# per-delegation table status.

defined_text = {
	name = GetGenevaStatusText
	text = {
		trigger = { has_global_flag = geneva_conference_in_session }
		localization_key = GENEVA_CONF_STATUS_SESSION
	}
	text = {
		trigger = {
			has_global_flag = geneva_conference_concluded
			check_variable = { global.geneva_outcome = 1 }
		}
		localization_key = GENEVA_CONF_STATUS_SETTLED
	}
	text = {
		trigger = {
			has_global_flag = geneva_conference_concluded
			check_variable = { global.geneva_outcome = 2 }
		}
		localization_key = GENEVA_CONF_STATUS_COLLAPSED
	}
	text = {
		localization_key = GENEVA_CONF_STATUS_NOT_CONVENED
	}
}
