# Deploy view: shows player's own division count vs limit
# Coloured green when under, red when over
defined_text = {
	name = CWIC_div_limit_count
	text = {
		trigger = { CWIC_has_exceeded_maximum_divisions = yes }
		localization_key = "CWIC_div_limit_count_over"
	}
	text = {
		localization_key = "CWIC_div_limit_count_ok"
	}
}
