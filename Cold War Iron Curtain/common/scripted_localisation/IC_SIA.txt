# Allows you to create your own dynamic keys
# to be called in localization.
# defined_text -> this is it, we're defining the text
# text -> a discrete entry that can be picked to display in loc.
# trigger -> determines if a text entry will be picked or not.
# (The triggers need to be valid for the scope the key is called in
# (eg Root or From.From).) 
# localization_key -> points to the localization key
# that'll be used if trigger passes
defined_text = { #
	name = SIA_Triangle_Cur_Selection
	text = {
		trigger = {
            has_country_flag = SIA_Triangle_Phibun_Selected_flag
		}
		localization_key = SIA_Triangle_Phibun_Selected_tt
	}
	text = {
		trigger = {
            has_country_flag = SIA_Triangle_Sarit_Selected_flag
		}
		localization_key = SIA_Triangle_Sarit_Selected_tt
	}
	text = {
		trigger = {
            has_country_flag = SIA_Triangle_Phao_Selected_flag
		}
		localization_key = SIA_Triangle_Phao_Selected_tt
	}
	text = {
		localization_key = SIA_Triangle_None_Selected_tt
	}
}
