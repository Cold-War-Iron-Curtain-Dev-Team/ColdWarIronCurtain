# Scripted localisation for the CEFEO (FRE) Metropole Patience readout.
# Bands mirror the thresholds fre_update_metropole_tier uses to swap the
# subsidy spirits (FRE_Indochina_Effects.txt); exhausted case first, fallback last.

defined_text = {
	name = GetFREMetropolePatience
	text = {
		trigger = { FRE = { check_variable = { FRE_Metropole_Patience < 0.01 } } }
		localization_key = FRE_PATIENCE_BAND_EXHAUSTED
	}
	text = {
		trigger = { FRE = { check_variable = { FRE_Metropole_Patience < 33 } } }
		localization_key = FRE_PATIENCE_BAND_CUT
	}
	text = {
		trigger = { FRE = { check_variable = { FRE_Metropole_Patience < 66 } } }
		localization_key = FRE_PATIENCE_BAND_REDUCED
	}
	text = {
		localization_key = FRE_PATIENCE_BAND_FULL
	}
}

defined_text = {
	name = GetFREOperationObjective
	text = {
		trigger = { FRE = { check_variable = { FRE_op_current = 2 } } }
		localization_key = FRE_OP_OBJ_HIRONDELLE
	}
	text = {
		trigger = { FRE = { check_variable = { FRE_op_current = 3 } } }
		localization_key = FRE_OP_OBJ_MOUETTE
	}
	text = {
		trigger = { FRE = { check_variable = { FRE_op_current = 4 } } }
		localization_key = FRE_OP_OBJ_BROCHET
	}
	text = {
		trigger = { FRE = { check_variable = { FRE_op_current = 5 } } }
		localization_key = FRE_OP_OBJ_BRETAGNE
	}
	text = {
		trigger = { FRE = { check_variable = { FRE_op_current = 6 } } }
		localization_key = FRE_OP_OBJ_ADOLPHE
	}
	text = {
		trigger = { FRE = { check_variable = { FRE_op_current = 7 } } }
		localization_key = FRE_OP_OBJ_CAMARGUE
	}
	text = {
		localization_key = FRE_OP_OBJ_NONE
	}
}

defined_text = {
	name = GetFREOperationContestStatus
	text = {
		trigger = {
			FRE = {
				OR = {
					check_variable = { FRE_op_current = 5 }
					check_variable = { FRE_op_current = 7 }
				}
				has_country_flag = FRE_Operation_Contested
			}
		}
		localization_key = FRE_OP_CONTEST_ENGAGED
	}
	text = {
		trigger = {
			FRE = {
				OR = {
					check_variable = { FRE_op_current = 5 }
					check_variable = { FRE_op_current = 7 }
				}
			}
		}
		localization_key = FRE_OP_CONTEST_WAITING
	}
	text = {
		localization_key = FRE_OP_CONTEST_NONE
	}
}
