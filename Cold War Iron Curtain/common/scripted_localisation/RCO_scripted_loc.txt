# Lists the player's open regime change targets (ROOT.RCO_targets,
# rebuilt by RCO_sync), one per line. Used by RCO_alert_tt.
defined_text = {
	name = RCO_target_list
	text = {
		trigger = { set_temp_variable = { rco_i = 0 } }
		localization_key = "[RCO_target_list_inner]"
	}
}
defined_text = {
	name = RCO_target_list_inner
	text = {
		trigger = {
			check_variable = { rco_i < RCO_targets^num }
			set_temp_variable = { rco_v = RCO_targets^rco_i }
			add_to_temp_variable = { rco_i = 1 }
		}
		localization_key = "RCO_target_line"
	}
	text = { localization_key = "" }
}
