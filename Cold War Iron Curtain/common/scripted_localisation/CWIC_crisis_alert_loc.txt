# Lists the country's active crises (CWIC_active_crises) one per line, with
# severity. Sets temp var i, which CWIC_ACTIVE_CRISIS_TITLE and
# CWIC_ACTIVE_CRISIS_SEVERITY read. Used by CWIC_crisis_alert_tt.
defined_text = {
	name = CWIC_crisis_alert_list
	text = {
		trigger = { set_temp_variable = { cwic_ca_i = 0 } }
		localization_key = "[CWIC_crisis_alert_list_inner]"
	}
}
defined_text = {
	name = CWIC_crisis_alert_list_inner
	text = {
		trigger = {
			check_variable = { cwic_ca_i < CWIC_active_crises^num }
			set_temp_variable = { i = cwic_ca_i }
			add_to_temp_variable = { cwic_ca_i = 1 }
		}
		localization_key = "CWIC_crisis_alert_line"
	}
	text = { localization_key = "" }
}
