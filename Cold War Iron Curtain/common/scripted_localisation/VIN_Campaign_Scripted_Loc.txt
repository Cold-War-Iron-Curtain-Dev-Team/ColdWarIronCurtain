defined_text = {
	name = GetVINCampaignHoldProgress
	text = {
		trigger = {
			OR = {
				check_variable = { VIN_camp_current = 5 }
				check_variable = { VIN_camp_current = 6 }
				check_variable = { VIN_camp_current = 7 }
				check_variable = { VIN_camp_current = 8 }
				check_variable = { VIN_camp_current = 10 }
				check_variable = { VIN_camp_current = 11 }
			}
		}
		localization_key = VIN_CAMP_HOLD_PROGRESS
	}
	text = { localization_key = VIN_CAMP_HOLD_PROGRESS_NONE }
}

defined_text = {
	name = GetVINSouthernAidStatus
	text = {
		trigger = {
			OR = {
				NOT = { country_exists = NLF }
				NOT = { date > 1950.1.1 }
				has_country_flag = VIN_Post_DBP_NLF_Stand_Down_Ordered
				has_country_flag = VIN_Post_DBP_Funded_NLF
				NLF = { has_war = no }
			}
		}
		localization_key = VIN_SOUTHERN_AID_INACTIVE
	}
	text = {
		trigger = { check_variable = { VIN_Campaign_Supply > 49 } }
		localization_key = VIN_SOUTHERN_AID_ACTIVE
	}
	text = { localization_key = VIN_SOUTHERN_AID_STARVED }
}
