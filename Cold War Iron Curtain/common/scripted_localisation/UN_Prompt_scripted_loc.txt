defined_text = {
	name = GetUNPromptDescription
	text = {
		trigger = {
			UNS = {
				has_country_flag = Current_UN_Vote
				NOT = { has_country_flag = UN_Vote_Cleanup }
			}
		}
		localization_key = UNPromptDesc
	}
	text = { localization_key = UNPromptVoteClosed }
}

# Where the vote stands. A resolution passes if Yes outnumbers No and no permanent member vetoes.
defined_text = {
	name = GetUNPromptVoteStatus
	text = {
		trigger = { check_variable = { UNS.un_veto_count > 0 } }
		localization_key = UNPromptStatusVetoed
	}
	text = {
		trigger = { check_variable = { UNS.un_yes_count > UNS.un_no_count } }
		localization_key = UNPromptStatusPassing
	}
	text = {
		trigger = { check_variable = { UNS.un_yes_count < UNS.un_no_count } }
		localization_key = UNPromptStatusFailing
	}
	text = { localization_key = UNPromptStatusTied }
}
