defined_text = {
	name = GetUNSeatRole
	text = {
		trigger = { has_idea = UN_Security_Council_Member_Perm }
		localization_key = UN_SEAT_PERMANENT
	}
	text = {
		localization_key = UN_SEAT_ELECTED
	}
}

defined_text = {
	name = GetUNSeatVote
	text = {
		trigger = { has_country_flag = UN_VETO_ONGOING }
		localization_key = UN_SEAT_VETO
	}
	text = {
		trigger = { has_country_flag = UN_YES_ONGOING }
		localization_key = UN_SEAT_YES
	}
	text = {
		trigger = { has_country_flag = UN_NO_ONGOING }
		localization_key = UN_SEAT_NO
	}
	text = {
		trigger = { has_country_flag = UN_ABSTAIN_ONGOING }
		localization_key = UN_SEAT_ABSTAIN
	}
	text = {
		trigger = {
			UNS = { has_country_flag = Current_UN_Vote }

	}
}

# A bloc leader who is not a UN member cannot carry anyone, and it is worth
# saying so before the player spends political power lobbying them.
defined_text = {
	name = GetUNBlocStanding
	text = {
		trigger = { NOT = { has_idea = un_member } }
		localization_key = UN_BLOC_NOT_MEMBER
	}
	text = {
		trigger = { has_country_flag = un_sc_elected }
		localization_key = UN_BLOC_ON_COUNCIL
	}
	text = {
		trigger = { has_idea = UN_Security_Council_Member_Perm }
		localization_key = UN_BLOC_PERMANENT
	}
	text = {
		localization_key = UN_BLOC_ASSEMBLY_ONLY
	}
}
