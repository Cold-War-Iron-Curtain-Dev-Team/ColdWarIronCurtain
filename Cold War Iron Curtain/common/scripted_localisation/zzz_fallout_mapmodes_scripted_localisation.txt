	}

		trigger = {
			is_claimed_by = ROOT
		}
		localization_key = fallout_core_map_mode_tt_claim
	}

		trigger = {
			is_owned_by = ROOT
		}
		localization_key = fallout_core_map_mode_tt_owned
	}

defined_text = {
	name = GetSeparatismLevel

	text = {
		trigger = {
			check_variable = { FROM.separatism > 49.9 } 
		}
		localization_key = separatism_map_mode_tooltip_separatism
	}
	text = {
		trigger = {
			check_variable = { FROM.separatism < 50 } 
		}
		localization_key = separatism_map_mode_tooltip_unrest
	}
	text = {
		trigger = {
			always = yes
		}
		localization_key = ""
	}
}

defined_text = {
	name = GetFalloutCoreMapModeTooltipDelayed
	text = {
		trigger = {
			is_owned_by = ROOT
			NOT = { is_controlled_by = ROOT }
		}
		localization_key = fallout_core_map_mode_tt_occupied_by_enemy
	}
	text = {
		trigger = {
			is_core_of = ROOT
			NOT = { is_owned_by = ROOT }
			is_controlled_by = ROOT
		}
		localization_key = fallout_core_map_mode_tt_core_not_owned_but_controlled
	}
	text = {
		trigger = {
			OR = {
				is_core_of = ROOT
				is_claimed_by = ROOT
			}
			NOT = { is_owned_by = ROOT }
		}
		localization_key = fallout_core_map_mode_tt_core_not_owned
	}
	text = {
		trigger = {
			is_controlled_by = ROOT
			NOT = { is_owned_by = ROOT }
		}
		localization_key = fallout_core_map_mode_tt_occupied_info
	}
	text = {
		trigger = {
			always = yes
		}
		localization_key = ""
	}
}

defined_text = {
	name = GetFalloutStabilityMapModeTooltipDelayed

	text = {
		trigger = {
			set_temp_variable = { temp = modifier@stability_weekly }
			set_temp_variable = { temp_mult = modifier@stability_weekly_factor }
			add_to_temp_variable = { temp_mult = 1 }
			multiply_temp_variable = { temp = temp_mult }
			NOT = { check_variable = { temp = 0 } }
		}
		localization_key = fallout_stability_map_mode_tt
	}
	text = {
		trigger = {
			always = yes
		}
		localization_key = ""
	}
}
