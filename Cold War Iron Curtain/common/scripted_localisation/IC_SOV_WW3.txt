defined_text = {
	name = get_paris_capital
	text = {
		trigger = {
			16 = {
				controller = {
					OR = {
						original_tag = SOV
						is_in_faction_with = SOV
					}
				}
			}
			NOT = {
				check_variable = { province_controllers^11506 = FRA }
			}
		}
		localization_key = paris_capital_green
	}
	text = {
		trigger = {
			FRA = {
				capital_scope = {
					is_controlled_by = FRA
				}
			}
		}
		localization_key = paris_capital_red
	}
}
