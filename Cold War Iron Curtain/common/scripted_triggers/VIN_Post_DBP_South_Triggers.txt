# Recorded DBP result plus a living, armed, ground-holding southern resistance at war.
vin_post_dbp_southern_war_active_trigger = {
	has_global_flag = geneva_dbp_outcome_applied
	NOT = { has_global_flag = Indochina_War_Over }
	NOT = { has_global_flag = geneva_conference_in_session }
	NOT = { has_global_flag = geneva_conference_concluded }
	country_exists = NLF
	NLF = {
		has_capitulated = no
		num_of_controlled_states > 0
		num_divisions > 0
	}
	OR = {
		AND = { country_exists = VIE  VIE = { has_war_with = NLF } }
		AND = { country_exists = FRE  FRE = { has_war_with = NLF } }
		AND = { country_exists = FRA  FRA = { has_war_with = NLF } }
	}
}

vin_post_dbp_southern_resistance_defeated_trigger = {
	OR = {
		NOT = { country_exists = NLF }
		NLF = { has_capitulated = yes }
		NLF = { num_of_controlled_states < 1 }
		NLF = { num_divisions < 1 }
	}
}

# The historical camp fall and a broken southern front leave no campaign to defer talks.
vin_post_dbp_auto_geneva_eligible_trigger = {
	has_global_flag = geneva_dbp_outcome_applied
	check_variable = { global.geneva_outcome_dbp = 2 }
	vin_post_dbp_southern_resistance_defeated_trigger = yes
	NOT = { has_global_flag = Indochina_War_Over }
	NOT = { has_global_flag = geneva_conference_announced }
	NOT = { has_global_flag = geneva_conference_in_session }
	NOT = { has_global_flag = geneva_conference_concluded }
	NOT = { has_country_flag = VIN_Post_DBP_Auto_Geneva_Launched }
}
