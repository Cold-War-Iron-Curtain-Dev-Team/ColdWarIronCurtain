# The faction provides alignment/access while no-call policy separates the wars.

indochina_french_command_active_trigger = {
	country_exists = FRE
	country_exists = VIE
	FRE = {
		has_country_flag = FRE_French_Indochina_Command_Established
		is_faction_leader = yes
	}
	VIE = { is_in_faction_with = FRE }
}

indochina_french_northern_command_member_trigger = {
	OR = {
		tag = FRE
		tag = TAI
		tag = NUN
		tag = TAM
		tag = THO
	}
	OR = {
		tag = FRE
		is_in_faction_with = FRE
	}
}

indochina_french_southern_command_member_trigger = {
	OR = {
		tag = VIE
		tag = CCC
		tag = FUL
	}
	is_in_faction_with = FRE
}

vin_nlf_coordination_active_trigger = {
	country_exists = VIN
	country_exists = NLF
	VIN = { has_country_flag = VIN_NLF_Coordination_Channel }
	NLF = {
		has_country_flag = NLF_Hanoi_Coordination
		is_in_faction_with = VIN
	}
}
