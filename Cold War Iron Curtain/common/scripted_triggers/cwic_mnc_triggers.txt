cwic_mnc_country_open_for_investment = {
	OR = {
		has_free_market_bonus_max = yes
		has_moderate_free_market_bonus = yes
		has_socialist_free_market_bonus = yes
		check_variable = { economic_x_axis > 12 }
		check_variable = { economic_x_axis = 12 }
	}
}

	has_country_flag = cwic_allows_foreign_concessions
}

	AND = {
		check_variable = { CWIC_corporate_risk_index < 175 }
		check_variable = { CWIC_nationalization_pressure_index < 4 }
		check_variable = { CWIC_instability_index < 2.5 }
	}
}
