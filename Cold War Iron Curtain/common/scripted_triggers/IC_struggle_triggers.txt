		AND = {
			check_variable = { global.geneva_outcome_highland > 1 }	# the highland recorder has gone against the federation

			1761 = { is_owned_by = VIN }	# Hoang Lien Son and Hoa Binh both gone: there is no northern position left

# =============================================================================
# Washington's three Indochina policies
# -----------------------------------------------------------------------------
# Three genuinely different American policies, each belonging to a different
# military situation. Each trigger below is a conjunction, not a grab-bag, and
# each applies to the AI and the player alike.
#
# Every clause reads a recorder that geneva_conf_world_leverage_pulse refreshes
# daily, so all three are safe in trigger context.
# =============================================================================

# --- Policy one: install our own man in Saigon -------------------------------
# Only worth doing while Cuong De is a live proposition: either he already sits
# in Saigon (the focus is recognition and money rather than a coup, and its
# payload branches accordingly) or his tree is running and Diem has not taken
# the succession.
USA_indochina_back_cuongde_trigger = {
	country_exists = VIE
	OR = {
		VIE = {
			has_country_leader = {
				character = VIE_Cuong_De
				ruling_only = yes
			}
		}
		AND = {
			VIE = { has_focus_tree = VIE_CuongDe_50 }
			NOT = { USA_indochina_diem_governs_trigger = yes }
		}
	}
}

# --- Policy two: back the war and press it harder ----------------------------
# Washington escalates when the war looks winnable: the southern insurgency is
# broken and Hanoi has taken nothing that matters. Deliberately the mirror of the
# settlement window - the two must never be true at the same time.
USA_indochina_direct_action_trigger = {
	OR = {
		has_global_flag = Southern_VietMinh_Defeated	# set by the NLF recorder in geneva_conf_world_leverage_pulse
		NOT = { country_exists = NLF }
		NLF = { num_of_controlled_states < 2 }
	}
	NOT = { check_variable = { global.geneva_outcome_dbp = 2 } }	# the camp has not fallen
	NOT = { check_variable = { global.geneva_outcome_highland > 0 } }	# no highland polity has gone over to Hanoi
	NOT = { 1766 = { is_owned_by = VIN } }	# Hoa Binh still stands
	NOT = { geneva_vie_side_losing_trigger = yes }	# and Saigon's camp is still ahead on points
}

# --- Policy three: cut the losses and get everyone to the table --------------
# Historically this followed a military catastrophe rather than a calendar.
# Dien Bien Phu alone is enough; short of that the collapse has to be general,
# so it takes two independent signals.
USA_indochina_settlement_window_trigger = {
	OR = {
		check_variable = { global.geneva_outcome_dbp = 2 }	# the camp at Dien Bien Phu fell - on its own, decisive, at any date
		AND = {
			date > 1953.1.1
			indochina_collapse_signal_count_trigger = yes
			indochina_collapse_second_signal_trigger = yes
		}
	}
}

# The two halves of the "at least two independent signals" test. Split so no
# single signal can satisfy both: the first covers what Hanoi has taken on the
# ground, the second covers whether the war itself is lost.
indochina_collapse_signal_count_trigger = {
	OR = {
		check_variable = { global.geneva_outcome_highland > 0 }	# Tai, Meo or Nung folded into Hanoi's orbit

		check_variable = { global.geneva_outcome_pathet_lao = 1 }	# the Pathet Lao overran the Kingdom outright
	}
}
