# Centralized Geneva Conference Trigger
# This is the single source of truth for when Geneva Conference should be available
indochina_struggle_ending_geneva_trigger = {
	# Geneva Conference can be triggered when:
	# 1. Phase reaches Low Tension (8) - de-escalation path
	# 2. OR date is after 1954.7.1 (historical trigger)
	# 3. AND struggle hasn't ended yet
	NOT = { has_global_flag = Indochina_War_Over }
	OR = {
		check_variable = { global.Indochina_War_Active_Phase = 8 }
		AND = {
			date > 1954.7.1
			check_variable = { global.Indochina_War_Active_Phase < 9 }
		}
	}
}

	NOT = { has_global_flag = Geneva_Conference }
