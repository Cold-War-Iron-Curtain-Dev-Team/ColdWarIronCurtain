indochina_struggle_ending_communist_victory_trigger = {
	NOT = { has_global_flag = Indochina_War_Over }
	NOT = { has_global_flag = Geneva_Conference }
	NOT = { has_global_flag = diem_ho_chi_minh_reunified }
	tag = VIN
	OR = {
		# Standard conditions: Score-based victory
		AND = {
			set_temp_variable = { StruggleInvolvedNationsIndochinaTotalAntiCommunistScoreTest = FRA.StruggleInvolvedNationsIndochinaTotalAntiCommunistScore }
			multiply_temp_variable = { StruggleInvolvedNationsIndochinaTotalAntiCommunistScoreTest = 2 }
			check_variable = { FRA.StruggleInvolvedNationsIndochinaCommunistScore > StruggleInvolvedNationsIndochinaTotalAntiCommunistScoreTest }
			check_variable = { FRA.StruggleInvolvedNationsIndochinaCommunistScore > 1000 }
		}
		AND = {
			has_war_with = VIE
			OR = {
				owns_state = 286
				VIE = {
					OR = {
						has_capitulated = yes
						exists = no
					}
				}
			}
		}
	}
}

indochina_struggle_ending_southern_victory_trigger = {
	NOT = { has_global_flag = Indochina_War_Over }
	NOT = { has_global_flag = Geneva_Conference }
	tag = VIE
	NOT = { has_government = communism }
	OR = {
		AND = {
			OR = {
				has_war_with = VIN
				AND = {
					has_opinion = {
						target = VIN
						value < -50
					}
					VIN = {
						has_opinion = {
							target = VIE
							value < -50
						}
					}
				}
			}
			OR = {
				check_variable = { global.Indochina_War_Active_Phase = 3 }
				check_variable = { global.Indochina_War_Active_Phase = 4 }
				check_variable = { global.Indochina_War_Active_Phase = 6 }
				check_variable = { global.Indochina_War_Active_Phase = 7 }
			}
			set_temp_variable = { StruggleInvolvedNationsIndochinaCommunistTest = FRA.StruggleInvolvedNationsIndochinaCommunistScore }
			multiply_temp_variable = { StruggleInvolvedNationsIndochinaCommunistTest = 2 }
			check_variable = { FRA.StruggleInvolvedNationsIndochinaTotalAntiCommunistScore > StruggleInvolvedNationsIndochinaCommunistTest }
			check_variable = { FRA.StruggleInvolvedNationsIndochinaProIndependenceScore > FRA.StruggleInvolvedNationsIndochinaProFranceScore }
			check_variable = { FRA.StruggleInvolvedNationsIndochinaProIndependenceScore > FRA.StruggleInvolvedNationsIndochinaProEthnicScore }
			check_variable = { FRA.StruggleInvolvedNationsIndochinaProIndependenceScore > 1000 }
		}
		AND = {
			has_war_with = VIN
			OR = {
				owns_state = 1760
				VIN = { has_capitulated = yes }
			}
		}
		# VIN already fully defeated and annexed: has_war_with can no longer be
		# true, so accept unification by ownership of both Hanoi and Saigon.
		AND = {
			NOT = { country_exists = VIN }
			owns_state = 1760
			owns_state = 286
		}
	}
}
