# Project MKNAOMI (Card 10), Project MKOFTEN (Card 11), Subproject 68 (Card 12),
# Project ARTICHOKE (Card 13), and Project GRILL FLAME (Card 14) triggers.

###########################################################################
# Card 10: Project MKNAOMI (1955-1970, CIA / Fort Detrick SOD)
###########################################################################

USA_mkultra_can_start_mknaomi_trigger = {
	USA_mkultra_research_start_allowed_trigger = yes
	USA_mkultra_has_free_slot_trigger = yes
	has_country_flag = USA_mkultra_mknaomi_unlocked
	check_variable = { USA_mkultra_mknaomi_state = 1 }
	NOT = { has_country_flag = USA_mkultra_mknaomi_paid }
	NOT = { has_country_flag = USA_mkultra_mknaomi_1970_order_fired }
}

USA_mkultra_can_afford_mknaomi_trigger = {
	if = {
		limit = { check_variable = { USA_mkultra_capacity = 2 } }
		has_political_power > 39
	}
	else = {
		has_political_power > 34
	}
}

USA_mkultra_ai_can_afford_mknaomi_trigger = {
	if = {
		limit = { USA_mkultra_major_crisis_trigger = yes }
		if = {
			limit = { check_variable = { USA_mkultra_capacity = 2 } }
			has_political_power > 189
		}
		else = {
			has_political_power > 184
		}
	}
	else = {
		if = {
			limit = { check_variable = { USA_mkultra_capacity = 2 } }
			has_political_power > 139
		}
		else = {
			has_political_power > 134
		}
	}
}

USA_mkultra_can_replicate_mknaomi_trigger = {
	USA_mkultra_research_start_allowed_trigger = yes
	check_variable = { USA_mkultra_mknaomi_state = 3 }
	NOT = { has_country_flag = USA_mkultra_mknaomi_replication_paid }
	NOT = { has_country_flag = USA_mkultra_mknaomi_archived }
	NOT = { has_country_flag = USA_mkultra_mknaomi_1970_order_fired }
}

USA_mkultra_can_apply_mknaomi_finding_trigger = {
	USA_mkultra_enabled_trigger = yes
	check_variable = { USA_mkultra_mknaomi_state = 7 }
	has_country_flag = USA_mkultra_mknaomi_replication_record_awarded
	NOT = { has_country_flag = USA_mkultra_mknaomi_finding_applied }
}

	USA_mkultra_enabled_trigger = yes

		check_variable = { USA_mkultra_artichoke_stage = 3 }

	}
	has_country_flag = USA_mkultra_artichoke_record_awarded

	NOT = { has_country_flag = USA_mkultra_artichoke_finding_applied }
