# Operation Pollux uses the actual Lai Chau-to-Dien Bien Phu line rather than
# a state target or arbitrary adjacent border. These helpers accept control by
# any member of the French-aligned command side.

fre_pollux_french_aligned_controls_lai_chau_trigger = {
	OR = {
		AND = { country_exists = FRE  FRE = { controls_province = 13765 } }
		AND = { country_exists = FRA  FRA = { controls_province = 13765 } }
		AND = { country_exists = TAI  TAI = { controls_province = 13765 } }
		AND = { country_exists = VIE  VIE = { controls_province = 13765 } }
	}
}

fre_pollux_french_aligned_controls_route_pavie_trigger = {
	OR = {
		AND = { country_exists = FRE  FRE = { controls_province = 13762 } }
		AND = { country_exists = FRA  FRA = { controls_province = 13762 } }
		AND = { country_exists = TAI  TAI = { controls_province = 13762 } }
		AND = { country_exists = VIE  VIE = { controls_province = 13762 } }
	}
}

fre_pollux_corridor_open_trigger = {
	fre_pollux_french_aligned_controls_lai_chau_trigger = yes
	fre_pollux_french_aligned_controls_route_pavie_trigger = yes
	fre_dbp_french_aligned_controls_camp_trigger = yes
}

fre_pollux_live_window_trigger = {
	NOT = { has_global_flag = Indochina_War_Over }
	NOT = { has_global_flag = geneva_conference_concluded }
	check_variable = { global.Indochina_War_Active_Phase < 9 }
	country_exists = VIN
	country_exists = TAI
	has_country_flag = FRE_Dien_Bien_Phu_Established
	VIN = { NOT = { has_country_flag = VIN_Campaign_Dien_Bien_Active } }
	fre_dbp_french_aligned_controls_camp_trigger = yes
}
