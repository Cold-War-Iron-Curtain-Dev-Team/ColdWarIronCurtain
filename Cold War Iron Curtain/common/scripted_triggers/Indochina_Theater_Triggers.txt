# The struggle is live and the router is not already running. Every entry point
# tests this; the routing latch is what makes the router safe to call from an
# on_action that its own payload will re-fire.
ic_failsafe_active_trigger = {
	has_global_flag = indochina_struggle_initialized
	has_global_flag = ic_failsafe_theatre_present
	NOT = { has_global_flag = Indochina_War_Over }
	NOT = { has_global_flag = ic_failsafe_routing }
	check_variable = { global.Indochina_War_Active_Phase < 100 }
}

# A conference that is genuinely running. Outranks the decisive rungs, the way
# the wrap-up timer always behaved.
ic_failsafe_geneva_live_trigger = {
	has_global_flag = Geneva_Conference
	OR = {
		has_global_flag = geneva_conference_announced
		has_global_flag = geneva_conference_in_session
	}
}

# The latch is set but nothing is running behind it and nothing ever concluded:
# the pipeline broke somewhere between unlock and announce. This is the state
# that makes the theatre unresolvable, and the only one the router clears.
ic_failsafe_geneva_latched_dead_trigger = {
	has_global_flag = Geneva_Conference
	NOT = { has_global_flag = geneva_conference_announced }
	NOT = { has_global_flag = geneva_conference_in_session }
	NOT = { has_global_flag = geneva_conference_concluded }
	NOT = { has_global_flag = Geneva_Conference_Negotiations_Complete }
}

# A principal has been knocked out, or a single non-French power holds both
# capitals. Holding Hanoi and Saigon is deliberately NOT decisive on its own for
# France: the CEFEO owning both while the Viet Minh is still in the field is an
# ordinary state of the war, not a result.
ic_failsafe_militarily_decided_trigger = {
	NOT = { has_global_flag = Indochina_War_Over }
	OR = {
		NOT = { country_exists = VIN }
		VIN = { has_capitulated = yes }
		NOT = { country_exists = VIE }
		VIE = { has_capitulated = yes }
		AND = {
			1760 = { owner = { owns_state = 286 } }
			1760 = {
				NOT = { is_owned_by = FRE }
				NOT = { is_owned_by = FRA }
			}
		}
	}
}

# Over-approximates on purpose: any human anywhere near the ending buys the long
# grace period rather than the short one.

	OR = {
		AND = { country_exists = VIN  VIN = { is_ai = no } }
		AND = { country_exists = VIE  VIE = { is_ai = no } }
		AND = { country_exists = FRA  FRA = { is_ai = no } }
		AND = { country_exists = FRE  FRE = { is_ai = no } }
		AND = { country_exists = PQC  PQC = { is_ai = no } }
		AND = { country_exists = USA  USA = { is_ai = no } }
		AND = { country_exists = CCC  CCC = { is_ai = no } }
		AND = { country_exists = FUL  FUL = { is_ai = no } }
	}
}

	NOT = { has_global_flag = Indochina_War_Over }
	OR = {
		date < 1954.7.21
		ic_failsafe_geneva_live_trigger = yes
		AND = {
			date < 1957.1.1
			country_exists = VIN
			VIN = { has_country_flag = VIN_Post_DBP_Focus_Geneva_Queued }
		}
		AND = {
			date < 1955.1.1
			OR = {
				ic_failsafe_vin_arc_unfinished_trigger = yes
				has_global_flag = Laos_Raid_Active
			}
		}
	}
}

# The router's preflight and postflight both move states and subjects. During a
# normal campaign lull they must not run merely because an unrelated on_action
# called the router while no theatre war happened to be live. Exceptional map
# breaks, earned endings and a dead carrier/conference may still route before
# the scripted arc's date guard expires.
ic_failsafe_route_may_mutate_trigger = {
	OR = {
		NOT = { ic_failsafe_theatre_unfinished_trigger = yes }
		ic_failsafe_militarily_decided_trigger = yes
		ic_failsafe_carrier_lost_trigger = yes
		ic_failsafe_ending_unclicked_trigger = yes
		ic_failsafe_geneva_latched_dead_trigger = yes
		has_global_flag = USA_VIN_Reunification_Complete
	}
}

ic_ending_communist_ok = {
	NOT = { has_global_flag = Indochina_War_Over }
	NOT = { has_global_flag = diem_ho_chi_minh_reunified }
	tag = VIN
	OR = {
		AND = {
			set_temp_variable = { ic_anticomm_x2 = global.ic_score_anticomm }
			multiply_temp_variable = { ic_anticomm_x2 = 2 }
			check_variable = { global.ic_score_comm > ic_anticomm_x2 }
			check_variable = { global.ic_score_comm > 1000 }
		}
		AND = {
			has_war_with = VIE
			OR = {
				owns_state = 286
				VIE = {
					OR = {
						has_capitulated = yes
						exists = no
					}
				}
			}
		}
	}
}

			check_variable = { global.ic_score_proind > global.ic_score_proeth }
			check_variable = { global.ic_score_proind > 1000 }
		}
		AND = {
			has_war_with = VIN

				owns_state = 1760

			}
		}
		AND = {
