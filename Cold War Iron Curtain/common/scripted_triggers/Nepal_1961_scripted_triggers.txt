### NEPAL 1961 CRISIS SCRIPTED TRIGGERS
### =========================================================

### Optional helper (recommended to run before trigger checks)
NEP_compute_margin = {
	set_temp_variable = { nep_margin = NEP_rank_1_value }
	subtract_from_temp_variable = { nep_margin = NEP_rank_2_value }
}


### =========================================================
### 👑 ROYAL CONSOLIDATION (PANCHAYAT COUP)
### =========================================================
NEP_trigger_royal_consolidation = {
	OR = {
		check_variable = { NEP_rank_1_id = 1 }
		AND = {
			check_variable = { NEP_rank_2_id = 1 }
			check_variable = { nep_margin < 5 }
		}
	}
	check_variable = { NEP_rank_1_value > 28 }
}

		check_variable = { NEP_rank_1_id = 3 }
