# Shared gates for the CEFEO set-piece operation focuses
# (Lorraine, Hirondelle, Mouette, Brochet), plus the separate Castor airhead.
# These used to be five verbatim ~40-line available blocks in
# FRE_50s_Indochina.txt; keep them in sync here instead.

# No operation currently in progress.
FRE_no_operation_running_trigger = {
	NOT = {
		OR = {
			has_country_flag = FRE_Limited_Operation_Pending
			has_country_flag = FRE_Limited_Operation_Active
			has_country_flag = FRE_Mission_Operation_Lorraine_Activated
			has_country_flag = FRE_Mission_Operation_Hirondelle_Activated
			has_country_flag = FRE_Mission_Operation_Mouette_Activated
			has_country_flag = FRE_Mission_Operation_Brochet_Activated
			has_country_flag = FRE_Mission_Operation_Castor_Activated
			has_country_flag = FRE_Mission_Operation_Pollux_Activated
			has_country_flag = FRE_Mission_Operation_Atlante_Activated
		}
	}
}

FRE_hirondelle_terminal_trigger = { OR = { has_country_flag = FRE_Operation_Hirondelle_Result_Clean has_country_flag = FRE_Operation_Hirondelle_Result_Costly has_country_flag = FRE_Operation_Hirondelle_Result_Aborted has_country_flag = FRE_Operation_Hirondelle_Result_Failure has_country_flag = FRE_Operation_Hirondelle_Result_Superseded } }
FRE_mouette_terminal_trigger = { OR = { has_country_flag = FRE_Operation_Mouette_Result_Clean has_country_flag = FRE_Operation_Mouette_Result_Costly has_country_flag = FRE_Operation_Mouette_Result_Aborted has_country_flag = FRE_Operation_Mouette_Result_Failure has_country_flag = FRE_Operation_Mouette_Result_Superseded } }
FRE_brochet_terminal_trigger = { OR = { has_country_flag = FRE_Operation_Brochet_Result_Clean has_country_flag = FRE_Operation_Brochet_Result_Costly has_country_flag = FRE_Operation_Brochet_Result_Aborted has_country_flag = FRE_Operation_Brochet_Result_Failure has_country_flag = FRE_Operation_Brochet_Result_Superseded } }

FRE_limited_operation_window_trigger = {
	NOT = { has_global_flag = Indochina_War_Over }
	NOT = { has_global_flag = geneva_conference_concluded }
	check_variable = { global.Indochina_War_Active_Phase < 9 }
	country_exists = VIN
	NOT = { has_war_with = VIN }	# each renovated operation must own a fresh, isolated war
	VIN = { VIN_no_campaign_running_trigger = yes }
}

# Exact-objective operation predicates. These are triggers, not effects: they
# are consumed inside `limit` blocks by the daily operation resolver.
fre_operation_raw_objective_met = {
	OR = {
		AND = {
			check_variable = { FRE_op_current = 2 }
			OR = { controls_province = 9948 FRA = { controls_province = 9948 } AND = { country_exists = TAI TAI = { controls_province = 9948 } } AND = { country_exists = VIE VIE = { controls_province = 9948 } } }
			OR = { controls_province = 13761 FRA = { controls_province = 13761 } AND = { country_exists = TAI TAI = { controls_province = 13761 } } AND = { country_exists = VIE VIE = { controls_province = 13761 } } }
		}
		AND = { check_variable = { FRE_op_current = 3 } OR = { controls_province = 11909 FRA = { controls_province = 11909 } AND = { country_exists = TAI TAI = { controls_province = 11909 } } AND = { country_exists = VIE VIE = { controls_province = 11909 } } } }
		AND = { check_variable = { FRE_op_current = 4 } OR = { controls_province = 13755 FRA = { controls_province = 13755 } AND = { country_exists = TAI TAI = { controls_province = 13755 } } AND = { country_exists = VIE VIE = { controls_province = 13755 } } } }
	}
}

fre_operation_objective_met = {
	OR = {
		AND = { check_variable = { FRE_op_current = 2 } check_variable = { FRE_op_objective_hold_days > 2 } }
		AND = { check_variable = { FRE_op_current = 3 } check_variable = { FRE_op_objective_hold_days > 4 } }
		AND = { check_variable = { FRE_op_current = 4 } check_variable = { FRE_op_days_elapsed > 34 } fre_operation_raw_objective_met = yes }
	}
}

fre_operation_theatre_closed = {
	OR = {
		has_global_flag = Indochina_War_Over
		has_global_flag = geneva_conference_concluded
		check_variable = { global.Indochina_War_Active_Phase > 8 }
		NOT = { country_exists = VIN }
	}
}

# No operation resolved within the last 60 days.
FRE_operation_cooldown_ready_trigger = {
	NOT = {
		OR = {
			AND = {
				has_country_flag = FRE_Operation_Cooldown_Lorraine
				NOT = { has_country_flag = { flag = FRE_Operation_Cooldown_Lorraine days > 59 } }
			}
			AND = {
				has_country_flag = FRE_Operation_Cooldown_Hirondelle
				NOT = { has_country_flag = { flag = FRE_Operation_Cooldown_Hirondelle days > 59 } }
			}
			AND = {
				has_country_flag = FRE_Operation_Cooldown_Mouette
				NOT = { has_country_flag = { flag = FRE_Operation_Cooldown_Mouette days > 59 } }
			}
			AND = {
				has_country_flag = FRE_Operation_Cooldown_Brochet
				NOT = { has_country_flag = { flag = FRE_Operation_Cooldown_Brochet days > 59 } }
			}
			AND = {
				has_country_flag = FRE_Operation_Cooldown_Castor
				NOT = { has_country_flag = { flag = FRE_Operation_Cooldown_Castor days > 59 } }
			}
			AND = {
				has_country_flag = FRE_Operation_Cooldown_Atlante
				NOT = { has_country_flag = { flag = FRE_Operation_Cooldown_Atlante days > 59 } }
			}
		}
	}
}

# The AI is banking political power for a pending CEFEO mission.
#
# The prep missions activate on one focus flag but only become available on a
# second one from a later focus, and HOI4's decision AI does not save up. This
# gate zeroes the competing decisions until the reserve is intact. Keyed on
# _Activated rather than on availability precisely so the AI banks through that
# gap; human players are never gated, since the reserve lives in ai_will_do.
FRE_mission_pp_reserved_trigger = {
	is_ai = yes
	check_variable = { political_power < 160 }
	OR = {
		AND = {
			has_country_flag = FRE_Mission_Assess_RC4_Activated
			NOT = { has_country_flag = FRE_Mission_Assess_RC4_Completed }
		}
		AND = {
			has_country_flag = FRE_Mission_Address_Revers_Report_Activated
			NOT = { has_country_flag = FRE_Mission_Address_Revers_Report_Completed }
		}
		AND = {
			has_country_flag = FRE_Mission_De_Lattre_Arrival_Activated
			NOT = { has_country_flag = FRE_Mission_De_Lattre_Arrival_Completed }
		}
		AND = {
			has_country_flag = FRE_Mission_Prepare_Na_San_Activated
			NOT = { has_country_flag = FRE_Mission_Prepare_Na_San_Completed }
		}
		AND = {
			has_country_flag = FRE_Mission_Prepare_Vinh_Yen_Activated
			NOT = { has_country_flag = FRE_Mission_Prepare_Vinh_Yen_Completed }
		}
		AND = {
			has_country_flag = FRE_Mission_Prepare_Mao_Khe_Activated
			NOT = { has_country_flag = FRE_Mission_Prepare_Mao_Khe_Completed }
		}
	}
}

# An operation focus is waiting on a war chest the AI cannot yet afford.
# Used to push the extraordinary-credits decision up the AI's list, since the
# operations are paid in War Credits and starving that starves the whole chain.
FRE_operation_credits_short_trigger = {
	is_ai = yes
	check_variable = { FRE_War_Credits < 120 }
	OR = {
		AND = { has_country_flag = FRE_Operation_Hirondelle_Launched NOT = { FRE_hirondelle_terminal_trigger = yes } }
		AND = { has_country_flag = FRE_Operation_Mouette_Launched NOT = { FRE_mouette_terminal_trigger = yes } }
		AND = { has_country_flag = FRE_Operation_Brochet_Launched NOT = { FRE_brochet_terminal_trigger = yes } }
	}
}
