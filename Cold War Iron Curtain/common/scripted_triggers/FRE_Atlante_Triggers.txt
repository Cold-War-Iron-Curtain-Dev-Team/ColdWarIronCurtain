fre_atlante_french_aligned_controls_north_objective_trigger = {
	OR = {
		AND = { country_exists = FRE  FRE = { controls_province = 4255 } }
		AND = { country_exists = FRA  FRA = { controls_province = 4255 } }
		AND = { country_exists = VIE  VIE = { controls_province = 4255 } }
	}
}

fre_atlante_french_aligned_controls_south_objective_trigger = {
	OR = {
		AND = { country_exists = FRE  FRE = { controls_province = 1300 } }
		AND = { country_exists = FRA  FRA = { controls_province = 1300 } }
		AND = { country_exists = VIE  VIE = { controls_province = 1300 } }
	}
}

fre_atlante_full_objective_trigger = {
	fre_atlante_french_aligned_controls_north_objective_trigger = yes
	fre_atlante_french_aligned_controls_south_objective_trigger = yes
}
