# Outcome gates for the FRE_50s_Indochina finale focuses.
# Scores live on FRA scope (see FRE focus rewards, which write FRA = { add_to_variable ... }).
# These three regions partition the score-space so exactly one ending is always valid:
#   Victory : AntiCommunist decisively dominant
#   Defeat  : Communist decisively dominant
#   Geneva  : everything in between (NOT victory AND NOT defeat)

# Decisive anti-communist result -> Proclaim Victory in Indochina
FRE_indochina_military_victory_trigger = {
	check_variable = { FRA.StruggleInvolvedNationsIndochinaTotalAntiCommunistScore > 1000 }
	set_temp_variable = { fre_ending_comm_x2 = FRA.StruggleInvolvedNationsIndochinaCommunistScore }
	multiply_temp_variable = { fre_ending_comm_x2 = 2 }
	check_variable = { FRA.StruggleInvolvedNationsIndochinaTotalAntiCommunistScore > fre_ending_comm_x2 }
	FRE_indochina_ground_situation_favourable_trigger = yes
}

# The scores are a political ledger, not a map, so a proclamation of victory has
# to survive the map as well as the ledger: none of the ground below may have
# gone over to Hanoi. Nothing is locked out by this - the finales partition the
# space, so a failure here routes to FRE_The_Geneva_Accords.
FRE_indochina_ground_situation_favourable_trigger = {
	NOT = { check_variable = { global.geneva_outcome_dbp = 2 } }	# the camp fell
	NOT = { has_global_flag = Indochina_Geneva_Panic }	# Paris was panicked to the table by a collapse
	country_exists = VIE
	671 = { NOT = { vin_aligned_state_trigger = yes } }	# Tay Bac Bo and the valley
	1761 = { NOT = { vin_aligned_state_trigger = yes } }	# Hoang Lien Son
	1766 = { NOT = { vin_aligned_state_trigger = yes } }	# Hoa Binh
}

# Decisive communist result -> The Fall of Saigon
FRE_indochina_defeat_trigger = {
	check_variable = { FRA.StruggleInvolvedNationsIndochinaCommunistScore > 1000 }
	set_temp_variable = { fre_ending_anti_x2 = FRA.StruggleInvolvedNationsIndochinaTotalAntiCommunistScore }
	multiply_temp_variable = { fre_ending_anti_x2 = 2 }
	check_variable = { FRA.StruggleInvolvedNationsIndochinaCommunistScore > fre_ending_anti_x2 }
}

# Stalemate / negotiated outcome -> The Geneva Accords (always the fallback)
FRE_indochina_stalemate_trigger = {
	NOT = { FRE_indochina_military_victory_trigger = yes }
	NOT = { FRE_indochina_defeat_trigger = yes }
}
