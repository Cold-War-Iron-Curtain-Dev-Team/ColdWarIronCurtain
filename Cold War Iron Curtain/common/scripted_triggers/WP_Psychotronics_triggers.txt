CZE_psych_enabled_trigger = {
	original_tag = CZE
	NOT = {
		has_game_rule = {
			rule = WP_PSYCHOTRONICS_MODE
			option = WP_PSYCHOTRONICS_OFF
		}
	}
}

CZE_psych_charter_active_trigger = {
	CZE_psych_enabled_trigger = yes
	check_variable = { CZE_psych_charter_state = 1 }
}

CZE_psych_can_open_trigger = {
	CZE_psych_enabled_trigger = yes
	has_government = communism
	date > 1966.12.31
	check_variable = { CZE_psych_charter_state = 0 }
	NOT = { has_country_flag = CZE_psych_shelved }
	OR = {
		NOT = { has_country_flag = WP_psych_directorate_participant }
		WP_psych_can_start_joint_work_trigger = yes
	}
}

CZE_psych_can_commission_trigger = {
	CZE_psych_charter_active_trigger = yes
	check_variable = { CZE_psych_demo_stage = 1 }
	check_variable = { CZE_psych_slot = 0 }
	NOT = { has_country_flag = CZE_psych_shelved }
	NOT = { has_country_flag = CZE_psych_demo_paid }
	OR = {
		NOT = { has_country_flag = WP_psych_directorate_participant }
		WP_psych_can_start_joint_work_trigger = yes
	}
}

CZE_psych_can_archive_trigger = {
	CZE_psych_charter_active_trigger = yes
	check_variable = { CZE_psych_demo_stage = 3 }
	NOT = { has_country_flag = CZE_psych_shelved }
	NOT = { has_country_flag = CZE_psych_claim_rejected }
}

CZE_psych_can_forward_trigger = {

	NOT = { has_country_flag = CZE_psych_shelved }
