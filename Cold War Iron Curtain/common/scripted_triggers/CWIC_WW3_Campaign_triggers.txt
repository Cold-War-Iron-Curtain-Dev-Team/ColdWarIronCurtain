# Country scope. Political successors created by the WW3 capitulation scripts
# sometimes join the Soviet sphere without joining the native faction immediately.
cwic_ww3_soviet_aligned = {
    OR = {
        original_tag = SOV
        is_in_faction_with = SOV
        is_subject_of = SOV
        AND = {
            has_idea = Warsaw_Treaty_Organization_member
            check_variable = { Sphere_Leader = SOV.id }
        }
    }
}

cwic_ww3_bloc_at_war = {
    any_country = {
        cwic_ww3_soviet_aligned = yes
        any_enemy_country = {
            has_country_flag = cwic_ww3_opponent
            NOT = { cwic_ww3_soviet_aligned = yes }
        }
    }
}

cwic_ww3_north_america_secured = {
    custom_trigger_tooltip = {
        tooltip = CWIC_WW3_WASHINGTON_TT
        SOV = {
            OR = {
                controls_province = 13344
                any_country = { cwic_ww3_soviet_aligned = yes controls_province = 13344 }
            }
        }
    }
    custom_trigger_tooltip = {
        tooltip = CWIC_WW3_US_DEFEATED_TT
        OR = {
            has_country_flag = cwic_ww3_us_defeated
            USA = { has_capitulated = yes }
            USA = { cwic_ww3_soviet_aligned = yes }
            NOT = { country_exists = USA }
        }
    }
}

    has_global_flag = WW3_Started
