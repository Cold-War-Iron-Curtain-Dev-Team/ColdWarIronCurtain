FRE_highlands_window_open_trigger = {
	NOT = { has_global_flag = Indochina_War_Over }
	NOT = { has_global_flag = geneva_conference_concluded }
	check_variable = { global.Indochina_War_Active_Phase < 9 }
	country_exists = NLF
}

FRE_highlands_dbp_terminal_trigger = {
	country_exists = VIN
	VIN = {
		OR = {
			has_country_flag = VIN_Campaign_Dien_Bien_Result_Clean
			has_country_flag = VIN_Campaign_Dien_Bien_Result_Costly
			has_country_flag = VIN_Campaign_Dien_Bien_Result_Aborted
			has_country_flag = VIN_Campaign_Dien_Bien_Result_Failure
			has_country_flag = VIN_Campaign_Dien_Bien_Result_Superseded
		}
	}
}

FRE_highlands_mang_yang_due_trigger = {
	FRE_highlands_window_open_trigger = yes
	NOT = { has_global_flag = FRE_Highlands_Mang_Yang_Launched }
	date > 1954.5.31
	OR = {
		FRE_highlands_dbp_terminal_trigger = yes
		date > 1954.6.30
	}
}

FRE_highlands_chu_dreh_due_trigger = {
	FRE_highlands_window_open_trigger = yes
	NOT = { has_global_flag = FRE_Highlands_Chu_Dreh_Launched }
	OR = {
		date > 1954.7.9
		AND = {
			has_global_flag = FRE_Highlands_Mang_Yang_Resolved
			check_variable = { FRE_Highlands_Aftermath_Days > 13 }
		}
	}
}

FRE_highlands_enemy_holds_dak_doa_trigger = {
	OR = {
		NLF = { controls_province = 1328 }
		AND = { country_exists = VIN VIN = { controls_province = 1328 } }
	}
}

FRE_highlands_enemy_holds_pleiku_trigger = {
	OR = {
		NLF = { controls_province = 16442 }
		AND = { country_exists = VIN VIN = { controls_province = 16442 } }
	}
}

FRE_highlands_dak_doa_nlf_result_trigger = {
	NLF = {
		OR = {
			has_country_flag = VIN_Campaign_Dak_Doa_Result_Clean
			has_country_flag = VIN_Campaign_Dak_Doa_Result_Costly
		}
	}
}
