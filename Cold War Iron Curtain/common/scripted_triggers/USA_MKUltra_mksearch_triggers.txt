# MKSEARCH continuation state gates. Full pre-click affordability is checked by
# the decisions; effects reuse these state-only triggers after native cost payment.

USA_mkultra_mksearch_window_open_trigger = {
	has_country_flag = USA_mkultra_mksearch_remit
	NOT = { has_country_flag = USA_mkultra_mksearch_funding_ended }
}

USA_mkultra_can_start_mksearch_trigger = {
	USA_mkultra_research_start_allowed_trigger = yes
	USA_mkultra_mksearch_window_open_trigger = yes
	USA_mkultra_has_free_slot_trigger = yes
	check_variable = { USA_mkultra_mksearch_stage = 1 }
	NOT = { has_country_flag = USA_mkultra_mksearch_paid }
}

USA_mkultra_can_replicate_mksearch_trigger = {
	USA_mkultra_research_start_allowed_trigger = yes
	USA_mkultra_mksearch_window_open_trigger = yes
	check_variable = { USA_mkultra_mksearch_stage = 3 }
	NOT = { has_country_flag = USA_mkultra_mksearch_replication_paid }
	NOT = { has_country_flag = USA_mkultra_mksearch_archived }
}

USA_mkultra_can_apply_mksearch_inventory_trigger = {
	USA_mkultra_enabled_trigger = yes
	check_variable = { USA_mkultra_mksearch_stage = 5 }
	has_country_flag = USA_mkultra_mksearch_replication_record_awarded
	NOT = { has_country_flag = USA_mkultra_mksearch_inventory_earned }
}

USA_mkultra_mksearch_stage_open_trigger = {
	OR = {
		check_variable = { USA_mkultra_mksearch_stage = 2 }
		check_variable = { USA_mkultra_mksearch_stage = 3 }
		check_variable = { USA_mkultra_mksearch_stage = 4 }
		check_variable = { USA_mkultra_mksearch_stage = 5 }
	}
}
