# Phase 3: Capacity & Slot Triggers
USA_mkultra_has_free_slot_trigger = {
	OR = {
		check_variable = { USA_mkultra_slot_1_state = 0 }
		AND = {
			check_variable = { USA_mkultra_capacity = 2 }
			check_variable = { USA_mkultra_slot_2_state = 0 }
		}
	}
}

	has_country_flag = USA_mkultra_missing_baseline_record_awarded
	OR = {
		has_country_flag = USA_mkultra_countermeasure_record_awarded
		has_country_flag = USA_mkultra_midnight_climax_record_awarded
		has_country_flag = USA_mkultra_hypnosis_record_awarded
		has_country_flag = USA_mkultra_behavioral_record_awarded
		has_country_flag = USA_mkultra_mksearch_record_awarded
		has_country_flag = USA_mkultra_missing_baseline_renewal_record_awarded
		has_country_flag = USA_mkultra_missing_baseline_replication_record_awarded
	}
}
