# Southern command contract: exactly one of the five leader flags is set on NLF at a time.
NLF_southern_command_settled_trigger = {
	OR = {
		has_country_flag = Nguyen_Binh_Leader
		has_country_flag = Le_Duan_Leader
		has_country_flag = Le_Duc_Tho_Leader
		has_country_flag = Tran_quoc_Buu_Leader
		has_country_flag = Kieu_Cong_Cung_Leader
	}
}

NLF_orthodox_southern_command_trigger = {
	OR = {
		has_country_flag = Le_Duan_Leader
		has_country_flag = Le_Duc_Tho_Leader
	}
}

NLF_united_front_southern_command_trigger = {
	OR = {
		has_country_flag = Nguyen_Binh_Leader
		has_country_flag = Tran_quoc_Buu_Leader
		has_country_flag = Kieu_Cong_Cung_Leader
	}
}

# The 1951 recall window only means anything while the southern front still exists.
VIN_binh_course_live_trigger = {
	country_exists = NLF
	NOT = { has_global_flag = Indochina_War_Over }
}

VIN_binh_outcome_recorded_trigger = {
	OR = {
		has_country_flag = Binh_Outcome
		AND = {
			country_exists = NLF
			NLF = { has_country_flag = Binh_Outcome }
		}
	}
}
