###########################################################################
# Regime Change Opportunities (RCO) - triggers
#
# An opportunity is open when the TARGET country carries its flag
# (RCO_<sender>_<name>), raised by any content: event, focus, decision...
# See the guide at the top of events/RCO_events.txt.
#
# Scope: THIS = target country, ROOT = sender (SOV or USA).
#
# ADDING AN OPPORTUNITY: add one line here, under its sender,
# and add that country's tag to RCO_sync in RCO_effects.txt.
###########################################################################

# Any opportunity open on THIS for ROOT - drives the diplomacy button and the alert.
RCO_any = {
	OR = {
		AND = {
			ROOT = { original_tag = SOV }
			OR = {
				has_country_flag = RCO_sov_greece
				has_country_flag = RCO_sov_china
				has_country_flag = RCO_sov_wgr49
				has_country_flag = RCO_sov_stalin_note
				has_country_flag = RCO_sov_yug
			}
		}
		AND = {
			ROOT = { original_tag = USA }
			OR = {
				always = no		# first US opportunity goes here
			}
		}
	}
}
