###########################################################################
# Regime Change Opportunities (RCO) - one trigger per opportunity
#
# Scope: THIS = target country, ROOT = sender.
# Each trigger = "the button should exist on this target right now":
#   target tag check + ROOT sender tag check + ROOT "not used" flag
#   + live conditions.
# The "done" flag is on the sender, so once used the button vanishes from
# every target of that opportunity at once (PDG and GRE, etc.).
#
# RULES
# - Two triggers must NEVER be true for the same target at the same time
#   (RCO_fire only runs the first matching branch).
#   Only shared target today: WGR, used by RCO_SOV_wgr49 (date < 1951)
#   and RCO_SOV_stalin_note (date > 1952) - the date windows keep them
#   apart. Check this again whenever a new trigger reuses a target.
# - Never put a bare "if" inside an OR here (it is true when its limit is
#   false, so the whole OR passes).
#
# ADDING AN OPPORTUNITY:
#   1. write its trigger here
#   2. add one line to RCO_any below
#   3. add one else_if in RCO_fire (scripted_effects/RCO_effects.txt)
###########################################################################

# --- Greece: Percentage agreement ------------------------------------------
RCO_SOV_greece = {
	OR = {
		original_tag = PDG
		original_tag = GRE
	}
	ROOT = {
		original_tag = SOV
		NOT = { has_country_flag = SOV_Percentage_agreement }
	}
	country_exists = PDG
	PDG = { surrender_progress > 0.4 }
}

# --- China: Manchurian intervention ----------------------------------------
RCO_SOV_china = {
	OR = {
		original_tag = PRC
		original_tag = CHI
	}
	ROOT = {
		original_tag = SOV
		NOT = { has_country_flag = SOV_Manchurian_Interventon }
	}
	country_exists = PRC
	PRC = { surrender_progress > 0.79 }
}

# --- West Germany 1949 ----------------------------------------------------
RCO_SOV_wgr49 = {
	original_tag = WGR
	ROOT = {
		original_tag = SOV
		NOT = { has_country_flag = SOV_WGR49_interfered }
	}
	date < 1951.1.1
}

# --- Stalin Note ----------------------------------------------------------
RCO_SOV_stalin_note = {
	OR = {
		original_tag = WGR
		original_tag = DDR
	}
	ROOT = {
		original_tag = SOV
		NOT = { has_country_flag = SOV_Stalin_Note }
		has_country_leader = { character = SOV_Joseph_Stalin ruling_only = yes }
	}
	date > 1952.1.1
	NOT = {
		WGR = { has_country_leader = { character = WGR_Maximillian_Reimann ruling_only = yes } }
	}
}

# --- Yugoslavia (was already in aaa_WoO_scripted_triggers.txt) ------------
RCO_SOV_Yug = {
	original_tag = YUG
	ROOT = {
		original_tag = SOV
		NOT = { has_country_flag = SOV_YUG_Invasion }
	}
	date < 1953.1.1
	country_exists = PDG
	PDG = { owns_state = 47 }
}

###########################################################################
# Aggregate - one line per opportunity
###########################################################################
RCO_any = {
	OR = {
		RCO_SOV_greece = yes
		RCO_SOV_china = yes
		RCO_SOV_wgr49 = yes
		RCO_SOV_stalin_note = yes
		RCO_SOV_Yug = yes
	}
}
