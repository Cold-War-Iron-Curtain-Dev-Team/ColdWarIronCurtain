# ============================================================
# CWIC Macro Crisis Scripted Triggers (v2)
# First-pass canonical crisis checks using existing econ variables
# where available. Kept intentionally modular so compound crises,
# monthly evaluators, and modifiers can all reference the same names.
#
# Notes:
# - Several crises are fully grounded in live variables already present
#   in your uploaded files: fiscal, inflation, finance, governance,
#   crop failure, and food security.
# - Economic contraction still wants a dedicated real GDP growth / GDPpc
#   trend variable to be truly clean. Placeholder support is included.
# - Energy crisis now references real oil/electricity demand and food/power
#   pulse variables, but still lacks a perfect electricity production balance
#   variable in the uploaded scripts.
# ============================================================


# ------------------------------------------------------------
# Shared helpers
# ------------------------------------------------------------

CWIC_has_gdp_growth_series = {
    OR = {
        has_variable = CWIC_real_gdp_growth_yoy
        has_variable = CWIC_gdp_growth_yoy
        has_variable = CWIC_gdp_growth_mom_annualized
    }
}

CWIC_has_food_accounting = {
    has_variable = FoodProdTotal
    has_variable = ConsumptionTotal
    has_variable = FoodBalanceTotal
    has_variable = food_penalty
}

CWIC_has_energy_accounting = {
    has_variable = oil_consumption
    has_variable = electricity_consumption
}

CWIC_has_macro_finance_accounting = {
    has_variable = grossdomesticproduct
    has_variable = currentMoney
    has_variable = currentDebt
    has_variable = currentDebtPayments
    has_variable = Credit_Rating
}

# ------------------------------------------------------------
# 1) Fiscal Crisis
# ------------------------------------------------------------

CWIC_is_fiscal_crisis_low = {
    OR = {
        has_country_flag = CWIC_DEFAULTED
        has_country_flag = CWIC_MARKET_LOCKOUT
        AND = {
            has_variable = pulse_deficit
            has_variable = grossdomesticproduct
            check_variable = { pulse_deficit > 0 }
            check_variable = { grossdomesticproduct > 0 }
            # deficit > ~4% GDP proxy
            check_variable = { pulse_deficit > 0.04 }
            check_variable = { austerityCheck > 0 }
        }
        AND = {
            has_variable = currentDebtPayments
            has_variable = grossdomesticproduct
            check_variable = { grossdomesticproduct > 0 }
            # debt service > ~5% GDP proxy
            check_variable = { currentDebtPayments > 0.05 }
        }
    }
}

CWIC_is_fiscal_crisis_medium = {
    OR = {
        has_country_flag = CWIC_DEFAULTED
        has_country_flag = CWIC_MARKET_LOCKOUT
        AND = {
            has_variable = pulse_deficit
            has_variable = grossdomesticproduct
            check_variable = { pulse_deficit > 0.08 }
            check_variable = { austerityCheck > 0 }
        }
        AND = {
            has_variable = currentDebtPayments
            has_variable = grossdomesticproduct
            check_variable = { currentDebtPayments > 0.10 }
        }
        AND = {
            has_variable = Credit_Rating
            check_variable = { Credit_Rating < 5 }
        }
    }
}

CWIC_is_fiscal_crisis_high = {
    OR = {
        has_country_flag = CWIC_DEFAULTED
        has_country_flag = CWIC_MARKET_LOCKOUT
        AND = {
            has_variable = pulse_deficit
            has_variable = grossdomesticproduct
            check_variable = { pulse_deficit > 0.12 }
        }
        AND = {
            has_variable = currentDebtPayments
            check_variable = { currentDebtPayments > 0.15 }
        }
        AND = {
            has_variable = Credit_Rating
            check_variable = { Credit_Rating < 4 }
        }
        AND = {
            has_variable = currentMoney
            has_variable = currentDebtPayments
            check_variable = { currentDebtPayments > 0 }
            check_variable = { currentMoney < currentDebtPayments }
        }
    }
}

CWIC_is_fiscal_crisis_extreme = {
    OR = {
        has_country_flag = CWIC_DEFAULTED
        AND = {
            has_country_flag = CWIC_MARKET_LOCKOUT
            has_variable = Credit_Rating
            check_variable = { Credit_Rating < 3 }
        }
        AND = {
            has_variable = pulse_deficit
            check_variable = { pulse_deficit > 0.18 }
        }
        AND = {
            has_variable = currentDebtPayments
            check_variable = { currentDebtPayments > 0.20 }
        }
    }
}

CWIC_is_fiscal_crisis = {
    OR = {
        CWIC_is_fiscal_crisis_low = yes
        CWIC_is_fiscal_crisis_medium = yes
        CWIC_is_fiscal_crisis_high = yes
        CWIC_is_fiscal_crisis_extreme = yes
    }
}

# ------------------------------------------------------------
# 2) Economic Contraction Crisis
# ------------------------------------------------------------
# Wants a real growth trend variable. Placeholder support included.
# Fallback path uses weak credit + soft/negative inflation + low money.
# ------------------------------------------------------------

CWIC_is_economic_contraction_crisis_low = {
    OR = {
        AND = {
            has_variable = CWIC_real_gdp_growth_yoy
            check_variable = { CWIC_real_gdp_growth_yoy < 0 }
        }
        AND = {
            has_variable = CWIC_gdp_growth_yoy
            check_variable = { CWIC_gdp_growth_yoy < 0 }
        }
        AND = {
            has_variable = CWIC_gdp_growth_mom_annualized
            check_variable = { CWIC_gdp_growth_mom_annualized < 0 }
        }
        AND = {
            has_variable = Credit_Rating
            has_variable = CWIC_inflation_mom
            check_variable = { Credit_Rating < 6 }
            check_variable = { CWIC_inflation_mom < 0.002 }
        }
    }
}

CWIC_is_economic_contraction_crisis_medium = {
    OR = {
        AND = {
            has_variable = CWIC_real_gdp_growth_yoy
            check_variable = { CWIC_real_gdp_growth_yoy < -0.02 }
        }
        AND = {
            has_variable = CWIC_gdp_growth_yoy
            check_variable = { CWIC_gdp_growth_yoy < -0.02 }
        }
        AND = {
            has_variable = CWIC_gdp_growth_mom_annualized
            check_variable = { CWIC_gdp_growth_mom_annualized < -0.02 }
        }
        AND = {
            has_variable = Credit_Rating
            has_variable = CWIC_inflation_mom
            has_variable = currentMoney
            check_variable = { Credit_Rating < 5 }
            check_variable = { CWIC_inflation_mom < 0.001 }
            check_variable = { currentMoney < 0 }
        }
    }
}

CWIC_is_economic_contraction_crisis_high = {
    OR = {
        AND = {
            has_variable = CWIC_real_gdp_growth_yoy
            check_variable = { CWIC_real_gdp_growth_yoy < -0.04 }
        }
        AND = {
            has_variable = CWIC_gdp_growth_yoy
            check_variable = { CWIC_gdp_growth_yoy < -0.04 }
        }
        AND = {
            has_variable = CWIC_gdp_growth_mom_annualized
            check_variable = { CWIC_gdp_growth_mom_annualized < -0.04 }
        }
        AND = {
            has_variable = Credit_Rating
            has_variable = CWIC_inflation_mom
            check_variable = { Credit_Rating < 4 }
            check_variable = { CWIC_inflation_mom < 0.0005 }
        }
    }
}

CWIC_is_economic_contraction_crisis_extreme = {
    OR = {
        AND = {
            has_variable = CWIC_real_gdp_growth_yoy
            check_variable = { CWIC_real_gdp_growth_yoy < -0.06 }
        }
        AND = {
            has_variable = CWIC_gdp_growth_yoy
            check_variable = { CWIC_gdp_growth_yoy < -0.06 }
        }
        AND = {
            has_variable = CWIC_gdp_growth_mom_annualized
            check_variable = { CWIC_gdp_growth_mom_annualized < -0.06 }
        }
    }
}

CWIC_is_economic_contraction_crisis = {
    OR = {
        CWIC_is_economic_contraction_crisis_low = yes
        CWIC_is_economic_contraction_crisis_medium = yes
        CWIC_is_economic_contraction_crisis_high = yes
        CWIC_is_economic_contraction_crisis_extreme = yes
    }
}

# ------------------------------------------------------------
# 3) Inflationary Crisis
# Based directly on CWIC_inflation_mom monthly rate.
# 0.005 ~= ~6% annualized, 0.01 ~= ~12%, 0.02 ~= ~24%+
# ------------------------------------------------------------

CWIC_is_inflationary_crisis_low = {
    OR = {
        AND = {
            has_variable = CWIC_inflation_mom
            check_variable = { CWIC_inflation_mom > 0.005 }
        }
        AND = {
            has_variable = CWIC_black_market_pressure
            check_variable = { CWIC_black_market_pressure > 0.25 }
        }
    }
}

CWIC_is_inflationary_crisis_medium = {
    OR = {
        AND = {
            has_variable = CWIC_inflation_mom
            check_variable = { CWIC_inflation_mom > 0.010 }
        }
        AND = {
            has_variable = CWIC_black_market_pressure
            check_variable = { CWIC_black_market_pressure > 0.50 }
        }
        has_country_flag = CWIC_fx_pegged_currency_break
    }
}

CWIC_is_inflationary_crisis_high = {
    OR = {
        AND = {
            has_variable = CWIC_inflation_mom
            check_variable = { CWIC_inflation_mom > 0.020 }
        }
        AND = {
            has_variable = CWIC_black_market_pressure
            check_variable = { CWIC_black_market_pressure > 0.75 }
        }
        AND = {
            has_country_flag = CWIC_fx_pegged_currency_break
            has_variable = CWIC_inflation_mom
            check_variable = { CWIC_inflation_mom > 0.012 }
        }
    }
}

CWIC_is_inflationary_crisis_extreme = {
    OR = {
        AND = {
            has_variable = CWIC_inflation_mom
            check_variable = { CWIC_inflation_mom > 0.035 }

            has_country_flag = CWIC_fx_pegged_currency_break
            has_variable = CWIC_black_market_pressure
            check_variable = { CWIC_black_market_pressure > 0.85 }
        }
    }
}

CWIC_is_inflationary_crisis = {

        CWIC_is_inflationary_crisis_low = yes
        CWIC_is_inflationary_crisis_medium = yes
        CWIC_is_inflationary_crisis_high = yes
        CWIC_is_inflationary_crisis_extreme = yes
    }
}

# ------------------------------------------------------------
# 4) Deflationary Crisis
# Negative/very low inflation plus contraction or tight money.
# ------------------------------------------------------------

CWIC_is_deflationary_crisis_low = {
    AND = {
        has_variable = CWIC_inflation_mom
        check_variable = { CWIC_inflation_mom < 0.001 }
        OR = {
            CWIC_is_economic_contraction_crisis_low = yes
            AND = {
                has_variable = Interest_Rate
                check_variable = { Interest_Rate > 1.10 }
            }
        }
    }
}

CWIC_is_deflationary_crisis_medium = {
    AND = {
        has_variable = CWIC_inflation_mom
        check_variable = { CWIC_inflation_mom < 0 }
        OR = {
            CWIC_is_economic_contraction_crisis_medium = yes
            AND = {
                has_variable = Interest_Rate
                check_variable = { Interest_Rate > 1.20 }
                has_variable = Credit_Rating
                check_variable = { Credit_Rating < 6 }
            }
        }
    }
}

CWIC_is_deflationary_crisis_high = {
    AND = {
        has_variable = CWIC_inflation_mom
        check_variable = { CWIC_inflation_mom < -0.002 }
        OR = {
            CWIC_is_economic_contraction_crisis_high = yes
            AND = {
                has_variable = Interest_Rate
                check_variable = { Interest_Rate > 1.30 }
            }
        }
    }
}

CWIC_is_deflationary_crisis_extreme = {
    AND = {
        has_variable = CWIC_inflation_mom
        check_variable = { CWIC_inflation_mom < -0.004 }
        CWIC_is_economic_contraction_crisis_high = yes
    }
}

CWIC_is_deflationary_crisis = {
    OR = {
        CWIC_is_deflationary_crisis_low = yes
        CWIC_is_deflationary_crisis_medium = yes
        CWIC_is_deflationary_crisis_high = yes
        CWIC_is_deflationary_crisis_extreme = yes
    }
}

# ------------------------------------------------------------
# 5) Energy Crisis
# Uses real consumption variables from industrial pulse.
# Electricity side is still a proxy until a national electricity
# production / balance variable is formalized.
# ------------------------------------------------------------

CWIC_is_energy_crisis_low = {
    OR = {
        AND = {
            has_variable = oil_consumption
            check_variable = { oil_consumption > 5 }
            # resource@oil can go negative elsewhere in the system
            has_resources_in_country = { resource = oil amount < 5  }
        }
        AND = {
            has_variable = electricity_consumption
            check_variable = { electricity_consumption > 10 }
            OR = {
                check_variable = { fossil_fuel_plant_quantity < 2 }
                check_variable = { nuclear_powerplant_quantity < 1 }
            }
        }
    }
}

CWIC_is_energy_crisis_medium = {
    OR = {
        AND = {
            has_variable = oil_consumption
            check_variable = { oil_consumption > 10 }
            has_resources_in_country = { resource = oil amount < 3  }
        }
        AND = {
            has_variable = electricity_consumption
            check_variable = { electricity_consumption > 20 }
            check_variable = { fossil_fuel_plant_quantity < 2 }
            check_variable = { nuclear_powerplant_quantity < 1 }
        }
    }
}

CWIC_is_energy_crisis_high = {
    OR = {
        AND = {
            has_variable = oil_consumption
            check_variable = { oil_consumption > 15 }
            has_resources_in_country = { resource = oil amount < 1  }
        }
        AND = {
            has_variable = electricity_consumption
            check_variable = { electricity_consumption > 30 }
            check_variable = { fossil_fuel_plant_quantity < 1 }
            check_variable = { nuclear_powerplant_quantity < 1 }
            check_variable = { nuclear_reactor_quantity < 1 }
        }
    }
}

CWIC_is_energy_crisis_extreme = {
    OR = {
        AND = {
            has_variable = oil_consumption
            check_variable = { oil_consumption > 20 }
            has_resources_in_country = { resource = oil amount < 0  }
        }
        AND = {
            has_variable = electricity_consumption
            check_variable = { electricity_consumption > 40 }
            check_variable = { fossil_fuel_plant_quantity < 1 }
            check_variable = { nuclear_powerplant_quantity < 1 }
            check_variable = { nuclear_reactor_quantity < 1 }
            check_variable = { synthetic_refinery_quantity < 1 }
        }
    }
}

CWIC_is_energy_crisis = {
    OR = {
        CWIC_is_energy_crisis_low = yes
        CWIC_is_energy_crisis_medium = yes
        CWIC_is_energy_crisis_high = yes
        CWIC_is_energy_crisis_extreme = yes
    }
}
