# Durable bookmark selectors. The global flags are written by current bookmark
# definitions; has_start_date keeps the bridge compatible with older saves.
cwic_started_from_1949_bookmark = {
	OR = {
		has_global_flag = cwic_bookmark_1949_start
		AND = {
			NOT = { has_start_date < 1949.5.23 }
			has_start_date < 1949.5.24
		}
	}
}

cwic_started_from_1980_bookmark = {
	OR = {
		has_global_flag = cwic_bookmark_1980_start
		AND = {
			NOT = { has_start_date < 1980.1.1 }
			has_start_date < 1980.1.2
		}
	}
}
