USA_congress_house_support_needed = {
	OR = {
		AND = {
			has_variable = usa_vote_house_required
			check_variable = { supseat < usa_vote_house_required }
		}
		AND = {
			NOT = { has_variable = usa_vote_house_required }
			check_variable = { supseat < 218 }
		}
	}
}

USA_congress_can_lobby = {
	tag = USA
	has_country_flag = law_passing
	has_political_power > 49.999
	OR = {
		AND = {
			check_variable = { lobby_check = 1 }
			check_variable = { moderate_dem > 0 }
			NOT = { has_country_flag = moderate_dem_sup }
		}
		AND = {
			check_variable = { lobby_check = 2 }
			check_variable = { progressive_dem > 0 }
			NOT = { has_country_flag = progressive_dem_sup }
		}
		AND = {
			check_variable = { lobby_check = 3 }
			check_variable = { dixiecrat_dem > 0 }
			NOT = { has_country_flag = dixiecrat_dem_sup }
		}
		AND = {
			check_variable = { lobby_check = 4 }
			check_variable = { moderate_gop > 0 }
			NOT = { has_country_flag = moderate_gop_sup }
		}
		AND = {
			check_variable = { lobby_check = 5 }
			check_variable = { conservative_gop > 0 }
			NOT = { has_country_flag = conservative_gop_sup }
		}
		AND = {
			check_variable = { lobby_check = 6 }
			check_variable = { liberal_gop > 0 }
			NOT = { has_country_flag = liberal_gop_sup }
		}
		AND = {
			check_variable = { lobby_check = 10 }
			USA_congress_house_support_needed = yes
		}
	}
}

	OR = {
		check_variable = { senator^0 = 1 }
		check_variable = { senator^1 = 1 }
		check_variable = { senator^2 = 1 }
	}
}

	OR = {
		check_variable = { senator^0 = 2 }
		check_variable = { senator^1 = 2 }
		check_variable = { senator^2 = 2 }
	}
}
USA_congress_state_is_territory = {
	OR = {
		check_variable = { id = 686.id }
		AND = {
			OR = {
				check_variable = { id = 463.id }
				check_variable = { id = 629.id }
			}
			USA = { NOT = { has_country_flag = ak_hi_state } }
		}
	}
}

	OR = {
		USA_congress_state_is_territory = yes

			USA_congress_has_dem_senator = no
			USA_congress_has_gop_senator = no
		}
	}
}
