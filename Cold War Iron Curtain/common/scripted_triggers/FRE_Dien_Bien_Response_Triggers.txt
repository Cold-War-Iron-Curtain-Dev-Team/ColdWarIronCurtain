# Dien Bien Phu response geography. These helpers deliberately test exact
# province control across the CEFEO side rather than state control: the primary
# VIN campaign is won only at the camp, and Operation Condor must reopen a real
# route after the valley has first been invested.

fre_dbp_french_aligned_controls_camp_trigger = {
	OR = {
		AND = { country_exists = FRE  FRE = { controls_province = 4529 } }
		AND = { country_exists = FRA  FRA = { controls_province = 4529 } }
		AND = { country_exists = TAI  TAI = { controls_province = 4529 } }
		AND = { country_exists = VIE  VIE = { controls_province = 4529 } }
	}
}

fre_dbp_french_aligned_controls_approach_trigger = {
	OR = {
		AND = { country_exists = FRE  FRE = { OR = { controls_province = 13754  controls_province = 10103 } } }
		AND = { country_exists = FRA  FRA = { OR = { controls_province = 13754  controls_province = 10103 } } }
		AND = { country_exists = TAI  TAI = { OR = { controls_province = 13754  controls_province = 10103 } } }
		AND = { country_exists = VIE  VIE = { OR = { controls_province = 13754  controls_province = 10103 } } }
	}
}

fre_dbp_french_aligned_controls_ring_trigger = {
	OR = {
		AND = {
			country_exists = FRE
			FRE = { OR = { controls_province = 13765  controls_province = 13757  controls_province = 13758  controls_province = 13760  controls_province = 13762  controls_province = 13764 } }
		}
		AND = {
			country_exists = FRA
			FRA = { OR = { controls_province = 13765  controls_province = 13757  controls_province = 13758  controls_province = 13760  controls_province = 13762  controls_province = 13764 } }
		}
		AND = {
			country_exists = TAI
			TAI = { OR = { controls_province = 13765  controls_province = 13757  controls_province = 13758  controls_province = 13760  controls_province = 13762  controls_province = 13764 } }
		}
		AND = {
			country_exists = VIE
			VIE = { OR = { controls_province = 13765  controls_province = 13757  controls_province = 13758  controls_province = 13760  controls_province = 13762  controls_province = 13764 } }
		}
	}
}

fre_dbp_condor_corridor_open_trigger = {
	fre_dbp_french_aligned_controls_camp_trigger = yes
	fre_dbp_french_aligned_controls_approach_trigger = yes
	fre_dbp_french_aligned_controls_ring_trigger = yes
}
