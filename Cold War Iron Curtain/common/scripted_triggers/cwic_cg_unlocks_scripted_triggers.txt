

# ------------------------------------------------------------
# CWIC Consumer Goods Tier Unlocks
# Capability variables assumed (country scope):
#   modifier@cap_light_industry
#   modifier@cap_precision_manufacturing
#   modifier@cap_automotive
#   modifier@cap_electronics
#   modifier@cap_management
#
# Tiers: 0 (baseline) .. 5 (highest)
# These are intentionally conservative; tune thresholds after playtests.
# ------------------------------------------------------------

# -------------------
# cg_basic unlocks
# -------------------
cwic_can_produce_cg_basic_tier_1 = {
  check_variable = { modifier@cap_light_industry > 5 }
}
cwic_can_produce_cg_basic_tier_2 = {
  check_variable = { modifier@cap_light_industry > 12 }
  check_variable = { modifier@cap_management > 5 }
}
cwic_can_produce_cg_basic_tier_3 = {
  check_variable = { modifier@cap_light_industry > 17 }
  check_variable = { modifier@cap_management > 12 }
  check_variable = { modifier@cap_precision_manufacturing > 7 }
}
cwic_can_produce_cg_basic_tier_4 = {
  check_variable = { modifier@cap_light_industry > 24 }
  check_variable = { modifier@cap_management > 15 }
  check_variable = { modifier@cap_precision_manufacturing > 12 }
}
cwic_can_produce_cg_basic_tier_5 = {
  check_variable = { modifier@cap_light_industry > 30 }
  check_variable = { modifier@cap_management > 30 }
  check_variable = { modifier@cap_precision_manufacturing > 20 }
}
