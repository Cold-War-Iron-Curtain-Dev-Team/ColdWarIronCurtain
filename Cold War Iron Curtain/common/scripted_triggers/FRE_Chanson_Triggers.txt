FRE_chanson_window_trigger = {
	custom_trigger_tooltip = {
		tooltip = FRE_chanson_window_trigger_TT
		NOT = { has_global_flag = Indochina_War_Over }
		NOT = { has_global_flag = geneva_conference_concluded }
		check_variable = { global.Indochina_War_Active_Phase < 9 }
	}
}

FRE_chanson_roll_due_trigger = {
	FRE_chanson_window_trigger = yes
	date > 1951.7.30
	date < 1951.9.1
	NOT = { has_country_flag = FRE_CEFEO_Dissolved }
	NOT = { has_country_flag = FRE_Chanson_Killed }
	NOT = { has_country_flag = FRE_Chanson_Survived }
}

ic_chanson_survived_trigger = {
	country_exists = FRE
	FRE = { has_country_flag = FRE_Chanson_Survived }
}

		tooltip = FRE_chanson_pacification_available_trigger_TT
		has_country_flag = FRE_Chanson_Survived
		NOT = { has_country_flag = FRE_Chanson_Pacification_Done }

	}
