# Charter ending triggers (design 13).

USA_mkultra_can_fund_church_response_trigger = {
	OR = {
		AND = {
			check_variable = { USA_mkultra_church_severity = 3 }
			has_political_power > 74
		}
		AND = {
			check_variable = { USA_mkultra_church_severity = 2 }
			has_political_power > 49
		}
		AND = {
			check_variable = { USA_mkultra_church_severity = 1 }
			has_political_power > 24
		}
	}
}

USA_mkultra_can_fund_senate_response_trigger = {
	OR = {
		AND = {
			check_variable = { USA_mkultra_senate_severity = 3 }
			has_political_power > 39
		}
		AND = {
			check_variable = { USA_mkultra_senate_severity = 2 }
			has_political_power > 24
		}
		AND = {
			check_variable = { USA_mkultra_senate_severity = 1 }
			has_political_power > 14
		}
	}
}

# A CIA finding that was actually retained or applied, not merely eligible.
USA_mkultra_cia_has_finding_trigger = {
	OR = {
		has_country_flag = USA_mkultra_administrative_consolidation_applied
		has_country_flag = USA_mkultra_countermeasure_lessons_earned
		has_country_flag = USA_mkultra_reliable_appraisal_earned
		has_country_flag = USA_mkultra_screening_finding_applied
		has_country_flag = USA_mkultra_mksearch_inventory_earned
	}
}

# Design 11.2: without a funded appraisal the file can never be explained; with
# one, only the narrow-discrepancy profile stays unresolved.
USA_mkultra_early_assessment_unresolved_trigger = {
	OR = {
		AND = {
			has_country_flag = USA_mkultra_early_assessment_resolved
			OR = {
				NOT = { has_country_flag = USA_mkultra_early_appraisal_completed }
				check_variable = { USA_mkultra_early_assessment_profile = 3 }
			}
		}
		AND = {
			has_country_flag = USA_mkultra_early_assessment_committed
			has_country_flag = USA_mkultra_early_assessment_cancelled
		}
		AND = {
			has_country_flag = USA_mkultra_early_assessment_committed
			OR = {
				check_variable = { USA_mkultra_early_assessment_stage = 1 }
				check_variable = { USA_mkultra_early_assessment_stage = 2 }
			}
		}
	}
}
