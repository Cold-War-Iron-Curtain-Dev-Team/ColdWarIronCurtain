vin_lower_laos_live_window_trigger = {
	NOT = { has_global_flag = Indochina_War_Over }
	NOT = { has_global_flag = geneva_conference_concluded }
	check_variable = { global.Indochina_War_Active_Phase < 9 }
}

vin_lower_laos_atlante_cue_trigger = {
	OR = {
		date > 1954.02.14
		AND = {
			country_exists = FRE
			FRE = { has_country_flag = FRE_Operation_Atlante_Launched }
		}
	}
}

vin_lower_laos_available_trigger = {
	date > 1953.12.14
	vin_lower_laos_live_window_trigger = yes
	vin_lower_laos_atlante_cue_trigger = yes
	NOT = { has_country_flag = VIN_Lower_Laos_Diversion_Active }
	NOT = { has_country_flag = VIN_Lower_Laos_Offensive_Launched }
}

vin_lower_laos_fre_garrison_present_trigger = {
	OR = {
		divisions_in_state = { state = 1796 size > 0 }
		divisions_in_state = { state = 1187 size > 0 }
		AND = {
			country_exists = LOS
			LOS = { has_country_flag = LOS_Associated_Army_Raised }
		}
		# The Bolovens airlift is raised under Royal Lao colours, so count the
		# Royal Lao divisions in Champasak once CEFEO has paid for it.
		AND = {
			has_country_flag = FRE_Lower_Laos_Reinforced
			country_exists = LOS
			LOS = { divisions_in_state = { state = 1187 size > 0 } }
		}
	}
}

vin_lower_laos_terminal_trigger = {
	OR = {
		has_country_flag = VIN_Lower_Laos_Result_Pressure
		has_country_flag = VIN_Lower_Laos_Result_Contained
		has_country_flag = VIN_Lower_Laos_Result_Superseded
	}
}
