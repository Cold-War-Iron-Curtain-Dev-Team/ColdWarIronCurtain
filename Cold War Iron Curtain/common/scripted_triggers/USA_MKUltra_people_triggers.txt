# Personnel windows follow the historical tenures. Each appointment is once per
# campaign: the _served flag is set when the spirit is removed.

USA_mkultra_dulles_window_trigger = {
	date > 1953.2.25
	date < 1961.11.29
}

USA_mkultra_helms_window_trigger = {
	date > 1966.6.29
	date < 1973.2.2
}

USA_mkultra_gottlieb_window_trigger = {
	date > 1953.4.12
	date < 1973.1.1
}

USA_mkultra_can_appoint_dulles_trigger = {
	USA_mkultra_enabled_trigger = yes
	USA_mkultra_dulles_window_trigger = yes
	NOT = { has_idea = USA_mkultra_dci_dulles }
	NOT = { has_country_flag = USA_mkultra_dulles_served }
}

USA_mkultra_can_appoint_helms_trigger = {
	USA_mkultra_enabled_trigger = yes
	USA_mkultra_helms_window_trigger = yes
	NOT = { has_idea = USA_mkultra_dci_helms }
	NOT = { has_country_flag = USA_mkultra_helms_served }
}

USA_mkultra_can_appoint_gottlieb_trigger = {
	USA_mkultra_charter_active_trigger = yes
	USA_mkultra_gottlieb_window_trigger = yes
	NOT = { has_idea = USA_mkultra_tss_gottlieb }
	NOT = { has_country_flag = USA_mkultra_gottlieb_served }
}

# Read by the 1973 records disposition (event 320) once its hook line is in:
# Helms ordered the MKUltra files destroyed in January 1973.
USA_mkultra_helms_in_office_trigger = {
	has_idea = USA_mkultra_dci_helms
}

# Preserving the files means overruling Helms: 10 PP on top of the usual 10.
USA_mkultra_helms_permits_preserve_trigger = {
	OR = {
		NOT = { USA_mkultra_helms_in_office_trigger = yes }
		has_political_power > 19
	}
}

USA_mkultra_olson_incident_due_trigger = {
	date > 1953.11.27
	USA_mkultra_charter_active_trigger = yes
	NOT = { has_country_flag = USA_mkultra_olson_incident_recorded }
}

USA_mkultra_rockefeller_olson_due_trigger = {
	date > 1975.6.9
	has_country_flag = USA_mkultra_olson_incident_recorded
	NOT = { has_country_flag = USA_mkultra_rockefeller_olson_fired }
}
