# Expansion dossier cards, indexes 9-20. Cards 1-8 keep their original checks in
# USA_MKUltra_gui_effects.txt and USA_MKUltra_gui.txt; these triggers are the
# only place new cards are revealed.
#  9 Early Assessment (Gateway Expanded episode)
# 10 Project MKNAOMI (TSS / Fort Detrick Special Operations Division liaison)
# 11 Project MKOFTEN (ORD / Edgewood Pharmacology & Anomaly Canvass)
# 12 Bluebird/Artichoke (pre-charter interrogation study)
# 13 MKDELTA (pre-charter overseas field program)
# 16 Frank Olson file (read-only)

USA_mkultra_card_9_revealed_trigger = {
	check_variable = { USA_mkultra_early_assessment_stage > 0 }
}

	OR = {
		has_country_flag = USA_mkultra_mknaomi_unlocked
		check_variable = { USA_mkultra_mknaomi_state > 0 }
	}
}
