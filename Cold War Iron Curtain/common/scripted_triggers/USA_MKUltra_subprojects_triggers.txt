# Pre-charter subprojects (cards 12-13). Neither needs the CIA charter or a slot:
# Bluebird/Artichoke and MKDELTA predate MKUltra's April 1953 authorization.

# Bluebird (April 1950), renamed Artichoke in August 1951; folded into MKUltra.
USA_mkultra_bluebird_window_trigger = {
	date > 1950.4.19
	date < 1954.1.1
}

# MKDELTA approved October 1952; its overseas work was absorbed by 1954.
USA_mkultra_mkdelta_window_trigger = {
	date > 1952.10.19
	date < 1955.1.1
}

USA_mkultra_can_start_bluebird_trigger = {
	USA_mkultra_enabled_trigger = yes
	USA_mkultra_bluebird_window_trigger = yes
	NOT = { has_country_flag = USA_mkultra_feature_cutoff }
	check_variable = { USA_mkultra_bluebird_stage = 1 }
	NOT = { has_country_flag = USA_mkultra_bluebird_paid }
}

USA_mkultra_can_start_mkdelta_trigger = {
	USA_mkultra_enabled_trigger = yes
	USA_mkultra_mkdelta_window_trigger = yes
	NOT = { has_country_flag = USA_mkultra_feature_cutoff }
	NOT = { check_variable = { USA_mkultra_inquiry_state = 1 } }
	check_variable = { USA_mkultra_mkdelta_stage = 1 }
	NOT = { has_country_flag = USA_mkultra_mkdelta_paid }
}
