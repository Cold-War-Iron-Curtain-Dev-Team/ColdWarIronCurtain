# Espionage completion, stage 1.

# Create Group: five operatives picked and named, 25 PP and $50M in hand, and
# room under the squad cap (two, plus one for every three agency upgrades).
CWIC_espionage_can_create_group_trigger = {
	check_variable = { operative_number_step = 5 }
	check_variable = { name_verification = 1 }
	has_political_power > 24
	NOT = { check_variable = { currentMoney < 0.05 } }
	OR = {
		check_variable = { num_espionage_paramilitary_squads < 2 }
		AND = {
			agency_upgrade_number > 2
			check_variable = { num_espionage_paramilitary_squads < 3 }
		}
		AND = {
			agency_upgrade_number > 5
			check_variable = { num_espionage_paramilitary_squads < 4 }
		}
		AND = {
			agency_upgrade_number > 8
			check_variable = { num_espionage_paramilitary_squads < 5 }
		}
	}
}

# Stage 2.
CWIC_espionage_can_replace_trigger = {
	has_political_power > 4
	NOT = { check_variable = { currentMoney < 0.002 } }
}

# Warsaw Pact services with their own Black Operations desk (SOV has its own data).
CWIC_espionage_warsaw_pact_service_trigger = {
	OR = {
		tag = DDR
		tag = POL
		tag = CZE
		tag = HUN
		tag = ROM
		tag = BUL
		tag = ALB
	}
}
