# Triggers for the Hypnosis & Conditioning Claims and Behavioral Assessment
# dossiers. Affordability includes every scripted charge made on top of the
# native decision cost, so a purchase never leaves PP negative.

# Front-grant stage: 35 native, +5 expansion surcharge.
USA_mkultra_can_afford_front_grant_stage_trigger = {
	if = {
		limit = { check_variable = { USA_mkultra_capacity = 2 } }
		has_political_power > 39
	}
	else = {
		has_political_power > 34
	}
}

# Midnight Climax stages cost 25 natively and inherit only the +5 expansion fee.
USA_mkultra_can_afford_high_risk_stage_trigger = {
	if = {
		limit = { check_variable = { USA_mkultra_capacity = 2 } }
		has_political_power > 29
	}
	else = { has_political_power > 24 }
}

# Replication stage: 30 native, +10 under compartmentalization, +5 surcharge.
USA_mkultra_can_afford_replication_stage_trigger = {
	if = {
		limit = {
			check_variable = { USA_mkultra_information_policy = 2 }
			check_variable = { USA_mkultra_capacity = 2 }
		}
		has_political_power > 44
	}
	else_if = {
		limit = { check_variable = { USA_mkultra_information_policy = 2 } }
		has_political_power > 39
	}
	else_if = {
		limit = { check_variable = { USA_mkultra_capacity = 2 } }
		has_political_power > 34
	}
	else = {
		has_political_power > 29
	}
}

# Behavioral Assessment and MKSEARCH can apply the Reliable Internal Appraisal
# finding before their scripted surcharge, making the full purchase 5 PP cheaper.
USA_mkultra_can_afford_discounted_replication_stage_trigger = {
	if = {
		limit = { NOT = { has_country_flag = USA_mkultra_reliable_appraisal_earned } }
		USA_mkultra_can_afford_replication_stage_trigger = yes
	}
	else_if = {
		limit = {
			check_variable = { USA_mkultra_information_policy = 2 }
			check_variable = { USA_mkultra_capacity = 2 }
		}
		has_political_power > 39
	}
	else_if = {
		limit = { check_variable = { USA_mkultra_information_policy = 2 } }
		has_political_power > 34
	}
	else_if = {
		limit = { check_variable = { USA_mkultra_capacity = 2 } }
		has_political_power > 29
	}
	else = { has_political_power > 24 }
}

# Hypnosis & Conditioning Claims
USA_mkultra_can_start_hypnosis_trigger = {
	USA_mkultra_research_start_allowed_trigger = yes
	USA_mkultra_has_free_slot_trigger = yes
	check_variable = { USA_mkultra_hypnosis_stage = 1 }
	NOT = { has_country_flag = USA_mkultra_hypnosis_paid }
}

USA_mkultra_can_renew_hypnosis_trigger = {
	USA_mkultra_research_start_allowed_trigger = yes
	check_variable = { USA_mkultra_hypnosis_stage = 3 }
	NOT = { has_country_flag = USA_mkultra_hypnosis_renewal_paid }
	NOT = { has_country_flag = USA_mkultra_hypnosis_renewal_foregone }
	NOT = { has_country_flag = USA_mkultra_hypnosis_archived }
}

USA_mkultra_can_replicate_hypnosis_trigger = {
	USA_mkultra_research_start_allowed_trigger = yes
	OR = {
		check_variable = { USA_mkultra_hypnosis_stage = 3 }
		check_variable = { USA_mkultra_hypnosis_stage = 5 }
	}
	NOT = { has_country_flag = USA_mkultra_hypnosis_replication_paid }
	NOT = { has_country_flag = USA_mkultra_hypnosis_archived }
}

USA_mkultra_can_apply_reliable_appraisal_trigger = {
	USA_mkultra_enabled_trigger = yes
	check_variable = { USA_mkultra_hypnosis_stage = 7 }
	has_country_flag = USA_mkultra_hypnosis_replication_record_awarded
	NOT = { has_country_flag = USA_mkultra_reliable_appraisal_earned }
}

# Behavioral Assessment

	USA_mkultra_has_free_slot_trigger = yes
	check_variable = { USA_mkultra_behavioral_stage = 1 }
	NOT = { has_country_flag = USA_mkultra_behavioral_paid }
}

USA_mkultra_can_replicate_behavioral_trigger = {
	USA_mkultra_research_start_allowed_trigger = yes
	check_variable = { USA_mkultra_behavioral_stage = 3 }
	NOT = { has_country_flag = USA_mkultra_behavioral_replication_paid }
	NOT = { has_country_flag = USA_mkultra_behavioral_archived }
}

USA_mkultra_can_apply_screening_finding_trigger = {
	USA_mkultra_enabled_trigger = yes
	check_variable = { USA_mkultra_behavioral_stage = 5 }
	has_country_flag = USA_mkultra_behavioral_replication_record_awarded
	check_variable = { USA_mkultra_behavioral_profile = 1 }
	NOT = { has_country_flag = USA_mkultra_screening_finding_applied }
}

# AI: front-grant work only with the standard reserve (design 14.2).
USA_mkultra_ai_can_afford_front_grant_trigger = {
	if = {
		limit = { USA_mkultra_major_crisis_trigger = yes }
		if = {
			limit = { check_variable = { USA_mkultra_capacity = 2 } }
			has_political_power > 189
		}
		else = { has_political_power > 184 }
	}
	else = {
		if = {
			limit = { check_variable = { USA_mkultra_capacity = 2 } }
			has_political_power > 139
		}
		else = { has_political_power > 134 }
	}
}

USA_mkultra_ai_can_afford_replication_stage_trigger = {
	if = {
		limit = { USA_mkultra_major_crisis_trigger = yes }
		if = {
			limit = {
				check_variable = { USA_mkultra_information_policy = 2 }
				check_variable = { USA_mkultra_capacity = 2 }
			}
			has_political_power > 194
		}
		else_if = {
			limit = { check_variable = { USA_mkultra_information_policy = 2 } }
			has_political_power > 189
		}
		else_if = {
			limit = { check_variable = { USA_mkultra_capacity = 2 } }
			has_political_power > 184
		}
		else = { has_political_power > 179 }
	}
	else = {
		if = {
			limit = {
				check_variable = { USA_mkultra_information_policy = 2 }
				check_variable = { USA_mkultra_capacity = 2 }
			}
			has_political_power > 144
		}
		else_if = {
			limit = { check_variable = { USA_mkultra_information_policy = 2 } }
			has_political_power > 139
		}
		else_if = {
			limit = { check_variable = { USA_mkultra_capacity = 2 } }
			has_political_power > 134
		}
		else = { has_political_power > 129 }
	}
}
