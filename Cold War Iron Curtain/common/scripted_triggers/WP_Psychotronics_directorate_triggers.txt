# Shared Warsaw Pact psychotronics directorate triggers. National programs use
# their own slots; only central appraisals compete for WP_psych_review_slot.

WP_psych_historical_network_enabled_trigger = {
	NOT = {
		has_game_rule = {
			rule = WP_PSYCHOTRONICS_MODE
			option = WP_PSYCHOTRONICS_OFF
		}
	}
}

WP_psych_directorate_enabled_trigger = {
	SOV_psych_enabled_trigger = yes
	check_variable = { global.WP_psych_directorate_state > 0 }
	check_variable = { global.WP_psych_directorate_state < 4 }
}

WP_psych_is_member_tag_trigger = {
	OR = {
		original_tag = CZE
		original_tag = BUL
		original_tag = POL
		original_tag = HUN
		original_tag = ROM
		original_tag = DDR
	}
}

WP_psych_joint_eligible_trigger = {
	WP_psych_historical_network_enabled_trigger = yes
	WP_psych_is_member_tag_trigger = yes
	has_government = communism
	country_exists = SOV
	is_in_faction_with = SOV
}

WP_psych_has_archive_access_trigger = {
	OR = {
		AND = {
			original_tag = SOV
			SOV_psych_enabled_trigger = yes
			has_country_flag = WP_psych_directorate_participant
		}
		AND = {
			WP_psych_historical_network_enabled_trigger = yes
			has_country_flag = WP_psych_directorate_participant
		}
		has_country_flag = WP_psych_archive_access
	}
}

WP_psych_can_start_joint_work_trigger = {
	WP_psych_has_archive_access_trigger = yes
	check_variable = { global.WP_psych_directorate_state = 2 }
	check_variable = { global.WP_psych_scrutiny < 50 }
	check_variable = { global.WP_psych_coordination > 29 }
	NOT = { has_global_flag = WP_psych_funding_crisis_active }
	if = {
		limit = { NOT = { original_tag = SOV } }
		WP_psych_joint_eligible_trigger = yes
	}
}

SOV_psych_can_form_directorate_trigger = {
	original_tag = SOV
	SOV_psych_enabled_trigger = yes
	WP_psych_historical_network_enabled_trigger = yes
	SOV_psych_has_service_trigger = yes
	has_government = communism
	check_variable = { global.WP_psych_directorate_state = 0 }
	OR = {
		has_global_flag = cwic_cze_psych_file_sent
		has_global_flag = cwic_bul_psych_file_sent
		AND = { country_exists = CZE CZE = { WP_psych_joint_eligible_trigger = yes check_variable = { CZE_psych_demo_stage = 4 } } }
		AND = { country_exists = BUL BUL = { WP_psych_joint_eligible_trigger = yes check_variable = { BUL_psych_demo_stage = 4 } } }
		date > 1971.12.31
	}
}
