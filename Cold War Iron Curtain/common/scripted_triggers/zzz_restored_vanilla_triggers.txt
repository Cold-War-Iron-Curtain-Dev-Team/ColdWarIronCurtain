# Vanilla China/Spain scripted triggers restored for CWIC.
#
# The mod ships common/scripted_triggers/CHI_scripted_triggers.txt and
# SPR_scripted_triggers.txt as 0-byte files. Same filename as vanilla means they
# REPLACE the vanilla files, so every trigger vanilla defined there vanished --
# which is why common/factions/goals/*.txt logged
#   Invalid trigger 'is_literally_china' / 'is_independent_china_or_warlord'
#   / 'SPR_scw_in_progress' / 'CHI_northern_mandate_states'
#   / 'CHI_southern_mandate_states' / 'CHI_greater_game_states'.
#
# This file has a different name, so it merges instead of overriding.
# Only the triggers the mod actually calls are restored.
#
# The three CHI_*_states triggers use strategic-region IDs. CWIC replace_path's
# map/strategicregions and renumbers them, so vanilla's IDs point at the wrong
# places here (vanilla 228 = South Indochina, CWIC 228 = Congo). They have been
# remapped to CWIC's own regions by name -- review the lists if the intended
# footprint differs.
#
# NOTE: the vanilla tag lists include Chinese warlord cliques that CWIC does not
# define (GDC, SND, NXM, GSM, XIC, KUM, RNG, HBC). Those entries are omitted -- referencing an
# undefined tag would just trade one load error for another. Re-add them here if
# those tags are ever added to common/country_tags.

is_literally_china = {
	OR = {
		original_tag = CHI
		original_tag = PRC
		original_tag = GXC
		original_tag = YUN
		original_tag = SHX
		original_tag = XSM
		original_tag = SIK
		original_tag = SIC
		original_tag = KHM
	}
}
