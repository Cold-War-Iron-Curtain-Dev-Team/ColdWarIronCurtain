# ============================================================================
# Favoritism Minigame - AI direction control
# ----------------------------------------------------------------------------
# The favoritism minigame (Alt_Favoritism = challenger vs Main_Favoritism =
# incumbent) is a GENERIC, SEQUENTIAL mechanic reused across several distinct
# struggles. Only one is ever live at a time, identified by the Main_* / Opposition_*
# flags. These two triggers tell the AI which way to steer whichever struggle is
# active, consulting the game rule that owns that struggle. Humans never affected.
#
#   Struggle              | Live when                 | Governing game rule
#   ----------------------|---------------------------|---------------------------
#   Bao-Dai premiership   | Main_Bao_Dai, opp = a PM  | VIE_ai_pm_outcome
#   Cuong-De takeover     | Opposition_Cuong_De       | VIE_ai_cuong_de_takeover
#
#   pursue_challenger -> raise Alt_Favoritism  -> alt-hist outcome
#                        (Huu Wins / Long Stays / Cuong-De takes over)
#   pursue_historical -> keep Alt_Favoritism low -> historical outcome
#                        (Huu dismissed / Long retires / Cuong-De faction loses)
#
# The Leroy/Hinh French-coup struggles are intentionally left on default here and
# are governed separately by VIE_french_coup. Once Cuong-De is the incumbent
# (Main_Cuong_De) his succession content runs on CD_Recruits/events, not these
# triggers, so it is not wired here.
# ============================================================================
VIE_ai_pm_pursue_challenger = {
	is_ai = yes
	OR = {
		# --- Bao-Dai premiership: KEEP the current PM (entrench him) ---
		# "Who entrenches" rule VIE_ai_pm_outcome. The chosen PM is made to STAY;
		# every PM before him retires (see pursue_historical) so the chain reaches him.
		# Only Long and Huu resolve via the favoritism minigame; the others resign/
		# rotate through their own event-driven outcomes.
		AND = {
			has_country_flag = VIE_Main_Bao_Dai
			OR = {
				AND = {
					has_country_flag = VIE_Opposition_Nguyen_Phan_Long
					has_game_rule = { rule = VIE_ai_pm_outcome option = VIE_AI_PM_LONG }
				}
				AND = {
					has_country_flag = VIE_Opposition_Tran_Van_Huu
					has_game_rule = { rule = VIE_ai_pm_outcome option = VIE_AI_PM_HUU }
				}
			}
		}
		# --- Cuong-De takeover (Bao-Dai vs the returning Cuong-De faction) ---
		AND = {
			has_country_flag = VIE_Opposition_Cuong_De
			OR = {
				has_game_rule = { rule = VIE_ai_cuong_de_takeover option = VIE_AI_CD_CHALLENGER }
				AND = {
					has_game_rule = { rule = VIE_ai_cuong_de_takeover option = DEFAULT }
					is_historical_focus_on = no
				}
			}
		}
	}
}

VIE_ai_pm_pursue_historical = {
	is_ai = yes
	OR = {
		# --- Bao-Dai premiership: RETIRE the current PM (rotate toward the chosen one) ---
		# On DEFAULT nobody is entrenched, so every favoritism PM retires and the chain
		# rotates through the historical succession (Long -> Huu -> Tam -> Loc -> Diem era).
		# When Long/Huu is the entrench target, the *other* one still retires to reach him.
		AND = {
			has_country_flag = VIE_Main_Bao_Dai
			OR = {
				AND = {
					has_country_flag = VIE_Opposition_Nguyen_Phan_Long
					NOT = { has_game_rule = { rule = VIE_ai_pm_outcome option = VIE_AI_PM_LONG } }
				}
				AND = {
					has_country_flag = VIE_Opposition_Tran_Van_Huu
					NOT = { has_game_rule = { rule = VIE_ai_pm_outcome option = VIE_AI_PM_HUU } }
				}
			}
		}
		# --- Cuong-De takeover (Bao-Dai vs the returning Cuong-De faction) ---
		AND = {
			has_country_flag = VIE_Opposition_Cuong_De
			OR = {
				has_game_rule = { rule = VIE_ai_cuong_de_takeover option = VIE_AI_CD_HISTORICAL }
				AND = {
					has_game_rule = { rule = VIE_ai_cuong_de_takeover option = DEFAULT }
					is_historical_focus_on = yes
				}
			}
		}
	}
}
