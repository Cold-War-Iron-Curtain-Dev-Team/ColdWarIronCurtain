	# Mission: Operation Lorraine
	FRE_Mission_Operation_Lorraine = {
		icon = generic_offensive_operations
		
		allowed = {
			tag = FRE
		}
		
		activation = {
			has_country_flag = FRE_Mission_Operation_Lorraine_Activated
			NOT = { has_country_flag = FRE_Mission_Operation_Lorraine_Completed }
		}
		
		available = {
			has_country_flag = FRE_Operation_Lorraine_Launched
			country_exists = VIN
			NOT = { has_country_flag = FRE_Mission_Operation_Lorraine_Completed }
			NOT = { has_country_flag = FRE_Mission_Operation_Lorraine_Failed }
			custom_trigger_tooltip = {
				tooltip = FRE_Mission_Operation_Lorraine_Available_tt
				always = yes
			}
		}
		
		selectable_mission = yes
		days_mission_timeout = 90
		
		cost = 150
		
		fire_only_once = yes
		
		complete_effect = {
			# Start border war with VIN for Operation Lorraine
			# Find a state that FRE owns (border wars use state owner, not controller)
			# If FRE doesn't own states, fall back to random chance system
			if = {
				limit = {
					country_exists = VIN
					any_owned_state = {
						OR = {
							state = 1280
							state = 1281
							state = 1760
							state = 1761
						}
						any_neighbor_state = {
							is_owned_by = VIN
						}
					}
				}
				# Find a FRE-owned state for attacker
				random_owned_state = {
					limit = {
						OR = {
							state = 1280
							state = 1281
							state = 1760
							state = 1761
						}
						any_neighbor_state = {
							is_owned_by = VIN
						}
					}
					set_temp_variable = { attacker_state = THIS.id }
					# Find a VIN-owned neighbor state for defender
					random_neighbor_state = {
						limit = {
							is_owned_by = VIN
						}
						set_temp_variable = { defender_state = THIS.id }
					}
				}
				start_border_war = {
					change_state_after_war = no
					combat_width = 80
					minimum_duration_in_days = 21
					attacker = {
						state = var:attacker_state
						num_provinces = 6
						on_win = FRE_Events.24
						on_lose = FRE_Events.25
						on_cancel = FRE_Events.25
						modifier = 0.3
					}
					defender = {
						state = var:defender_state
						num_provinces = 6
						on_win = FRE_Events.25
						on_lose = FRE_Events.24
						on_cancel = FRE_Events.25
					}
				}
			}
			else = {
				# If border war can't start, use random chance based on metrics
				fre_mission_calculate_battle_prep_success = yes
				set_temp_variable = { mission_failure_chance = 100 }
				subtract_from_temp_variable = { mission_failure_chance = mission_success_chance }
				random_list = {
					temp_var:mission_success_chance = {
						country_event = { id = FRE_Events.24 days = 1 }
					}
					temp_var:mission_failure_chance = {
						country_event = { id = FRE_Events.25 days = 1 }
					}
				}
			}
		}
		
		timeout_effect = {
			country_event = { id = FRE_Events.25 days = 1 }
		}
		
		ai_will_do = {
			factor = 100
		}
	}
