# =============================================================================
# CEFEO lifeline: requests to the Metropole
# =============================================================================
# Each request costs PP and a little Metropole Patience, then fires an
# FRA-side event (FRE_Metropole namespace) where Paris grants, half-grants or
# refuses. If FRA is gone, a reduced automatic grant is applied instead so the
# corps is never soft-locked.
# =============================================================================

FRE_Metropole_Support_category = {

	FRE_Request_Equipment_Convoy = {
		icon = generic_industry

		allowed = {
			tag = FRE
		}

		visible = {
			NOT = { has_country_flag = FRE_CEFEO_Dissolved }
		}

		available = {
			NOT = { has_country_flag = FRE_Metropole_Request_Pending }
		}

		cost = 25

		fire_only_once = no
		days_re_enable = 60

		complete_effect = {
			set_temp_variable = { fre_patience_delta = -3 }
			fre_metropole_patience_change = yes
			if = {
				limit = { country_exists = FRA }
				# Timed so a vanished FRA can never wedge the request system
			set_country_flag = { flag = FRE_Metropole_Request_Pending days = 30 value = 1 }
				FRA = {
					country_event = { id = FRE_Metropole.1 days = 3 }
				}
			}
			else = {
				# No Metropole left to ask: scrape together theatre reserves
				add_equipment_to_stockpile = {
					type = infantry_equipment_1
					amount = 6000
					producer = FRA
				}
				add_equipment_to_stockpile = {
					type = support_equipment
					amount = 750
					producer = FRA
				}
			}
		}

		ai_will_do = {
			factor = 10
			modifier = {
				factor = 0
				FRE_mission_pp_reserved_trigger = yes	# bank it for the mission instead
			}
			modifier = {
