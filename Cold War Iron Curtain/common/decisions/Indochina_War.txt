	Indochina_Wrapup_Timer = {
		allowed = {
			always = no
		}

		available = {
			always = no
		}
		activation = {
			# Never activates on its own: ic_pulse calls activate_mission when
			# indochina_struggle_all_participants_ai_trigger passes (all
			# Indochina participants + France AI controlled) inside the 1954
			# Geneva window or after 1957 as a stalemate fallback.
			always = no
		}
		is_good = no

		selectable_mission = no
		days_mission_timeout = 60

		cancel_trigger = {
			# A player-driven ending (or hot-joined player) resolved the war
			# while the timer was running.
			OR = {
			has_global_flag = Indochina_War_Over
				NOT = { indochina_struggle_all_participants_ai_trigger = yes }	# a human took one of the stake countries mid-timer: they play it out themselves
			}
		}

		cancel_effect = {
			clr_global_flag = Indochina_War_Timer_Active	# ic_pulse can re-arm later if the map goes all-AI again
		}

		timeout_effect = {
			if = {
				# A decisive set-piece is still being fought out: give the battle
				# another 60 days to produce its own result, four times over.
				limit = {
					NOT = { has_global_flag = Indochina_War_Over }
					indochina_war_still_being_fought_trigger = yes
					check_variable = { global.Indochina_Wrapup_Extensions < 4 }
				}
				add_to_variable = { global.Indochina_Wrapup_Extensions = 1 }
				activate_mission = Indochina_Wrapup_Timer
				log = "[GetDateText]: Indochina wrap-up timer extended - a campaign, the Laos raid or the Dien Bien Phu siege is still running"
			}
			else_if = {
				limit = {
					NOT = { has_global_flag = Indochina_War_Over }
				}
				# Resolve with whichever decisive ending the AI war has already
				# earned; otherwise default to the historical Geneva Accords
				# partition. Each ending effect handles cleanup, sets
				# Indochina_War_Over and fires the (major) news event itself.
				if = {
					limit = { has_global_flag = USA_VIN_Reunification_Complete }
					indochina_struggle_american_north_vietnam = yes
				}
				else_if = {
					limit = {
						country_exists = VIN
						VIN = { indochina_struggle_ending_communist_victory_trigger = yes }
					}
					indochina_struggle_communist_victory = yes
				}
				else_if = {
					limit = {
						country_exists = VIE
						VIE = { indochina_struggle_ending_southern_victory_trigger = yes }
					}
					indochina_struggle_southern_victory = yes
				}
				else_if = {
					limit = {
						country_exists = PQC
						PQC = { indochina_struggle_ending_kuomintang_victory_trigger = yes }
					}
					indochina_struggle_kuomintang_victory = yes
				}
				else_if = {
					limit = {
						country_exists = VIE
						VIE = { indochina_struggle_ending_dan_quoc_peace_trigger = yes }
					}
					indochina_struggle_dan_quoc_peace = yes
				}
				else_if = {
					limit = {
						country_exists = FRA
						FRA = { indochina_struggle_ending_federal_vietnam_trigger = yes }
					}
					indochina_struggle_federal_vietnam = yes
				}
				else_if = {
					limit = {
						country_exists = FRA
						FRA = { indochina_struggle_ending_balkanized_vietnam_trigger = yes }
					}
					indochina_struggle_balkanized_vietnam = yes
				}
				else_if = {
					# No decisive ending earned: convene the Geneva Conference
					# headlessly and let the delegations settle (or wreck) it.
					# Re-checked here, not just at arming: a human who joined a stake
					# country during the 60 days must not have the conference concluded
					# over their head.
					limit = { indochina_struggle_all_participants_ai_trigger = yes }
					geneva_conf_auto_resolve = yes
				}
				else = {
					clr_global_flag = Indochina_War_Timer_Active
					set_global_flag = Indochina_Geneva_Pursuit	# settle through the normal gated path instead, so the human plays the conference
				}
			}
		}
		complete_effect = {
		}
		ai_will_do = {
			factor = 0
		}
	}
