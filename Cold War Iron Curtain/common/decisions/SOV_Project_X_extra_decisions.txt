# Project X can stall after resonance: an under-documented or failed result
# leaves the anomaly record below the 30 the prototype needs, and nothing else
# raises it. This lets the programme be wound down instead of left open.
WP_PSYCH_DIRECTORATE = {
	SOV_project_x_wind_down = {
		icon = generic_research
		cost = 0
		fire_only_once = yes
		visible = {
			SOV_project_x_enabled_trigger = yes
			check_variable = { SOV_project_x_stage = 5 }
		}
		available = {
			SOV_project_x_enabled_trigger = yes
			check_variable = { SOV_project_x_stage = 5 }
		}
		complete_effect = {
			SOV_project_x_terminate = yes
		}
		ai_will_do = {
			factor = 0
			modifier = {
				add = 10
				check_variable = { global.WP_psych_anomaly_record < 30 }
			}
		}
	}
}
