##############################################################################
# USSR DEBUG BREAK-UP TOOLS
#
# Scenario tools for testing the Soviet collapse. Nothing here fires on its
# own and no AI will ever take these (ai_will_do = 0 everywhere).
#
# Enable with the toggle in Political Actions, then use the "USSR Debug"
# category. Every decision is repeatable and costs nothing.
#
# Relies on cores set in history/states:
#   RUS = 103 RSFSR states   TTR BSK KOM CHV MRD MRI KLY = the seven ASSRs
#   GSR / KSR have no cores at start - their decisions grant them first.
##############################################################################

debug_decisions = {

	ussr_debug_enable_tools = {
		icon = generic_political_actions

		allowed = { original_tag = SOV }
		visible = {
			NOT = { has_country_flag = ussr_debug_tools_enabled }
		}
		available = { always = yes }
		cost = 0
		fire_only_once = no

		complete_effect = {
			set_country_flag = ussr_debug_tools_enabled
		}
		ai_will_do = { factor = 0 }
	}

	ussr_debug_disable_tools = {

		allowed = { original_tag = SOV }

		fire_only_once = no

		complete_effect = {

		}
		ai_will_do = { factor = 0 }
	}
}
