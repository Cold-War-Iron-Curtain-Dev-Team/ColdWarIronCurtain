# Toggle to show/hide the Foreign Investment Invitation decisions.
# The 'cwic_mnc_invitations' category hides itself when the flag 'cwic_mnc_invitations_off' is set
# (see common/decisions/categories/00_decision_categories.txt).
#
# HIDE button -> lives in the Foreign Investment Invitations tab itself.
# SHOW button -> lives in the 'cwic_mnc_invitations_restore' tab, which only appears while the
#                invitations are hidden (same display name), so the tab looks like it collapsed
#                to a single "Show" button. (political_actions is not rendered in this mod.)

cwic_mnc_invitations = {

	cwic_mnc_hide_invitations = {
		icon = decision_trade

		allowed = { always = yes }
		cost = 0
		fire_only_once = no

		visible = {
			has_variable = economic_x_axis
			NOT = { has_country_flag = cwic_mnc_invitations_off }
		}
		available = { always = yes }

		complete_effect = {
			set_country_flag = cwic_mnc_invitations_off
		}

		ai_will_do = { factor = 0 }
	}
}

cwic_mnc_invitations_restore = {

	cwic_mnc_show_invitations = {
		icon = decision_trade

		allowed = { always = yes }
		cost = 0
		fire_only_once = no

		visible = {
			has_country_flag = cwic_mnc_invitations_off
		}
		available = { always = yes }

		complete_effect = {
			clr_country_flag = cwic_mnc_invitations_off
		}

		ai_will_do = { factor = 0 }
	}
}
