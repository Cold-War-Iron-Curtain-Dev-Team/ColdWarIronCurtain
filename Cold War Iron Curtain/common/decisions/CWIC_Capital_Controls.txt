    CWIC_capital_controls_lift = {
        icon = decision_liberalization
        cost = 75

        available = {
            OR = {
                has_country_flag = CWIC_capital_controls_light
                has_country_flag = CWIC_capital_controls_heavy
                has_country_flag = CWIC_fx_controls_dual
                has_country_flag = CWIC_price_controls_basic
            }
        }

        complete_effect = {

            # Base bounce
            add_to_variable = { CWIC_inflation_mom = 0.0015 }

            # Extra bounce if heavy controls were in place
            if = {
                limit = { has_country_flag = CWIC_capital_controls_heavy }
                add_to_variable = { CWIC_inflation_mom = 0.0025 }
            }

            # Extra bounce from black market overhang
            if = {
                limit = { check_variable = { CWIC_black_market_pressure > 0.40 } }
                add_to_variable = { CWIC_inflation_mom = 0.0010 }
            }
            if = {
                limit = { check_variable = { CWIC_black_market_pressure > 0.70 } }
                add_to_variable = { CWIC_inflation_mom = 0.0015 }
            }

            # Clear control flags
            clr_country_flag = CWIC_capital_controls_light
            clr_country_flag = CWIC_capital_controls_heavy
            clr_country_flag = CWIC_fx_controls_dual
            clr_country_flag = CWIC_price_controls_basic

            # Undo some structural drag from controls (tunable)
            add_to_variable = { CWIC_inflation_bias_mom = 0.0005 }
        }

        ai_will_do = {
            factor = 1

            modifier = {
                factor = 6
                check_variable = { CWIC_inflation_mom < 0.005 }
            }
            modifier = {
                factor = 0.2
                check_variable = { CWIC_inflation_mom > 0.01 }
            }
            modifier = {
                factor = 0.3
                check_variable = { CWIC_black_market_pressure > 0.70 }
            }
        }
    }

        icon = decision_liberalization
        cost = 150

        available = {
            check_variable = { CWIC_reform_cooldown = 0 }

            # don't allow during hyperinflation collapse unless you want it
            check_variable = { CWIC_inflation_mom < 0.06 }     # under ~100% annual
            check_variable = { CWIC_black_market_pressure < 0.80 }
        }

        complete_effect = {

            # reduce CPI index somewhat (rebasing / redenomination)
            # (Do NOT set to 1 or you break long-run comparisons)
            multiply_variable = { CWIC_cpi_index = 0.50 }

            # immediate inflation break
            subtract_from_variable = { CWIC_inflation_mom = 0.010 }

            # credibility improvement (reduce chronic inflation)
            subtract_from_variable = { CWIC_inflation_bias_mom = 0.001 }

            # suppress black market via conversion & enforcement
            subtract_from_variable = { CWIC_black_market_pressure = 0.20 }

            # short-term disruption
            add_to_variable = { CWIC_productivity_mult = -0.020 }

            # reform aftershock window
            set_country_flag = CWIC_currency_reform_recent
            set_variable = { CWIC_reform_cooldown = 24 }  # 2 years
        }

        ai_will_do = {
            factor = 1

            # strongly prefer if inflation is very high but not total hypercollapse
            modifier = { factor = 8 check_variable = { CWIC_inflation_mom > 0.015 } }
            modifier = { factor = 12 check_variable = { CWIC_inflation_mom > 0.030 } }

            # avoid if black market is extremely high (reform will be undermined)
            modifier = { factor = 0.2 check_variable = { CWIC_black_market_pressure > 0.70 } }
        }
    }
