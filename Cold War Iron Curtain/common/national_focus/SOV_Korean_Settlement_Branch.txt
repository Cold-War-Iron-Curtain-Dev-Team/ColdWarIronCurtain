# ============================================================================
#  THE KOREAN SETTLEMENT - post-war Korea side branch for the SOV (Stalin) tree.
#
#  Appears when the Korean War ends and replaces SOV_Korean_War_Branch.txt in
#  place - the wartime branch's allow_branch now excludes KOREA_CEASEFIRE_UN, and
#  this branch occupies the same coordinates.
#
#  OUTCOME DETECTION - no new flags needed. All three endings are already
#  produced by common/on_actions/wars_on_actions.txt and are distinguishable
#  from the state they leave behind:
#
#    NORTH WINS   - the KPA takes the whole peninsula except state 1004, which
#                   is marked with the state flag JEJU. What is left of the
#                   South survives there as a rump and receives the country flag
#                   lost_in_korean_war from korean_surrender.5.
#                   Trigger: KOR = { has_country_flag = lost_in_korean_war }
#
#    SOUTH WINS   - KOR annexes the KPA outright. This is the only ending in
#                   which the KPA ceases to exist.
#                   Trigger: NOT = { country_exists = KPA }
#
#    DIVISION     - negotiated ceasefire (decisions/korean_war.txt,
#                   USA_FP_50s.txt, America_1950s_Expansion.txt). Both Koreas
#                   survive along the line.
#                   Trigger: both exist, and the South did not lose.
#
#  All three set KOREA_CEASEFIRE_UN and clear korean_war, so the root gates on
#  that plus korean_war_begun - a ceasefire flag with no war behind it should
#  not open this branch.
#
#  The three arms are mutually exclusive and share coordinates, so only one is
#  ever drawn.
#
#  NOTE ON THE EXILED SOUTH: where the rump government ends up varies by
#  playthrough, so it is referred to generically throughout - never by island.
#
#  SOV_kw_the_armistice_at_panmunjom was moved here from the wartime branch. It
#  was only ever available after the ceasefire, so leaving it behind would have
#  orphaned it. Its ID, payload and localisation are unchanged.
#
#  Shared focuses; only the root is listed in the SOV_Stalin focus_tree header.
# ============================================================================

# ---- ROOT -------------------------------------------------------------------
shared_focus = {
	id = SOV_kwp_the_korean_settlement
	icon = GFX_focus_korean_war
	text_icon = Soviet_focus_style
	cost = 3
	x = 35
	y = 0
	search_filters = { FOCUS_FILTER_POLITICAL }

	allow_branch = {
		has_global_flag = korean_war_begun
		has_global_flag = KOREA_CEASEFIRE_UN
	}
	available = {
		has_global_flag = KOREA_CEASEFIRE_UN
	}

	completion_reward = {
		add_political_power = 40
		custom_effect_tooltip = SOV_kwp_the_korean_settlement_tt
	}
}


# ############################################################################
#  ARM A - HISTORICAL DIVISION. Both Koreas survive along the line.
# ############################################################################

shared_focus = {
	id = SOV_kw_the_armistice_at_panmunjom
	icon = GFX_Hold_Conference_on_Neutrality
	text_icon = Soviet_focus_style
	cost = 10
	relative_position_id = SOV_kwp_the_korean_settlement
	x = 0
	y = 1
	prerequisite = { focus = SOV_kwp_the_korean_settlement }
	# The war-ended gate is repeated here, not just on the root. A focus that
	# declares its own allow_branch is evaluated on that block alone rather than
	# inheriting the parent's hidden state, so without these two lines this arm
	# renders from 1949 - both Koreas exist and the South has not lost yet.
	allow_branch = {
		has_global_flag = korean_war_begun
		has_global_flag = KOREA_CEASEFIRE_UN
		country_exists = KPA
		country_exists = KOR
		NOT = { KOR = { has_country_flag = lost_in_korean_war } }
	}
	available = {
		has_global_flag = KOREA_CEASEFIRE_UN
	}
	search_filters = { FOCUS_FILTER_POLITICAL }

	completion_reward = {
		add_political_power = 75
		add_stability = 0.05
		add_named_threat = { threat = -10 name = SOV_kw_panmunjom_threat }
		add_to_variable = {
			bonus_fpp_modifier = 0.5
			tooltip = bonus_fpp_modifier_tt
		}
		if = {
			limit = { country_exists = KPA }
			KPA = {
				set_temp_variable = { money_to_gain = 60 }
				add_money_with_tooltip_effect = yes
				add_stability = 0.05
			}
		}
		news_event = sov_korea_branch.2
		custom_effect_tooltip = SOV_kw_the_armistice_at_panmunjom_tt
	}
}
