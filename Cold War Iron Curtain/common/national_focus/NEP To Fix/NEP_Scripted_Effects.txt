NEP_Monthly_Factional_Display = {
    if = {
        limit = {
            check_variable = {
                Sphere_Leader = RAJ.id
            }
        }
        set_variable = { NEP_Foriegn_Ring = 7 }
    }
    else_if = {
        limit = {
            check_variable = {
                Sphere_Leader = PRC.id
            }
        }
        set_variable = { NEP_Foriegn_Ring = 8 }
    }
    else = {
        set_variable = { NEP_Foriegn_Ring = 10 }
    }
    if = {
        limit = {
            has_stability > 0.74
        }
        set_variable = { NEP_Stability_Ring = 5 }
    }
    else_if = {
        limit = {
            has_stability > 0.39
        }
        set_variable = { NEP_Stability_Ring = 4 }
    }
    else = {
        set_variable = { NEP_Stability_Ring = 6 }
    }
}

NEP_determine_top_two_factions = {
	set_temp_variable = { nep_top_1 = 0 }
	set_temp_variable = { nep_top_2 = 0 }
	set_temp_variable = { nep_top_1_value = -1 }
	set_temp_variable = { nep_top_2_value = -1 }

	# Monarchist
	set_temp_variable = { tmp_val = NEP_shah_monarchy_percent }
	add_to_temp_variable = { tmp_val = NEP_gorkha_nationalist_percent }

	if = {
		limit = { check_variable = { tmp_val > nep_top_1_value } }
		set_temp_variable = { nep_top_2 = nep_top_1 }
		set_temp_variable = { nep_top_2_value = nep_top_1_value }
		set_temp_variable = { nep_top_1 = 1 }
		set_temp_variable = { nep_top_1_value = tmp_val }
	}
	else_if = {
		limit = { check_variable = { tmp_val > nep_top_2_value } }
		set_temp_variable = { nep_top_2 = 1 }
		set_temp_variable = { nep_top_2_value = tmp_val }
	}

	# Democratic
	set_temp_variable = { tmp_val = NEP_nepali_congress_percent }
	add_to_temp_variable = { tmp_val = NEP_tanka_prasad_singh_faction_percent }

	if = {
		limit = { check_variable = { tmp_val > nep_top_1_value } }
		set_temp_variable = { nep_top_2 = nep_top_1 }
		set_temp_variable = { nep_top_2_value = nep_top_1_value }
		set_temp_variable = { nep_top_1 = 2 }
		set_temp_variable = { nep_top_1_value = tmp_val }
	}
	else_if = {
		limit = { check_variable = { tmp_val > nep_top_2_value } }
		set_temp_variable = { nep_top_2 = 2 }
		set_temp_variable = { nep_top_2_value = tmp_val }
	}

	# Rana
	if = {
		limit = { check_variable = { NEP_rana_percent > nep_top_1_value } }
		set_temp_variable = { nep_top_2 = nep_top_1 }
		set_temp_variable = { nep_top_2_value = nep_top_1_value }
		set_temp_variable = { nep_top_1 = 3 }
		set_temp_variable = { nep_top_1_value = NEP_rana_percent }
	}
	else_if = {
		limit = { check_variable = { NEP_rana_percent > nep_top_2_value } }
		set_temp_variable = { nep_top_2 = 3 }
		set_temp_variable = { nep_top_2_value = NEP_rana_percent }
	}

	# Communist
	if = {
		limit = { check_variable = { NEP_communist_party_nepal_percent > nep_top_1_value } }
		set_temp_variable = { nep_top_2 = nep_top_1 }
		set_temp_variable = { nep_top_2_value = nep_top_1_value }
		set_temp_variable = { nep_top_1 = 4 }
		set_temp_variable = { nep_top_1_value = NEP_communist_party_nepal_percent }
	}
	else_if = {
		limit = { check_variable = { NEP_communist_party_nepal_percent > nep_top_2_value } }
		set_temp_variable = { nep_top_2 = 4 }
		set_temp_variable = { nep_top_2_value = NEP_communist_party_nepal_percent }
	}
}

	set_temp_variable = { nep_bop_right_strength = 0 }
	set_temp_variable = { nep_bop_total_strength = 0 }
	set_temp_variable = { nep_bop_value = 0 }

	if = {
		limit = {
			has_country_flag = nep_bop_left_rana
			has_country_flag = nep_bop_right_dem
		}
		set_temp_variable = { nep_bop_left_strength = NEP_rana_percent }
		set_temp_variable = { nep_bop_right_strength = NEP_nepali_congress_percent }
		add_to_temp_variable = { nep_bop_right_strength = NEP_tanka_prasad_singh_faction_percent }
	}
	else_if = {
		limit = {
			has_country_flag = nep_bop_left_rana
			has_country_flag = nep_bop_right_shah
		}
		set_temp_variable = { nep_bop_left_strength = NEP_rana_percent }
		set_temp_variable = { nep_bop_right_strength = NEP_shah_monarchy_percent }
		add_to_temp_variable = { nep_bop_right_strength = NEP_gorkha_nationalist_percent }
	}
	else_if = {
		limit = {
			has_country_flag = nep_bop_left_rana
			has_country_flag = nep_bop_right_cpn
		}
		set_temp_variable = { nep_bop_left_strength = NEP_rana_percent }
		set_temp_variable = { nep_bop_right_strength = NEP_communist_party_nepal_percent }
	}
	else_if = {
		limit = {
			has_country_flag = nep_bop_left_dem
			has_country_flag = nep_bop_right_shah
		}
		set_temp_variable = { nep_bop_left_strength = NEP_nepali_congress_percent }
		add_to_temp_variable = { nep_bop_left_strength = NEP_tanka_prasad_singh_faction_percent }
		set_temp_variable = { nep_bop_right_strength = NEP_shah_monarchy_percent }
		add_to_temp_variable = { nep_bop_right_strength = NEP_gorkha_nationalist_percent }

		limit = {
			has_country_flag = nep_bop_left_dem
			has_country_flag = nep_bop_right_cpn

	else_if = {
		limit = {
			has_country_flag = nep_bop_left_shah
			has_country_flag = nep_bop_right_cpn
		}
		set_temp_variable = { nep_bop_left_strength = NEP_shah_monarchy_percent }
		add_to_temp_variable = { nep_bop_left_strength = NEP_gorkha_nationalist_percent }
		set_temp_variable = { nep_bop_right_strength = NEP_communist_party_nepal_percent }

	add_to_temp_variable = { nep_bop_total_strength = nep_bop_right_strength }

	if = {

		set_power_balance = {

		}
	}
	else = {
		set_power_balance = {
