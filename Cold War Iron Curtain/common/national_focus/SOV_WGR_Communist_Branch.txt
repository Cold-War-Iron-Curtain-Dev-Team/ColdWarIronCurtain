# ============================================================================
#  A RED BONN - 1949 West German election side branch for the SOV (Stalin) tree.
#
#  Entry chain:
#    SOV_Stalin_The_Zhdanov_Doctrine -> sov_german_question.1
#      -> SOV_WGR.1   (Moscow: interfere or stay out)
#         -> SOV_WGR.2   (Bonn/occupation authorities: expose or dismiss)
#            -> WGRElection.1 option c   (KPD victory, loads WGR_Reimann_49)
#               -> sets SOV_WGR49_kpd_victory on SOV, which reveals this branch.
#
#  If the KPD does not win, the flag is never set and nothing here appears.
#
#  This branch shares its coordinates with the Stalin Notes branch
#  (SOV_Stalin_Notes_Branch.txt). They can never both be active: a KPD victory
#  in 1949 suppresses germanquestion.1 in ic_pulse, so the 1952 Notes arc never
#  starts, and the Notes root additionally checks NOT = SOV_WGR49_kpd_victory.
#  This branch does no election work of its own - it exists purely to hook a
#  communist West Germany into the Soviet foreign-policy systems: the sphere
#  (add_to_sphere / sphere_tag_list), influence percentages, FPP power
#  projection (bonus_fpp_modifier), the budget (money_to_gain), market access
#  and the faction.
#
#  Shared focuses; only the root is listed in the SOV_Stalin focus_tree header.
# ============================================================================

# ---- ROOT -------------------------------------------------------------------
shared_focus = {
	id = SOV_wge_a_red_bonn
	icon = GFX_Dream_of_United_Germany
	text_icon = Soviet_focus_style
	cost = 3
	# Sits in the same region as the Stalin Notes branch, which it is mutually
	# exclusive with. Root at 42 puts this branch's leftmost column at 40 - the
	# same column the Notes branch starts at, and two columns clear of the main
	# tree's right edge (x = 37), so the player barely has to scroll.
	x = 42
	y = 0
	search_filters = { FOCUS_FILTER_POLITICAL }

	allow_branch = { has_country_flag = SOV_WGR49_kpd_victory }
	available = { has_country_flag = SOV_WGR49_kpd_victory }

	completion_reward = {
		add_political_power = 50
		add_to_variable = { bonus_fpp_modifier = 0.5 tooltip = bonus_fpp_modifier_tt }
		custom_effect_tooltip = SOV_wge_a_red_bonn_tt
	}
}

# ---- RECOGNITION ------------------------------------------------------------
shared_focus = {
	id = SOV_wge_recognize_the_reimann_government
	icon = GFX_Erection_Problem
	text_icon = Soviet_focus_style
	cost = 5
	relative_position_id = SOV_wge_a_red_bonn
	x = 0
	y = 1
	prerequisite = { focus = SOV_wge_a_red_bonn }
	available = { country_exists = WGR }
	search_filters = { FOCUS_FILTER_POLITICAL }

	completion_reward = {
		add_political_power = 40
		if = {
			limit = { country_exists = WGR }
			WGR = {
				add_opinion_modifier = { target = SOV modifier = large_increase }
				set_temp_variable = { percent_change = -15 }
				set_temp_variable = { tag_index = USA.id }
				set_temp_variable = { influence_target = THIS.id }
				change_influence_percentage = yes
				set_temp_variable = { percent_change = 15 }
				set_temp_variable = { tag_index = SOV.id }
				set_temp_variable = { influence_target = THIS.id }
				change_influence_percentage = yes
				WGR_Increase_KPD_Approval = yes
			}
			add_opinion_modifier = { target = WGR modifier = large_increase }
		}
		news_event = sov_german_question.4
		custom_effect_tooltip = SOV_wge_recognize_the_reimann_government_tt
	}
}

# ---- (i) THE SPHERE ---------------------------------------------------------
shared_focus = {
	id = SOV_wge_bonn_into_the_sphere
	icon = GFX_Warsaw_Pact
	text_icon = Soviet_focus_style
	cost = 8
	relative_position_id = SOV_wge_recognize_the_reimann_government
	x = -2
	y = 1
	prerequisite = { focus = SOV_wge_recognize_the_reimann_government }
	available = { country_exists = WGR }
	search_filters = { FOCUS_FILTER_POLITICAL }

	completion_reward = {
		add_to_variable = { bonus_fpp_modifier = 1 tooltip = bonus_fpp_modifier_tt }
		if = {
			limit = { country_exists = WGR }
			WGR = {
				set_temp_variable = { New_sphere_leader = SOV }
				add_to_sphere = yes
				WGR_Increase_KPD_State_Control = yes
				WGR_Increase_Communist_Influence = yes
			}
		}
		if = {
			limit = {
				is_faction_leader = yes
				country_exists = WGR
				NOT = { is_in_faction_with = WGR }
			}
			add_to_faction = WGR
		}
		custom_effect_tooltip = SOV_wge_bonn_into_the_sphere_tt
	}
}

	relative_position_id = SOV_wge_recognize_the_reimann_government

		country_exists = WGR
		country_exists = DDR
	}
	search_filters = { FOCUS_FILTER_POLITICAL }

	completion_reward = {
		add_political_power = 75
		add_to_variable = { bonus_fpp_modifier = 0.5 tooltip = bonus_fpp_modifier_tt }
		if = {
			limit = { country_exists = DDR }
			DDR = {
				increase_SED_party_power = yes
				increase_party_legitimacy = yes
				decrease_public_discontent = yes
				if = {
					limit = { country_exists = WGR }
					add_opinion_modifier = { target = WGR modifier = large_increase }
				}
			}
		}
		if = {
			limit = { country_exists = WGR }
			WGR = {
				add_opinion_modifier = { target = DDR modifier = large_increase }
				WGR_Increase_KPD_Approval = yes
			}
		}
		news_event = sov_german_question.5
		custom_effect_tooltip = SOV_wge_two_states_one_road_tt
	}
}
