add_namespace = CWIC_MNC

country_event = {
	id = CWIC_MNC.1000
	is_triggered_only = yes

	title = CWIC_MNC.1000.t
	desc = CWIC_MNC.1000.d
	immediate = {
		# If invite expired/removed, do nothing
		if = {
			limit = { NOT = { has_variable = cwic_invite_exxon_hydrocarbons } }
			hidden_effect = { }
		}
	}

	option = {
		name = CWIC_MNC.1000.a

		trigger = {
			NOT = { check_variable = { CWIC_nationalization_pressure_index > 4.5 } }
			OR = {
				NOT = { has_variable = corp_exxon_presence_hydrocarbons }
				check_variable = { corp_exxon_presence_hydrocarbons < 5 }
			}
		}

		custom_effect_tooltip = cwic_mnc_invite_accept_tt
		hidden_effect = {
			set_temp_variable = { cwic_mnc_before = 0 }
			if = {
				limit = { has_variable = corp_exxon_presence_hydrocarbons }
				set_temp_variable = { cwic_mnc_before = corp_exxon_presence_hydrocarbons }
			}
			cwic_corp_exxon_invest_presence_hydrocarbons = yes
			set_variable = { cwic_mnc_last_corp = 0 }
			set_variable = { cwic_mnc_last_sector = 1 }
			if = {
				limit = {
					has_variable = corp_exxon_presence_hydrocarbons
					check_variable = { corp_exxon_presence_hydrocarbons > cwic_mnc_before }
				}
				# The invitation is fulfilled: one invite buys one presence level.
				clear_variable = cwic_invite_exxon_hydrocarbons
				set_variable = { cwic_mnc_last_presence = corp_exxon_presence_hydrocarbons }
				cwic_mnc_apply_sector_presence_effects_country = yes
				set_variable = { cwic_mnc_last_tier = CWIC_mnc_presence_tier_hydrocarbons }
				country_event = { id = CWIC_MNC.1100 }
			}
			else = {
				# Corporation lacked the capital this month; the invite stays open
				# but loses time, so a broke corporation cannot hold it forever.
				subtract_from_variable = { cwic_invite_exxon_hydrocarbons = 12 }
				if = {
					limit = { check_variable = { cwic_invite_exxon_hydrocarbons < 1 } }
					clear_variable = cwic_invite_exxon_hydrocarbons
					set_temp_variable = { cwic_mnc_lapsed = 1 }
				}
				else = {
					set_temp_variable = { cwic_mnc_lapsed = 0 }
					set_country_flag = cwic_has_pending_mnc_invite
				}
				set_variable = { cwic_mnc_last_lapsed = cwic_mnc_lapsed }
				country_event = { id = CWIC_MNC.1101 }
			}
		}
	}

	option = {
		name = CWIC_MNC.1000.b
		hidden_effect = {

			# Longer cooldown on refusal
			set_country_flag = { flag = cwic_invite_cd_exxon_hydrocarbons days = 180 }

			# Corp perceives uncertainty (small shock timer bump)
			add_to_variable = { global.CWIC_CORP_exxon_shock_timer = 2 }
			clamp_variable = { var = global.CWIC_CORP_exxon_shock_timer min = 0 max = 120 }

			# Declining ends the invitation
			clear_variable = cwic_invite_exxon_hydrocarbons
		}
	}
}

	title = CWIC_MNC.1066.t
	desc = CWIC_MNC.1066.d
	immediate = {
		# If invite expired/removed, do nothing
		if = {
			limit = { NOT = { has_variable = cwic_invite_toshiba_electronics } }
			hidden_effect = { }
		}
	}

	option = {
		name = CWIC_MNC.1066.a

		trigger = {
			NOT = { check_variable = { CWIC_nationalization_pressure_index > 4.5 } }
			OR = {
				NOT = { has_variable = corp_toshiba_presence_electronics }
				check_variable = { corp_toshiba_presence_electronics < 5 }
			}
		}

		custom_effect_tooltip = cwic_mnc_invite_accept_tt
		hidden_effect = {
			set_temp_variable = { cwic_mnc_before = 0 }
			if = {
				limit = { has_variable = corp_toshiba_presence_electronics }
				set_temp_variable = { cwic_mnc_before = corp_toshiba_presence_electronics }
			}
			cwic_corp_toshiba_invest_presence_electronics = yes
			set_variable = { cwic_mnc_last_corp = 66 }
			set_variable = { cwic_mnc_last_sector = 5 }
			if = {
				limit = {
					has_variable = corp_toshiba_presence_electronics
					check_variable = { corp_toshiba_presence_electronics > cwic_mnc_before }
				}
				# The invitation is fulfilled: one invite buys one presence level.
				clear_variable = cwic_invite_toshiba_electronics
				set_variable = { cwic_mnc_last_presence = corp_toshiba_presence_electronics }
				cwic_mnc_apply_sector_presence_effects_country = yes
				set_variable = { cwic_mnc_last_tier = CWIC_mnc_presence_tier_electronics }
				country_event = { id = CWIC_MNC.1100 }
			}
			else = {
				# Corporation lacked the capital this month; the invite stays open
				# but loses time, so a broke corporation cannot hold it forever.
				subtract_from_variable = { cwic_invite_toshiba_electronics = 12 }
				if = {
					limit = { check_variable = { cwic_invite_toshiba_electronics < 1 } }
					clear_variable = cwic_invite_toshiba_electronics
					set_temp_variable = { cwic_mnc_lapsed = 1 }
				}
				else = {
					set_temp_variable = { cwic_mnc_lapsed = 0 }
					set_country_flag = cwic_has_pending_mnc_invite
				}
				set_variable = { cwic_mnc_last_lapsed = cwic_mnc_lapsed }
				country_event = { id = CWIC_MNC.1101 }
			}
		}
	}

	option = {
		name = CWIC_MNC.1066.b
		hidden_effect = {

			# Longer cooldown on refusal
			set_country_flag = { flag = cwic_invite_cd_toshiba_electronics days = 180 }

			# Corp perceives uncertainty (small shock timer bump)
			add_to_variable = { global.CWIC_CORP_toshiba_shock_timer = 2 }
			clamp_variable = { var = global.CWIC_CORP_toshiba_shock_timer min = 0 max = 120 }

			# Declining ends the invitation
			clear_variable = cwic_invite_toshiba_electronics
		}
	}
}
