add_namespace = CWIC_MNC

country_event = {
	id = CWIC_MNC.1000
	is_triggered_only = yes

	title = CWIC_MNC.1000.t
	desc = CWIC_MNC.1000.d
	immediate = {
		# If invite expired/removed, do nothing
		if = {
			limit = { NOT = { has_variable = cwic_invite_exxon_hydrocarbons } }
			hidden_effect = { }
		}
	}

	option = {
		name = CWIC_MNC.1000.a

		trigger = {
			NOT = { check_variable = { CWIC_nationalization_pressure_index > 4.5 } }

			OR = {
				check_variable = { global.CWIC_CORP_exxon_budget_remaining > global.CWIC_CORP_BUDGET_MIN }
				check_variable = { global.CWIC_CORP_exxon_budget_remaining = global.CWIC_CORP_BUDGET_MIN }
			}
			OR = {
				check_variable = { global.CWIC_CORP_exxon_cash > global.CWIC_CORP_BUDGET_MIN }
				check_variable = { global.CWIC_CORP_exxon_cash = global.CWIC_CORP_BUDGET_MIN }
			}

			OR = {
				NOT = { has_variable = corp_exxon_presence_hydrocarbons }
				check_variable = { corp_exxon_presence_hydrocarbons < 5 }
			}
		}

		hidden_effect = {
			cwic_corp_exxon_invest_presence_hydrocarbons = yes
		}

		# Shorten or clear the invite after acceptance
		subtract_from_variable = { cwic_invite_exxon_hydrocarbons = 6 }
		clamp_variable = { var = cwic_invite_exxon_hydrocarbons min = 0 max = 999 }
		if = {
			limit = {
				OR = {
					check_variable = { cwic_invite_exxon_hydrocarbons < 1 }
					check_variable = { cwic_invite_exxon_hydrocarbons = 0 }
				}
			}
			clear_variable = cwic_invite_exxon_hydrocarbons
		}
	}

	option = {
		name = CWIC_MNC.1000.b
		hidden_effect = {

			# Longer cooldown on refusal
			set_country_flag = cwic_invite_cd_exxon_hydrocarbons
			clr_country_flag = { flag = cwic_invite_cd_exxon_hydrocarbons days = 180 }

			# Corp perceives uncertainty (small shock timer bump)
			add_to_variable = { global.CWIC_CORP_exxon_shock_timer = 2 }
			clamp_variable = { var = global.CWIC_CORP_exxon_shock_timer min = 0 max = 120 }

			# Reduce invite duration more sharply
			subtract_from_variable = { cwic_invite_exxon_hydrocarbons = 12 }
			clamp_variable = { var = cwic_invite_exxon_hydrocarbons min = 0 max = 999 }
			if = {
				limit = {
					OR = {
						check_variable = { cwic_invite_exxon_hydrocarbons < 1 }
						check_variable = { cwic_invite_exxon_hydrocarbons = 0 }
					}
				}
				clear_variable = cwic_invite_exxon_hydrocarbons
			}
		}
	}
}
