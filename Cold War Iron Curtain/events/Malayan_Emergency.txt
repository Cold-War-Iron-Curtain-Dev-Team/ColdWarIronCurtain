#Border campaign callbacks. Win events fire to the side that won and apply the state change; loss events apply only the country penalty.
country_event = {
	id = malayan_emergency.1
	title = malayan_emergency.1.t
	desc = malayan_emergency.1.d
	picture = GFX_MLA_Power_Struggle.1
	is_triggered_only = yes

	immediate = {
		every_state = {
			limit = { OR = { has_state_flag = MLA_liberation_campaign has_state_flag = MAL_clearing_campaign } }
			set_state_controller_to = ROOT
			if = {
				limit = { ROOT = { tag = MLA } }
				add_to_variable = { emergency_control = 30 }
			}
			else = {
				add_to_variable = { emergency_control = -30 }
			}
			clamp_variable = { var = emergency_control min = -100 max = 100 }
			MAL_emergency_set_control_frame = yes
			clr_state_flag = MLA_liberation_campaign
			clr_state_flag = MAL_clearing_campaign
		}
		MLA = { MAL_emergency_recount = yes }
	}

	option = {
		name = malayan_emergency.1.a
	}
}
