# ----------------------------------------------------------------
# Dien Bien Phu fallback - exact-province Operation Castor normally establishes
# the French airhead. This chain only runs when the CEFEO never fortified it.
# ----------------------------------------------------------------
country_event = {
    id = SWF_Indochina_War.14
    hidden = yes

    is_triggered_only = yes

    immediate = {
        if = {
            limit = {
                tag = VIN
                country_exists = TAI
                NOT = { owns_state = 671 }
                NOT = { date > 1956.1.1 }	# hard stop: past this the war is decided elsewhere
                VIN_indochina_campaign_window_trigger = yes
                NOT = {
                    AND = {
                        country_exists = FRE
                        FRE = { has_country_flag = FRE_Dien_Bien_Phu_Established }
                    }
                }
            }
            if = {
                limit = { # Castor may still establish the camp; wait only while its result is unresolved
                    country_exists = FRE
                    FRE = {
                        OR = {
                            has_country_flag = FRE_Mission_Operation_Castor_Activated
                            AND = {
                                has_country_flag = FRE_Operation_Castor_Launched
                                NOT = { date > 1954.9.1 }
                                NOT = {
                                    OR = {
                                        has_country_flag = FRE_Operation_Castor_Result_Success
                                        has_country_flag = FRE_Operation_Castor_Result_Failure
                                        has_country_flag = FRE_Operation_Castor_Result_Superseded
                                    }
                                }
                            }
                        }
                    }
                }
                country_event = { id = SWF_Indochina_War.14 days = 60 }
            }
            else_if = {
                limit = {	# the divisions are not in position, or another campaign is still in the field
                    OR = {
                        NOT = { owns_state = 1761 }
                        NOT = { VIN_no_campaign_running_trigger = yes }
                        NOT = { VIN_campaign_cooldown_ready_trigger = yes }
                    }
                }
                country_event = { id = SWF_Indochina_War.14 days = 45 }
            }
            else = {
                country_event = { id = SWF_Indochina_War.17 days = 1 }
            }
        }
    }
}

country_event = {
    id = SWF_Indochina_War.17
    title = SWF_Indochina_War.17.t
    desc = SWF_Indochina_War.17.d
    picture = GFX_report_event_tanks_vietnam

    is_triggered_only = yes

    trigger = {	# re-checked on arrival: a scheduled copy must not fire into a valley the French have since taken
        tag = VIN
        country_exists = TAI
        NOT = { owns_state = 671 }
        owns_state = 1761
        VIN_indochina_campaign_window_trigger = yes
        VIN_no_campaign_running_trigger = yes
        VIN_campaign_cooldown_ready_trigger = yes
        NOT = {
            AND = {
                country_exists = FRE
                FRE = { has_country_flag = FRE_Dien_Bien_Phu_Established }
            }
        }
    }

    option = {	# take the valley
        name = SWF_Indochina_War.17.a
        ai_chance = { base = 85 }
        if = {
            limit = {
                country_exists = FRE
            }
            fre_tai_highlands_relief = yes
            fre_dbp_fortify_camp = yes	# the valley is dug in: province 4529 is the objective, not the state
        }
        set_country_flag = VIN_Campaign_Dien_Bien_Active
        set_temp_variable = { vin_camp_id = 3 }
        vin_campaign_begin = yes
        custom_effect_tooltip = VIN_Camp_War_Dien_Bien_tt
        vin_campaign_declare_war = yes
        if = {
            limit = {
                NOT = { has_global_flag = Indochina_War_Over }
            }
            FRA = {
                add_to_variable = {
					StruggleInvolvedNationsIndochinaCommunistScore = 15
                    tooltip = Add_to_Communist_Victory_Points_TT
                }
                add_to_variable = {
					global.Indochina_War_Next_Phase_A_Points = 25
                    tooltip = Add_to_Escalation_Phase_TT
                }
            }
        }
    }

    option = {	# leave the highlands alone
        name = SWF_Indochina_War.17.b
        ai_chance = { base = 15 }
    }
}
