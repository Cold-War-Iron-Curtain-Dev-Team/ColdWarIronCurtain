# ------------------------------------------------------------------
country_event = {
	id = VIE_VIN_Reunification_50s.2
	title = VIE_VIN_Reunification_50s.2.t
	desc = VIE_VIN_Reunification_50s.2.d
	picture = GFX_report_event_vietnam_reunification

	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = VIE_VIN_Reunification_50s.2.a
		ai_chance = { base = 15 }

		set_country_flag = VIE_reunification_election_accepted
		add_stability = -0.02
		add_opinion_modifier = {
			target = VIN
			modifier = medium_increase
		}

		VIN = {
			set_country_flag = VIN_reunification_election_approved
			add_opinion_modifier = {
				target = VIE
				modifier = medium_increase
			}
		}
	}

	option = {
		name = VIE_VIN_Reunification_50s.2.b
		ai_chance = { base = 85 }

		set_country_flag = VIE_reunification_election_refused
		add_stability = 0.02
		add_opinion_modifier = {
			target = VIN
			modifier = medium_decrease
		}

		VIN = {
			set_country_flag = VIN_reunification_election_cancelled
			add_opinion_modifier = {
				target = VIE
				modifier = medium_decrease
			}
		}
	}
}
