add_namespace = FRE_Pollux

country_event = {
	id = FRE_Pollux.1
	title = FRE_Pollux.1.t
	desc = FRE_Pollux.1.d
	picture = GFX_report_event_generic_soldiers

	is_triggered_only = yes

	option = {
		name = FRE_Pollux.1.a
		trigger = {
			check_variable = { FRE_War_Credits > 29 }
			OR = {
				has_equipment = { transport_plane_equipment > 2 }
				has_country_flag = FRE_Air_Assets_Delivered
			}
		}
		ai_chance = {
			base = 80
			modifier = { factor = 10  is_historical_focus_on = yes }
		}
		custom_effect_tooltip = FRE_Pollux_Historical_tt
		fre_pollux_begin_historical = yes
	}

	option = {
		name = FRE_Pollux.1.b
		trigger = { check_variable = { FRE_War_Credits > 79 } }
		ai_chance = {
			base = 15
			modifier = { factor = 3  has_country_flag = FRE_Hoa_Binh_Mobile_Groups_Preserved }
			modifier = {
				factor = 0
				is_historical_focus_on = yes
				check_variable = { FRE_War_Credits > 29 }
				OR = {
					has_equipment = { transport_plane_equipment > 2 }
					has_country_flag = FRE_Air_Assets_Delivered
				}
			}
		}
		custom_effect_tooltip = FRE_Pollux_Escort_tt
		fre_pollux_begin_escort = yes
	}

	option = {
		name = FRE_Pollux.1.c
		trigger = {
			check_variable = { FRE_War_Credits > 119 }
			OR = {
				has_equipment = { transport_plane_equipment > 7 }
				has_country_flag = FRE_Air_Assets_Delivered
			}
		}
		ai_chance = {
			base = 5
			modifier = { factor = 4  check_variable = { FRE_Metropole_Patience > 65 } }
			modifier = {
				factor = 0
				is_historical_focus_on = yes
				check_variable = { FRE_War_Credits > 29 }
				OR = {
					has_equipment = { transport_plane_equipment > 2 }
					has_country_flag = FRE_Air_Assets_Delivered
				}
			}
		}
		custom_effect_tooltip = FRE_Pollux_Airlift_tt
		fre_pollux_begin_airlift = yes
	}

	option = {
		name = FRE_Pollux.1.e
		trigger = {
			OR = {
				check_variable = { FRE_War_Credits < 30 }
				AND = {
					check_variable = { FRE_War_Credits < 80 }
					NOT = { has_equipment = { transport_plane_equipment > 2 } }
					NOT = { has_country_flag = FRE_Air_Assets_Delivered }
				}
			}
		}
		ai_chance = { base = 100 }
		custom_effect_tooltip = FRE_Pollux_Unfunded_tt
		fre_pollux_abandon_unfunded = yes
	}
}
