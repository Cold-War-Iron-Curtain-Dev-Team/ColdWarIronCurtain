country_event = {
	id = DDR_1956.3
	title = DDR_1956.3.t
	desc = DDR_1956.3.d
	picture = GFX_report_event_berlin_protest

	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		set_country_flag = DDR_1956_unrest_resolved
	}

	# --------------------------------------------------------
	# No major DDR break: existing leadership survives.
	# --------------------------------------------------------

	option = {
		name = DDR_1956.3.a

		trigger = {
			NOT = {
				DDR_1956_high_unrest = yes
			}
		}

		set_country_flag = DDR_1956_unrest_contained

		# The crisis leaves a mark, but does not force a new tree.
		add_stability = -0.02
	}


	# --------------------------------------------------------
	# Kádár-equivalent route:
	# high unrest + weak intelligentsia
	# --------------------------------------------------------

	option = {
		name = DDR_1956.3.b

		trigger = {
			DDR_1956_high_unrest = yes

			NOT = {
				DDR_1956_high_intelligentsia_strength = yes
			}
		}

		set_country_flag = DDR_1956_steinhoff_path

		# Insert your existing Steinhoff leader handoff here,
		# if it is not already handled by the new focus tree.
		#

		load_focus_tree = GDR_1956_Steinhoff
        create_country_leader = {
            name = "Karl STeinhoff"
            picture = "Karl_Steinhoff.dds"
            ideology = liberal_communism
        }
        

		increase_stasi_power = yes
	}


	# --------------------------------------------------------
	# Gomułka-equivalent route:
	# high unrest + strong intelligentsia
	# --------------------------------------------------------

	option = {
		name = DDR_1956.3.c

		trigger = {
			DDR_1956_high_unrest = yes
			DDR_1956_high_intelligentsia_strength = yes
		}

		set_country_flag = DDR_1956_ackermann_path

		# Insert your existing Ackermann leader handoff here,
		# if it is not already handled by the new focus tree.
		#
		# promote_character = DDR_Anton_Ackermann
        create_country_leader = {
            name = "Anton Ackermann"
            picture = "Anton_Ackermann.dds"
            ideology = marxism_leninism
        }

		load_focus_tree = GDR_1956_Ackerman

		decrease_stasi_power = yes
	}
}
