###########################################################################
#  The German Question — entry event for the Stalin Notes side branch.
#  Auto-fires once for the USSR from ~March 1952 while Stalin leads (see
#  common/on_actions/SOV_German_Question_on_actions.txt). Setting the flag
#  SOV_german_question_active reveals the side branch on the Stalin tree.
###########################################################################

add_namespace = germanquestion

country_event = {
	id = germanquestion.1
	title = germanquestion.1.t
	desc = germanquestion.1.d
	picture = GFX_StalinNotes_1

	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = germanquestion.1.a		# Address the German question
		set_country_flag = SOV_german_question_active
		mark_focus_tree_layout_dirty = yes
		complete_national_focus = SOV_sn_the_german_question
		add_political_power = 25
		ai_chance = { factor = 10 }
	}
	option = {
		name = germanquestion.1.b		# Not yet — reconsider later
		set_country_flag = SOV_german_question_active
		hidden_effect = {
			# Branch still appears; the player simply hasn't committed. Kept as a
			# separate option purely for flavor / roleplay.
			mark_focus_tree_layout_dirty = yes
			complete_national_focus = SOV_sn_the_german_question
		}
		ai_chance = { factor = 1 }
	}
}
