country_event = {
    id = BRU_sultan.3
    immediate = { log = "[GetDateText]: [Root.GetTag]: id = BRU_sultan.3" }
    title = BRU_sultan.3.t
    desc = BRU_sultan.3.d
    picture = GFX_BRU_sultan.3
    is_triggered_only = yes
    fire_only_once = no
    option = {
        name = BRU_sultan.3.a
		if = {
			limit = {
				has_country_flag = BRU_Heir_Saifuddien
			}
			country_event = { id = BRU_Sultan.5 days = 1 }
		}
		if = {
			limit = {
				has_country_flag = BRU_Heir_Ehsani
			}
			country_event = { id = BRU_Sultan.6 days = 1 }
		}
		if = {
			limit = {
				has_country_flag = BRU_Heir_MacBryan
			}
			country_event = { id = BRU_MacBryan.3 days = 1 }
		}
		if = {
			limit = {
				AND = {
					NOT = { has_country_flag = BRU_Heir_Ehsani }
					NOT = { has_country_flag = BRU_Heir_MacBryan }
					NOT = { has_country_flag = BRU_Heir_Saifuddien }
				}
			}
			country_event = { id = BRU_Sultan.7 days = 1 }
			hidden_effect = { recruit_character = BRU_Gerald_MacBryan }
		}
		complete_national_focus = BRU_Sultan_Death
    }
}
