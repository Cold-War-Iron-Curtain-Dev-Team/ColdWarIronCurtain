#1952.12.1
country_event = {
	id = mex.1
 	immediate = {log = "[GetDateText]: [Root.GetTag]: id = mex.1"}
    title = "Adolfo Ruiz Cortines Wins 1952 Mexican Election"
	desc = mex.1.d
	
	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		set_popularities = { 
			communism = 15
			Social_Democratic = 15
			Liberal_Conservatism = 10
			conservative = 25
			centrist = 35
		}
		
		set_politics = {
			ruling_party = centrist
			last_election = "1952.12.1"
			election_frequency = 72
			elections_allowed = yes
		}
		
		create_country_leader = {
			name = "Adolfo Ruiz Cortines"
			picture = "Adolfo_Ruiz_Cortines_50s.dds"
			desc = POLITICS_ADOLFO_RUIZ_CORTINES_DESC
			ideology = mexican_constitutionalism
		}
		
		news_event = news.1674
	}

	option= {
		name = mex.1.a
 	}
}
