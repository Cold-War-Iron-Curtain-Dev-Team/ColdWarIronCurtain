				add_to_variable = {

				add_to_variable = {
					global.Indochina_War_Next_Phase_B_Points = 50
					tooltip = Add_to_De_Escalation_Phase_TT
				}
			}
		}
		if = {
			limit = { country_exists = LOS }
			LOS = {
				add_stability = -0.05
				add_popularity = { ideology = communism popularity = 0.08 }
			}
		}
	}
}
