# Early completion records a contingency; the live Northwest launch reoffers it.
country_event = {
	id = FRE_Northwest.1
	title = FRE_Northwest.1.t
	desc = {
		text = FRE_Northwest.1.d_active
		trigger = {
			country_exists = VIN
			VIN = { has_country_flag = VIN_Campaign_Tay_Bac_Active }
		}
	}
	desc = {
		text = FRE_Northwest.1.d_planned
		trigger = {
			OR = {
				NOT = { country_exists = VIN }
				VIN = { NOT = { has_country_flag = VIN_Campaign_Tay_Bac_Active } }
			}
		}
	}
	picture = GFX_report_event_generic_soldiers
	is_triggered_only = yes

	option = {
		name = FRE_Northwest.1.a
		trigger = {
			country_exists = VIN
			VIN = { has_country_flag = VIN_Campaign_Tay_Bac_Active }
			check_variable = { FRE_War_Credits > 99 }
		}
		ai_chance = {
			base = 80
			modifier = {
				factor = 0.25
				VIN = { check_variable = { VIN_camp_days_elapsed > 89 } }
			}
		}
		custom_effect_tooltip = FRE_Lorraine_Deep_Thrust_tt
		fre_lorraine_commit_deep_thrust = yes
	}

	option = {
		name = FRE_Northwest.1.b
		trigger = {
			country_exists = VIN
			VIN = { has_country_flag = VIN_Campaign_Tay_Bac_Active }
			check_variable = { FRE_War_Credits > 59 }
		}
		ai_chance = { base = 15 }
		custom_effect_tooltip = FRE_Lorraine_Limited_Raid_tt
		fre_lorraine_commit_limited_raid = yes
	}

	option = {
		name = FRE_Northwest.1.c
		trigger = {
			country_exists = VIN
			VIN = { has_country_flag = VIN_Campaign_Tay_Bac_Active }
			check_variable = { FRE_War_Credits > 29 }
		}
		ai_chance = {
			base = 5
			modifier = {
				factor = 8
				check_variable = { FRE_War_Credits < 60 }
			}
		}
		custom_effect_tooltip = FRE_Lorraine_Withhold_tt
		fre_lorraine_withhold_for_na_san = yes
	}

	option = {
		name = FRE_Northwest.1.e
		trigger = {
			country_exists = VIN
			VIN = { has_country_flag = VIN_Campaign_Tay_Bac_Active }
			check_variable = { FRE_War_Credits < 30 }
		}
		ai_chance = { base = 100 }
		custom_effect_tooltip = FRE_Lorraine_No_Reserve_tt
		fre_lorraine_decline_response = yes
	}

	option = {
		name = FRE_Northwest.1.d
		trigger = {
			OR = {
				NOT = { country_exists = VIN }
				VIN = { NOT = { has_country_flag = VIN_Campaign_Tay_Bac_Active } }
			}
		}
		custom_effect_tooltip = FRE_Lorraine_Planned_tt
	}
}
