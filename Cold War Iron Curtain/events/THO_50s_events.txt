# ============================================================================
# Viet Bac Autonomous Region
# Hanoi's post-settlement decision on the Tay and Nung marches, mirroring the
# Tai-Meo chain in MEO_50s_events.txt.
#   THO.1 - fired in VIN by vin_viet_bac_question_check. Hanoi decides.
#   THO.2 - fired in VIN by THO.1's first option. Constitutes the region.
# Both are plain country_events on purpose: major = yes on a country_event
# broadcasts to every country in the game.
# Refusing at THO.1 ends the chain outright - VIN keeps 1280, 1768, 881 and
# 1767 and no half-built country is left behind.
# ============================================================================
add_namespace = THO

country_event = {
	id = THO.1
	title = THO.1.t
	desc = THO.1.d
	picture = GFX_report_event_generic_rally

	is_triggered_only = yes

	option = {
		name = THO.1.a	# the historical answer - a region under Hanoi's supervision
		ai_chance = { factor = 90 }
		set_country_flag = VIN_viet_bac_zone_granted
		country_event = { id = THO.2 days = 3 }
	}
	option = {
		name = THO.1.b	# govern the marches as ordinary provinces
		ai_chance = { factor = 10 }
		set_country_flag = VIN_viet_bac_zone_refused
	}
}

country_event = {
	id = THO.2
	title = THO.2.t
	desc = THO.2.d
	picture = GFX_report_event_generic_rally

	is_triggered_only = yes

	option = {
		name = THO.2.a
		custom_effect_tooltip = THO_viet_bac_establishment_TT
		vin_establish_viet_bac_zone = yes
	}
}
