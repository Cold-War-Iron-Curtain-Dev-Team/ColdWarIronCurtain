add_namespace = VIN_Camp

# ============================================================================
# Viet Minh campaign commitment prompt. Driven by VIN_Campaign_Effects.txt.
# The campaign's border war is opened by the focus itself; this decides how
# much of the regular army marches with it.
# ============================================================================

country_event = {
	id = VIN_Camp.1 # How much of the army goes forward
	title = VIN_Camp.1.t
	desc = VIN_Camp.1.d
	picture = GFX_report_event_tanks_vietnam

	is_triggered_only = yes

	option = {
		name = VIN_Camp.1.a	# Light
		ai_chance = {
			factor = 1
			modifier = {
				factor = 4
				check_variable = { VIN_Campaign_Supply < 90 }
			}
		}
		trigger = {
			check_variable = { VIN_Campaign_Supply > VIN_camp_price_light }
		}
		custom_effect_tooltip = VIN_Camp_Commit_Light_tt
		set_temp_variable = { vin_camp_tier = 0 }
		vin_campaign_commit = yes
	}

	option = {
		name = VIN_Camp.1.b	# Standard
		ai_chance = {
			factor = 6
			modifier = {
				factor = 0
				check_variable = { VIN_Campaign_Supply < VIN_camp_price_standard }
			}
		}
		trigger = {
			check_variable = { VIN_Campaign_Supply > VIN_camp_price_standard }
		}
		custom_effect_tooltip = VIN_Camp_Commit_Standard_tt
		set_temp_variable = { vin_camp_tier = 1 }
		vin_campaign_commit = yes
	}

	option = {
		name = VIN_Camp.1.c	# Maximum
		ai_chance = {
			factor = 20	# an AI Hanoi that commits light pays for a campaign and loses it
			modifier = {
				factor = 0
				check_variable = { VIN_Campaign_Supply < VIN_camp_price_maximum }
			}
			modifier = {
				factor = 4
				check_variable = { VIN_camp_current = 3 }	# everything goes to Dien Bien Phu
			}
		}
		trigger = {
			check_variable = { VIN_Campaign_Supply > VIN_camp_price_maximum }
		}
		custom_effect_tooltip = VIN_Camp_Commit_Maximum_tt
		set_temp_variable = { vin_camp_tier = 2 }
		vin_campaign_commit = yes
	}

	option = {
		name = VIN_Camp.1.e	# The porters cannot carry it: the campaign goes forward unsupported
		trigger = {
			NOT = { check_variable = { VIN_Campaign_Supply > VIN_camp_price_light } }	# exact complement of option a
		}
		custom_effect_tooltip = VIN_Camp_Commit_None_tt
		set_variable = { VIN_camp_commitment_tier = 0 }
	}
}
