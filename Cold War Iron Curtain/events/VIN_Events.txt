country_event = {
	id = Binh_Cung_Leadership.5
	title = Binh_Cung_Leadership.5.t
	desc = Binh_Cung_Leadership.5.d
	picture = GFX_Binh_Cung_Leadership_5

	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = Binh_Cung_Leadership.5.a
		ai_chance = {
			base = 70
		}
		add_stability = -0.05
		set_country_flag = Cung_Defection
		if = {
			limit = { has_character = NLF_Kieu_Cong_Cung }
			NLF_Kieu_Cong_Cung = {
				set_nationality = VIE
				remove_country_leader_role = { ideology = syndicalism }
				set_portraits = {
					army = { large = "GFX_Kieu_Cong_Cung_ARVN" }
				}
			}
		}
		if = {
			limit = { country_exists = NLF }
			add_opinion_modifier = {
				target = NLF
				modifier = small_decrease
			}
			NLF = {
				add_opinion_modifier = {
					target = VIE
					modifier = small_decrease
				}
				if = {
					limit = { has_country_flag = Kieu_Cong_Cung_Leader }
					set_temp_variable = { nlf_command_choice = 2 }
					if = {
						limit = { date > 1953.3.01 }
						set_temp_variable = { nlf_command_choice = 3 }
					}
					nlf_set_southern_command = yes
				}
				country_event = Binh_Cung_Leadership.6
			}
		}
		if = {
			limit = { country_exists = VIN }
			VIN = { country_event = Binh_Cung_Leadership.6 }
		}
	}

	option = {
		name = Binh_Cung_Leadership.5.b
		ai_chance = {
			base = 30
		}
		add_stability = 0.05
		add_political_power = 25
		if = {
			limit = { country_exists = NLF }
			NLF = {
				if = {
					limit = { has_country_flag = Kieu_Cong_Cung_Leader }
					set_temp_variable = { nlf_command_choice = 2 }
					if = {
						limit = { date > 1953.3.01 }
						set_temp_variable = { nlf_command_choice = 3 }
					}
					nlf_set_southern_command = yes
				}
				if = {
					limit = { has_character = NLF_Kieu_Cong_Cung }
					NLF_Kieu_Cong_Cung = { retire = yes }
				}
			}
		}
	}
}
