# 1. The Liquidation of the KGB & The MBRF (January 1992)
country_event = {
	id = RUS_FSB90.1
	title = RUS_FSB90.1.t
	desc = RUS_FSB90.1.d
	picture = GFX_report_event_soviet_purge_trial
	is_triggered_only = yes

	immediate = {
		set_country_flag = RUS_mbrf_resolved
	}

	option = {
		name = RUS_FSB90.1.a
		add_political_power = 50
		add_stability = 0.02
		add_ideas = RUS_mbrf_ministry_of_security
		set_country_flag = RUS_mbrf_formed
		ai_chance = { factor = 85 }
	}
	option = {
		name = RUS_FSB90.1.b
		add_political_power = -30
		add_stability = -0.04
		if = {
			limit = { check_variable = { Russian_Chaos < 95 } }
			add_to_variable = { Russian_Chaos = 5 }
		}
		set_country_flag = RUS_mbrf_weakened
		ai_chance = { factor = 15 }
	}
}
