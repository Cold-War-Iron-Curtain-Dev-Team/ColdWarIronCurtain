add_namespace = cwic_esp

# Captured operative: one event per squad seat, so several captures from one
# operation each get their own answer. The decisions in CWIC_ESPIONAGE_PERSONNEL
# offer the same choices later for anyone who waits.
country_event = {
	id = cwic_esp.1
	title = cwic_esp.1.t
	desc = cwic_esp.captured.d
	picture = GFX_report_event_portugal_navy_mutiny_prisoners
	is_triggered_only = yes
	option = {
		name = cwic_esp.captured.a
		trigger = {
			NOT = { has_war_with = var:CWIC_esp_mia_target_1 }
			has_political_power > 24
			NOT = { check_variable = { currentMoney < 0.005 } }
		}
		ai_chance = { factor = 70 }
		custom_effect_tooltip = CWIC_ESP_EXCHANGE_TT
		add_political_power = -25
		subtract_from_variable = { currentMoney = 0.005 }
		set_temp_variable = { CWIC_esp_seat = 1 }
		CWIC_espionage_exchange_operative = yes
	}
	option = {
		name = cwic_esp.captured.b
		ai_chance = { factor = 30 }
		custom_effect_tooltip = CWIC_ESP_DISAVOW_TT
		subtract_from_variable = { network_strength^CWIC_esp_mia_target_1 = 0.05 }
		clamp_variable = { var = network_strength^CWIC_esp_mia_target_1 min = 0 max = 1 }
		set_temp_variable = { CWIC_esp_seat = 1 }
		CWIC_espionage_disavow_operative = yes
	}
	option = {
		name = cwic_esp.captured.c
		ai_chance = { factor = 0 }
		custom_effect_tooltip = CWIC_ESP_WAIT_TT
	}
}
