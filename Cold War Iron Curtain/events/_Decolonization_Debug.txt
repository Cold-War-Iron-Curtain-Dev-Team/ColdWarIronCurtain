#----------------------------------------------
# BRITISH-TRUCIAL SHEIKHDOMS TREATY EXPIRED
# ROOT = ENG
# BUG FIX: Original only freed UAE, QAT, BHR. The 6 individual Trucial
# sheikhdoms (DUB, SHR, FUJ, RAS, UMM, AJM) remained ENG subjects.
# Fix: free all 7 emirates then annex the 6 into UAE to form the federation.
# QAT and BHR become independent separately (historically they chose not to join).
#----------------------------------------------
country_event = {
	id = debug_decolon.16
	title = decolonization.16.t
	desc = decolonization.16.d
	is_triggered_only = yes
	fire_only_once = yes
	hidden = yes
	immediate = {
		log = "[GetDateText]: [Root.GetTag]: DEBUG id = debug_decolon.16 (Trucial States/UAE formation)"

		# Free UAE (Abu Dhabi) first — it becomes the federation core
		if = {
			limit = { UAE = { is_subject_of = ROOT } }
			set_autonomy = {
				target = UAE
				autonomy_state = autonomy_free
			}
		}

		# Free the 6 individual Trucial sheikhdoms that were missing from original event
		if = {
			limit = { DUB = { is_subject_of = ROOT } }
			set_autonomy = { target = DUB autonomy_state = autonomy_free }
		}
		if = {
			limit = { SHR = { is_subject_of = ROOT } }
			set_autonomy = { target = SHR autonomy_state = autonomy_free }
		}
		if = {
			limit = { FUJ = { is_subject_of = ROOT } }
			set_autonomy = { target = FUJ autonomy_state = autonomy_free }
		}
		if = {
			limit = { RAS = { is_subject_of = ROOT } }
			set_autonomy = { target = RAS autonomy_state = autonomy_free }
		}
		if = {
			limit = { UMM = { is_subject_of = ROOT } }
			set_autonomy = { target = UMM autonomy_state = autonomy_free }
		}
		if = {
			limit = { AJM = { is_subject_of = ROOT } }
			set_autonomy = { target = AJM autonomy_state = autonomy_free }
		}

		# Absorb the 6 Trucial sheikhdoms — route depends on whether QAT_UGE has formed
		if = {
			# QAT-led Gulf federation: QAT_UGE absorbs the Trucial States instead of UAE
			limit = { QAT = { has_cosmetic_tag = QAT_UGE } }
			QAT = {
				if = { limit = { country_exists = DUB } annex_country = { target = DUB transfer_troops = no } }
				if = { limit = { country_exists = SHR } annex_country = { target = SHR transfer_troops = no } }
				if = { limit = { country_exists = FUJ } annex_country = { target = FUJ transfer_troops = no } }
				if = { limit = { country_exists = RAS } annex_country = { target = RAS transfer_troops = no } }
				if = { limit = { country_exists = UMM } annex_country = { target = UMM transfer_troops = no } }
				if = { limit = { country_exists = AJM } annex_country = { target = AJM transfer_troops = no } }
			}
		}
		else_if = {
			# BHR-led Gulf federation: BHR (as QAT_UGE) absorbs the Trucial States
			limit = { BHR = { has_cosmetic_tag = QAT_UGE } }
			BHR = {
				if = { limit = { country_exists = DUB } annex_country = { target = DUB transfer_troops = no } }
				if = { limit = { country_exists = SHR } annex_country = { target = SHR transfer_troops = no } }
				if = { limit = { country_exists = FUJ } annex_country = { target = FUJ transfer_troops = no } }
				if = { limit = { country_exists = RAS } annex_country = { target = RAS transfer_troops = no } }
				if = { limit = { country_exists = UMM } annex_country = { target = UMM transfer_troops = no } }
				if = { limit = { country_exists = AJM } annex_country = { target = AJM transfer_troops = no } }
			}
		}
		else = {
			# Normal path: UAE forms and absorbs the Trucial States
			UAE = {
				leave_faction = yes
				drop_cosmetic_tag = yes
				set_cosmetic_tag = UAE_unified
				conversion_to_rentier = yes
				add_ideas = un_member
				if = { limit = { country_exists = DUB } annex_country = { target = DUB transfer_troops = no } }
				if = { limit = { country_exists = SHR } annex_country = { target = SHR transfer_troops = no } }
				if = { limit = { country_exists = FUJ } annex_country = { target = FUJ transfer_troops = no } }
				if = { limit = { country_exists = RAS } annex_country = { target = RAS transfer_troops = no } }

		# is why "Senegal never left the Mali Federation". Recreate SEN from MLI's
