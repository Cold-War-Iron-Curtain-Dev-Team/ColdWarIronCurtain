add_namespace = VIE_Border_Settlement

country_event = {	# fires to the owner of the disputed state
	id = VIE_Border_Settlement.1
	title = VIE_Border_Settlement.1.t
	desc = VIE_Border_Settlement.1.d
	picture = GFX_French_Drawn_Borders.1

	is_triggered_only = yes

	option = {	# cede the province
		name = VIE_Border_Settlement.1.a

		ai_chance = {
			base = 10
			modifier = {
				factor = 3
				has_opinion = { target = VIE value > 25 }
			}
			modifier = {
				factor = 4
				strength_ratio = { tag = VIE ratio < 0.4 }
			}
			modifier = {
				factor = 2
				has_war = yes
			}
			modifier = {	# a long campaign of pressure wears the frontier down
				factor = 2
				check_variable = { event_target:VIE_border_demand_target.VIE_border_pressure > 74 }
			}
			modifier = {
				factor = 0.2
				strength_ratio = { tag = VIE ratio > 0.9 }
			}
			modifier = {
				factor = 0.25
				is_subject = yes	# a patron will not let us give ground away
			}
		}
		VIE = { country_event = { id = VIE_Border_Settlement.2 days = 1 } }
	}

	option = {	# refuse
		name = VIE_Border_Settlement.1.b

		ai_chance = {
			base = 90
			modifier = {
				factor = 0.3
				has_opinion = { target = VIE value > 50 }
			}
			modifier = {
				factor = 0.4
				check_variable = { event_target:VIE_border_demand_target.VIE_border_pressure > 74 }
			}
		}
		VIE = { country_event = { id = VIE_Border_Settlement.3 days = 1 } }
	}
}

country_event = {	# the frontier is ceded at the table
	id = VIE_Border_Settlement.2
	title = VIE_Border_Settlement.2.t
	desc = VIE_Border_Settlement.2.d
	picture = GFX_French_Drawn_Borders.1

	is_triggered_only = yes

	option = {
		name = VIE_Border_Settlement.2.a

		clr_country_flag = VIE_Border_Demand_Pending
		add_stability = 0.03
		add_political_power = 25
		var:VIE_border_demand_state = {	# resolve the owner before the state changes hands
			owner = {
				add_opinion_modifier = {
					target = VIE
					modifier = VIE_Border_Ceded_Peacefully
				}
				reverse_add_opinion_modifier = {
					target = VIE
					modifier = VIE_Border_Ceded_Peacefully
				}
			}
		}
		set_temp_variable = { vie_border_settle_state = VIE_border_demand_state }
		VIE_border_settle_target_state = yes
	}
}
