	id = NEP_1961_Crisis.1
	title = NEP_1961_Crisis.1.t
	desc = NEP_1961_Crisis.1.d
	picture = GFX_NEP_1961_Crisis_1
	fire_only_once = yes
	is_triggered_only = yes

	option = {
		name = NEP_1961_Crisis.1.a
		if = {
			limit = { NEP_trigger_royal_consolidation = yes }
			country_event = { id = NEP_1961_Crisis.10 days = 1 }
		}
		else_if = {
			limit = { NEP_trigger_rana_restoration = yes }
			country_event = { id = NEP_1961_Crisis.20 days = 1 }
		}
		else_if = {
			limit = { NEP_trigger_democratic_reform = yes }
			country_event = { id = NEP_1961_Crisis.30 days = 1 }
		}
		else_if = {
			limit = { NEP_trigger_nationalist_intervention = yes }
			country_event = { id = NEP_1961_Crisis.40 days = 1 }
		}
		else_if = {
			limit = { NEP_trigger_communist_chinese_aligned = yes }
			country_event = { id = NEP_1961_Crisis.50 days = 1 }
		}
		else_if = {
			limit = { NEP_trigger_communist_neutral = yes }
			country_event = { id = NEP_1961_Crisis.51 days = 1 }
		}
		else_if = {
			limit = { NEP_trigger_communist_indian_suppressed = yes }
			country_event = { id = NEP_1961_Crisis.52 days = 1 }
		}
		else = {
			country_event = { id = NEP_1961_Crisis.10 days = 1 }
		}
	}
}
