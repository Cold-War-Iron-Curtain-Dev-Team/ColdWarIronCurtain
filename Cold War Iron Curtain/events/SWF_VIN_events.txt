# This event is deliberately valid for both VIN and NLF, because the
# focus fires it in both country scopes. Hanoi selects the policy; the
# NLF receives a non-conflicting acknowledgement event.
country_event = {
	id = SWF_VIN.17
	title = SWF_VIN.17.t
	desc = SWF_VIN.17.d
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { tag = VIN }
			set_country_flag = VIN_southern_revisionist_question
		}
		else_if = {
			limit = { tag = NLF }
			set_country_flag = NLF_southern_revisionist_question
		}
	}

	option = {
		name = SWF_VIN.17.a
		trigger = {
			tag = VIN
		}
		ai_chance = {
			base = 20
			modifier = {
				factor = 0
				is_historical_focus_on = yes
			}
		}

		set_country_flag = VIN_southern_unity_prioritized
		add_political_power = -50
		if = {
			limit = { country_exists = NLF }
			NLF = {
				set_country_flag = NLF_southern_unity_prioritized
				add_war_support = 0.05
				if = {
					limit = { has_idea = NLF_fractured_command_structure }
					swap_ideas = {
						remove_idea = NLF_fractured_command_structure
						add_idea = NLF_guerilla_network
					}
				}
			}
		}
	}

	option = {
		name = SWF_VIN.17.b
		trigger = {
			tag = VIN
		}
		ai_chance = {
			base = 30
			modifier = {
				factor = 4
				is_historical_focus_on = yes
			}
		}

		set_country_flag = VIN_southern_orthodoxy_prioritized
		add_political_power = 25
		if = {
			limit = { country_exists = NLF }
			NLF = {
				set_country_flag = NLF_southern_orthodoxy_prioritized
				add_stability = -0.05
				if = {
					limit = { has_idea = NLF_vietminh_disarray }
					swap_ideas = {
						remove_idea = NLF_vietminh_disarray
						add_idea = NLF_rev_cohesion
					}
				}
				if = {
					limit = { NOT = { NLF_orthodox_southern_command_trigger = yes } }
					set_temp_variable = { nlf_command_choice = 2 }
					if = {
						limit = { date > 1953.3.01 }
						set_temp_variable = { nlf_command_choice = 3 }
					}
					nlf_set_southern_command = yes
				}
			}
		}
	}

	option = {
		name = SWF_VIN.17.c
		trigger = { tag = NLF }
		ai_chance = { factor = 1 }
	}

}
