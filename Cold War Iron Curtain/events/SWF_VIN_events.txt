# This event is deliberately valid for both VIN and NLF, because the
# focus fires it in both country scopes. Hanoi selects the policy; the
# NLF receives a non-conflicting acknowledgement event.
country_event = {
	id = SWF_VIN.17
	title = SWF_VIN.17.t
	desc = SWF_VIN.17.d
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { tag = VIN }
			set_country_flag = VIN_southern_revisionist_question
		}
		else_if = {
			limit = { tag = NLF }
			set_country_flag = NLF_southern_revisionist_question
		}
	}

	option = {
		name = SWF_VIN.17.a
		trigger = {
			tag = VIN
		}

		set_country_flag = VIN_southern_unity_prioritized
		NLF = {
			set_country_flag = NLF_southern_unity_prioritized
		}
	}

	option = {
		name = SWF_VIN.17.b
		trigger = {
			tag = VIN
		}

		set_country_flag = VIN_southern_orthodoxy_prioritized
		NLF = {
			set_country_flag = NLF_southern_orthodoxy_prioritized
		}
	}

	option = {
		name = SWF_VIN.17.c
		trigger = {
			tag = NLF
		}
	}
}
