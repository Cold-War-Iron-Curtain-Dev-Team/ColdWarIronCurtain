# Greek Elections 1950
country_event = {
	id = GRE_Elections.0 
 immediate = { }
	title = GRE_Elections.0.t
	desc = GRE_Elections.0.d
	picture = GFX_report_event_greek_election_1950

	fire_only_once = yes
	is_triggered_only = yes
	immediate = {
		complete_national_focus = GRE_Elections_of_1950
	}

	immediate = {
		create_country_leader = {
			name = "Sofoklis Venizelos"
			picture = "Sofoklis_Venizelos.dds"
			ideology = christian_democracy
		}
		set_global_flag = greek_elections_1950
	}

	# Christian_Democratic
	option = {
		name = GRE_Elections.0.a
		set_politics = {
			ruling_party = Christian_Democratic
		}
		complete_national_focus = GRE_Venizelos
		ai_chance = {
			factor = 10
			modifier = {
				is_historical_focus_on = yes
				factor = 100
			}
			modifier = {
				has_game_rule = {
					rule = GRE_1950_elections
					option = TSALDARIS_1950
				}
				factor = 0
			}
		}
	}

	# Conservative
	option = {
		name = GRE_Elections.0.b
		set_politics = {
			ruling_party = Liberal_Conservatism
		}
		complete_national_focus = GRE_Tsaldaris
		ai_chance = {
			factor = 10
			modifier = {
				has_game_rule = {
					rule = GRE_1950_elections
					option = VENIZELOS_1950
				}
				factor = 0
			}
		}
	}

}
